1//! Compile block graphs for fault-tolerant quantum circuits.
2//!
3//! Workspace crates are available as [`graph`], [`compile`], [`ir`], and other
4//! modules. Import common types and entry points with [`prelude`].
5//! Use [`Result`] for workflows spanning these modules; [`Error`] preserves
6//! their typed failures and supports automatic conversion with `?`.
7//!
8//! The whole pipeline in one call:
9//!
10//! ```
11//! use bloq::prelude::*;
12//!
13//! let stim_text = compile_to_stim(&GalleryItem::CNOT.build(), 3)?;
14//! assert!(stim_text.contains("QUBIT_COORDS"));
15//! # Ok::<(), bloq::Error>(())
16//! ```
17//!
18//! Spelled out, when the intermediate [`Bloq`](bloq_ir::Bloq) is wanted or a
19//! non-default [`CompileConfig`](bloq_compile::CompileConfig) applies:
20//!
21//! ```
22//! use bloq::prelude::*;
23//!
24//! let graph = GalleryItem::CNOT.build();
25//! let bloq = compile(&graph, 3)?;
26//! let stim_text = emit_bloq_stim(&bloq)?;
27//! assert!(stim_text.contains("QUBIT_COORDS"));
28//! # Ok::<(), bloq::Error>(())
29//! ```
30//!
31//! Compiling many related graphs with one configuration can use a
32//! [`CompileContext`](bloq_compile::CompileContext) to reuse their circuit
33//! templates.
34//!
35//! Default features are empty, so library use pulls in no CLI dependencies.
36//! The separate `bloq-cli` package provides the `bloq` command-line binary
37//! and the shared runner used by Python. `vm` exposes physical execution. `gltf` and
38//! `verify` forward to
39//! `bloq_graph/gltf` and `bloq_stim/verify` respectively. `graph-verify` exposes
40//! graph-level logical verification without requiring the native Stim backend.
41//! `hotpath` and `hotpath-alloc` add opt-in compiler-pipeline profiling.
42

57/// Compile `graph` at `distance` and emit it as Stim circuit text.
58///
59/// Uses a fresh template cache. Equivalent to
60/// [`bloq_compile::compile`] followed by [`bloq_stim::emit_bloq_stim`].
61///
62/// # Errors
63///
64/// Wraps [`compile::CompileError`] if `distance` is invalid or `graph` does not
65/// compile, and [`stim::StimEmissionError`] if the compiled program uses a
66/// node the static Stim backend cannot express (dynamic `RepeatUntilSuccess` or
67/// retry regions or unpinned quantum membership — use `bloq_vm` for dynamic execution).
68///
69/// # Examples
70///
71/// ```
72/// let stim_text = bloq::compile_to_stim(&bloq::graph::GalleryItem::CNOT.build(), 3)?;
73/// assert!(stim_text.contains("QUBIT_COORDS"));
74/// # Ok::<(), bloq::Error>(())
75/// ```
76pub fn compile_to_stim(graph: &BlockGraph, distance: u32) -> Result<String> {
77    let bloq = bloq_compile::compile(graph, distance)?;
78    Ok(bloq_stim::emit_bloq_stim(&bloq)?)
79}

136    fn module_stim(source: &str, distance: u32) -> Result<String> {
137        use crate::prelude::*;
138
139        let source = BlockGraph::from_text(source)?;
140        let config = CompileConfig::try_new(distance)?;
141        let artifacts = CompileContext::new(config).compile(&source)?;
142        let program = Bloq::from_binary(&artifacts.bloq.to_binary())?;
143        let program = Bloq::from_text(&program.to_text())?;
144        program.validate()?;
145        Ok(emit_bloq_stim(&program)?)
146    }

148    #[test]
149    fn facade_result_composes_modules_compilation_and_codecs() -> Result {
150        use std::error::Error as _;
151
152        let source = graph::GalleryItem::CNOT.build().to_blog_text();
153        assert!(module_stim(&source, 3)?.contains("QUBIT_COORDS"));
154        let error = module_stim("invalid BLOG", 3).unwrap_err();
155        assert!(error.is::<graph::BlockGraphError>());
156        assert!(error.source().unwrap().is::<graph::BlockGraphError>());
157        assert!(matches!(
158            module_stim(&source, 4).unwrap_err().downcast_ref(),
159            Some(compile::InvalidDistance(4))
160        ));
161        Ok(())
162    }

164    #[test]
165    fn facade_error_retains_typed_sources_and_resource_failures() {
166        use std::error::Error as _;
167
168        fn assert_thread_safe<E: std::error::Error + Send + Sync + 'static>() {}
169        assert_thread_safe::<Error>();
170
171        let resource = utils::boolean::BooleanResourceError {
172            resource: "Boolean steps",
173            observed: 2,
174            limit: 1,
175        };
176        let source = compile::CompileError::from(resource);
177        let message = source.to_string();
178        let error = Error::from(source);
179        assert_eq!(error.to_string(), message);
180        let source = error
181            .source()
182            .unwrap()
183            .downcast_ref::<compile::CompileError>()
184            .unwrap();
185        assert!(source.is_resource_limited());
186        assert!(
187            matches!(source, compile::CompileError::BooleanResource(actual) if *actual == resource)
188        );
189        assert!(error.is::<compile::CompileError>());
190        assert!(error.downcast_ref::<graph::ParseError>().is_none());
191
192        let error = Error::from(std::io::Error::other("output write failed"));
193        assert_eq!(error.to_string(), "output write failed");
194        assert!(error.source().unwrap().is::<std::io::Error>());
195    }

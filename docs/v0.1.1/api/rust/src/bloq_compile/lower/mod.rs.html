97        TemplateInstanceId(id)

101impl BloqLowerContext<'_> {
102    fn site_node(&self, source: ChunkSiteSource) -> NodeIndex {
103        match source {
104            ChunkSiteSource::Block(block_pos) => self.input.plan.node_by_block()[&block_pos],
105            ChunkSiteSource::TemporalPipe(pipe) => self.input.plan.node_by_temporal_pipe()[&pipe],
106            ChunkSiteSource::SpatialPort(port) => self.input.plan.node_by_spatial_port()[&port],
107            // A wall lives on the component node holding both its endpoints.
108            ChunkSiteSource::SpatialPipe(pipe) => self.input.plan.node_by_block()[&pipe.src],
109        }
110    }
111
112    /// The lowered template instance that `source` resolves to on its plan node.
113    fn instance_for_site(&self, source: ChunkSiteSource) -> &LoweredTemplateInstance {
114        let node = self.site_node(source);
115        self.node_instances[node.index()]
116            .iter()
117            .find(|instance| instance.site_source == source)
118            .expect("observable insertion site has a lowered template instance")
119    }
120}

122/// Source cut ownership needed after physical/semantic assembly is released.
123/// Operator inference still runs on optimized IR; the expanded source graph,
124/// signature maps, readout plan, and Boolean caches are no longer needed then.
125pub(crate) struct LogicalInterface {
126    inputs: crate::FxMap<TemplateInstanceId, IVec3>,
127    outputs: crate::FxMap<TemplateInstanceId, IVec3>,
128    multiplex: Vec<(IVec3, TemplateInstanceId, IVec2)>,
129    distance: u32,
130}
131
132impl LogicalInterface {
133    pub(crate) fn bind(
134        bloq: &Bloq,
135        graph: &BlockGraph,
136        instance_by_site: &crate::FxMap<ChunkSiteSource, TemplateInstanceId>,
137        spatial_ports: &SpatialPortExpansionMap,
138        distance: u32,
139    ) -> Self {
140        let inputs = instance_by_site
141            .iter()
142            .filter_map(|(source, &instance)| match source {
143                ChunkSiteSource::SpatialPort(port)
144                    if spatial_ports[port].role.has_input_boundary() =>
145                {
146                    Some((instance, *port))
147                }
148                ChunkSiteSource::Block(port)
149                    if !spatial_ports.contains_key(port)
150                        && graph.get_block(*port).is_some_and(|block| {
151                            block.kind() == BlockKind::Port
152                                && match block.port_role().expect("Port blocks expose a role") {
153                                    bloq_graph::PortRole::Input
154                                    | bloq_graph::PortRole::Multiplex => true,
155                                    bloq_graph::PortRole::Output => false,
156                                    bloq_graph::PortRole::Auto => graph
157                                        .neighbor_positions(*port)
158                                        .iter()
159                                        .all(|neighbor| neighbor.z > port.z),
160                                }
161                        }) =>
162                {
163                    Some((instance, *port))
164                }
165                _ => None,
166            })
167            .collect();
168        let frame_ports: crate::FxSet<IVec3> = bloq
169            .output_frames()
170            .into_iter()
171            .map(|frame| frame.port)
172            .collect();
173        let outputs = instance_by_site
174            .iter()
175            .filter_map(|(source, &instance)| match source {
176                ChunkSiteSource::SpatialPort(port)
177                    if frame_ports.contains(port)
178                        && spatial_ports[port].role != bloq_graph::PortRole::Multiplex =>
179                {
180                    Some((instance, *port))
181                }
182                ChunkSiteSource::Block(port)
183                    if frame_ports.contains(port) && !spatial_ports.contains_key(port) =>
184                {
185                    Some((instance, *port))
186                }
187                _ => None,
188            })
189            .collect();
190
191        let multiplex = spatial_ports
192            .values()
193            .filter(|port| port.role == bloq_graph::PortRole::Multiplex)
194            .map(|port| {
195                (
196                    port.source,
197                    instance_by_site[&ChunkSiteSource::SpatialPort(port.source)],
198                    port.output_qubit
199                        .expect("Multiplex expansions allocate an output qubit"),
200                )
201            })
202            .collect();
203        Self {
204            inputs,
205            outputs,
206            multiplex,
207            distance,
208        }
209    }
210}

212/// Shared finalization after every physical member and semantic recipe is bound.
213pub(crate) fn finish_program(
214    bloq: &mut Bloq,
215    interface: LogicalInterface,
216) -> Result<(), CompileError> {
217    hotpath::measure_block!(
218        "finish.occupancy_edges",
219        schedule::induce_occupancy_order_edges(bloq)
220    )?;
221    hotpath::measure_block!("finish.optimize", bloq.optimize())
222        .map_err(|_| BloqValidationError::CyclicGraph)?;
223    let inputs = hotpath::measure_block!(
224        "finish.logical_inputs",
225        logical_inputs(bloq, interface.inputs, interface.distance)
226    )?;
227    bloq.set_logical_inputs(inputs);
228    let outputs = hotpath::measure_block!("finish.logical_outputs", {
229        logical_outputs(
230            bloq,
231            interface.outputs,
232            interface.multiplex,
233            interface.distance,
234        )
235    })?;
236    bloq.set_logical_outputs(outputs);
237    Ok(())
238}

240/// Persist the complete input patches an execution hook may initialize.
241///
242/// Both ordinary temporal (`+Z`) Ports and compiler-expanded spatial input
243/// Ports land here. The bound interface is the authoritative source→instance
244/// map; the executor must not reconstruct it from fused-node provenance.
245fn logical_inputs(
246    bloq: &Bloq,
247    ports_by_instance: crate::FxMap<TemplateInstanceId, IVec3>,
248    distance: u32,
249) -> Result<Vec<LogicalInput>, CompileError> {
250    let (placements, mut faces) = logical_faces(bloq, bloq_ir::BoundaryFace::Input);
251
252    let mut inputs = Vec::new();
253    for (instance, port) in ports_by_instance {
254        let (template, offset) = placements[&instance];
255        let Some((x, z)) = boundary_operators(
256            &bloq.templates()[template],
257            faces.remove(&instance).unwrap_or_default(),
258            bloq_ir::BoundaryFace::Input,
259            offset,
260            distance,
261        )?
262        else {
263            continue;
264        };
265
266        inputs.push(LogicalInput {
267            port,
268            instance,
269            x,
270            z,
271        });
272    }
273    inputs.sort_by_key(|input| input.port.to_array());
274    Ok(inputs)
275}

277/// Persist the complete terminal logical operators the executor needs.
278/// Compiler geometry stays here; consumers read only the resulting IR maps.
279fn logical_outputs(
280    bloq: &Bloq,
281    ports_by_instance: crate::FxMap<TemplateInstanceId, IVec3>,
282    multiplex: Vec<(IVec3, TemplateInstanceId, IVec2)>,
283    distance: u32,
284) -> Result<Vec<LogicalOutput>, CompileError> {
285    let (placements, mut faces) = logical_faces(bloq, bloq_ir::BoundaryFace::Output);
286
287    let mut outputs = Vec::new();
288    for (instance, port) in ports_by_instance {
289        let (template, offset) = placements[&instance];
290        let Some((x, z)) = boundary_operators(
291            &bloq.templates()[template],
292            faces.remove(&instance).unwrap_or_default(),
293            bloq_ir::BoundaryFace::Output,
294            offset,
295            distance,
296        )?
297        else {
298            continue;
299        };
300        outputs.push(LogicalOutput {
301            port,
302            instance,
303            x,
304            z,
305        });
306    }
307    for (port, instance, output_qubit) in multiplex {
308        let qubit =
309            bloq_circuit::checked_translate_coordinate(output_qubit, placements[&instance].1)?;
310        for (_, node) in bloq.quantum_nodes() {
311            for other in &node.instances {
312                if other.id == instance {
313                    continue;
314                }
315                for local in bloq.templates()[other.template_id].qubits() {
316                    if bloq_circuit::checked_translate_coordinate(*local, other.offset)? == qubit {
317                        return Err(CompileError::MultiplexOutputCoordinateUnavailable { port });
318                    }
319                }
320            }
321        }
322        outputs.push(LogicalOutput {
323            port,
324            instance,
325            x: PauliMap::from_unique_entries([(qubit, Pauli::X)]),
326            z: PauliMap::from_unique_entries([(qubit, Pauli::Z)]),
327        });
328    }
329    outputs.sort_by_key(|output| output.port.to_array());
330    Ok(outputs)
331}

333type LogicalFaces = crate::FxMap<TemplateInstanceId, (Option<PauliMap>, Option<PauliMap>)>;
334
335/// Complete observed boundary operators in template coordinates, then place them.
336fn boundary_operators(
337    template: &bloq_ir::lowering::BloqTemplate,
338    observed: (Option<PauliMap>, Option<PauliMap>),
339    face: bloq_ir::BoundaryFace,
340    offset: IVec2,
341    distance: u32,
342) -> Result<Option<(PauliMap, PauliMap)>, CompileError> {
343    let x = observed.0.map(|map| rebase_map(&map, offset)).transpose()?;
344    let z = observed.1.map(|map| rebase_map(&map, offset)).transpose()?;
345    complete_logical_pair(x, z)
346        .or_else(|| template_logical_pair(template, face, distance))
347        .map(|(x, z)| Ok((x.try_translated(offset)?, z.try_translated(offset)?)))
348        .transpose()
349}
350
351fn logical_faces(
352    bloq: &Bloq,
353    face: bloq_ir::BoundaryFace,
354) -> (
355    crate::FxMap<TemplateInstanceId, (TemplateId, IVec2)>,
356    LogicalFaces,
357) {
358    let mut placements = crate::FxMap::default();
359    let mut faces: LogicalFaces = crate::FxMap::default();
360    bloq.walk(|cx| {
361        if let Some(quantum) = cx.node.try_quantum() {
362            placements.extend(
363                quantum
364                    .instances
365                    .iter()
366                    .map(|instance| (instance.id, (instance.template_id, instance.offset))),
367            );
368        }
369        if let Some(ClassicalNode::Observable { operators, .. }) = cx.node.try_classical() {
370            for operator in operators.iter().filter(|operator| operator.face == face) {
371                let entry = faces.entry(operator.instance).or_default();
372                match homogeneous_pauli(&operator.operator) {
373                    Some(bloq_circuit::Pauli::X) => {
374                        entry.0.get_or_insert_with(|| operator.operator.clone());
375                    }
376                    Some(bloq_circuit::Pauli::Z) => {
377                        entry.1.get_or_insert_with(|| operator.operator.clone());
378                    }
379                    _ => {}
380                }
381            }
382        }
383        WalkControl::Continue
384    });
385    (placements, faces)
386}

388#[hotpath::measure]
389fn template_logical_pair(
390    template: &bloq_ir::lowering::BloqTemplate,
391    face: bloq_ir::BoundaryFace,
392    distance: u32,
393) -> Option<(PauliMap, PauliMap)> {
394    use crate::block::fixed_bulk::observable::logical_line_operator;
395
396    let maps = template
397        .boundary_flows
398        .iter()
399        .filter_map(|flow| match face {
400            bloq_ir::BoundaryFace::Input if flow.start.is_empty() && !flow.end.is_empty() => {
401                Some(&flow.end)
402            }
403            bloq_ir::BoundaryFace::Output if !flow.start.is_empty() && flow.end.is_empty() => {
404                Some(&flow.start)
405            }
406            _ => None,
407        })
408        .collect::<Vec<_>>();
409    [Basis::Z, Basis::X].into_iter().find_map(|top_basis| {
410        let x = logical_line_operator(distance, Basis::X, top_basis, Pauli::X);
411        let z = logical_line_operator(distance, Basis::Z, top_basis, Pauli::Z);
412        maps.iter()
413            .all(|stabilizer| {
414                !pauli_maps_anticommute(&x, stabilizer) && !pauli_maps_anticommute(&z, stabilizer)
415            })
416            .then_some((x, z))
417    })
418}
419
420fn pauli_maps_anticommute(a: &PauliMap, b: &PauliMap) -> bool {
421    a.iter().fold(false, |odd, (coordinate, pauli)| {
422        odd ^ b
423            .get(coordinate)
424            .is_some_and(|other| pauli.anticommutes(*other))
425    })
426}
427
428fn homogeneous_pauli(map: &PauliMap) -> Option<bloq_circuit::Pauli> {
429    let pauli = *map.iter().next()?.1;
430    map.iter()
431        .all(|(_, candidate)| *candidate == pauli)
432        .then_some(pauli)
433}
434
435fn rebase_map(map: &PauliMap, offset: IVec2) -> Result<PauliMap, CompileError> {
436    map.iter()
437        .map(|(coordinate, pauli)| {
438            let x = coordinate.x.checked_sub(offset.x).ok_or(
439                CompileError::LogicalBoundaryRebaseOverflow {
440                    coordinate: *coordinate,
441                    offset,
442                },
443            )?;
444            let y = coordinate.y.checked_sub(offset.y).ok_or(
445                CompileError::LogicalBoundaryRebaseOverflow {
446                    coordinate: *coordinate,
447                    offset,
448                },
449            )?;
450            Ok((IVec2::new(x, y), *pauli))
451        })
452        .collect()
453}
454
455fn complete_logical_pair(x: Option<PauliMap>, z: Option<PauliMap>) -> Option<(PauliMap, PauliMap)> {
456    match (x, z) {
457        (Some(x), Some(z)) => Some((x, z)),
458        (Some(x), None) => perpendicular_line(&x, Pauli::X).map(|z| (x, z)),
459        (None, Some(z)) => perpendicular_line(&z, Pauli::Z).map(|x| (x, z)),
460        (None, None) => None,
461    }
462}
463
464fn perpendicular_line(known: &PauliMap, known_pauli: Pauli) -> Option<PauliMap> {
465    use crate::block::fixed_bulk::observable::logical_line_operator;
466
467    let distance = known.len() as u32;
468    let (known_basis, other_basis, other_pauli) = match known_pauli {
469        Pauli::X => (Basis::X, Basis::Z, Pauli::Z),
470        _ => (Basis::Z, Basis::X, Pauli::X),
471    };
472    let top_basis = [Basis::Z, Basis::X]
473        .into_iter()
474        .find(|&top| &logical_line_operator(distance, known_basis, top, known_pauli) == known)?;
475    Some(logical_line_operator(
476        distance,
477        other_basis,
478        top_basis,
479        other_pauli,
480    ))
481}

483/// The program under construction, the pool it copies templates from, and the
484/// dedup table between them. Bundled because instantiating a template always
485/// needs all three, and region building threads them through every layer.
486pub(super) struct TemplateRemapper<'a> {
487    bloq: &'a mut Bloq,
488    program_template_ids: &'a mut crate::FxMap<LoweringTemplateId, TemplateId>,
489    pool: &'a LoweringTemplatePool,
490}
491
492impl<'a> TemplateRemapper<'a> {
493    fn new(
494        bloq: &'a mut Bloq,
495        program_template_ids: &'a mut crate::FxMap<LoweringTemplateId, TemplateId>,
496        pool: &'a LoweringTemplatePool,
497    ) -> Self {
498        Self {
499            bloq,
500            program_template_ids,
501            pool,
502        }
503    }
504
505    /// Lazily copy a pool template into the program, dedup'd by
506    /// `LoweringTemplateId`, and return its program-local `TemplateId`.
507    pub(super) fn remap(&mut self, lowering_template_id: LoweringTemplateId) -> TemplateId {
508        let (bloq, pool) = (&mut *self.bloq, self.pool);
509        *self
510            .program_template_ids
511            .entry(lowering_template_id)
512            .or_insert_with(|| {
513                bloq.add_shared_template(pool[lowering_template_id].program_template.clone())
514            })
515    }
516
517    /// The pool being copied from, for callers that read a template's side
518    /// tables (gateways, seams) rather than instantiate it.
519    pub(super) fn pool(&self) -> &LoweringTemplatePool {
520        self.pool
521    }

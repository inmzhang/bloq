17pub(crate) fn lower_clifford_proxy(
18    input: PhysicalInput<'_>,
19    stabilizers: &StabilizerGenerators,
20) -> Result<Bloq, CompileError> {
21    check_resource(
22        "readout IDs",
23        stabilizers.generators.len(),
24        u32::MAX as usize,
25    )?;
26    let mut placed = PlacedPlan::new(input)?;
27    placed.compose_detectors()?;
28    let (observables, reference_offsets) = resolve_observables(stabilizers, &placed.context())?;
29    let PhysicalProgram {
30        mut bloq,
31        placements,
32        node_map,
33    } = placed.emit(stabilizers.generators.len() as u32)?;
34    debug_assert!(
35        placements.selectives.is_empty(),
36        "proxy selectives are pinned"
37    );
38    let rows = statically_resolved_row_indices(&stabilizers.generators, input.compiled);
39    let mut materialized = materialize_observable_nodes(observables, &rows, &mut bloq, &node_map);
40    materialized.reference_offsets = reference_offsets;
41    emit_frames(input, stabilizers, &placements, &materialized, &mut bloq)?;
42    let interface = super::LogicalInterface::bind(
43        &bloq,
44        input.graph,
45        &placements.instances,
46        input.spatial_ports,
47        input.layout.distance(),
48    );
49    drop((materialized, placements, node_map));
50    super::finish_program(&mut bloq, interface)?;
51    #[cfg(debug_assertions)]
52    super::validate_bloq_qubit_layout_for_source(&bloq, input.graph)?;
53    Ok(bloq)
54}

56/// The proxy retains its pinned public basis for distance tests. Complete
57/// surfaces still supply private parity, including rows with no public member.
58fn emit_frames(
59    input: PhysicalInput<'_>,
60    stabilizers: &StabilizerGenerators,
61    placements: &PhysicalPlacements,
62    materialized: &MaterializedObservables,
63    bloq: &mut Bloq,
64) -> Result<(), CompileError> {
65    let outputs = stabilizers.zx_graph.output_ports();
66    let (rows, surfaces): (Vec<_>, Vec<_>) = RuntimeStabilizerBasis::from_generators(stabilizers)
67        .with_t_nodes_as_ports()
68        .into_output_correction_surfaces(&outputs)?
69        .into_iter()
70        .unzip();
71    let correction = bloq_graph::solve_output_correction_symbolic(&outputs, &rows)?;
72    let pipe_templates = input
73        .plan
74        .graph()
75        .node_weights()
76        .filter_map(|node| {
77            let NodeProvenance::TemporalPipe { pipe } = node.provenance else {
78                return None;
79            };
80            node.template
81                .map(|template| (pipe, input.temporal_templates[&template]))
82        })
83        .collect();
84    let context = LocalSurfaceContext {
85        graph: input.graph,
86        compiled: input.compiled,
87        spatial_ports: input.spatial_ports,
88        spatial_templates: input.spatial_port_templates,
89        wall_templates: input.wall_templates,
90        pipe_templates: &pipe_templates,
91        pool: input.template_pool,
92        instances: &placements.instances,
93        selectives: &placements.selectives,
94        layout: input.layout,
95    };
96    let mut owners = bloq
97        .nodes()
98        .flat_map(|(id, node)| {
99            node.try_quantum().into_iter().flat_map(move |quantum| {
100                quantum
101                    .instances
102                    .iter()
103                    .map(move |instance| (instance.id, id))
104            })
105        })
106        .collect::<crate::FxMap<_, _>>();
107    for (site, node) in &placements.regions {
108        if let Some(instance) = placements.instances.get(&ChunkSiteSource::Block(*site)) {
109            owners.insert(*instance, *node);
110        }
111    }
112    let mut terms = Vec::with_capacity(rows.len());
113    for (index, (row, surface)) in rows.iter().zip(surfaces).enumerate() {
114        let mut positions = surface
115            .interior_nodes
116            .iter()
117            .chain(&surface.port_stabilizer)
118            .filter_map(|(&pos, &pauli)| (pauli != Pauli::I).then_some(pos))
119            .chain(
120                surface
121                    .interior_edges
122                    .iter()
123                    .filter(|(_, pauli)| **pauli != Pauli::I)
124                    .flat_map(|(&(left, right), _)| [left, right]),
125            )
126            .collect::<Vec<_>>();
127        positions.sort_unstable_by_key(glam::IVec3::to_array);
128        positions.dedup();
129        let (records, _, mut sign) = context.resolve(
130            &positions,
131            &surface,
132            None,
133            &StabilizerRowKind::Logical,
134            index as u32,
135            None,
136        )?;
137        let predecessors = records
138            .iter()
139            .map(|record| owners[&record.instance])
140            .collect::<crate::FxSet<_>>();
141        let parity = bloq.add_node(BloqNode::classical(ClassicalNode::observable_fragment(
142            records,
143            vec![],
144        )));
145        for owner in predecessors {
146            bloq.add_edge(owner, parity, BloqEdge::Order);
147        }
148        for (slot, &ordinal) in row.readout_ordinals.iter().enumerate() {
149            let ordinal = ordinal as u32;
150            let observable = materialized
151                .observable_by_index
152                .get(&ordinal)
153                .copied()
154                .expect("frame member has a public readout");
155            bloq.add_edge(observable, parity, BloqEdge::flip(slot as u32));
156            sign ^= materialized.reference_offsets.contains(&ordinal);
157        }
158        terms.push((parity, sign));
159    }
160    for pair in correction.frames() {
161        for (basis, rows) in [(Basis::X, &pair.x), (Basis::Z, &pair.z)] {
162            let mut leaves = ExprLeaves::default();
163            let mut sign = false;
164            let mut operands = Vec::new();
165            for &row in rows {
166                let (input, offset) = &terms[row];
167                sign ^= offset;
168                operands.push(leaves.input(*input));
169            }
170            let mut expr = ClassicalExpr::xor(operands);
171            if sign {
172                expr = ClassicalExpr::Not(Box::new(expr));
173            }
174            let frame = bloq.add_node(
175                BloqNode::classical(ClassicalNode::Compute { expr }).with_provenance(
176                    NodeProvenance::OutputFrame {
177                        port: pair.output,
178                        basis,
179                    },
180                ),
181            );
182            leaves.wire(bloq, frame);
183        }
184    }
185    Ok(())
186}

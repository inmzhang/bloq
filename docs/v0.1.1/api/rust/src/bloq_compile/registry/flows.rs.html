35#[derive(Clone)]
36struct FlowRow {
37    body: BooleanRow,
38    /// Canonical X^x Z^z phase, rather than the Hermitian-Pauli phase.
39    phase: [DecisionId; 2],
40    parity: BTreeMap<InstanceMeasurement, DecisionId>,
41    centers: BTreeMap<[i32; 2], DecisionId>,
42    discard: DecisionId,
43    restart: DecisionId,
44    latent: bool,
45}
46
47impl FlowRow {
48    fn empty(active: DecisionId) -> Self {
49        Self {
50            body: BooleanRow::new(active),
51            phase: [ZERO; 2],
52            parity: BTreeMap::new(),
53            centers: BTreeMap::new(),
54            discard: ZERO,
55            restart: ZERO,
56            latent: false,
57        }
58    }
59
60    fn multiply(
61        &mut self,
62        other: &Self,
63        factor: DecisionId,
64        diagram: &mut BooleanDecisionDiagram,
65    ) -> Result<(), CompileError> {
66        if factor == ZERO {
67            return Ok(());
68        }
69        diagram.charge(
70            other
71                .body
72                .terms()
73                .len()
74                .saturating_add(other.parity.len())
75                .saturating_add(other.centers.len()),
76        )?;
77        let low = diagram.apply(BooleanOp::And, factor, other.phase[0])?;
78        let high = diagram.apply(BooleanOp::And, factor, other.phase[1])?;
79        let carry = diagram.apply(BooleanOp::And, self.phase[0], low)?;
80        self.phase[0] = diagram.apply(BooleanOp::Xor, self.phase[0], low)?;
81        self.phase[1] = diagram.apply(BooleanOp::Xor, self.phase[1], high)?;
82        self.phase[1] = diagram.apply(BooleanOp::Xor, self.phase[1], carry)?;
83        for &(column, x) in other
84            .body
85            .terms()
86            .iter()
87            .filter(|(column, _)| *column % 2 == 0)
88        {
89            let x = diagram.apply(BooleanOp::And, factor, x)?;
90            let phase = diagram.apply(BooleanOp::And, self.body.get(column + 1), x)?;
91            self.phase[1] = diagram.apply(BooleanOp::Xor, self.phase[1], phase)?;
92        }
93        self.body.xor_scaled(&other.body, factor, diagram)?;
94        self.multiply_annotations(other, factor, diagram)
95    }
96
97    fn multiply_annotations(
98        &mut self,
99        other: &Self,
100        factor: DecisionId,
101        diagram: &mut BooleanDecisionDiagram,
102    ) -> Result<(), CompileError> {
103        for (&measurement, &coefficient) in &other.parity {
104            let delta = diagram.apply(BooleanOp::And, factor, coefficient)?;
105            let current = self.parity.get(&measurement).copied().unwrap_or(ZERO);
106            let next = diagram.apply(BooleanOp::Xor, current, delta)?;
107            if next == ZERO {
108                self.parity.remove(&measurement);
109            } else {
110                self.parity.insert(measurement, next);
111            }
112        }
113        let discarded = diagram.apply(BooleanOp::And, factor, other.discard)?;
114        self.discard = diagram.apply(BooleanOp::Or, self.discard, discarded)?;
115        let restarted = diagram.apply(BooleanOp::And, factor, other.restart)?;
116        self.restart = diagram.apply(BooleanOp::Or, self.restart, restarted)?;
117        let mut centered = self
118            .centers
119            .values()
120            .copied()
121            .try_fold(ZERO, |sum, bit| diagram.apply(BooleanOp::Or, sum, bit))?;
122        for (&center, &coefficient) in &other.centers {
123            let absent = diagram.negate(centered)?;
124            let selected = diagram.apply(BooleanOp::And, factor, coefficient)?;
125            let selected = diagram.apply(BooleanOp::And, absent, selected)?;
126            let previous = self.centers.get(&center).copied().unwrap_or(ZERO);
127            self.centers
128                .insert(center, diagram.apply(BooleanOp::Or, previous, selected)?);
129            centered = diagram.apply(BooleanOp::Or, centered, selected)?;
130        }
131        Ok(())
132    }
133
134    fn close_matching(
135        source: &Self,
136        previous: &Self,
137        selected: DecisionId,
138        diagram: &mut BooleanDecisionDiagram,
139    ) -> Result<Self, CompileError> {
140        // Source is the raw Hermitian PauliMap from source(); its low phase bit
141        // is the Y parity. Selected includes same_body, so the bodies cancel
142        // and their cross phase is exactly the selected source low bit.
143        diagram.charge(
144            source
145                .parity
146                .len()
147                .saturating_add(source.centers.len())
148                .saturating_add(previous.parity.len())
149                .saturating_add(previous.centers.len()),
150        )?;
151        let mut closed = Self::empty(selected);
152        let source_low = diagram.apply(BooleanOp::And, selected, source.phase[0])?;
153        let source_high = diagram.apply(BooleanOp::And, selected, source.phase[1])?;
154        let previous_low = diagram.apply(BooleanOp::And, selected, previous.phase[0])?;
155        let previous_high = diagram.apply(BooleanOp::And, selected, previous.phase[1])?;
156        let carry = diagram.apply(BooleanOp::And, source_low, previous_low)?;
157        closed.phase[0] = diagram.apply(BooleanOp::Xor, source_low, previous_low)?;
158        closed.phase[1] = diagram.apply(BooleanOp::Xor, source_high, previous_high)?;
159        closed.phase[1] = diagram.apply(BooleanOp::Xor, closed.phase[1], carry)?;
160        closed.phase[1] = diagram.apply(BooleanOp::Xor, closed.phase[1], source_low)?;
161        closed.multiply_annotations(source, selected, diagram)?;
162        closed.multiply_annotations(previous, selected, diagram)?;
163        Ok(closed)
164    }
165
166    fn same_body(
167        &self,
168        other: &Self,
169        diagram: &mut BooleanDecisionDiagram,
170    ) -> Result<DecisionId, CompileError> {
171        diagram.charge(
172            self.body
173                .terms()
174                .len()
175                .saturating_add(other.body.terms().len()),
176        )?;
177        if self
178            .body
179            .terms()
180            .iter()
181            .chain(other.body.terms())
182            .all(|&(_, bit)| bit == ONE)
183        {
184            return Ok(if self.body.terms() == other.body.terms() {
185                ONE
186            } else {
187                ZERO
188            });
189        }
190        let mut equal = ONE;
191        for &(column, _) in self.body.terms().iter().chain(other.body.terms()) {
192            let difference = diagram.apply(
193                BooleanOp::Xor,
194                self.body.get(column),
195                other.body.get(column),
196            )?;
197            let same = diagram.negate(difference)?;
198            equal = diagram.apply(BooleanOp::And, equal, same)?;
199            if equal == ZERO {
200                break;
201            }
202        }
203        Ok(equal)
204    }
205
206    fn merge_disjoint(
207        &mut self,
208        row: Self,
209        diagram: &mut BooleanDecisionDiagram,
210        domain: DecisionId,
211    ) -> Result<(), CompileError> {
212        let overlap = diagram.apply(BooleanOp::And, self.body.active, row.body.active)?;
213        if diagram.constrain(overlap, domain)? != ZERO {
214            return Err(super::assembly("duplicate open boundary flow"));
215        }
216        let mut selected =
217            FlowRow::empty(diagram.apply(BooleanOp::Or, self.body.active, row.body.active)?);
218        selected.multiply(self, self.body.active, diagram)?;
219        selected.multiply(&row, row.body.active, diagram)?;
220        // Equal bodies stay fixed rather than carrying their guard.
221        let active = selected.body.active;
222        selected.body = self.body.clone();
223        selected.body.active = active;
224        selected.latent = row.latent;
225        *self = selected;
226        Ok(())
227    }
228}

230fn witness_terms(row: &BooleanRow, first_column: usize) -> &[(usize, DecisionId)] {
231    let terms = row.terms();
232    &terms[terms.partition_point(|&(column, _)| column < first_column)..]
233}

236pub(super) struct GuardedFlowEngine {
237    qubits: crate::FxMap<[i32; 2], usize>,
238    rows: Vec<FlowRow>,
239    constant: crate::FxMap<Vec<(usize, DecisionId)>, Vec<usize>>,
240    conditional: Vec<usize>,
241    latent: Vec<usize>,
242    vacant: usize,
243}

245impl GuardedFlowEngine {
246    fn source_body(&mut self, paulis: &PauliMap, active: DecisionId) -> FlowRow {
247        let mut row = FlowRow::empty(active);
248        let mut phase = 0u8;
249        for (qubit, pauli) in paulis {
250            let next = self.qubits.len();
251            let index = *self.qubits.entry(qubit.to_array()).or_insert(next);
252            for (axis, pauli_axis) in [Pauli::X, Pauli::Z].into_iter().enumerate() {
253                if *pauli & pauli_axis {
254                    row.body.set(2 * index + axis, ONE);
255                }
256            }
257            phase = (phase + u8::from(*pauli == Pauli::Y)) % 4;
258        }
259        row.phase = [
260            if phase & 1 != 0 { ONE } else { ZERO },
261            if phase & 2 != 0 { ONE } else { ZERO },
262        ];
263        row
264    }
265
266    fn source(
267        &mut self,
268        flow: &BoundaryFlow,
269        output: bool,
270        diagram: &mut BooleanDecisionDiagram,
271    ) -> Result<FlowRow, CompileError> {
272        let paulis = if output { &flow.end } else { &flow.start };
273        diagram.charge(
274            paulis
275                .len()
276                .saturating_add(flow.measurements.len())
277                .saturating_add(1),
278        )?;
279        let mut row = self.source_body(paulis, flow.guard);
280        if flow.sign {
281            row.phase[1] = diagram.negate(row.phase[1])?;
282        }
283        for &measurement in &flow.measurements {
284            if row.parity.remove(&measurement).is_none() {
285                row.parity.insert(measurement, ONE);
286            }
287        }
288        if let Some(center) = flow.center {
289            row.centers.insert(center, ONE);
290        }
291        match flow.marker {
292            FlowMarker::Detector => {}
293            FlowMarker::Discard => row.discard = ONE,
294            FlowMarker::Restart => row.restart = ONE,
295        }
296        Ok(row)
297    }

299    #[hotpath::measure]
300    pub(super) fn append(
301        &mut self,
302        flows: &[BoundaryFlow],
303        diagram: &mut BooleanDecisionDiagram,
304        domain: DecisionId,
305    ) -> Result<Vec<GuardedCheck>, CompileError> {
306        let mut output = Vec::new();
307        if flows.is_empty() {
308            return Ok(output);
309        }
310        let mut active = ZERO;
311        let mut incoming = Vec::new();
312        let mut creators = Vec::new();
313        for flow in flows {
314            active = diagram.apply(BooleanOp::Or, active, flow.guard)?;
315            if flow.start.is_empty() {
316                creators.push(self.source(flow, true, diagram)?);
317            } else {
318                incoming.push((flow, self.source(flow, false, diagram)?));
319            }
320        }
321        let mut exact = active;
322        diagram.charge(self.latent.len())?;
323        for &index in &self.latent {
324            let absent = diagram.negate(self.rows[index].body.active)?;
325            exact = diagram.apply(BooleanOp::And, exact, absent)?;
326        }
327        let mut incoming_guards = crate::FxMap::default();
328        diagram.charge(incoming.len())?;
329        let mut matches = Vec::with_capacity(incoming.len());
330        for (flow, row) in &incoming {
331            diagram.charge(row.body.terms().len())?;
332            let guard = incoming_guards.entry(row.body.terms()).or_insert(ZERO);
333            let duplicate = diagram.apply(BooleanOp::And, row.body.active, *guard)?;
334            if diagram.constrain(duplicate, domain)? != ZERO {
335                return Err(super::assembly(
336                    "incoming flow boundaries are not independent",
337                ));
338            }
339            *guard = diagram.apply(BooleanOp::Or, *guard, row.body.active)?;
340            let constant = self
341                .constant
342                .get(row.body.terms())
343                .map(Vec::as_slice)
344                .unwrap_or_default();
345            diagram.charge(constant.len().saturating_add(self.conditional.len()))?;
346            let mut candidates = constant
347                .iter()
348                .map(|&index| (index, ONE))
349                .collect::<Vec<_>>();
350            for &index in &self.conditional {
351                let same = self.rows[index].same_body(row, diagram)?;
352                if same != ZERO {
353                    candidates.push((index, same));
354                }
355            }
356            // Preserve multiplication and detector-center precedence.
357            candidates.sort_unstable_by_key(|&(index, _)| index);
358            if !flow.skip_unmatched {
359                let mut present = ZERO;
360                for &(index, same) in &candidates {
361                    let old = &self.rows[index];
362                    if !old.latent {
363                        let matched = diagram.apply(BooleanOp::And, old.body.active, same)?;
364                        present = diagram.apply(BooleanOp::Or, present, matched)?;
365                    }
366                }
367                let absent = diagram.negate(row.body.active)?;
368                let valid = diagram.apply(BooleanOp::Or, absent, present)?;
369                exact = diagram.apply(BooleanOp::And, exact, valid)?;
370            }
371            matches.push(candidates);
372        }
373        exact = diagram.constrain(exact, domain)?;
374        let not_exact = diagram.negate(exact)?;
375        let general = diagram.apply(BooleanOp::And, active, not_exact)?;
376        let general = diagram.constrain(general, domain)?;
377        // Constant consumers only change matched rows. Keep their slots stable
378        // until enough are empty to amortize compaction. Relays and span changes
379        // retain the full ordered path below, including detector-center order.
380        let incremental = general == ZERO
381            && self.conditional.is_empty()
382            && self.latent.is_empty()
383            && incoming.iter().all(|(flow, _)| flow.end.is_empty());
384        let mut old = if incremental {
385            Vec::new()
386        } else {
387            std::mem::take(&mut self.rows)
388        };
389        if exact != ZERO {
390            for ((flow, row), candidates) in incoming.iter().zip(matches) {
391                let mut combined = FlowRow::empty(ZERO);
392                // Consuming matches changes availability, never their bodies.
393                for (index, same) in candidates {
394                    let previous = if incremental {
395                        &mut self.rows[index]
396                    } else {
397                        &mut old[index]
398                    };
399                    let selected = diagram.apply(BooleanOp::And, exact, row.body.active)?;
400                    let selected = diagram.apply(BooleanOp::And, selected, previous.body.active)?;
401                    let selected = diagram.apply(BooleanOp::And, selected, same)?;
402                    if selected == ZERO {
403                        continue;
404                    }
405                    let closed = FlowRow::close_matching(row, previous, selected, diagram)?;
406                    if combined.body.active == ZERO {
407                        // The closed row's coefficients already imply selected.
408                        combined = closed;
409                    } else {
410                        combined.multiply(&closed, selected, diagram)?;
411                        combined.body.active =
412                            diagram.apply(BooleanOp::Or, combined.body.active, selected)?;
413                    }
414                    let unselected = diagram.negate(selected)?;
415                    previous.body.active =
416                        diagram.apply(BooleanOp::And, previous.body.active, unselected)?;
417                    if incremental {
418                        previous.body.active = diagram.constrain(previous.body.active, domain)?;
419                        if previous.body.active == ZERO {
420                            diagram.charge(row.body.terms().len())?;
421                            self.constant.remove(row.body.terms());
422                            *previous = FlowRow::empty(ZERO);
423                            self.vacant += 1;
424                        }
425                    }
426                }
427                if combined.body.active == ZERO {
428                    continue;
429                }
430                if flow.end.is_empty() {
431                    self.checks(combined, diagram, domain, &mut output)?;
432                } else {
433                    // The start relation owns measurements, markers, centers,
434                    // and flow sign. The end adds only its canonical Y phase.
435                    diagram.charge(flow.end.len().saturating_add(1))?;
436                    let end = self.source_body(&flow.end, flow.guard);
437                    combined.multiply(&end, combined.body.active, diagram)?;
438                    self.rows.push(combined);
439                }
440            }
441        }
442        let mut transitioning = Vec::new();
443        for mut row in old {
444            if general != ZERO {
445                let selected = diagram.apply(BooleanOp::And, row.body.active, general)?;
446                if selected != ZERO {
447                    let mut selected_row = row.clone();
448                    selected_row.body.active = selected;
449                    transitioning.push(selected_row);
450                    let keep = diagram.negate(general)?;
451                    row.body.active = diagram.apply(BooleanOp::And, row.body.active, keep)?;
452                }
453            }
454            if row.body.active != ZERO {
455                self.rows.push(row);
456            }
457        }
458        if general != ZERO {
459            let mut measured = Vec::new();
460            for (flow, mut row) in incoming {
461                row.body.active = diagram.apply(BooleanOp::And, row.body.active, general)?;
462                if row.body.active == ZERO {
463                    continue;
464                }
465                if !flow.end.is_empty() {
466                    return Err(super::assembly(
467                        "stabilizer-span transitions require boundary consumer flows",
468                    ));
469                }
470                if flow.skip_unmatched {
471                    let mut matched = ZERO;
472                    for old in &transitioning {
473                        let same = old.same_body(&row, diagram)?;
474                        let same = diagram.apply(BooleanOp::And, same, old.body.active)?;
475                        matched = diagram.apply(BooleanOp::Or, matched, same)?;
476                    }
477                    row.body.active = diagram.apply(BooleanOp::And, row.body.active, matched)?;
478                }
479                if row.body.active != ZERO {
480                    measured.push(row);
481                }
482            }
483            self.transition(transitioning, measured, diagram, domain, &mut output)?;
484        }
485        for creator in creators {
486            if creator.body.terms().is_empty() {
487                self.checks(creator, diagram, domain, &mut output)?;
488            } else if incremental {
489                self.insert_constant(creator, diagram, domain)?;
490            } else {
491                self.rows.push(creator);
492            }
493        }
494        if !incremental {
495            self.compact(diagram, domain)?;
496        } else if self.vacant > self.rows.len() / 2 {
497            self.compact_constant(diagram, domain)?;
498        }
499        Ok(output)
500    }

502    fn insert_constant(
503        &mut self,
504        mut row: FlowRow,
505        diagram: &mut BooleanDecisionDiagram,
506        domain: DecisionId,
507    ) -> Result<(), CompileError> {
508        row.body.active = diagram.constrain(row.body.active, domain)?;
509        if row.body.active == ZERO {
510            return Ok(());
511        }
512        diagram.charge(row.body.terms().len().saturating_add(1))?;
513        if let Some(indices) = self.constant.get(row.body.terms()) {
514            self.rows[indices[0]].merge_disjoint(row, diagram, domain)?;
515        } else {
516            self.constant
517                .insert(row.body.terms().to_vec(), vec![self.rows.len()]);
518            self.rows.push(row);
519        }
520        Ok(())
521    }

523    #[hotpath::measure]
524    fn matrix(
525        &self,
526        sources: &[FlowRow],
527        diagram: &mut BooleanDecisionDiagram,
528    ) -> Result<Vec<BooleanRow>, CompileError> {
529        diagram.charge(sources.len())?;
530        diagram.charge(sources.iter().fold(0usize, |terms, source| {
531            terms.saturating_add(source.body.terms().len().saturating_add(1))
532        }))?;
533        let width = self.qubits.len() * 2;
534        Ok(sources
535            .iter()
536            .enumerate()
537            .map(|(index, source)| {
538                let mut row = source.body.clone();
539                row.set(width + index, ONE);
540                row
541            })
542            .collect())
543    }

545    fn product(
546        &self,
547        combination: &BooleanRow,
548        sources: &[FlowRow],
549        first_source: usize,
550        diagram: &mut BooleanDecisionDiagram,
551    ) -> Result<FlowRow, CompileError> {
552        let width = self.qubits.len() * 2;
553        let witnesses = witness_terms(combination, width);
554        diagram.charge(witnesses.len())?;
555        let split = witnesses.partition_point(|&(column, _)| column < width + first_source);
556        let mut result = FlowRow::empty(combination.active);
557        // Preserve the requested cyclic source order while skipping every zero
558        // witness. Cross-group Pauli products and center precedence are ordered.
559        for &(column, coefficient) in witnesses[split..].iter().chain(&witnesses[..split]) {
560            let coefficient = diagram.apply(BooleanOp::And, combination.active, coefficient)?;
561            result.multiply(&sources[column - width], coefficient, diagram)?;
562        }
563        Ok(result)
564    }

566    #[hotpath::measure]
567    fn transition(
568        &mut self,
569        old: Vec<FlowRow>,
570        incoming: Vec<FlowRow>,
571        diagram: &mut BooleanDecisionDiagram,
572        domain: DecisionId,
573        output: &mut Vec<GuardedCheck>,
574    ) -> Result<(), CompileError> {
575        let width = self.qubits.len() * 2;
576        let old_len = old.len();
577        let mut sources = old;
578        sources.extend(incoming);
579        let mut combined = BooleanRowSpace::new(self.matrix(&sources, diagram)?, diagram)?;
580        diagram.charge(combined.columns().len())?;
581        let columns = combined
582            .columns()
583            .filter(|&column| column < width)
584            .collect::<BTreeSet<_>>();
585        for column in columns {
586            combined.eliminate_column(column, diagram)?;
587        }
588        for dependency in combined.into_rows(diagram)? {
589            let witnesses = witness_terms(&dependency, width);
590            diagram.charge(witnesses.len())?;
591            let mut has_old = ZERO;
592            let mut has_new = ZERO;
593            for &(column, coefficient) in witnesses {
594                let present = if column < width + old_len {
595                    &mut has_old
596                } else {
597                    &mut has_new
598                };
599                *present = diagram.apply(BooleanOp::Or, *present, coefficient)?;
600            }
601            let both = diagram.apply(BooleanOp::And, has_old, has_new)?;
602            let missing = diagram.negate(both)?;
603            let invalid = diagram.apply(BooleanOp::And, dependency.active, missing)?;
604            if diagram.constrain(invalid, domain)? != ZERO {
605                return Err(super::assembly(
606                    "boundary flow generators are not independent",
607                ));
608            }
609            // The measured product precedes the old product, exactly as in the
610            // ordinary flow engine. Cross-group generators need not commute.
611            let row = self.product(&dependency, &sources, old_len, diagram)?;
612            self.checks(row, diagram, domain, output)?;
613        }
614        let old = &sources[..old_len];
615        let mut surviving = BooleanRowSpace::new(self.matrix(old, diagram)?, diagram)?;
616        for measured in &sources[old_len..] {
617            diagram.charge(measured.body.terms().len())?;
618            // Distribute the measurement activation over the symplectic sum:
619            // active & XOR(x_i z'_i, z_i x'_i). The guard still gates every term.
620            let terms = measured
621                .body
622                .terms()
623                .iter()
624                .map(|&(column, value)| {
625                    diagram
626                        .apply(BooleanOp::And, measured.body.active, value)
627                        .map(|weight| (column ^ 1, weight))
628                })
629                .collect::<Result<Vec<_>, _>>()?;
630            surviving.eliminate_linear(&terms, diagram)?;
631        }
632        let surviving = surviving
633            .into_rows(diagram)?
634            .iter()
635            .map(|row| self.product(row, old, 0, diagram))
636            .collect::<Result<Vec<_>, _>>()?;
637        sources.drain(..old_len);
638        let measured = sources.len();
639        sources.extend(surviving);
640        let mut rows = BooleanRowSpace::new(self.matrix(&sources, diagram)?, diagram)?;
641        diagram.charge(rows.columns().len())?;
642        let columns = rows
643            .columns()
644            .filter(|&column| column < width)
645            .collect::<BTreeSet<_>>();
646        for column in columns {
647            let mut pivot = rows.eliminate_column(column, diagram)?;
648            let witnesses = witness_terms(&pivot, width + measured);
649            diagram.charge(witnesses.len())?;
650            let from_old = witnesses
651                .iter()
652                .try_fold(ZERO, |value, &(_, coefficient)| {
653                    diagram.apply(BooleanOp::Or, value, coefficient)
654                })?;
655            pivot.active = diagram.apply(BooleanOp::And, pivot.active, from_old)?;
656            if pivot.active == ZERO {
657                continue;
658            }
659            let mut row = self.product(&pivot, &sources, 0, diagram)?;
660            row.latent = true;
661            self.rows.push(row);
662        }
663        Ok(())
664    }

666    #[hotpath::measure]
667    fn compact_constant(
668        &mut self,
669        diagram: &mut BooleanDecisionDiagram,
670        domain: DecisionId,
671    ) -> Result<(), CompileError> {
672        // Incremental appends keep one constant, nonlatent row per key.
673        // Constrain first so a resource error cannot leave the index half remapped.
674        diagram.charge(self.rows.len())?;
675        let actives = self
676            .rows
677            .iter()
678            .map(|row| diagram.constrain(row.body.active, domain))
679            .collect::<Result<Vec<_>, _>>()?;
680        diagram.charge(self.rows.iter().fold(0usize, |work, row| {
681            if row.body.terms().is_empty() {
682                work
683            } else {
684                work.saturating_add(row.body.terms().len().saturating_add(1))
685            }
686        }))?;
687        let mut next = 0;
688        let mut actives = actives.into_iter();
689        let constant = &mut self.constant;
690        self.rows.retain_mut(|row| {
691            let active = actives.next().expect("one activity per boundary row");
692            if active == ZERO {
693                if !row.body.terms().is_empty() {
694                    constant.remove(row.body.terms());
695                }
696                return false;
697            }
698            row.body.active = active;
699            let indices = constant
700                .get_mut(row.body.terms())
701                .expect("incremental boundary row has a constant key");
702            debug_assert_eq!(indices.len(), 1);
703            indices[0] = next;
704            next += 1;
705            true
706        });
707        self.vacant = 0;
708        Ok(())
709    }

711    #[hotpath::measure]
712    fn compact(
713        &mut self,
714        diagram: &mut BooleanDecisionDiagram,
715        domain: DecisionId,
716    ) -> Result<(), CompileError> {
717        let mut indices = crate::FxMap::default();
718        let mut rows: Vec<FlowRow> = Vec::new();
719        diagram.charge(self.rows.len())?;
720        for mut row in std::mem::take(&mut self.rows) {
721            row.body.active = diagram.constrain(row.body.active, domain)?;
722            if row.body.active == ZERO {
723                continue;
724            }
725            diagram.charge(row.body.terms().len())?;
726            let key = (row.latent, row.body.terms().to_vec());
727            if let Some(&index) = indices.get(&key) {
728                let previous: &mut FlowRow = &mut rows[index];
729                previous.merge_disjoint(row, diagram, domain)?;
730            } else {
731                indices.insert(key, rows.len());
732                rows.push(row);
733            }
734        }
735        self.rows = rows;
736        self.constant.clear();
737        self.conditional.clear();
738        self.latent.clear();
739        self.vacant = 0;
740        for (index, row) in self.rows.iter().enumerate() {
741            diagram.charge(row.body.terms().len().saturating_add(1))?;
742            if row.body.terms().iter().all(|&(_, bit)| bit == ONE) {
743                self.constant
744                    .entry(row.body.terms().to_vec())
745                    .or_default()
746                    .push(index);
747            } else {
748                self.conditional.push(index);
749            }
750            if row.latent {
751                self.latent.push(index);
752            }
753        }
754        Ok(())
755    }

757    fn checks(
758        &self,
759        row: FlowRow,
760        diagram: &mut BooleanDecisionDiagram,
761        domain: DecisionId,
762        output: &mut Vec<GuardedCheck>,
763    ) -> Result<(), CompileError> {
764        let conflict = diagram.apply(BooleanOp::And, row.discard, row.restart)?;
765        let conflict = diagram.apply(BooleanOp::And, row.body.active, conflict)?;
766        if diagram.constrain(conflict, domain)? != ZERO {
767            return Err(super::assembly(
768                "stabilizer-span transition mixes discard and restart components",
769            ));
770        }
771        let keep = diagram.negate(row.discard)?;
772        let active = diagram.apply(BooleanOp::And, row.body.active, keep)?;
773        let active = diagram.constrain(active, domain)?;
774        if active == ZERO {
775            return Ok(());
776        }
777        let imaginary = diagram.apply(BooleanOp::And, active, row.phase[0])?;
778        if diagram.constrain(imaginary, domain)? != ZERO {
779            return Err(super::assembly("non-Hermitian completed boundary flow"));
780        }
781        let mut centers = row
782            .centers
783            .iter()
784            .map(|(&center, &guard)| (Some(center), guard))
785            .collect::<Vec<_>>();
786        let centered = row
787            .centers
788            .values()
789            .copied()
790            .try_fold(ZERO, |value, guard| {
791                diagram.apply(BooleanOp::Or, value, guard)
792            })?;
793        centers.push((None, diagram.negate(centered)?));
794        for (center, guard) in centers {
795            let guard = diagram.apply(BooleanOp::And, active, guard)?;
796            for restart in [false, true] {
797                let kind = if restart {
798                    row.restart
799                } else {
800                    diagram.negate(row.restart)?
801                };
802                let guard = diagram.apply(BooleanOp::And, guard, kind)?;
803                let guard = diagram.constrain(guard, domain)?;
804                if guard == ZERO {
805                    continue;
806                }
807                let mut parity = NodeDetectorParity::default();
808                let mut contributions = BTreeMap::<DecisionId, NodeDetectorParity>::new();
809                for (&measurement, &coefficient) in &row.parity {
810                    let coefficient = diagram.apply(BooleanOp::And, guard, coefficient)?;
811                    let coefficient = diagram.constrain(coefficient, domain)?;
812                    if coefficient == ZERO {
813                        continue;
814                    }
815                    let term = NodeDetectorParity::from_measurements([measurement]);
816                    if coefficient == guard {
817                        parity.xor_assign(&term);
818                    } else {
819                        contributions
820                            .entry(coefficient)
821                            .or_default()
822                            .xor_assign(&term);
823                    }
824                }
825                let sign = diagram.apply(BooleanOp::And, guard, row.phase[1])?;
826                let sign = diagram.constrain(sign, domain)?;
827                if sign == guard {
828                    parity = parity.with_sign(true);
829                } else if sign != ZERO {
830                    contributions
831                        .entry(sign)
832                        .or_default()
833                        .xor_assign(&NodeDetectorParity::default().with_sign(true));
834                }
835                if !parity.terms().is_empty() || parity.sign() || !contributions.is_empty() {
836                    output.push(GuardedCheck {
837                        guard,
838                        parity,
839                        contributions,
840                        center,
841                        restart,
842                    });
843                }
844            }
845        }
846        Ok(())
847    }

849    pub(super) fn finish(
850        self,
851        diagram: &mut BooleanDecisionDiagram,
852        domain: DecisionId,
853    ) -> Result<(), CompileError> {
854        for row in &self.rows {
855            if !row.latent && diagram.constrain(row.body.active, domain)? != ZERO {
856                return Err(super::assembly("unterminated boundary flows"));
857            }
858        }
859        Ok(())
860    }

877        let mut basis = BTreeMap::<InstanceMeasurement, NodeDetectorParity>::new();
878        for mut parity in parities {
879            loop {
880                let pivot = parity.measurements().next();
881                let Some(pivot) = pivot else {
882                    break;
883                };
884                if let Some(row) = basis.get(&pivot) {
885                    parity.xor_assign(row);
886                } else {
887                    basis.insert(pivot, parity);
888                    break;
889                }
890            }
891        }
892        let pivots = basis.keys().copied().rev().collect::<Vec<_>>();
893        for pivot in pivots {
894            let row = basis[&pivot].clone();
895            for (_, earlier) in basis.range_mut(..pivot) {
896                if earlier.measurements().any(|term| term == pivot) {
897                    earlier.xor_assign(&row);
898                }
899            }
900        }
901        basis.into_values().collect()
902    }

1242            // The two X consumers overlap only outside the reachable domain.
1243            let consumers = [
1244                flow(a, &x, &empty, 4, false),
1245                flow(b, &x, &empty, 5, true),
1246                flow(ONE, &z, &empty, 6, true),
1247            ];
1248            let mut checks = engine.append(&consumers, &mut diagram, domain).unwrap();
1249            let mut remainder = flow(ONE, &x, &empty, 7, false);
1250            remainder.skip_unmatched = true;
1251            checks.extend(engine.append(&[remainder], &mut diagram, domain).unwrap());
1252            engine.finish(&mut diagram, domain).unwrap();
1253            for mask in 0..3 {
1254                let enabled = |root| diagram.evaluate(root, |index| mask & (1 << index) != 0);
1255                let actual = checks
1256                    .iter()
1257                    .filter(|check| enabled(check.guard))
1258                    .map(|check| {
1259                        let mut parity = check.parity.clone();
1260                        for (&guard, contribution) in &check.contributions {
1261                            if enabled(guard) {
1262                                parity.xor_assign(contribution);
1263                            }
1264                        }
1265                        assert!(!check.restart);
1266                        (parity, check.center)
1267                    })
1268                    .collect::<Vec<_>>();
1269                let (x_records, x_sign, x_center) = match mask {
1270                    0 => (vec![0, 7], true, 7),
1271                    1 => (vec![0, 2, 4], false, 4),
1272                    2 => (vec![0, 5], false, 5),
1273                    _ => unreachable!(),
1274                };
1275                let z_records = if mask == 2 { vec![1, 3, 6] } else { vec![1, 6] };
1276                let expected = [
1277                    (
1278                        NodeDetectorParity::from_measurements(
1279                            x_records.into_iter().map(measurement),
1280                        )
1281                        .with_sign(x_sign),
1282                        Some([x_center, 0]),
1283                    ),
1284                    (
1285                        NodeDetectorParity::from_measurements(
1286                            z_records.into_iter().map(measurement),
1287                        )
1288                        .with_sign(true),
1289                        Some([6, 0]),
1290                    ),
1291                ];
1292                assert_eq!(actual.len(), expected.len(), "mask {mask}");
1293                for check in expected {
1294                    assert!(actual.contains(&check), "mask {mask}: {actual:?}");
1295                }
1296            }
1297            // The same consumer overlap must be rejected when it is reachable.
1298            assert!(matches!(
1299                GuardedFlowEngine::default().append(&consumers, &mut diagram, ONE),
1300                Err(CompileError::BranchAssembly { reason })
1301                    if reason == "incoming flow boundaries are not independent"
1302            ));
1303        }

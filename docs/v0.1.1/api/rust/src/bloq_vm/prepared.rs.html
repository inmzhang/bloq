29/// One quantum node's merged op stream, in emit order, with the map that
30/// resolves its measurement records to global record ids. A build-time
31/// intermediate: [`PreparedProgramCore`] translates it and keeps only the
32/// result.
33struct NodeOpStream {
34    /// The enclosing region chain (empty at the top level).
35    scope: LevelPath,
36    /// The quantum node this stream belongs to (level-local id).
37    node: BloqNodeId,
38    /// The node's merged emission-plan entry-body ops, already translated to
39    /// global coordinates (instance offsets applied at lowering).
40    ops: Vec<Op>,
41    /// Merged-local measurement id to global record id. Preparation proves this
42    /// map covers every record produced or consumed by `ops`.
43    local_to_global: FxHashMap<u32, u32>,
44}
45
46/// Flattened facts the whole-program walk runs on.
47pub(super) struct PreparedProgramCore {
48    pub(super) bloq: Bloq,
49    pub(super) coord_index: FxHashMap<IVec2, u32>,
50    pub(super) global_meas: FxHashMap<InstanceMeasurement, u32>,
51    pub(super) instance_offset: FxHashMap<TemplateInstanceId, IVec2>,
52    pub(super) record_count: u32,
53    /// Every quantum node's stream in engine-instruction form, in emit order.
54    /// Translating a [`NodeOpStream`] is shot-independent, so it happens once
55    /// here and every shot replays the result; the streams themselves are not
56    /// retained, since nothing downstream reads them back.
57    prepared: Vec<PreparedOps>,
58    node_index: FxHashMap<ScopedNode, usize>,
59    /// Deterministic SEM-ORD schedule for every level, keyed by its scope.
60    emit_orders: FxHashMap<LevelPath, Vec<BloqNodeId>>,
61    /// Boundary maps translated to engine indices once for all shots.
62    boundary_operators: FxHashMap<PauliMap, PauliString>,
63    selected: RefCell<FxHashMap<SelectionKey, Rc<SelectedOps>>>,
64}
65
66impl PreparedProgramCore {
67    pub(super) fn build(bloq: &Bloq) -> Result<Self, ExecError> {
68        // The op streams are built from the flattened program. SEM-MERGE runs
69        // on that graph; flattening preserves the scoped node addresses used by
70        // `collect_streams`. Full validation is an explicit caller operation.
71        let mut bloq = bloq.clone();
72        bloq.flatten()?;
73        let plans = bloq.emission_plans(&InstantiationOptions::default())?;
74        Self::assemble(bloq, &plans)
75    }
76
77    /// Assemble the shared core from an already-flattened `bloq`, reading each
78    /// quantum node's stream out of its emission plan.
79    fn assemble(bloq: Bloq, plans: &ValidatedPlans) -> Result<Self, ExecError> {
80        let mut global_meas = FxHashMap::default();
81        let mut instance_offset = FxHashMap::default();
82        let mut instance_templates = FxHashMap::default();
83        let mut record_count = 0;
84        let mut streams_by_node = FxHashMap::default();
85        collect_streams(
86            &bloq,
87            bloq.top(),
88            &LevelPath::default(),
89            plans,
90            &mut global_meas,
91            &mut instance_offset,
92            &mut instance_templates,
93            &mut record_count,
94            &mut streams_by_node,
95        )?;
96        for (_, level) in bloq.levels() {
97            for (_, quantum) in level.quantum_nodes() {
98                bloq.check_detector_bundle_bindings(quantum, |instance| {
99                    instance_templates.get(&instance).copied()
100                })?;
101            }
102        }
103
104        let mut node_streams = Vec::with_capacity(streams_by_node.len());
105        let mut emit_orders = FxHashMap::default();
106        order_streams(
107            bloq.top(),
108            &LevelPath::default(),
109            &mut streams_by_node,
110            &mut node_streams,
111            &mut emit_orders,
112        )?;
113        if !streams_by_node.is_empty() {
114            return Err(ExecError::MalformedGraph(
115                "prepared stream has no scheduled quantum node",
116            ));
117        }
118        let node_index = node_streams
119            .iter()
120            .enumerate()
121            .map(|(index, stream)| ((stream.scope.clone(), stream.node), index))
122            .collect();
123
124        let coord_index: FxHashMap<IVec2, u32> = bloq
125            .sorted_layout_coords()?
126            .iter()
127            .enumerate()
128            .map(|(index, &coord)| {
129                u32::try_from(index)
130                    .map(|index| (coord, index))
131                    .map_err(|_| ExecError::IdOverflow("qubit"))
132            })
133            .collect::<Result<_, _>>()?;
134
135        // Instance offsets were already applied at lowering, so the whole
136        // program shares one un-shifted layout — the same one every shot runs
137        // against, which is why the translation below can be hoisted out of the
138        // shot loop entirely.
139        let layout = LayoutCtx {
140            coord_to_index: &coord_index,
141            offset: IVec2::ZERO,
142            qubit_count: coord_index.len(),
143        };
144        let prepared = node_streams
145            .iter()
146            .map(|stream| {
147                // `collect_streams` proved the remap total over this stream's
148                // records, so the lookup cannot miss.
149                prepare_ops(&stream.ops, &layout, &|local| {
150                    stream.local_to_global[&local]
151                })
152            })
153            .collect::<Result<Vec<_>, _>>()?;
154        let mut boundary_operators = FxHashMap::default();
155        prepare_boundary_operators(bloq.top(), &layout, &mut boundary_operators)?;
156
157        Ok(Self {
158            bloq,
159            coord_index,
160            global_meas,
161            instance_offset,
162            record_count,
163            prepared,
164            node_index,
165            emit_orders,
166            boundary_operators,
167            selected: RefCell::default(),
168        })
169    }

171    pub(super) fn selected_ops(
172        &self,
173        scope: &LevelPath,
174        id: BloqNodeId,
175        node: &BloqNode,
176        inputs: &FxHashMap<u32, bool>,
177    ) -> Result<Rc<SelectedOps>, ExecError> {
178        let mut values = inputs
179            .iter()
180            .map(|(&slot, &value)| (slot, value))
181            .collect::<Vec<_>>();
182        values.sort_unstable();
183        let key = ((scope.clone(), id), values);
184        if let Some(selected) = self.selected.borrow().get(&key) {
185            return Ok(Rc::clone(selected));
186        }
187        let selected = node
188            .select_quantum_members(|slot| inputs.get(&slot).copied())
189            .map_err(ExecError::InvalidCircuit)?;
190        let plan = selected
191            .emission_plan(self.bloq.templates())
192            .map_err(ExecError::InvalidCircuit)?;
193        let mut local_to_global = FxHashMap::default();
194        let mut aliases = Vec::new();
195        for (local, sources) in plan.grouped_measurements() {
196            let canonical = sources
197                .iter()
198                .map(|source| self.global_meas[source])
199                .min()
200                .expect("measurement group is nonempty");
201            local_to_global.insert(local, canonical);
202            aliases.extend(
203                sources
204                    .iter()
205                    .map(|source| (self.global_meas[source], canonical)),
206            );
207        }
208        let ops = plan
209            .circuit
210            .body(plan.circuit.entry_body())
211            .expect("entry body exists")
212            .ops();
213        validate_record_remap(ops, &local_to_global)?;
214        let layout = LayoutCtx {
215            coord_to_index: &self.coord_index,
216            offset: IVec2::ZERO,
217            qubit_count: self.coord_index.len(),
218        };
219        let prepared = Rc::new(SelectedOps {
220            quantum: selected.expect_quantum().clone(),
221            ops: prepare_ops(ops, &layout, &|local| local_to_global[&local])?,
222            aliases,
223        });
224        let mut cache = self.selected.borrow_mut();
225        // ponytail: 64 plans retained across the program; use LRU if measured
226        // mask churn warrants retaining hot plans across an eviction.
227        if cache.len() >= 64 {
228            cache.clear();
229        }
230        cache.insert(key, Rc::clone(&prepared));
231        Ok(prepared)
232    }

234    /// The engine-instruction form of a quantum node's stream.
235    pub(super) fn prepared_ops(
236        &self,
237        scope: &LevelPath,
238        node: BloqNodeId,
239    ) -> Result<&PreparedOps, ExecError> {
240        self.node_index
241            .get(&(scope.clone(), node))
242            .and_then(|&index| self.prepared.get(index))
243            .ok_or(ExecError::MalformedGraph(
244                "quantum node has no prepared stream",
245            ))
246    }
247
248    /// The prepared deterministic schedule for one level.
249    pub(super) fn emit_order(&self, scope: &LevelPath) -> Result<&[BloqNodeId], ExecError> {
250        self.emit_orders
251            .get(scope)
252            .map(Vec::as_slice)
253            .ok_or(ExecError::MalformedGraph(
254                "level has no prepared emit order",
255            ))
256    }
257
258    /// A boundary map already translated to the engine's dense layout.
259    pub(super) fn boundary_operator(&self, map: &PauliMap) -> Result<&PauliString, ExecError> {
260        self.boundary_operators
261            .get(map)
262            .ok_or(ExecError::MalformedGraph(
263                "boundary map has no prepared engine operator",
264            ))
265    }
266}

268#[expect(
269    clippy::too_many_arguments,
270    reason = "recursive preparation threads one shared walk state"
271)]
272fn collect_streams(
273    bloq: &Bloq,
274    level: &SubGraph,
275    scope: &LevelPath,
276    plans: &ValidatedPlans,
277    global_meas: &mut FxHashMap<InstanceMeasurement, u32>,
278    instance_offset: &mut FxHashMap<TemplateInstanceId, IVec2>,
279    instance_templates: &mut FxHashMap<TemplateInstanceId, TemplateId>,
280    next_record: &mut u32,
281    out: &mut FxHashMap<ScopedNode, NodeOpStream>,
282) -> Result<(), ExecError> {
283    // Node-id order deliberately preserves the established global record layout.
284    for (id, node) in level.nodes() {
285        match &node.kind {
286            BloqNodeKind::Quantum(quantum) => {
287                for instance in &quantum.instances {
288                    instance_offset.insert(instance.id, instance.offset);
289                    instance_templates.insert(instance.id, instance.template_id);
290                }
291                if !quantum.guards.is_empty() {
292                    for instance in &quantum.instances {
293                        let template = bloq
294                            .templates()
295                            .get(instance.template_id)
296                            .ok_or(ExecError::MalformedGraph("missing selected template"))?;
297                        for record in template.circuit.meas_registry().records() {
298                            let global = *next_record;
299                            *next_record = global
300                                .checked_add(1)
301                                .ok_or(ExecError::IdOverflow("record"))?;
302                            global_meas.insert(
303                                InstanceMeasurement {
304                                    instance: instance.id,
305                                    measurement: record.id,
306                                },
307                                global,
308                            );
309                        }
310                    }
311                    continue;
312                }
313                // The plan contains the materialized merge used for execution.
314                let plan = plans.get(scope, id).ok_or(ExecError::MalformedGraph(
315                    "quantum node has no emission plan",
316                ))?;
317                // Mint one global record per merged-local id, in the canonical
318                // grouping order (ascending local; see `grouped_measurements`), so
319                // the established global record layout is preserved exactly.
320                let mut local_to_global = FxHashMap::default();
321                for (local, sources) in plan.grouped_measurements() {
322                    let global = *next_record;
323                    *next_record = global
324                        .checked_add(1)
325                        .ok_or(ExecError::IdOverflow("record"))?;
326                    local_to_global.insert(local, global);
327                    for measurement in sources {
328                        global_meas.insert(measurement, global);
329                    }
330                }
331                let ops = plan
332                    .circuit
333                    .body(plan.circuit.entry_body())
334                    .expect("entry body exists")
335                    .ops()
336                    .to_vec();
337                validate_record_remap(&ops, &local_to_global)?;
338                if out
339                    .insert(
340                        (scope.clone(), id),
341                        NodeOpStream {
342                            scope: scope.clone(),
343                            node: id,
344                            ops,
345                            local_to_global,
346                        },
347                    )
348                    .is_some()
349                {
350                    return Err(ExecError::MalformedGraph(
351                        "duplicate scoped quantum node identity",
352                    ));
353                }
354            }
355            BloqNodeKind::Region(region) => {
356                for (selector, body) in region.bodies() {
357                    collect_streams(
358                        bloq,
359                        body,
360                        &scope.child(id, selector),
361                        plans,
362                        global_meas,
363                        instance_offset,
364                        instance_templates,
365                        next_record,
366                        out,
367                    )?;
368                }
369            }
370            BloqNodeKind::Classical(_) => {}
371        }
372    }
373    Ok(())
374}

376fn order_streams(
377    level: &SubGraph,
378    scope: &LevelPath,
379    by_node: &mut FxHashMap<ScopedNode, NodeOpStream>,
380    out: &mut Vec<NodeOpStream>,
381    emit_orders: &mut FxHashMap<LevelPath, Vec<BloqNodeId>>,
382) -> Result<(), ExecError> {
383    let emit_order = level.deterministic_emit_order()?;
384    for &id in &emit_order {
385        let node = level.node(id).expect("emit order yields live nodes");
386        match &node.kind {
387            BloqNodeKind::Quantum(quantum) if quantum.guards.is_empty() => out.push(
388                by_node
389                    .remove(&(scope.clone(), id))
390                    .ok_or(ExecError::MalformedGraph(
391                        "scheduled quantum node has no prepared stream",
392                    ))?,
393            ),
394            BloqNodeKind::Region(region) => {
395                for (selector, body) in region.bodies() {
396                    order_streams(body, &scope.child(id, selector), by_node, out, emit_orders)?;
397                }
398            }
399            BloqNodeKind::Quantum(_) | BloqNodeKind::Classical(_) => {}
400        }
401    }
402    emit_orders.insert(scope.clone(), emit_order);
403    Ok(())
404}

406fn prepare_boundary_operators(
407    level: &SubGraph,
408    layout: &LayoutCtx<'_>,
409    out: &mut FxHashMap<PauliMap, PauliString>,
410) -> Result<(), ExecError> {
411    for (_, node) in level.nodes() {
412        if let Some(ClassicalNode::Observable { operators, .. }) = node.try_classical() {
413            for operator in operators {
414                if !out.contains_key(&operator.operator) {
415                    let prepared = prepare_boundary_operator(&operator.operator, layout)?;
416                    out.insert(operator.operator.clone(), prepared);
417                }
418            }
419        }
420        if let Some(region) = node.try_region() {
421            for (_, body) in region.bodies() {
422                prepare_boundary_operators(body, layout, out)?;
423            }
424        }
425    }
426    Ok(())
427}

437    fn nested_bloq() -> (Bloq, BloqNodeId, BloqNodeId, BloqNodeId, u32) {
438        let mut circuit = CoordCircuit::new();
439        let measurement = circuit.measure(PauliBasis::Z, [IVec2::ZERO])[0];
440        let mut bloq = Bloq::new();
441        let template = bloq.add_template(BloqTemplate::new(circuit));
442
443        let top = bloq.add_node(quantum_node(template, 0, IVec2::ZERO));
444        let mut body = SubGraph::new();
445        let nested = body.add_node(quantum_node(template, 1, IVec2::new(1, 0)));
446        let region = bloq.add_node(BloqNode::region(RegionNode::RepeatUntilSuccess {
447            restart_condition: ClassicalExpr::Const(false),
448            restart_source: None,
449            body,
450        }));
451        bloq.add_edge(top, region, BloqEdge::Order);
452        (bloq, top, region, nested, measurement)
453    }

455    /// Two nodes at different levels share a level-local id, so only the scope
456    /// tells their streams apart. Each must get its own instructions (their
457    /// instances sit at different coordinates) and its own global record.
458    #[test]
459    fn scoped_node_index_distinguishes_level_local_ids() {
460        let (bloq, top, region, nested, _measurement) = nested_bloq();
461        assert_eq!(top, nested);
462        let core = PreparedProgramCore::build(&bloq).unwrap();
463        let top_scope = LevelPath::default();
464        let nested_scope = top_scope.child(region, bloq_ir::BodySelector::Body);
465
466        let top_stream = core.prepared_ops(&top_scope, top).unwrap();
467        let nested_stream = core.prepared_ops(&nested_scope, nested).unwrap();
468        assert_ne!(top_stream, nested_stream);
469        assert_eq!(top_stream.record_ids(), [0]);
470        assert_eq!(nested_stream.record_ids(), [1]);
471        assert_eq!(core.emit_order(&top_scope).unwrap(), [top, region]);
472        assert_eq!(core.emit_order(&nested_scope).unwrap(), [nested]);
473    }

475    #[test]
476    fn record_allocation_rejects_exhaustion_without_aliasing() {
477        for guarded in [false, true] {
478            let (mut bloq, top, ..) = nested_bloq();
479            if guarded {
480                bloq.top_mut()
481                    .node_mut(top)
482                    .unwrap()
483                    .expect_quantum_mut()
484                    .guards
485                    .push(bloq_ir::QuantumGuard::default());
486            }
487            let plans = bloq
488                .emission_plans(&InstantiationOptions::default())
489                .unwrap();
490            let mut next_record = u32::MAX;
491            let mut measurements = FxHashMap::default();
492            let result = collect_streams(
493                &bloq,
494                bloq.top(),
495                &LevelPath::default(),
496                &plans,
497                &mut measurements,
498                &mut FxHashMap::default(),
499                &mut FxHashMap::default(),
500                &mut next_record,
501                &mut FxHashMap::default(),
502            );
503            assert!(matches!(result, Err(ExecError::IdOverflow("record"))));
504            assert_eq!(next_record, u32::MAX);
505            assert!(measurements.is_empty());
506        }

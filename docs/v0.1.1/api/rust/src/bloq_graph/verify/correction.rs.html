40impl InternalCorrection {
41    /// Cover anonymous coordinates by complete corrected boundary-sign classes.
42    ///
43    /// With resource legs open, exposing the native phase bits as outcome legs
44    /// gives one Clifford stabilizer tensor. The resolved basis must span its
45    /// full correlation space, including private rows: its closed kernel then
46    /// gives complete affine support, and nonzero outcome slices have equal
47    /// norm. Complete boundary signs identify their rays; equal coordinate
48    /// fibers therefore have equal Kraus weight, not just equal directions.
49    /// A fixed resource contraction preserves within-class unit-phase equality,
50    /// while different classes may acquire different norms or vanish.
51    pub(super) fn representatives(
52        &self,
53        outputs: &[IVec3],
54        actions: &[ActionNode],
55        values: &FxHashMap<Var, bool>,
56        feedback_values: &FxHashMap<usize, bool>,
57        resolved: &ResolvedCorrection,
58        limit: usize,
59    ) -> Result<Vec<FxHashMap<Var, bool>>, FeedbackInferenceError> {
60        let ResolvedCorrection {
61            basis,
62            phases,
63            readouts,
64        } = resolved;
65        let surfaces = basis.resolved_surfaces();
66        let columns = surfaces
67            .iter()
68            .map(|surface| pauli_coeffs(&surface.paulis))
69            .collect::<Vec<_>>();
70        // Validate the actual readout recipe, not merely the existence of some
71        // C0 witness. A later fill must preserve its signed Clifford relation.
72        for (name, readout) in self.named_vars.keys().zip(readouts) {
73            if solve_coeff_combination(&columns, &pauli_coeffs(&readout.paulis)).is_none()
74                || basis
75                    .zx_graph()
76                    .materialize_stabilizer(readout.paulis.clone())
77                    .sign
78                    != readout.sign
79                || basis
80                    .zx_graph()
81                    .nodes()
82                    .iter()
83                    .filter(|node| node.is_output_port(basis.zx_graph()))
84                    .any(|node| {
85                        let pauli = readout.paulis.get(node.id);
86                        if node.role == crate::PortRole::Multiplex {
87                            pauli & Pauli::X
88                        } else {
89                            pauli != Pauli::I
90                        }
91                    })
92            {
93                return Err(FeedbackInferenceError::ReadoutConventionUnavailable(
94                    name.clone(),
95                ));
96            }
97        }
98        let variables = self.sampled_vars().collect::<Vec<_>>();
99        let variable_set = variables.iter().copied().collect::<BTreeSet<_>>();
100        let named = phases
101            .iter()
102            .map(|(key, parity)| {
103                (
104                    key.clone(),
105                    phase_value(parity, |variable| {
106                        !variable_set.contains(&variable) && values[&variable]
107                    }),
108                )
109            })
110            .collect::<BTreeMap<_, _>>();
111        let closed = basis.closed_stabilizer_surfaces();
112        let constraints = self.parity_columns(&closed, phases);
113        let mut rhs = CoeffVec::zeros(closed.len());
114        for (row, surface) in closed.iter().enumerate() {
115            let parity = named.iter().fold(surface.sign, |odd, (key, value)| {
116                odd ^ (*value && stabilizer_supports_measurement(surface, key))
117            }) ^ feedback_sign(surface, basis.zx_graph(), actions, feedback_values);
118            rhs.set_bit(row, parity);
119        }
120        let Some(witness) = solve_coeff_combination(&constraints, &rhs) else {
121            return Ok(Vec::new());
122        };
123        let directions = kernel_directions(&constraints);
124        let reference = self.apply_coordinate(phases, named.clone(), &witness);
125        let (correction_surfaces, correction) =
126            solve_evolved_correction(basis, outputs, |surface| {
127                phases
128                    .keys()
129                    .any(|key| stabilizer_supports_measurement(surface, key))
130            })?;
131        let correction_columns = self.parity_columns(&correction_surfaces, phases);
132        let mut constant = CoeffVec::zeros(correction_surfaces.len());
133        for (row, surface) in correction_surfaces.iter().enumerate() {
134            constant.set_bit(
135                row,
136                named.iter().fold(false, |odd, (key, value)| {
137                    odd ^ ((*value ^ reference[key])
138                        && stabilizer_supports_measurement(surface, key))
139                }),
140            );
141        }
142        // Discharge every runtime consistency condition on the affine domain,
143        // without treating a failed correction as a physically impossible shot.
144        let witness_rhs = combine_columns(&correction_columns, &witness, constant.clone());
145        correction.evaluate(
146            &(0..witness_rhs.len())
147                .map(|bit| witness_rhs.bit(bit))
148                .collect::<Vec<_>>(),
149        )?;
150        for direction in &directions {
151            let test = combine_columns(&correction_columns, direction, witness_rhs.clone());
152            correction.evaluate(&(0..test.len()).map(|bit| test.bit(bit)).collect::<Vec<_>>())?;
153        }
154        if directions.is_empty() {
155            checked_count(0, limit, "anonymous boundary classes")?;
156            return Ok(vec![
157                variables
158                    .iter()
159                    .enumerate()
160                    .map(|(index, &variable)| (variable, witness.bit(index)))
161                    .collect(),
162            ]);
163        }
164
165        let full_columns = self.parity_columns(&surfaces, phases);
166        let (bodies, extra_outputs, boundary_qubits) = boundary_bodies(basis, &surfaces, outputs)?;
167        let mut independent = Vec::<CoeffVec>::new();
168        for body in &bodies {
169            let column = pauli_coeffs(body);
170            if solve_coeff_combination(&independent, &column).is_none() {
171                independent.push(column);
172            }
173        }
174        if independent.len() != boundary_qubits
175            || bodies.iter().enumerate().any(|(index, body)| {
176                bodies[..index]
177                    .iter()
178                    .any(|other| !body.commutes_with(other))
179            })
180        {
181            return Err(FeedbackInferenceError::IncompleteCertificate {
182                rank: independent.len(),
183                expected: boundary_qubits,
184            });
185        }
186
187        let mut signatures = Vec::<CoeffVec>::new();
188        let mut selected = Vec::new();
189        for direction in directions {
190            let changes = combine_columns(
191                &correction_columns,
192                &direction,
193                CoeffVec::zeros(correction_surfaces.len()),
194            );
195            let frames = correction
196                .frames()
197                .iter()
198                .map(|pair| {
199                    let x = pair
200                        .x
201                        .iter()
202                        .fold(false, |odd, &row| odd ^ changes.bit(row));
203                    let z = pair
204                        .z
205                        .iter()
206                        .fold(false, |odd, &row| odd ^ changes.bit(row));
207                    match (x, z) {
208                        (false, false) => Pauli::I,
209                        (true, false) => Pauli::X,
210                        (false, true) => Pauli::Z,
211                        (true, true) => Pauli::Y,
212                    }
213                })
214                .collect::<Vec<_>>();
215            let native =
216                combine_columns(&full_columns, &direction, CoeffVec::zeros(surfaces.len()));
217            let mut signature = CoeffVec::zeros(bodies.len());
218            for (row, surface) in surfaces.iter().enumerate() {
219                let correction =
220                    outputs
221                        .iter()
222                        .zip(&frames)
223                        .fold(false, |odd, (position, frame)| {
224                            let node = basis
225                                .zx_graph()
226                                .node_at(*position)
227                                .expect("validated output");
228                            odd ^ frame.anticommutes(surface.paulis.get(node.id))
229                        });
230                signature.set_bit(row, native.bit(row) ^ correction);
231            }
232            for (extra, &output) in extra_outputs.iter().enumerate() {
233                signature.set_bit(
234                    surfaces.len() + extra,
235                    frames[output].anticommutes(Pauli::Z),
236                );
237            }
238            if solve_coeff_combination(&signatures, &signature).is_none() {
239                signatures.push(signature);
240                selected.push(direction);
241                checked_count(selected.len(), limit, "anonymous boundary classes")?;
242            }
243        }
244        let count = checked_count(selected.len(), limit, "anonymous boundary classes")?;
245        Ok((0..count)
246            .map(|mask| {
247                let mut coordinate = witness.clone();
248                for (bit, direction) in selected.iter().enumerate() {
249                    if mask & (1 << bit) != 0 {
250                        coordinate.xor_assign(direction);
251                    }
252                }
253                variables
254                    .iter()
255                    .enumerate()
256                    .map(|(index, &variable)| (variable, coordinate.bit(index)))
257                    .collect()
258            })
259            .collect())
260    }

262    fn parity_columns(
263        &self,
264        surfaces: &[crate::Stabilizer],
265        phases: &BTreeMap<MeasurementKey, Parity>,
266    ) -> Vec<CoeffVec> {
267        self.sampled_vars()
268            .map(|variable| {
269                let mut column = CoeffVec::zeros(surfaces.len());
270                for (row, surface) in surfaces.iter().enumerate() {
271                    column.set_bit(
272                        row,
273                        phases
274                            .iter()
275                            .filter(|(_, parity)| parity.iter().any(|value| value == variable))
276                            .fold(false, |odd, (key, _)| {
277                                odd ^ stabilizer_supports_measurement(surface, key)
278                            }),
279                    );
280                }
281                column
282            })
283            .collect()
284    }
285    pub(super) fn new(
286        zx: &ZXGraph,
287        named_phases: &BTreeMap<MeasurementKey, Parity>,
288        named_vars: &BTreeMap<String, Var>,
289        next_var: &mut usize,
290    ) -> Result<Self, VerifyLogicalError> {
291        let keys = measurement_keys(zx);
292        let mut measurement_vars = named_phases.clone();
293        let mut sampled = Vec::new();
294        for key in keys {
295            if measurement_vars.contains_key(&key)
296                || matches!(&key, MeasurementKey::Node { pos, .. }
297                if measurement_vars.keys().any(|named| matches!(named, MeasurementKey::Node { pos: named, .. } if named == pos)))
298            {
299                continue;
300            }
301            let variable = fresh_var(next_var)?;
302            measurement_vars.insert(key.clone(), Parity::single(variable));
303            sampled.push((key, variable));
304        }
305
306        let (initial_basis, readouts) = if named_vars.is_empty() {
307            let basis = RuntimeStabilizerBasis::from_zx_graph(zx.clone()).map_err(|source| {
308                VerifyLogicalError::RuntimeBasisInitializationFailed { source }
309            })?;
310            (basis, BTreeMap::new())
311        } else {
312            let mut stabilizers = zx.stabilizers().map_err(|source| {
313                VerifyLogicalError::RuntimeBasisInitializationFailed {
314                    source: source.into(),
315                }
316            })?;
317            let initial_basis = RuntimeStabilizerBasis::from_generators(&stabilizers);
318            // ZX conversion starts with syntax dependencies. Restore the same
319            // selected witnesses and fixing closure used by physical lowering.
320            let mut dag = zx.action_graph().clone();
321            dag.attach_readout_dependencies(&stabilizers.generators, Some(&stabilizers.zx_graph))?;
322            stabilizers
323                .prepare_analysis_readouts(&dag)
324                .map_err(
325                    |source| VerifyLogicalError::RuntimeBasisInitializationFailed { source },
326                )?;
327            let readouts = stabilizers
328                .generators
329                .into_iter()
330                .filter_map(|generator| {
331                    let name = generator.measurement_name()?.to_owned();
332                    Some((name, generator))
333                })
334                .collect();
335            (initial_basis, readouts)
336        };
337        if !sampled.is_empty() && !named_vars.is_empty() {
338            // A free native phase can cross a named row. Compensate through the
339            // same right-inverse that realizes named outcomes, so the action still
340            // reads its supplied outcome after unnamed measurements are sampled.
341            for (name, generator) in &readouts {
342                let named = named_vars[name];
343                for (key, variable) in &sampled {
344                    if !stabilizer_supports_measurement(&generator.stabilizer, key) {
345                        continue;
346                    }
347                    for (target, parity) in named_phases {
348                        if parity.iter().any(|value| value == named) {
349                            let target = measurement_vars
350                                .get_mut(target)
351                                .expect("named phase exists");
352                            *target = &*target + &Parity::single(*variable);
353                        }
354                    }
355                }
356            }
357        }
358        Ok(Self {
359            measurement_vars,
360            sampled,
361            initial_basis,
362            named_vars: named_vars.clone(),
363            readouts,
364        })
365    }

375    fn resolved_measurement_vars(
376        &self,
377        selectives: &FxHashMap<IVec3, MeasurementKey>,
378        rows: &[crate::Stabilizer],
379    ) -> Result<BTreeMap<MeasurementKey, Parity>, VerifyLogicalError> {
380        let mut phases = BTreeMap::<MeasurementKey, Parity>::new();
381        for (key, parity) in &self.measurement_vars {
382            let target = match key {
383                MeasurementKey::Node { pos, .. } => selectives.get(pos).unwrap_or(key),
384                MeasurementKey::Edge { .. } => key,
385            };
386            let entry = phases.entry(target.clone()).or_default();
387            *entry = &*entry + parity;
388        }
389        if self.named_vars.is_empty() {
390            return Ok(phases);
391        }
392        let columns = phases
393            .keys()
394            .map(|key| {
395                let mut column = CoeffVec::zeros(rows.len());
396                for (row, surface) in rows.iter().enumerate() {
397                    column.set_bit(row, stabilizer_supports_measurement(surface, key));
398                }
399                column
400            })
401            .collect::<Vec<_>>();
402        // A fill can move a later named readout onto different native records.
403        // Reparameterize the whole affine fiber, rather than preserving an
404        // obsolete pre-fill compensation or losing an anonymous direction.
405        let sampled = self.sampled_vars().collect::<BTreeSet<_>>();
406        for parity in phases.values_mut() {
407            *parity = Parity::new(
408                parity
409                    .iter()
410                    .filter(|variable| !sampled.contains(variable))
411                    .collect::<Vec<_>>(),
412                phase_constant(parity),
413            );
414        }
415        for (row, (name, &variable)) in self.named_vars.iter().enumerate() {
416            let mut rhs = CoeffVec::singleton(row, rows.len());
417            for (column, parity) in columns.iter().zip(phases.values()) {
418                if parity.iter().any(|value| value == variable) {
419                    rhs.xor_assign(column);
420                }
421            }
422            let adjustment = solve_coeff_combination(&columns, &rhs)
423                .ok_or_else(|| VerifyLogicalError::MeasurementPhaseUnavailable(name.clone()))?;
424            for (index, parity) in phases.values_mut().enumerate() {
425                if adjustment.bit(index) {
426                    *parity = &*parity + &Parity::single(variable);
427                }
428            }
429        }
430        let mut constant = CoeffVec::zeros(rows.len());
431        for (column, parity) in columns.iter().zip(phases.values()) {
432            if phase_constant(parity) {
433                constant.xor_assign(column);
434            }
435        }
436        let adjustment = solve_coeff_combination(&columns, &constant).ok_or_else(|| {
437            VerifyLogicalError::MeasurementPhaseUnavailable("affine readout offset".into())
438        })?;
439        for (index, parity) in phases.values_mut().enumerate() {
440            if adjustment.bit(index) {
441                *parity = parity.negated();
442            }
443        }
444        let directions = kernel_directions(&columns);
445        if directions.len() > self.sampled.len() {
446            return Err(VerifyLogicalError::MeasurementPhaseUnavailable(
447                "resolved anonymous coordinates".into(),
448            ));
449        }
450        for (direction, (_, variable)) in directions.iter().zip(&self.sampled) {
451            for (index, parity) in phases.values_mut().enumerate() {
452                if direction.bit(index) {
453                    *parity = &*parity + &Parity::single(*variable);
454                }
455            }
456        }
457        Ok(phases)
458    }

464    pub(super) fn resolve(
465        &self,
466        actions: &[ActionNode],
467        selectives: &FxHashMap<IVec3, PauliBasis>,
468    ) -> Result<ResolvedCorrection, VerifyLogicalError> {
469        let mut basis = self.initial_basis.clone();
470        let mut keys = FxHashMap::default();
471        let mut resolves = actions
472            .iter()
473            .filter_map(|action| match action.action {
474                Action::Resolve { target, .. } => Some((target.z, action.ordinal, target)),
475                _ => None,
476            })
477            .collect::<Vec<_>>();
478        resolves.sort_unstable_by_key(|&(z, ordinal, _)| (z, ordinal));
479        for (_, _, target) in resolves {
480            let (updated, key) = replay_fill(&basis, target, selectives[&target])?;
481            basis = updated;
482            keys.insert(target, key);
483        }
484        // Match physical readout lowering: keep the authored signed row when
485        // it needs no fills; otherwise use precisely its causal fills and
486        // accepted deadline. A fresh all-fills witness can change the label.
487        // SEM-READ reports the physical parity of these particular records:
488        // transport signs remain in the ZX tensor/closed constraints. Adding
489        // surface.sign again would invert signed readouts. The convention is
490        // anchored here, rather than assuming every equivalent witness has
491        // zero affine offset relative to the authored record.
492        let readouts = self
493            .readouts
494            .iter()
495            .map(|(name, generator)| {
496                if let Some(plan) = generator.readout_plan()
497                    && plan.sites.iter().any(|site| !selectives.contains_key(site))
498                {
499                    return Err(VerifyLogicalError::MeasurementPhaseUnavailable(
500                        name.clone(),
501                    ));
502                }
503                Ok(generator
504                    .named_readout()
505                    .expect("named row")
506                    .select(|site| {
507                        let NodeKind::Selective(kind) =
508                            self.zx_graph().node_at(site).expect("resolve node").kind
509                        else {
510                            unreachable!("resolve target is selective")
511                        };
512                        selectives[&site] == kind.pauli_if_true()
513                    })
514                    .clone())
515            })
516            .collect::<Result<Vec<_>, _>>()?;
517        let phases = self.resolved_measurement_vars(&keys, &readouts)?;
518        Ok(ResolvedCorrection {
519            basis,
520            phases,
521            readouts,
522        })
523    }

525    pub(super) fn apply(
526        &self,
527        graph: &mut QuizxGraph,
528        outputs: &[IVec3],
529        values: &FxHashMap<Var, bool>,
530        resolved: &ResolvedCorrection,
531        actions: &[ActionNode],
532        feedback_values: &FxHashMap<usize, bool>,
533    ) -> Result<(), VerifyLogicalError> {
534        let ResolvedCorrection {
535            basis,
536            phases: measurement_vars,
537            ..
538        } = resolved;
539        let measured = measurement_vars
540            .iter()
541            .map(|(key, parity)| {
542                (
543                    key.clone(),
544                    phase_value(parity, |variable| values[&variable]),
545                )
546            })
547            .collect::<BTreeMap<_, _>>();
548        let sampled = self.sampled_vars().collect::<BTreeSet<_>>();
549        let reference = measurement_vars
550            .iter()
551            .map(|(key, parity)| {
552                let value = phase_value(parity, |variable| {
553                    !sampled.contains(&variable) && values[&variable]
554                });
555                (key.clone(), value)
556            })
557            .collect::<BTreeMap<_, _>>();
558
559        let Some(reference) =
560            self.reference_outcomes(basis, measurement_vars, reference, actions, feedback_values)
561        else {
562            *graph.scalar_mut() = FScalar::zero();
563            return Ok(());
564        };
565        let (surfaces, correction) = solve_evolved_correction(basis, outputs, |surface| {
566            measured
567                .keys()
568                .any(|key| stabilizer_supports_measurement(surface, key))
569        })?;
570        let flips = surfaces
571            .iter()
572            .map(|surface| {
573                measured.iter().fold(false, |parity, (key, value)| {
574                    parity
575                        ^ ((*value ^ reference[key])
576                            && stabilizer_supports_measurement(surface, key))
577                })
578            })
579            .collect::<Vec<_>>();
580        match correction.evaluate(&flips) {
581            Ok(corrections) => insert_output_corrections(graph, outputs, &corrections),
582            Err(OutputCorrectionError::Inconsistent) => *graph.scalar_mut() = FScalar::zero(),
583            Err(
584                error @ (OutputCorrectionError::RhsLengthMismatch { .. }
585                | OutputCorrectionError::UnknownOutput { .. }),
586            ) => return Err(error.into()),
587        }
588        Ok(())
589    }

591    /// Correct within the authored named branch. Its all-zero unnamed outcome
592    /// assignment can be impossible: repeated merges must agree with the named
593    /// parity. Closed surfaces find a consistent reference with free bits zero.
594    fn reference_outcomes(
595        &self,
596        basis: &RuntimeStabilizerBasis,
597        measurement_vars: &BTreeMap<MeasurementKey, Parity>,
598        reference: BTreeMap<MeasurementKey, bool>,
599        actions: &[ActionNode],
600        feedback_values: &FxHashMap<usize, bool>,
601    ) -> Option<BTreeMap<MeasurementKey, bool>> {
602        let surfaces = basis.closed_stabilizer_surfaces();
603        let mut rhs = CoeffVec::zeros(surfaces.len());
604        for (index, surface) in surfaces.iter().enumerate() {
605            let parity = reference
606                .iter()
607                .fold(surface.sign, |parity, (key, &value)| {
608                    parity ^ (value && stabilizer_supports_measurement(surface, key))
609                })
610                ^ feedback_sign(surface, basis.zx_graph(), actions, feedback_values);
611            rhs.set_bit(index, parity);
612        }
613        // Each sampled variable includes its compensating native phases.
614        // Solving individual phase coordinates would change the named branch.
615        let columns = self.parity_columns(&surfaces, measurement_vars);
616        let solution = solve_coeff_combination(&columns, &rhs)?;
617        Some(self.apply_coordinate(measurement_vars, reference, &solution))
618    }

620    fn apply_coordinate(
621        &self,
622        phases: &BTreeMap<MeasurementKey, Parity>,
623        mut outcomes: BTreeMap<MeasurementKey, bool>,
624        coordinate: &CoeffVec,
625    ) -> BTreeMap<MeasurementKey, bool> {
626        for (key, value) in &mut outcomes {
627            let parity = &phases[key];
628            for (index, (_, variable)) in self.sampled.iter().enumerate() {
629                *value ^= coordinate.bit(index) && parity.iter().any(|value| value == *variable);
630            }
631        }
632        outcomes
633    }
634}

636fn combine_columns(columns: &[CoeffVec], selected: &CoeffVec, mut result: CoeffVec) -> CoeffVec {
637    for column in selected.iter_ones() {
638        result.xor_assign(&columns[column]);
639    }
640    result
641}

643fn pauli_coeffs(paulis: &crate::PauliString) -> CoeffVec {
644    let mut coefficients = CoeffVec::zeros(2 * paulis.len());
645    for (column, pauli) in paulis.iter_support() {
646        coefficients.set_bit(2 * column, pauli & Pauli::X);
647        coefficients.set_bit(2 * column + 1, pauli & Pauli::Z);
648    }
649    coefficients
650}

652fn kernel_directions(columns: &[CoeffVec]) -> Vec<CoeffVec> {
653    let mut basis = Vec::<CoeffVec>::new();
654    let mut selected = Vec::new();
655    let mut kernel = Vec::new();
656    for (index, column) in columns.iter().enumerate() {
657        if let Some(combination) = solve_coeff_combination(&basis, column) {
658            let mut direction = CoeffVec::singleton(index, columns.len());
659            for bit in combination.iter_ones() {
660                direction.set_bit(selected[bit], true);
661            }
662            kernel.push(direction);
663        } else {
664            selected.push(index);
665            basis.push(column.clone());
666        }
667    }
668    kernel
669}

671fn feedback_sign(
672    surface: &crate::Stabilizer,
673    zx: &ZXGraph,
674    actions: &[ActionNode],
675    values: &FxHashMap<usize, bool>,
676) -> bool {
677    actions
678        .iter()
679        .filter(|action| values.get(&action.ordinal) == Some(&true))
680        .fold(false, |odd, action| {
681            let Action::Feedback { targets, .. } = &action.action else {
682                return odd;
683            };
684            targets.iter().fold(odd, |odd, target| {
685                let Some((column, pauli)) = zx.feedback_column(target) else {
686                    return odd;
687                };
688                odd ^ surface.paulis.get(column).anticommutes(pauli)
689            })
690        })
691}

693fn boundary_bodies(
694    basis: &RuntimeStabilizerBasis,
695    surfaces: &[crate::Stabilizer],
696    outputs: &[IVec3],
697) -> Result<(Vec<crate::PauliString>, Vec<usize>, usize), FeedbackInferenceError> {
698    let zx = basis.zx_graph();
699    let open = zx
700        .nodes()
701        .iter()
702        .filter(|node| matches!(node.kind, NodeKind::Port | NodeKind::T))
703        .collect::<Vec<_>>();
704    let width = open
705        .iter()
706        .map(|node| {
707            if node.role == crate::PortRole::Multiplex {
708                2
709            } else {
710                1
711            }
712        })
713        .sum();
714    let mut bodies = Vec::with_capacity(surfaces.len() + open.len());
715    for surface in surfaces {
716        let mut body = crate::PauliString::new(width);
717        let mut column = 0;
718        for node in &open {
719            let pauli = surface.paulis.get(node.id);
720            if node.role == crate::PortRole::Multiplex {
721                // The Z-copy isometry maps X -> X⊗X, Z -> I⊗Z and adds Z⊗Z.
722                body.set(column, if pauli & Pauli::X { Pauli::X } else { Pauli::I });
723                column += 1;
724            }
725            body.set(column, pauli);
726            column += 1;
727        }
728        bodies.push(body);
729    }
730    let mut extras = Vec::new();
731    let mut column = 0;
732    for node in open {
733        if node.role == crate::PortRole::Multiplex {
734            let output = outputs
735                .iter()
736                .position(|&position| position == node.pos)
737                .ok_or_else(|| {
738                    FeedbackInferenceError::Binding(
739                        "multiplex boundary missing from output order".into(),
740                    )
741                })?;
742            let mut body = crate::PauliString::new(width);
743            body.set(column, Pauli::Z);
744            body.set(column + 1, Pauli::Z);
745            bodies.push(body);
746            extras.push(output);
747            column += 1;
748        }
749        column += 1;
750    }
751    Ok((bodies, extras, width))
752}

754pub(super) fn fresh_var(next: &mut usize) -> Result<Var, VerifyLogicalError> {
755    let variable = Var::try_from(*next).map_err(|_| VerifyLogicalError::TooManySymbols)?;
756    *next += 1;

760fn measurement_keys(zx: &ZXGraph) -> Vec<MeasurementKey> {
761    let mut keys = BTreeSet::new();
762    for node in zx.nodes() {
763        let pauli = match node.kind {
764            NodeKind::X => Pauli::Z,
765            NodeKind::Y => Pauli::Y,
766            NodeKind::Z => Pauli::X,
767            NodeKind::Selective(kind) => Pauli::from(kind.pauli_if_false()),
768            _ => continue,
769        };
770        let continues = zx.neighbors(node.id).is_some_and(|neighbors| {
771            neighbors.iter().any(|&neighbor| {
772                zx.nodes()[neighbor].pos.z > node.pos.z || zx.nodes()[neighbor].is_output_port(zx)
773            })
774        });
775        if !continues {
776            keys.insert(MeasurementKey::Node {
777                pos: node.pos,
778                pauli,
779            });
780        }
781    }
782    for edge in zx.edges().iter().filter(|edge| edge.n1 < edge.n2) {
783        let lhs = zx.nodes()[edge.n1];
784        let rhs = zx.nodes()[edge.n2];
785        if lhs.pos.z != rhs.pos.z
786            || !matches!(lhs.kind, NodeKind::X | NodeKind::Z)
787            || !matches!(rhs.kind, NodeKind::X | NodeKind::Z)
788        {
789            continue;
790        }
791        if (lhs.kind == NodeKind::X) == ((rhs.kind == NodeKind::X) ^ edge.hadamard) {
792            keys.insert(MeasurementKey::Edge {
793                src: lhs.pos,
794                dst: rhs.pos,
795                pauli: if lhs.kind == NodeKind::X {
796                    Pauli::X
797                } else {
798                    Pauli::Z
799                },
800            });
801        }
802    }
803    keys.into_iter().collect()
804}

806fn insert_output_corrections(
807    graph: &mut QuizxGraph,
808    outputs: &[IVec3],
809    corrections: &[(IVec3, Pauli)],
810) {
811    for &(position, pauli) in corrections {
812        let output = outputs
813            .iter()
814            .position(|candidate| *candidate == position)
815            .expect("correction names a requested output");
816        let boundary = graph.outputs()[output];
817        let vertex_types: &[VType] = match pauli {
818            Pauli::I => &[],
819            Pauli::X => &[VType::X],
820            Pauli::Y => &[VType::Z, VType::X],
821            Pauli::Z => &[VType::Z],
822        };
823        for &vertex_type in vertex_types {
824            let (neighbor, edge_type) = graph
825                .incident_edges(boundary)
826                .next()
827                .expect("an output boundary has one wire");
828            graph.remove_edge(boundary, neighbor);
829            let correction = graph.add_vertex_with_phase(vertex_type, Phase::from(1_i64));
830            graph.add_edge(boundary, correction);
831            graph.add_edge_with_type(correction, neighbor, edge_type);
832        }
833    }
834}

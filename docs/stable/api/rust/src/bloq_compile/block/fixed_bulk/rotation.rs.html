1//! Fixed-bulk-convention patch rotation patch construction.
2//!
3//! This module constructs the intermediate patch-rotation geometries directly
4//! on the fixed-bulk surface code lattice. Measure qubits stay on even
5//! coordinates, data qubits stay on odd coordinates, and stabilizer bases are
6//! always assigned by the fixed-bulk checkerboard, whose top-left bulk tile is
7//! Z.

61#[derive(Debug, Clone, Copy, PartialEq, Eq)]
62struct RotationLayout {
63    input_x_basis: Basis,
64    movement: IVec2,
65    geometry: RotationGeometry,
66}
67
68impl RotationLayout {
69    fn new(kind: PatchRotationKind, distance: u32) -> Self {
70        Self {
71            input_x_basis: kind.x_axis_boundary_basis(),
72            movement: kind.movement(),
73            geometry: RotationGeometry::new(distance),
74        }
75    }
76
77    fn local_to_global(self, p: IVec2, offset: IVec2) -> IVec2 {
78        (match self.movement.to_array() {
79            [0, 1] => p,
80            [0, -1] => IVec2::new(-p.x, -p.y),
81            [1, 0] => IVec2::new(p.y, p.x),
82            [-1, 0] => IVec2::new(-p.y, p.x),
83            _ => unreachable!("PatchRotationKind rejects non-cardinal movement"),
84        }) + offset
85    }
86}

88pub(super) fn compile_patch_rotation(
89    kind: PatchRotationKind,
90    connectivity: Connectivity,
91    distance: u32,
92) -> Result<CompiledTemplate, CompileError> {
93    let layout = RotationLayout::new(kind, distance);
94    // The spatial offset depends only on the layout, so resolve it once rather than
95    // per gateway key (the closure below runs once per observable key).
96    let offset = patch_rotation_offset(layout);
97    let patches = RotationPatches::new(layout, offset);
98    let chunks = patch_rotation_chunks(layout, distance, &patches)?;
99    let observable_gateway = resolve_gateway_measurements(
100        build_patch_rotation_gateway(layout, offset, connectivity, &patches),
101        &chunks,
102        |key| patch_rotation_operator_faces(layout, offset, key),
103    );
104    Ok(Arc::new(LoweringTemplate::from_chunks(
105        chunks,
106        observable_gateway,
107    )?))
108}

110fn patch_rotation_chunks(
111    layout: RotationLayout,
112    distance: u32,
113    patches: &RotationPatches,
114) -> Result<Vec<ChunkOrLoop>, CompileError> {
115    let init_basis = layout.input_x_basis.flip();
116    let meas_basis = init_basis.flip();
117    let grown_init_data = data_difference(&patches.grown, &patches.input, init_basis);
118    let grown_meas_data = data_difference_by_adjacent_tile_basis(&patches.grown, &patches.rotated);
119    let rotated_init_data =
120        data_difference_by_adjacent_tile_basis(&patches.rotated, &patches.grown);
121    let rotated_meas_data = data_difference(&patches.rotated, &patches.output, meas_basis);
122    let loop_repetitions = distance.saturating_sub(2);
123
124    Ok(vec![
125        ChunkOrLoop::Single(Box::new(make_patch_rotation_round_with_flow_boundaries(
126            &patches.grown,
127            &patches.input,
128            &patches.grown,
129            non_empty_data_map(&grown_init_data),
130            None,
131            FlowBoundaryData::default(),
132        )?)),
133        ChunkOrLoop::Loop {
134            body: vec![make_patch_rotation_round_with_flow_boundaries(
135                &patches.grown,
136                &patches.grown,
137                &patches.grown,
138                None,
139                None,
140                FlowBoundaryData::default(),
141            )?],
142            repetitions: loop_repetitions,
143        },
144        ChunkOrLoop::Single(Box::new(make_patch_rotation_round_with_flow_boundaries(
145            &patches.grown,
146            &patches.grown,
147            &patches.rotated,
148            None,
149            non_empty_data_map(&grown_meas_data),
150            FlowBoundaryData {
151                next_init_data: Some(&rotated_init_data),
152                ..FlowBoundaryData::default()
153            },
154        )?)),
155        ChunkOrLoop::Single(Box::new(make_patch_rotation_round_with_flow_boundaries(
156            &patches.rotated,
157            &patches.grown,
158            &patches.rotated,
159            non_empty_data_map(&rotated_init_data),
160            None,
161            FlowBoundaryData {
162                previous_meas_data: Some(&grown_meas_data),
163                ..FlowBoundaryData::default()
164            },
165        )?)),
166        ChunkOrLoop::Loop {
167            body: vec![make_patch_rotation_round_with_flow_boundaries(
168                &patches.rotated,
169                &patches.rotated,
170                &patches.rotated,
171                None,
172                None,
173                FlowBoundaryData::default(),
174            )?],
175            repetitions: loop_repetitions,
176        },
177        ChunkOrLoop::Single(Box::new(make_patch_rotation_round_with_flow_boundaries(
178            &patches.rotated,
179            &patches.rotated,
180            &patches.output,
181            None,
182            non_empty_data_map(&rotated_meas_data),
183            FlowBoundaryData::default(),
184        )?)),
185    ])
186}

188/// Boundary operators on a patch rotation's temporal faces.
189///
190/// Unlike a realignment (which flips X↔Z in place), a patch rotation keeps the
191/// logical basis and **moves** the operator's support: it enters on the input
192/// patch's middle line and leaves on the output patch's middle line, which is
193/// the same line turned 90° (the rotation swaps the patch's x- and y-axis
194/// boundaries). So `operator_in` equals the lower cube's `operator_out` and
195/// `operator_out` the upper cube's `operator_in`, and both seams cancel during
196/// lowering.
197///
198/// The lines are built in the template frame (through `local_to_global`, the
199/// same placement the patch tiles use), so the global-frame seam match lines
200/// them up with the neighbouring cubes despite the spatial move. Presence is
201/// gated on the key's temporal arms, mirroring `cube_operator_faces`.
202fn patch_rotation_operator_faces(
203    layout: RotationLayout,
204    offset: IVec2,
205    key: LocalStabilizer,
206) -> (PauliMap, PauliMap) {
207    let basis =
208        Basis::try_from(key.center_basis()).expect("a rotation flow key carries an X/Z center");
209    let pauli = Pauli::from(basis);
210
211    // The middle line runs along the axis whose like-basis boundaries it
212    // connects. The input patch carries `input_x_basis` on its x boundaries, so a
213    // basis-`b` line runs horizontally exactly when `b == input_x_basis`; the
214    // output patch flips that axis, turning the support 90°.
215    let operator_face = |present: bool, y0: i32, horizontal: bool| -> PauliMap {
216        if !present {
217            return PauliMap::empty();
218        }
219        standard_patch_line(layout, offset, y0, horizontal)
220            .into_iter()
221            .map(|q| (q, pauli))
222            .collect()
223    };
224    let input_horizontal = basis == layout.input_x_basis;
225    let operator_in = operator_face(key.has_arm(Direction::ZMINUS), 0, input_horizontal);
226    let operator_out = operator_face(
227        key.has_arm(Direction::ZPLUS),
228        layout.geometry.split_y + 1,
229        !input_horizontal,
230    );
231    (operator_in, operator_out)
232}

234/// Middle logical line of a standard `d×d` patch rooted at local row `y0`,
235/// expressed in the template frame: the shared middle line lifted to `y0` and
236/// pushed through the rotation's spatial move, matching the placement of the
237/// patch's own data qubits.
238fn standard_patch_line(
239    layout: RotationLayout,
240    offset: IVec2,
241    y0: i32,
242    horizontal: bool,
243) -> Vec<IVec2> {
244    let shift = IVec2::new(0, 2 * y0);
245    line_qubits(layout.geometry.width as u32, horizontal)
246        .into_iter()
247        .map(|q| layout.local_to_global(q + shift, offset))
248        .collect()
249}

251fn build_patch_rotation_gateway(
252    layout: RotationLayout,
253    offset: IVec2,
254    connectivity: Connectivity,
255    patches: &RotationPatches,
256) -> SelectedGateway {
257    let mut gateway = SelectedGateway::default();
258    let horizontal = Pauli::from(layout.input_x_basis);
259    let vertical = Pauli::from(layout.input_x_basis.flip());
260    let horizontal_measurements = horizontal_observable_measurements(layout, offset, patches);
261    let vertical_measurements = vertical_observable_measurements(layout, offset, patches);
262
263    for gateway_connectivity in patch_rotation_gateway_connectivity_subsets(connectivity) {
264        gateway.insert(
265            LocalStabilizer::new(horizontal, gateway_connectivity),
266            horizontal_measurements.clone(),
267        );
268        gateway.insert(
269            LocalStabilizer::new(vertical, gateway_connectivity),
270            vertical_measurements.clone(),
271        );
272    }
273
274    gateway
275}

277/// Every subset of the temporal pipe dirs (including the empty one), each as a
278/// Hadamard-free gateway-key connectivity.
279fn patch_rotation_gateway_connectivity_subsets(connectivity: Connectivity) -> Vec<Connectivity> {
280    let temporal_dirs = [Direction::ZMINUS, Direction::ZPLUS]
281        .into_iter()
282        .filter(|&dir| connectivity.has_pipe(dir))
283        .collect::<Vec<_>>();
284    (0..1usize << temporal_dirs.len())
285        .map(|mask| {
286            gateway_key_connectivity(
287                temporal_dirs
288                    .iter()
289                    .enumerate()
290                    .filter(|&(index, _)| mask & (1 << index) != 0)
291                    .map(|(_, &dir)| dir),
292            )
293        })
294        .collect()
295}

297fn horizontal_observable_measurements(
298    layout: RotationLayout,
299    offset: IVec2,
300    patches: &RotationPatches,
301) -> Vec<SelectedChunkMeasurements> {
302    let g = layout.geometry;
303    let mixed_right = VariantSpec::for_layout(layout).mixed_side.is_right();
304    let middle_line_end = if g.width % 4 == 1 {
305        g.width / 2 + 1
306    } else {
307        g.width / 2
308    };
309    let middle_line_xs = if mixed_right {
310        g.width - middle_line_end..g.width
311    } else {
312        0..middle_line_end
313    };
314
315    vec![
316        SelectedChunkMeasurements {
317            chunk_index: GROW_INIT_CHUNK,
318            qubits: selected_tile_measure_qubits_by_local_measure_coord(
319                &patches.grown,
320                layout,
321                offset,
322                layout.input_x_basis,
323                |m| in_horizontal_grown_observable_region(m, g, mixed_right),
324            ),
325        },
326        SelectedChunkMeasurements {
327            chunk_index: SHRINK_MEAS_CHUNK,
328            qubits: sorted_unique_coords(local_data_region(
329                layout,
330                offset,
331                middle_line_xs,
332                g.split_y..g.split_y + 1,
333            )),
334        },
335    ]
336}

338fn vertical_observable_measurements(
339    layout: RotationLayout,
340    offset: IVec2,
341    patches: &RotationPatches,
342) -> Vec<SelectedChunkMeasurements> {
343    let g = layout.geometry;
344    let mixed_right = VariantSpec::for_layout(layout).mixed_side.is_right();
345
346    vec![SelectedChunkMeasurements {
347        chunk_index: ROTATED_INIT_CHUNK,
348        qubits: selected_tile_measure_qubits_by_local_measure_coord(

350            layout,
351            offset,
352            layout.input_x_basis.flip(),

355    }]

358fn in_horizontal_grown_observable_region(
359    tile_coord: IVec2,
360    geometry: RotationGeometry,
361    mixed_right: bool,
362) -> bool {
363    let half_start = geometry.width / 2 + 1;
364    let in_bottom_patch_top_half = (0..=geometry.width).contains(&tile_coord.x)
365        && (half_start..=geometry.split_y).contains(&tile_coord.y);
366    let in_top_patch_side_half = if mixed_right {
367        (0..=geometry.width / 2).contains(&tile_coord.x)
368    } else {
369        (half_start..=geometry.width).contains(&tile_coord.x)
370    } && (geometry.split_y + 1..=geometry.height)
371        .contains(&tile_coord.y);
372    in_bottom_patch_top_half || in_top_patch_side_half
373}
374
375fn in_vertical_rotated_observable_region(
376    tile_coord: IVec2,
377    geometry: RotationGeometry,
378    mixed_right: bool,
379) -> bool {
380    let in_lower_patch_side_half = if mixed_right {
381        (0..=geometry.width / 2).contains(&tile_coord.x)
382    } else {
383        (geometry.width / 2 + 1..=geometry.width).contains(&tile_coord.x)
384    } && (0..=geometry.split_y).contains(&tile_coord.y);
385    let in_upper_patch_bottom_half = (0..=geometry.width).contains(&tile_coord.x)
386        && (geometry.split_y + 1..=geometry.split_y + 1 + geometry.width / 2)
387            .contains(&tile_coord.y);
388    in_lower_patch_side_half || in_upper_patch_bottom_half
389}

395/// Data qubits in `from` but not `without`, each mapped through `basis_of`.
396fn data_difference_with(
397    from: &Patch,
398    without: &Patch,
399    basis_of: impl Fn(IVec2) -> Basis,
400) -> FxMap<IVec2, Basis> {
401    let without = without.data_set();
402    from.data_set()
403        .into_iter()
404        .filter(|q| !without.contains(q))
405        .map(|q| (q, basis_of(q)))
406        .collect()
407}
408
409fn data_difference(from: &Patch, without: &Patch, basis: Basis) -> FxMap<IVec2, Basis> {
410    data_difference_with(from, without, |_| basis)
411}
412
413fn data_difference_by_adjacent_tile_basis(from: &Patch, without: &Patch) -> FxMap<IVec2, Basis> {
414    data_difference_with(from, without, |q| data_adjacent_tile_basis(from, q))
415}
416
417fn data_adjacent_tile_basis(patch: &Patch, data: IVec2) -> Basis {
418    patch
419        .tiles()
420        .iter()
421        .filter_map(|tile| {
422            let active = tile.active_data_qubits();
423            active.contains(&data).then_some((active.len(), tile))
424        })
425        .max_by_key(|(weight, tile)| (*weight, -tile.measure_qubit().y, -tile.measure_qubit().x))
426        .map(|(_, tile)| tile.basis())
427        .expect("data qubit from patch data set belongs to at least one tile")
428}

430fn patch_rotation_offset(layout: RotationLayout) -> IVec2 {
431    let geometry = layout.geometry;
432    let cell_shift = 2 * geometry.width;
433    let origin_shift = match layout.movement.to_array() {
434        [-1, 0] => IVec2::new(cell_shift, 0),
435        [0, -1] => IVec2::new(cell_shift, cell_shift),
436        _ => IVec2::ZERO,
437    };
438
439    [
440        IVec2::ZERO,
441        IVec2::new(2, 0),
442        IVec2::new(0, 2),
443        IVec2::new(2, 2),
444    ]
445    .into_iter()
446    .map(|phase| phase + origin_shift)
447    .find(|&offset| {
448        top_left_lattice_basis(layout, offset, geometry.width, geometry.height) == Basis::Z
449    })
450    .expect("the four phases enumerate every 2-periodic checkerboard parity, one of which is Z")
451}
452
453fn top_left_lattice_basis(layout: RotationLayout, offset: IVec2, width: i32, height: i32) -> Basis {
454    [
455        IVec2::new(0, 0),
456        IVec2::new(2 * width, 0),
457        IVec2::new(0, 2 * height),
458        IVec2::new(2 * width, 2 * height),
459    ]
460    .into_iter()
461    .map(|p| layout.local_to_global(p, offset))
462    .max_by_key(|p| (p.y, -p.x))
463    .map(checkerboard_basis)
464    .expect("rectangular lattice has corners")
465}

509#[derive(Debug, Clone, Copy)]
510struct BoundaryPolicy {
511    bottom: Basis,
512    top: Basis,
513    left_bottom: Basis,
514    left_top: Basis,
515    right_bottom: Basis,
516    right_top: Basis,
517    split_y: i32,
518}
519
520impl BoundaryPolicy {
521    /// The split-edge counterpart of [`EdgeBases::demanded`]: `None` when `m`
522    /// is interior or two edges demand different bases.
523    fn demanded(self, m: IVec2, rect: Rect) -> Option<Basis> {
524        let side = |bottom, top| {
525            if is_below_split_midline(m.y, self.split_y) {
526                bottom
527            } else {
528                top
529            }
530        };
531        let mut basis = None;
532        for edge_basis in [
533            (m.y == 0).then_some(self.bottom),
534            (m.y == 2 * rect.height).then_some(self.top),
535            (m.x == 0).then_some(side(self.left_bottom, self.left_top)),
536            (m.x == 2 * rect.width).then_some(side(self.right_bottom, self.right_top)),
537        ]
538        .into_iter()
539        .flatten()
540        {
541            if basis.is_some_and(|basis| basis != edge_basis) {
542                return None;
543            }
544            basis = Some(edge_basis);
545        }
546        basis
547    }
548
549    fn keeps(self, m: IVec2, rect: Rect, basis: Basis) -> bool {
550        match self.demanded(m, rect) {
551            Some(demanded) => basis == demanded,
552            None => !rect.on_edge(m),
553        }
554    }
555}

557#[derive(Debug, Clone, Copy, PartialEq, Eq)]
558struct VariantSpec {
559    mixed_side: MixedSide,
560}
561
562impl VariantSpec {
563    fn for_layout(layout: RotationLayout) -> Self {
564        // Canonical construction table before applying the movement transform.
565        //
566        // input X basis | movement | mixed side | grown extension | rotated extension
567        // X             | -X       | right      | left            | left
568        // X             | other    | left       | right           | right
569        // Z             | -X       | left       | right           | right
570        // Z             | other    | right      | left            | left
571        let mixed_side = match (layout.input_x_basis, layout.movement.to_array()) {
572            (Basis::X, [-1, 0]) => MixedSide::Right,
573            (Basis::X, _) | (Basis::Z, [-1, 0]) => MixedSide::Left,
574            (Basis::Z, _) => MixedSide::Right,
575        };
576        Self { mixed_side }
577    }
578
579    fn boundaries(
580        self,
581        input_x_basis: Basis,
582        stage: RotationStage,
583        geometry: RotationGeometry,
584    ) -> BoundaryPolicy {
585        let opposite = input_x_basis.flip();
586        let extended_basis = match stage {
587            RotationStage::Grown => input_x_basis,
588            RotationStage::Rotated => opposite,
589        };
590        match self.mixed_side {
591            MixedSide::Left => BoundaryPolicy {
592                bottom: opposite,
593                top: input_x_basis,
594                left_bottom: input_x_basis,
595                left_top: opposite,
596                right_bottom: extended_basis,
597                right_top: extended_basis,
598                split_y: geometry.split_y,
599            },
600            MixedSide::Right => BoundaryPolicy {
601                bottom: opposite,
602                top: input_x_basis,
603                left_bottom: extended_basis,
604                left_top: extended_basis,
605                right_bottom: input_x_basis,
606                right_top: opposite,
607                split_y: geometry.split_y,
608            },
609        }
610    }
611
612    fn schedule_rule(self, stage: RotationStage, geometry: RotationGeometry) -> ScheduleRule {
613        match (self.mixed_side, stage) {
614            (MixedSide::Left, RotationStage::Grown) => ScheduleRule::new(1, geometry.split_y),
615            (MixedSide::Left, RotationStage::Rotated) => ScheduleRule::new(-1, geometry.split_y),
616            (MixedSide::Right, RotationStage::Grown) => {
617                ScheduleRule::new(-1, geometry.split_y + geometry.width)
618            }
619            (MixedSide::Right, RotationStage::Rotated) => {
620                ScheduleRule::new(1, geometry.split_y - geometry.width)
621            }
622        }
623    }
624
625    fn cx_order(self, stage: RotationStage, schedule: CXSchedule) -> [Corner; 4] {
626        match (self.mixed_side, stage, schedule) {
627            (MixedSide::Left, RotationStage::Grown, CXSchedule::Horizontal)
628            | (MixedSide::Right, RotationStage::Rotated, CXSchedule::Horizontal) => {
629                [Corner::BR, Corner::BL, Corner::TR, Corner::TL]
630            }
631            (MixedSide::Left, RotationStage::Grown, CXSchedule::Vertical)
632            | (MixedSide::Right, RotationStage::Rotated, CXSchedule::Vertical) => {
633                [Corner::BR, Corner::TR, Corner::BL, Corner::TL]
634            }
635            (MixedSide::Left, RotationStage::Rotated, CXSchedule::Horizontal)
636            | (MixedSide::Right, RotationStage::Grown, CXSchedule::Horizontal) => {
637                [Corner::BL, Corner::BR, Corner::TL, Corner::TR]
638            }
639            (MixedSide::Left, RotationStage::Rotated, CXSchedule::Vertical)
640            | (MixedSide::Right, RotationStage::Grown, CXSchedule::Vertical) => {
641                [Corner::BL, Corner::TL, Corner::BR, Corner::TR]
642            }
643        }
644    }
645}

659#[derive(Debug, Clone, Copy)]
660struct ScheduleRule {
661    diagonal_sign: i32,
662    line_intercept: i32,
663}
664
665impl ScheduleRule {
666    const fn new(diagonal_sign: i32, line_intercept: i32) -> Self {
667        Self {
668            diagonal_sign,
669            line_intercept,
670        }
671    }
672
673    fn schedule(self, basis: Basis, m: IVec2, horizontal_hook_basis: Basis) -> CXSchedule {
674        let standard = tile_cx_schedule(basis, horizontal_hook_basis);
675        if self.uses_original_schedule(m) {
676            standard
677        } else {
678            opposite_schedule(standard)
679        }
680    }
681
682    fn uses_original_schedule(self, m: IVec2) -> bool {
683        data_y(m.y) <= self.diagonal_sign * data_x(m.x) + self.line_intercept
684    }
685}
686
687fn opposite_schedule(schedule: CXSchedule) -> CXSchedule {
688    match schedule {
689        CXSchedule::Horizontal => CXSchedule::Vertical,
690        CXSchedule::Vertical => CXSchedule::Horizontal,
691    }
692}

694fn data_x(measure_x: i32) -> i32 {
695    measure_x / 2
696}
697
698fn data_y(measure_y: i32) -> i32 {
699    measure_y / 2
700}

702fn is_below_split_midline(measure_y: i32, split_y: i32) -> bool {
703    measure_y < 2 * split_y + 1

706fn make_patch(
707    stage: RotationStage,
708    layout: RotationLayout,
709    offset: IVec2,
710    spec: VariantSpec,
711) -> Patch {
712    let geometry = layout.geometry;
713    let rect = Rect::new(geometry.width, geometry.height);
714    let boundaries = spec.boundaries(layout.input_x_basis, stage, geometry);
715    let schedule_rule = spec.schedule_rule(stage, geometry);
716
717    carve_patch(
718        rect.data_qubits(),
719        |m| {
720            boundaries.keeps(
721                m,
722                rect,
723                checkerboard_basis(layout.local_to_global(m, offset)),
724            )
725        },
726        |m| layout.local_to_global(m, offset),
727        |m, data| {
728            let global_m = layout.local_to_global(m, offset);
729            let basis = checkerboard_basis(global_m);
730            let schedule = schedule_rule.schedule(basis, m, layout.input_x_basis.flip());
731            let slots = spec.cx_order(stage, schedule).into_iter().map(|corner| {
732                let dq = m + corner.to_ivec2();
733                data.contains(&dq)
734                    .then_some(layout.local_to_global(dq, offset))
735            });
736            Tile::new(basis, global_m, slots)
737        },
738    )
739}
740
741fn make_standard_patch(layout: RotationLayout, offset: IVec2, y0: i32, x_basis: Basis) -> Patch {
742    let width = layout.geometry.width;
743    let rect = Rect::rooted(width, y0, width);
744    let boundaries = EdgeBases::standard(x_basis);
745
746    carve_patch(
747        rect.data_qubits(),
748        |m| {
749            boundaries.keeps(
750                m,
751                rect,
752                checkerboard_basis(layout.local_to_global(m, offset)),
753            )
754        },
755        |m| layout.local_to_global(m, offset),
756        |m, data| {
757            let global_m = layout.local_to_global(m, offset);
758            let basis = checkerboard_basis(global_m);
759            let slots = tile_cx_schedule(basis, x_basis.flip())
760                .compact()
761                .into_iter()
762                .map(|corner| {
763                    let dq = m + corner.to_ivec2();
764                    data.contains(&dq)
765                        .then_some(layout.local_to_global(dq, offset))
766                });
767            Tile::new(basis, global_m, slots)
768        },
769    )
770}

772fn data_qubit(x: i32, y: i32) -> IVec2 {
773    IVec2::new(2 * x + 1, 2 * y + 1)
774}

776fn local_data_region(
777    layout: RotationLayout,
778    offset: IVec2,
779    xs: impl Iterator<Item = i32> + Clone,
780    ys: impl Iterator<Item = i32>,
781) -> FxSet<IVec2> {
782    ys.flat_map(|y| {
783        xs.clone()
784            .map(move |x| layout.local_to_global(data_qubit(x, y), offset))
785    })
786    .collect()
787}

789fn selected_tile_measure_qubits_by_local_measure_coord(
790    patch: &Patch,
791    layout: RotationLayout,
792    offset: IVec2,
793    basis: Basis,
794    keep_local_measure: impl Fn(IVec2) -> bool,
795) -> Vec<IVec2> {
796    patch
797        .tiles()
798        .iter()
799        .filter(|tile| tile.basis() == basis)
800        .filter(|tile| {
801            let m = global_to_local(layout, offset, tile.measure_qubit());
802            debug_assert_eq!(m.x % 2, 0);
803            debug_assert_eq!(m.y % 2, 0);
804            keep_local_measure(m / 2)
805        })
806        .map(Tile::measure_qubit)
807        .collect()
808}

810fn global_to_local(layout: RotationLayout, offset: IVec2, p: IVec2) -> IVec2 {
811    let q = p - offset;
812    match layout.movement.to_array() {
813        [0, 1] => q,
814        [0, -1] => -q,
815        [1, 0] => IVec2::new(q.y, q.x),
816        [-1, 0] => IVec2::new(q.y, -q.x),
817        _ => unreachable!("PatchRotationKind rejects non-cardinal movement"),
818    }
819}

821fn make_patch_rotation_round_with_flow_boundaries(
822    patch: &Patch,
823    previous_patch: &Patch,
824    next_patch: &Patch,
825    init_data: Option<&FxMap<IVec2, Basis>>,
826    meas_data: Option<&FxMap<IVec2, Basis>>,
827    flow_boundaries: FlowBoundaryData<'_>,
828) -> Result<Chunk, CompileError> {
829    let round = emit_syndrome_round(
830        patch,
831        &sorted_batches(init_data, [Basis::Z, Basis::X]),
832        &sorted_batches(meas_data, [Basis::Z, Basis::X]),
833    )?;
834    // Precompute the neighbouring patches' boundary keys once so each tile
835    // below does a hash lookup instead of rescanning every neighbour tile.
836    let previous_outputs =
837        PatchBoundarySets::collapsed_boundaries(previous_patch, flow_boundaries.previous_meas_data);
838    let next_inputs =
839        PatchBoundarySets::collapsed_boundaries(next_patch, flow_boundaries.next_init_data);
840    let flow_context = ExactRoundFlowContext {
841        previous_outputs,
842        next_inputs,
843        init_data,
844        meas_data,
845        measurement_ids: &round.measurements,
846    };
847    let mut flows = Vec::new();
848    for tile in patch.tiles() {
849        flows.extend(tile_flows_with_exact_boundaries(tile, &flow_context)?);
850    }
851
852    Ok(Chunk {
853        circuit: round.circuit,
854        flows,
855    })
856}

858#[derive(Debug, Clone, Copy, Default)]
859struct FlowBoundaryData<'a> {
860    previous_meas_data: Option<&'a FxMap<IVec2, Basis>>,
861    next_init_data: Option<&'a FxMap<IVec2, Basis>>,
862}
863
864struct ExactRoundFlowContext<'a> {
865    previous_outputs: PatchBoundarySets,
866    next_inputs: PatchBoundarySets,
867    init_data: Option<&'a FxMap<IVec2, Basis>>,
868    meas_data: Option<&'a FxMap<IVec2, Basis>>,
869    measurement_ids: &'a MeasurementIndex,
870}
871
872/// Per-basis boundary keys exposed by a neighbouring patch.
873#[derive(Debug, Default)]
874struct PatchBoundarySets {
875    z: FxSet<PauliMap>,
876    x: FxSet<PauliMap>,
877}
878
879impl PatchBoundarySets {
880    /// Boundary keys the patch's tiles carry once `collapsed_data` is
881    /// measured/initialized away: the outputs a previous round still presents,
882    /// or equivalently the inputs a next round expects — the continuing
883    /// supports of the shared collapse rule either way.
884    fn collapsed_boundaries(patch: &Patch, collapsed_data: Option<&FxMap<IVec2, Basis>>) -> Self {
885        let mut sets = Self::default();
886        for tile in patch.tiles() {
887            let basis = tile.basis();
888            if let Some((key, _)) = collapse_matching_data_boundary(
889                &tile.active_data_qubits(),
890                basis,
891                Pauli::from(basis),
892                collapsed_data,
893            ) {
894                sets.for_basis_mut(basis).insert(key);
895            }
896        }
897        sets
898    }
899
900    fn contains(&self, basis: Basis, key: &PauliMap) -> bool {
901        match basis {
902            Basis::Z => self.z.contains(key),
903            Basis::X => self.x.contains(key),
904        }
905    }
906
907    fn for_basis_mut(&mut self, basis: Basis) -> &mut FxSet<PauliMap> {
908        match basis {
909            Basis::Z => &mut self.z,
910            Basis::X => &mut self.x,
911        }
912    }
913}
914
915fn tile_flows_with_exact_boundaries(
916    tile: &Tile,
917    context: &ExactRoundFlowContext<'_>,
918) -> Result<Vec<Flow>, CompileError> {
919    let mut flows = Vec::with_capacity(2);
920    let active_data = tile.active_data_qubits();
921    let basis = tile.basis();
922    let pauli = Pauli::from(basis);
923    let own_measurement = measurement_id(context.measurement_ids, tile.measure_qubit())?;
924
925    if let Some((start, _)) =
926        collapse_matching_data_boundary(&active_data, basis, pauli, context.init_data)
927    {
928        let compared_to_previous = context.previous_outputs.contains(basis, &start);
929        let initialized = start.is_empty();
930        if initialized || compared_to_previous {
931            flows.push(
932                Flow::new(start, PauliMap::empty())
933                    .with_measurements([own_measurement])
934                    .with_center(tile.measure_qubit()),
935            );
936        }
937    }
938
939    if let Some((end, mut measured_data)) =
940        collapse_matching_data_boundary(&active_data, basis, pauli, context.meas_data)
941    {
942        let continues_to_next = context.next_inputs.contains(basis, &end);
943        let finalized = end.is_empty();
944        if finalized || continues_to_next {
945            measured_data.sort_unstable_by_key(|q| (q.y, q.x));
946            let mut measurements = FlowMeasurements::with_capacity(measured_data.len() + 1);
947            for q in measured_data {
948                measurements.push(measurement_id(context.measurement_ids, q)?);
949            }
950            measurements.push(own_measurement);
951            flows.push(
952                Flow {
953                    measurements,
954                    ..Flow::new(PauliMap::empty(), end)
955                }
956                .with_center(tile.measure_qubit()),
957            );
958        }
959    }
960
961    Ok(flows)
962}
963
964fn measurement_id(measurement_ids: &MeasurementIndex, q: IVec2) -> Result<u32, CompileError> {
965    measurement_ids
966        .get(q)
967        .ok_or_else(|| CompileError::PatchRotationConstructionFailed {
968            reason: format!("flow references unmeasured qubit {q:?}"),
969        })
970}

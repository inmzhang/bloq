41/// Patch and collapsed-data maps for a cube-shaped block: spatial cubes get
42/// the tqec spatial template, regular cubes the boundary-basis classifier.
43/// `layer_schedule` is the CX-slot depth of the cube's z-layer spatial
44/// component, which every member shares (see [`LayerSchedule`]).
45fn build_cube_patch(
46    kind: CubeKind,
47    connectivity: Connectivity,
48    distance: u32,
49    layer_schedule: LayerSchedule,
50) -> grid::PatchData {
51    build_cube_patch_with_reverse(kind, connectivity, distance, layer_schedule, false)
52}

54fn build_cube_patch_with_reverse(
55    kind: CubeKind,
56    connectivity: Connectivity,
57    distance: u32,
58    layer_schedule: LayerSchedule,
59    reverse_schedule: bool,
60) -> grid::PatchData {
61    if kind.is_spatial() {
62        build_spatial_patch_with_reverse(
63            kind,
64            distance,
65            connectivity,
66            layer_schedule,
67            reverse_schedule,
68        )
69    } else {
70        build_regular_patch_with_reverse(
71            kind,
72            distance,
73            connectivity,
74            layer_schedule,
75            reverse_schedule,
76        )
77    }
78}

80/// Compile a cube block into init + bulk loop + meas chunks.
81fn compile_cube(
82    kind: CubeKind,
83    connectivity: Connectivity,
84    distance: u32,
85    rounds: u32,
86    layer_schedule: LayerSchedule,
87) -> Result<CompiledTemplate, CompileError> {
88    let (patch, init_data, meas_data) =
89        build_cube_patch(kind, connectivity, distance, layer_schedule);
90    let cube = make_surface_code_cube_chunks(&patch, Some(&init_data), &meas_data)?;
91    let (init, init_ids) = cube.init.expect("cube initialization was requested");
92    let repetitions = cube_bulk_repetitions(rounds);
93
94    let mut chunks = vec![ChunkOrLoop::Single(Box::new(init))];
95    // The stage that ends up last is the one whose recorded index the gateway
96    // needs; the extended schedule picks between the forward and reverse round.
97    let meas_ids;
98    if layer_schedule.is_extended() {
99        let (reverse_patch, _, _) =
100            build_cube_patch_with_reverse(kind, connectivity, distance, layer_schedule, true);
101        let reverse = make_surface_code_cube_chunks(&reverse_patch, None, &meas_data)?;
102        if repetitions >= 2 {
103            chunks.push(ChunkOrLoop::Loop {
104                body: vec![reverse.bulk.clone(), cube.bulk],
105                repetitions: repetitions / 2,
106            });
107        }
108        if repetitions % 2 == 1 {
109            chunks.push(ChunkOrLoop::Single(Box::new(reverse.bulk)));
110            chunks.push(ChunkOrLoop::Single(Box::new(cube.meas)));
111            meas_ids = cube.meas_ids;
112        } else {
113            chunks.push(ChunkOrLoop::Single(Box::new(reverse.meas)));
114            meas_ids = reverse.meas_ids;
115        }
116    } else {
117        // A two-round cube — `height=d/2` at `d=3`, say — is all init and meas with
118        // no bulk left over. Drop the loop rather than emit a zero-repetition
119        // one, which downstream stage indexing treats as a real stage.
120        if repetitions > 0 {
121            chunks.push(ChunkOrLoop::Loop {
122                body: vec![cube.bulk],
123                repetitions,
124            });
125        }
126        chunks.push(ChunkOrLoop::Single(Box::new(cube.meas)));
127        meas_ids = cube.meas_ids;
128    }
129
130    let measurement_chunk = stage_chunks(&chunks).count() - 1;
131    let observable_gateway = resolve_gateway_measurements_at(
132        build_gateway_from_patch(kind, connectivity, distance, &patch),
133        &chunks,
134        measurement_chunk,
135        [(INIT_CHUNK, init_ids), (measurement_chunk, meas_ids)]
136            .into_iter()
137            .collect(),
138        |key| cube_operator_faces(kind, distance, key),
139    );
140
141    Ok(Arc::new(LoweringTemplate::from_chunks(
142        chunks,
143        observable_gateway,
144    )?))
145}

147/// Compile a temporal seam's padding templates: `rounds` standard
148/// syndrome-extraction rounds over a plain `d x d` surface code face named by
149/// the basis on its `y`-normal edges, with no data init or readout.
150///
151/// The face is always on the compact four-slot schedule — idling a bare patch
152/// has no merges and no walls to lengthen a round for. The first round stays
153/// outside the loop so its seam-facing consumer flows survive into
154/// `boundary_flows` (they close against the node below the seam); the loop's
155/// top-open creators are the backend post-`REPEAT` lookback pattern (they close
156/// against the node above). One template per entry of `rounds`, sharing the
157/// patch's Theta(d^2) tile build across them.
158pub(crate) fn compile_seam_padding_rounds<const N: usize>(
159    y_face_basis: Basis,
160    distance: u32,
161    rounds: [u32; N],
162) -> Result<[Arc<bloq_ir::lowering::BloqTemplate>; N], CompileError> {
163    compile_patch_memory_padding_rounds(
164        &make_normal_surface_code_patch(distance, y_face_basis),
165        rounds,
166    )
167}

169/// The cube-derived padding builder the seam builder replaced, kept only so
170/// `seam_padding_matches_the_cube_derived_patch` can hold one to the other.
171#[cfg(test)]
172fn compile_memory_padding_rounds<const N: usize>(
173    kind: CubeKind,
174    connectivity: Connectivity,
175    distance: u32,
176    layer_schedule: LayerSchedule,
177    rounds: [u32; N],
178) -> Result<[Arc<bloq_ir::lowering::BloqTemplate>; N], CompileError> {
179    let (patch, _init_data, _meas_data) =
180        build_cube_patch(kind, connectivity, distance, layer_schedule);
181    compile_patch_memory_padding_rounds(&patch, rounds)
182}

184/// Compile one padding template per entry of `rounds`. The patch's memory chunk
185/// is identical across every round count — padding carries no init or
186/// measurement data to vary it — so it is built once and cloned into each.
187fn compile_patch_memory_padding_rounds<const N: usize>(
188    patch: &Patch,
189    rounds: [u32; N],
190) -> Result<[Arc<bloq_ir::lowering::BloqTemplate>; N], CompileError> {
191    if rounds.contains(&0) {
192        return Err(CompileError::PaddingRoundsZero);
193    }
194    let round = make_surface_code_chunk(patch, None, None)?;
195    let mut templates = Vec::with_capacity(N);
196    for count in rounds {
197        if count == 1 {
198            templates.push(single_round_padding_template(&round));
199            continue;
200        }
201        let chunks = vec![
202            ChunkOrLoop::Single(Box::new(round.clone())),
203            ChunkOrLoop::Loop {
204                body: vec![round.clone()],
205                repetitions: count - 1,
206            },
207        ];
208        // A padding template routes no observables: the seam's logical operator
209        // moves through it untouched, and observable evidence stays on the cubes.
210        let template =
211            LoweringTemplate::from_chunks(chunks, crate::block::gateway::ObservableGateway::new())?;
212        templates.push(template.program_template);
213    }
214    let Ok(templates) = <[_; N]>::try_from(templates) else {
215        unreachable!("one template pushed per requested round count")
216    };
217    Ok(templates)
218}

220fn single_round_padding_template(round: &Chunk) -> Arc<BloqTemplate> {
221    assert!(
222        round
223            .flows
224            .iter()
225            .all(|flow| flow.start.is_empty() != flow.end.is_empty()),
226        "a memory round has only boundary flows"
227    );
228    let boundary_flows = round.flows.clone();
229    #[cfg(debug_assertions)]
230    {
231        let mut sorted = boundary_flows.clone();
232        crate::block::compile::sort_boundary_flows(&mut sorted);
233        debug_assert_eq!(boundary_flows, sorted);
234    }
235    Arc::new(BloqTemplate::with_parts(
236        round.circuit.clone(),
237        Vec::new(),
238        Vec::new(),
239        boundary_flows,
240        Vec::new(),
241    ))
242}

244/// Bulk-loop repetitions for a cube of `rounds` syndrome rounds: the rounds left
245/// over once the initialization and measurement stages have taken one each.
246///
247/// Infallible because `signature::cube_rounds` has already rejected any height
248/// that resolves below two rounds, and it is the only producer of the round
249/// counts that reach here.
250pub(super) fn cube_bulk_repetitions(rounds: u32) -> u32 {
251    rounds
252        .checked_sub(2)
253        .expect("block signatures reject cubes with fewer than 2 rounds")
254}

256/// The repetition count of the template's bulk loop, or `None` when the
257/// schedule has no loop (e.g. a flattened two-round cube).
258///
259/// Shared by the `fixed_bulk` and `ybasis` test modules.
260#[cfg(test)]
261fn loop_repetitions(template: &LoweringTemplate) -> Option<u32> {
262    template
263        .program_template
264        .circuit
265        .entry_top_level_repeats()
266        .first()
267        .copied()
268}

270#[cfg(test)]
271fn split_ops_into_moments(ops: &[bloq_circuit::Op]) -> Vec<Vec<bloq_circuit::Op>> {
272    let mut moments = vec![Vec::new()];
273    for op in ops {
274        if matches!(op, bloq_circuit::Op::Tick) {
275            moments.push(Vec::new());
276        } else {
277            moments
278                .last_mut()
279                .expect("at least one moment")
280                .push(op.clone());
281        }
282    }
283    moments
284}

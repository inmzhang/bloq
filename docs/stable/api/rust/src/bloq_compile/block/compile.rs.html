23/// A block's whole compiled circuit plus the canonical detector, observable,
24/// and boundary-flow data the backend instantiates per placement.
25#[derive(Debug, Clone)]
26pub(crate) struct LoweringTemplate {
27    /// The whole circuit plus its canonical template-internal detectors, repeat
28    /// states, and residual open flows. The backend instantiates these per
29    /// placement by a cheap offset + measurement relabel, so internal detectors
30    /// never need replay. The residual (promoted onto the persisted template)
31    /// lives in [`BloqTemplate::boundary_flows`]. Shared across
32    /// identical placements via `Arc` so a repeated template lowers only once.
33    pub(crate) program_template: Arc<BloqTemplate>,
34    /// Observable routing, in template-measurement-id space (see
35    /// [`ObservableGateway::resolve_to_template_ids`]).
36    pub(crate) observable_gateway: ObservableGateway,
37}

39/// Drain the finished composition engine's boundary residual: the open flows
40/// the template does not close internally, fused into independent chains.
41///
42/// `build_program_template` appends every chunk in residual mode, so this is a
43/// read-off of that single pass rather than a second composition: internal
44/// detectors already fell out through `drain_completed`, and what the engine
45/// still holds (chains open, or closed against a boundary input) is the
46/// residual, already fused — matching *is* fusion, so a chain threaded across
47/// chunk seams becomes one `start -> end` flow. The result is sorted into a
48/// canonical order for reproducible emission; order does not affect correctness.
49///
50/// A loop's recurrence stays in `program_template.repeat_states` (backend-owned):
51/// `start_loop_detector_states` rewrites each carried key's seed into the loop
52/// state's `initial`, so the seed never reaches the residual, and a top-open loop
53/// creator falls out as a bare-measurement flow resolved by the backend's
54/// post-`REPEAT` lookback. A *moving* operator inside a loop body instead leaves
55/// a residual chain still carrying a `LoopState` term, refused with
56/// `LoopBoundaryMovingOperatorUnsupported`.
57#[hotpath::measure]
58fn drain_boundary_flows(
59    engine: FlowEngine<'_, DetectorTerm<u32>>,
60) -> Result<Vec<Flow>, CompileError> {
61    // Convert the fused residual to template-id `Flow`s. A surviving `LoopState`
62    // term is the moving-operator-in-loop case.
63    let residuals = engine.into_residual_flows();
64    let mut flows = Vec::with_capacity(residuals.len());
65    for residual in residuals {
66        let mut measurements = FlowMeasurements::new();
67        for term in residual.measurements {
68            match term {
69                DetectorTerm::Measurement(measurement) => measurements.push(measurement),
70                DetectorTerm::LoopState(_) => {
71                    return Err(CompileError::LoopBoundaryMovingOperatorUnsupported);
72                }
73            }
74        }
75        // Propagate the fused marker onto the persisted boundary flow so the
76        // instance-seam detector emission can suppress detectors (or route to a
77        // restart) for a selective block's opposite-basis stabilizers.
78        flows.push(
79            Flow {
80                measurements,
81                ..Flow::new(residual.start, residual.end)
82            }
83            .with_sign(residual.sign)
84            .with_center(residual.center)
85            .with_marker(residual.marker),
86        );
87    }
88
89    // `into_residual_flows` yields open chains in hash order; sort into a
90    // canonical order for reproducible detector emission. `center` and `marker`
91    // are the final tiebreakers so flows that agree on measurements/sign/faces
92    // (a `Detector` vs. `Restart` twin, or two centers) no longer fall to hash
93    // order.
94    sort_boundary_flows(&mut flows);
95    Ok(flows)
96}

98pub(super) fn sort_boundary_flows(flows: &mut [Flow]) {
99    flows.sort_unstable_by(|a, b| {
100        a.measurements
101            .cmp(&b.measurements)
102            .then_with(|| a.sign.cmp(&b.sign))
103            .then_with(|| pauli_map_cmp(&a.start, &b.start))
104            .then_with(|| pauli_map_cmp(&a.end, &b.end))
105            .then_with(|| {
106                a.center
107                    .map(|c| c.to_array())
108                    .cmp(&b.center.map(|c| c.to_array()))
109            })
110            .then_with(|| (a.marker as u8).cmp(&(b.marker as u8)))
111    });
112}

114/// Total order on `PauliMap`s by their coordinate-sorted entries, for
115/// deterministic residual ordering.
116fn pauli_map_cmp(a: &PauliMap, b: &PauliMap) -> std::cmp::Ordering {
117    a.iter()
118        .map(|(coord, pauli)| (coord.x, coord.y, *pauli as u8))
119        .cmp(
120            b.iter()
121                .map(|(coord, pauli)| (coord.x, coord.y, *pauli as u8)),
122        )
123}

125impl LoweringTemplate {
126    /// Build a template from its ordered chunks and observable gateway.
127    ///
128    /// Composes the per-chunk flows into the program template's internal
129    /// detectors and boundary residual, then canonicalizes the gateway from
130    /// chunk-local measurement ids to template ids so observable lowering never
131    /// has to translate.
132    #[hotpath::measure]
133    pub(crate) fn from_chunks(
134        chunks: Vec<ChunkOrLoop>,
135        observable_gateway: ObservableGateway,
136    ) -> Result<Self, CompileError> {
137        if chunks.iter().any(chunk_has_conditional_pauli) {
138            return Err(CompileError::ConditionalPauliFeedbackUnsupported);
139        }
140        Self::from_chunks_with_commuting_feedback(chunks, observable_gateway)
141    }
142
143    /// Build a template whose physical feedback is known to commute with every
144    /// persisted flow and boundary operator.
145    pub(crate) fn from_chunks_with_commuting_feedback(
146        chunks: Vec<ChunkOrLoop>,
147        mut observable_gateway: ObservableGateway,
148    ) -> Result<Self, CompileError> {
149        let mut chunks = normalize_chunks(chunks);
150        let ProgramTemplateBuild {
151            template: program_template,
152            measurement_remaps,
153        } = build_program_template(&mut chunks)?;
154
155        // Canonicalize the gateway from chunk-local ids to template ids so
156        // observable lowering never has to translate. The flow remap already
157        // maps `(chunk, local) -> template id`, so reuse it rather than build a
158        // second lookup table.
159        let resolve = |chunk_index: usize, local: u32| {
160            measurement_remaps
161                .get(chunk_index)
162                .and_then(|remap| remap.get(local as usize).copied())
163                .filter(|&id| id != MISSING_TEMPLATE_MEASUREMENT)
164                .expect("gateway measurement was moved into the template")
165        };
166        observable_gateway.resolve_to_template_ids(resolve);
167
168        Ok(Self {
169            program_template: Arc::new(program_template),
170            observable_gateway,
171        })
172    }
173}

175fn chunk_has_conditional_pauli(chunk: &ChunkOrLoop) -> bool {
176    match chunk {
177        ChunkOrLoop::Single(chunk) => circuit_has_conditional_pauli(&chunk.circuit),
178        ChunkOrLoop::Loop { body, .. } => body
179            .iter()
180            .any(|chunk| circuit_has_conditional_pauli(&chunk.circuit)),
181    }
182}
183
184fn circuit_has_conditional_pauli(circuit: &CoordCircuit) -> bool {
185    (0..circuit.body_count()).any(|body| {
186        circuit.body(BodyId(body as u32)).is_some_and(|body| {
187            body.ops()
188                .iter()
189                .any(|op| matches!(op, Op::ConditionalPauli(_)))
190        })
191    })
192}

201#[hotpath::measure]
202fn build_program_template(
203    chunks: &mut [ChunkOrLoop],
204) -> Result<ProgramTemplateBuild, CompileError> {
205    let circuit = CoordCircuit::new();
206    let entry = circuit.entry_body();
207    let mut entry_ops = Vec::new();
208    let mut state = ProgramTemplateState {
209        circuit,
210        detectors: Vec::new(),
211        repeat_states: Vec::new(),
212        restarts: Vec::new(),
213        measurement_remaps: Vec::new(),
214        engine: FlowEngine::new(),
215        next_loop_state: 0,
216    };
217
218    for chunk_or_loop in chunks {
219        match chunk_or_loop {
220            ChunkOrLoop::Single(chunk) => {
221                append_template_chunk_ops(
222                    &mut state,
223                    &mut chunk.circuit,
224                    &chunk.flows,
225                    TemplateDetectorScope::TopLevel,
226                    &mut entry_ops,
227                );
228            }
229            ChunkOrLoop::Loop { body, repetitions } => {
230                if *repetitions == 0 || body.is_empty() {
231                    continue;
232                }
233
234                append_separator_tick(&mut entry_ops);
235                let body_id = state.circuit.add_body(CircuitBody::new());
236                let mut body_ops = Vec::new();
237                // Split circuit ownership from the flow borrows retained by the engine.
238                let mut parts = body
239                    .iter_mut()
240                    .map(|chunk| (&mut chunk.circuit, chunk.flows.as_slice()));
241                let first = parts.next().expect("non-empty loop body");
242                let pending_loop_states = state
243                    .engine
244                    .start_loop_detector_states(
245                        &[OffsetFlows::new(first.1, glam::IVec2::ZERO)],
246                        &mut state.next_loop_state,
247                    )
248                    .expect("compiler-generated loops fit in the loop detector state id space");
249                for (circuit, flows) in std::iter::once(first).chain(parts) {
250                    append_template_chunk_ops(
251                        &mut state,
252                        circuit,
253                        flows,
254                        TemplateDetectorScope::RepeatBody { body: body_id },
255                        &mut body_ops,
256                    );
257                }
258                state.repeat_states.extend(
259                    state
260                        .engine
261                        .finish_loop_detector_states(pending_loop_states)
262                        .into_iter()
263                        .map(|state| TemplateRepeatState {
264                            body: body_id,
265                            state: state.state,
266                            initial: state.initial,
267                            next: state.next,
268                        }),
269                );
270                body_ops.push(Op::Tick);
271                *state
272                    .circuit
273                    .body_mut(body_id)
274                    .expect("repeat body was allocated above")
275                    .ops_mut() = body_ops;
276                entry_ops.push(Op::Repeat {
277                    body: body_id,
278                    repetitions: *repetitions,
279                });
280            }
281        }
282    }
283
284    *state
285        .circuit
286        .body_mut(entry)
287        .expect("entry body is created with the circuit")
288        .ops_mut() = entry_ops;
289    // The same engine that dropped the internal detectors still holds the
290    // boundary residual, so reading it off here needs no second composition.
291    let boundary_flows = drain_boundary_flows(state.engine)?;
292    // Cached templates never append detectors again. Do not retain the flow
293    // builder's geometric growth capacity for every signature.
294    state.detectors.shrink_to_fit();
295    Ok(ProgramTemplateBuild {
296        template: BloqTemplate::with_parts(
297            state.circuit,
298            state.detectors,
299            state.repeat_states,
300            boundary_flows,
301            state.restarts,
302        ),
303        measurement_remaps: state.measurement_remaps,
304    })
305}

317#[hotpath::measure]
318fn append_template_chunk_ops<'c>(
319    state: &mut ProgramTemplateState<'c>,
320    circuit: &mut CoordCircuit,
321    flows: &'c [Flow],
322    detector_scope: TemplateDetectorScope,
323    destination: &mut Vec<Op>,
324) {
325    let entry = circuit.entry_body();
326    let entry_destination = match detector_scope {
327        TemplateDetectorScope::TopLevel => state.circuit.entry_body(),
328        TemplateDetectorScope::RepeatBody { body } => body,
329    };
330    let (ops, measurement_map) = {
331        let mut builder = TemplateCircuitBuilder {
332            source: circuit,
333            entry_destination,
334            destination: &mut state.circuit,
335            measurement_map: Vec::new(),
336            body_map: crate::FxMap::default(),
337        };
338        let ops = builder.take_body_ops(entry);
339        (ops, builder.measurement_map)
340    };
341    append_separator_tick(destination);
342    destination.extend(ops);
343    append_template_chunk_flows(state, flows, detector_scope, &measurement_map);
344    state.measurement_remaps.push(measurement_map);
345}

347struct TemplateCircuitBuilder<'a, 'b> {
348    source: &'a mut CoordCircuit,
349    entry_destination: BodyId,
350    destination: &'b mut CoordCircuit,
351    measurement_map: Vec<u32>,
352    body_map: crate::FxMap<BodyId, BodyId>,
353}
354
355impl TemplateCircuitBuilder<'_, '_> {
356    fn take_body_ops(&mut self, body: BodyId) -> Vec<Op> {
357        let source_body = self
358            .source
359            .body_mut(body)
360            .expect("chunk operation references a body in its source circuit");
361        std::mem::take(source_body.ops_mut())
362            .into_iter()
363            .map(|op| self.remap_op(op))
364            .collect()
365    }
366
367    fn remap_op(&mut self, mut op: Op) -> Op {
368        match &mut op {
369            Op::Measure {
370                qubits,
371                measurements,
372                ..
373            } => {
374                for (measurement, &qubit) in measurements.iter_mut().zip(qubits.iter()) {
375                    *measurement = self.remap_measurement(*measurement, qubit);
376                }
377            }
378            Op::MPP {
379                products,
380                measurements,
381            } => {
382                for (measurement, product) in measurements.iter_mut().zip(products.iter()) {
383                    let qubit = product
384                        .representative_coord()
385                        .expect("MPP product is non-empty");
386                    *measurement = self.remap_measurement(*measurement, qubit);
387                }
388            }
389            Op::Repeat { body, .. } => *body = self.remap_body(*body),
390            Op::ConditionalPauli(corrections) => {
391                for correction in corrections {
392                    correction.control = self
393                        .template_measurement(correction.control)
394                        .expect("ConditionalPauli control precedes its measurement");
395                }
396            }
397            Op::Gate { .. }
398            | Op::Tick
399            | Op::Depolarize1 { .. }
400            | Op::Depolarize2 { .. }
401            | Op::PauliError { .. } => {}
402        }
403        op
404    }
405
406    fn remap_measurement(&mut self, source_measurement: u32, qubit: glam::IVec2) -> u32 {
407        if let Some(measurement) = self.template_measurement(source_measurement) {
408            return measurement;
409        }
410        let measurement = self.destination.reserve_measurement_id(qubit);
411        let local = source_measurement as usize;
412        if self.measurement_map.len() <= local {
413            self.measurement_map
414                .resize(local + 1, MISSING_TEMPLATE_MEASUREMENT);
415        }
416        self.measurement_map[local] = measurement;
417        measurement
418    }
419
420    fn template_measurement(&self, source_measurement: u32) -> Option<u32> {
421        self.measurement_map
422            .get(source_measurement as usize)
423            .copied()
424            .filter(|&measurement| measurement != MISSING_TEMPLATE_MEASUREMENT)
425    }
426
427    fn remap_body(&mut self, body: BodyId) -> BodyId {
428        // Preserve a malformed back-edge instead of turning a drained entry into
429        // an empty body; ordinary circuit validation must still reject the cycle.
430        if body == self.source.entry_body() {
431            return self.entry_destination;
432        }
433        if let Some(&body) = self.body_map.get(&body) {
434            return body;
435        }
436        let mapped = self.destination.add_body(CircuitBody::new());
437        self.body_map.insert(body, mapped);
438        let ops = self.take_body_ops(body);
439        *self
440            .destination
441            .body_mut(mapped)
442            .expect("template body was allocated above")
443            .ops_mut() = ops;
444        mapped
445    }
446}

448#[hotpath::measure]
449fn append_template_chunk_flows<'c>(
450    state: &mut ProgramTemplateState<'c>,
451    flows: &'c [Flow],
452    scope: TemplateDetectorScope,
453    measurement_map: &[u32],
454) {
455    // Residual mode, not Skip: a flow entering through the template boundary is
456    // kept as an open chain instead of dropped, so the same pass that closes the
457    // internal detectors below also accumulates the boundary residual
458    // `drain_boundary_flows` reads off at the end (§6.1.5).
459    state
460        .engine
461        .append_residual_group(
462            &[OffsetFlows::new(flows, glam::IVec2::ZERO)],
463            |_, &measurement| {
464                measurement_map
465                    .get(measurement as usize)
466                    .copied()
467                    .filter(|&measurement| measurement != MISSING_TEMPLATE_MEASUREMENT)
468                    .map(DetectorTerm::Measurement)
469                    .expect("compiler-authored template flow measurement was moved")
470            },
471        )
472        .expect("compiler-authored template chunks move every measurement their flows reference");
473    let completed = state.engine.drain_completed();
474    state.detectors.reserve(completed.len());
475    for detector in completed {
476        // A discarding chain participates in matching but emits no detector; see
477        // `FlowMarker::Discard` (a selective block's opposite-basis stabilizers).
478        if detector.marker == FlowMarker::Discard
479            || (detector.measurements.is_empty() && !detector.sign)
480        {
481            continue;
482        }
483        // A restarting chain is a RUS post-selection syndrome (U17b): it lands in
484        // the template's `restarts` side table (no coords — restart parities are
485        // never decoder food) instead of `detectors`. `TemplateRestart` carries no
486        // repeat-body scope, so a restart chain completing inside a loop body has
487        // no per-iteration representation yet; no current builder produces one.
488        // Hard assert (chunks are compiler-authored): a release fall-through would
489        // silently record a per-iteration syndrome as a whole-template parity.
490        if detector.marker == FlowMarker::Restart {
491            assert_eq!(
492                scope,
493                TemplateDetectorScope::TopLevel,
494                "restart chains inside repeat bodies are unsupported"
495            );
496            state.restarts.push(TemplateRestart {
497                parity: DetectorParity::from_term_buf(detector.measurements)
498                    .with_sign(detector.sign),
499            });
500            continue;
501        }
502        state.detectors.push(TemplateDetector {
503            scope,
504            parity: DetectorParity::from_term_buf(detector.measurements).with_sign(detector.sign),
505            coords: detector
506                .center
507                .map(|coord| smallvec::smallvec![coord.x as f64, coord.y as f64]),
508        });
509    }
510}

512fn append_separator_tick(ops: &mut Vec<Op>) {
513    if matches!(ops.last(), Some(Op::Tick | Op::Repeat { .. }) | None) {
514        return;
515    }
516    ops.push(Op::Tick);
517}

519fn normalize_chunks(chunks: Vec<ChunkOrLoop>) -> Vec<ChunkOrLoop> {
520    let mut normalized = Vec::with_capacity(chunks.len());
521    for chunk_or_loop in chunks {
522        match chunk_or_loop {
523            ChunkOrLoop::Single(chunk) => normalized.push(ChunkOrLoop::Single(chunk)),
524            ChunkOrLoop::Loop { repetitions: 0, .. } => {}
525            ChunkOrLoop::Loop {
526                body,
527                repetitions: 1,
528                ..
529            } => {
530                normalized.extend(
531                    body.into_iter()
532                        .map(|chunk| ChunkOrLoop::Single(Box::new(chunk))),
533                );
534            }
535            ChunkOrLoop::Loop { body, .. } if body.is_empty() => {}
536            ChunkOrLoop::Loop { body, repetitions } => {
537                normalized.push(ChunkOrLoop::Loop { body, repetitions });
538            }
539        }
540    }
541    normalized
542}

544/// The compilation's store of [`LoweringTemplate`]s, addressed by
545/// [`LoweringTemplateId`]. Identical placements share one entry via `Arc`.
546#[derive(Debug, Clone, Default)]
547pub(crate) struct LoweringTemplatePool {
548    templates: Vec<Arc<LoweringTemplate>>,
549}
550
551impl LoweringTemplatePool {
552    /// Add `template` and return its freshly assigned [`LoweringTemplateId`].
553    pub(crate) fn insert(&mut self, template: Arc<LoweringTemplate>) -> LoweringTemplateId {
554        debug_assert!(template.program_template.circuit.body_count() > 0);
555        let id = LoweringTemplateId(self.templates.len());
556        self.templates.push(template);
557        id
558    }
559
560    /// The template for `id`, or `None` if it is out of range.
561    pub(crate) fn get(&self, id: LoweringTemplateId) -> Option<&LoweringTemplate> {
562        self.templates.get(id.0).map(Arc::as_ref)
563    }
564
565    #[cfg(test)]
566    pub(crate) fn len(&self) -> usize {
567        self.templates.len()
568    }
569}

572    type Output = LoweringTemplate;
573
574    fn index(&self, index: LoweringTemplateId) -> &Self::Output {
575        self.templates[index.0].as_ref()
576    }

714        // A loop whose carried key K is left open at the top — nothing after the
715        // loop consumes it — must surface its reopening creator as boundary

725                .with_measurements([0])
726                .with_center(ivec2(0, 0)),
727        );
728        // loop body (chunk 1): the split surface code shape — a consumer that
729        // closes the per-round detector and a separate creator that re-opens K.
730        let mut body = measured_chunk(&[ivec2(1, 0)]);
731        body.flows.push(
732            Flow::new(k.clone(), PauliMap::empty())
733                .with_measurements([0])
734                .with_center(ivec2(1, 0)),
735        );
736        body.flows.push(
737            Flow::new(PauliMap::empty(), k.clone())
738                .with_measurements([0])
739                .with_center(ivec2(1, 0)),
740        );
741
742        let template = LoweringTemplate::from_chunks(
743            vec![
744                ChunkOrLoop::Single(Box::new(init)),
745                ChunkOrLoop::Loop {
746                    body: vec![body],
747                    repetitions: 3,
748                },
749            ],
750            ObservableGateway::new(),
751        )
752        .expect("top-open loop template compiles");
753
754        // The init seed (in the loop state's `initial`) and the per-round

767    fn boundary_flows_fuse_cross_chunk_chain() {
768        // A chain threaded across a chunk seam by a moving operator: chunk 0
769        // opens K, chunk 1 transforms K -> L and leaves L open at the top.
770        // Neither closes inside the template, so both flows survive — and must
771        // fuse into one boundary flow `∅ -> L` carrying both measurements. That
772        // is exactly the case the old `Vec<Vec<Flow>>` kept as two ordered
773        // groups; storing the fused chain lets instance lowering append it as one
774        // independent flow with no per-chunk ordering to preserve.
775        let k = PauliMap::from_iter([(ivec2(0, 0), Pauli::Z)]);
776        let l = PauliMap::from_iter([(ivec2(1, 0), Pauli::Z)]);
777
778        let mut first = measured_chunk(&[ivec2(0, 0)]);
779        first.flows.push(
780            Flow::new(PauliMap::empty(), k.clone())
781                .with_measurements([0])
782                .with_center(ivec2(0, 0)),
783        );
784        let mut second = measured_chunk(&[ivec2(1, 0)]);
785        second.flows.push(
786            Flow::new(k.clone(), l.clone())
787                .with_measurements([0])
788                .with_center(ivec2(1, 0)),
789        );
790
791        let template = LoweringTemplate::from_chunks(
792            vec![
793                ChunkOrLoop::Single(Box::new(first)),
794                ChunkOrLoop::Single(Box::new(second)),
795            ],
796            ObservableGateway::new(),
797        )
798        .expect("cross-chunk fusion template compiles");
799
800        assert_eq!(
801            template.program_template.boundary_flows.len(),
802            1,
803            "the threaded chain fuses to a single boundary flow"
804        );
805        let fused = &template.program_template.boundary_flows[0];
806        assert!(
807            fused.start.is_empty(),
808            "fused chain starts at the template top"
809        );
810        assert_eq!(fused.end, l, "fused chain ends on the surviving open key");
811        // Both rounds' measurements are spliced in (template ids 0 and 1).
812        // Order is irrelevant — `DetectorParity` canonicalizes on emission — so
813        // the test sorts before comparing.
814        let mut measurements = fused.measurements.to_vec();
815        measurements.sort_unstable();
816        assert_eq!(measurements, [0, 1]);
817    }

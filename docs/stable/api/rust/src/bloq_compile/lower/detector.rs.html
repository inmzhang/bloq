18impl<'ctx> BloqLowerContext<'ctx> {
19    /// Resolve the templates each node member instantiates, with the offsets
20    /// and measurement-map sites needed to read their flows in place.
21    ///
22    /// The detector pass walks template flows directly instead of per-node
23    /// translated copies: the engine's offset-aware keys make that exact, and
24    /// it skips materializing two translated `PauliMap`s per flow per node.
25    fn detector_node_context(
26        &self,
27        plan_node: NodeIndex,
28    ) -> Result<DetectorNodeContext<'ctx>, CompileError> {
29        let node = &self.input.plan.graph()[plan_node];
30        let node_instances = self
31            .node_instances
32            .get(plan_node.index())
33            .expect("plan node has lowered template instances");
34        let mut sources = Vec::new();
35        match &self.classified[plan_node.index()] {
36            NodeLowering::Selective { .. } => {
37                unreachable!("static detector lowering receives pinned selective templates")
38            }
39            // A T block lowers to a `RepeatUntilSuccess` region; its escape
40            // instance is the block's temporal face toward the +Z neighbour
41            // (U8's source-only residual). The escape template's boundary flows
42            // also carry the six Steane-side consumers already closed by the
43            // intra-body cultivation seam (`compose_instance_seam`) — at top
44            // level they have nothing to match, so this source skips unmatched
45            // inputs instead of erroring. The end-face creators stay live and
46            // the neighbour's consumers close them into ordinary top-level
47            // `NodeDetector`s referencing the in-body escape instance.
48            NodeLowering::TRegion { pos, escape, .. } => {
49                let [instance] = &node_instances[..] else {
50                    unreachable!("T plan node carries exactly its escape instance");
51                };
52                debug_assert_eq!(instance.site_source, ChunkSiteSource::Block(*pos));
53                sources.push(MemberFlowSource {
54                    template: &self.input.template_pool[*escape],
55                    offset: self.input.layout.offset(*pos)?,
56                    instance_id: instance.instance.id,
57                    skip_unmatched_inputs: true,
58                });
59            }
60            NodeLowering::TemporalPipe {
61                pipe,
62                template,
63                origin,
64            } => {
65                assert_eq!(
66                    node_instances.len(),
67                    1,
68                    "temporal pipe node has one lowered template instance"
69                );
70                let instance = &node_instances[0];
71                debug_assert_eq!(instance.site_source, ChunkSiteSource::TemporalPipe(*pipe));
72                sources.push(MemberFlowSource {
73                    template: &self.input.template_pool[*template],
74                    offset: self.input.layout.offset(*origin)?,
75                    instance_id: instance.instance.id,
76                    skip_unmatched_inputs: false,
77                });
78            }
79            NodeLowering::SpatialPort { source, template } => {
80                let [instance] = &node_instances[..] else {
81                    unreachable!("derived temporal Port has one instance");
82                };
83                debug_assert_eq!(instance.site_source, ChunkSiteSource::SpatialPort(*source));
84                sources.push(MemberFlowSource {
85                    template: &self.input.template_pool[*template],
86                    offset: self.input.layout.offset(*source)?,
87                    instance_id: instance.instance.id,
88                    skip_unmatched_inputs: false,
89                });
90            }
91            NodeLowering::Fixed { members, walls } => {
92                assert_eq!(
93                    node_instances.len(),
94                    members.len() + walls.len(),
95                    "block node has one lowered template instance per member and wall"
96                );
97                let (member_instances, wall_instances) = node_instances.split_at(members.len());
98                for (&(member, template), instance) in members.iter().zip(member_instances) {
99                    debug_assert_eq!(instance.site_source, ChunkSiteSource::Block(member.pos));
100                    sources.push(MemberFlowSource {
101                        template: &self.input.template_pool[template],
102                        offset: self.input.layout.offset(member.pos)?,
103                        instance_id: instance.instance.id,
104                        skip_unmatched_inputs: false,
105                    });
106                }
107                for (wall, instance) in walls.iter().zip(wall_instances) {
108                    debug_assert_eq!(
109                        instance.site_source,
110                        ChunkSiteSource::SpatialPipe(wall.pipe)
111                    );
112                    sources.push(MemberFlowSource {
113                        template: &self.input.template_pool[wall.template],
114                        offset: self.input.layout.offset(wall.pipe.src)?,
115                        instance_id: instance.instance.id,
116                        skip_unmatched_inputs: false,
117                    });
118                }
119            }
120        }
121        Ok(DetectorNodeContext {
122            plan_node: plan_node.index(),
123            layer: node.layer,
124            members: node.block_members(),
125            sources,
126        })
127    }
128}

130struct DetectorNodeContext<'a> {
131    plan_node: usize,
132    layer: i64,
133    members: &'a [SourceBlockRef],
134    sources: Vec<MemberFlowSource<'a>>,
135}

137/// One node member's compiled template, placed at the member's XY offset.
138struct MemberFlowSource<'a> {
139    template: &'a LoweringTemplate,
140    offset: IVec2,
141    instance_id: TemplateInstanceId,
142    /// Compose with `append_group_skipping_unmatched_inputs`: a consumer flow
143    /// with no matching open chain is dropped instead of erroring. Set only for
144    /// a T block's escape template at top level, whose Steane-side consumers
145    /// were already closed by the intra-body cultivation seam.
146    skip_unmatched_inputs: bool,
147}

149/// Composed boundary detectors grouped by the plan node the chains closed on.
150type ComposedDetectors = Vec<(usize, Vec<NodeDetector>)>;
151/// Composed restart syndromes grouped by the plan node the chains closed on.
152type ComposedRestarts = Vec<(usize, Vec<NodeRestart>)>;
153
154/// Pending side-table entries keyed by the plan node the chain closed on.
155struct DetectorLowerState<'c> {
156    detectors: ComposedDetectors,
157    restarts: ComposedRestarts,
158    engine: DetectorFlowEngine<'c>,
159}
160
161impl DetectorLowerState<'_> {
162    fn drain_node_chains(&mut self, plan_node: usize) {
163        let (detectors, restarts) = bloq_ir::lowering::drain_composed_chains(&mut self.engine);
164        if !detectors.is_empty() {
165            self.detectors.push((plan_node, detectors));
166        }
167        if !restarts.is_empty() {
168            self.restarts.push((plan_node, restarts));
169        }
170    }
171}

173// ==============================================================================
174// Boundary-only detector composition
175// ==============================================================================
176//
177// Detectors that close inside a single template are recovered downstream by the
178// backend (it instantiates `program_template.detectors`/`repeat_states` per
179// instance via a cheap offset + measurement relabel). So instance lowering only
180// composes the **boundary** detectors — flows the template left open that close
181// against a temporally adjacent node.
182//
183// Those open flows are precomputed per template as `boundary_flows` (see
184// `block/compile.rs::drain_boundary_flows`), already fused into mutually
185// independent chains. Here we replay only that residual through a per-component
186// engine in emission order, so an open created by a lower-z node is consumed by
187// the node above it. This pass composes nothing but `TopLevel` `NodeDetector`s
188// and allocates no loop state — a loop's recurrence is backend-owned, and its
189// sole cross-node round is resolved by the backend's post-`REPEAT` lookback.
190#[hotpath::measure]
191pub(super) fn attach_temporal_component_detectors(
192    plan: &LowerPlan,
193    context: &mut BloqLowerContext<'_>,
194) -> Result<(), CompileError> {
195    for mut component in plan.temporal_node_components() {
196        layer_major_order(&mut component, context)?;
197        attach_component_detectors(&component, context)?;
198    }
199    Ok(())
200}

202/// Layer-major, emit-order tie-broken order for one temporal component. Every
203/// plan edge ascends `layer`, so this is still a topological order — but plain
204/// emit order is not enough here: the engine keys open chains by 2D support only,
205/// and a bare toposort may interleave same-footprint branches out of z order (two
206/// T columns are DAG sources with no path between them, U17e), letting a lower
207/// column's creators clobber another column's still-open chains at the same XY
208/// offset. Layer-major ordering walks the component in time.
209#[hotpath::measure]
210fn layer_major_order(
211    nodes: &mut [NodeIndex],
212    context: &BloqLowerContext<'_>,
213) -> Result<(), CompileError> {
214    nodes.sort_by_cached_key(|node| {
215        (
216            context.input.plan.graph()[*node].layer,
217            context.emit_order[node.index()],
218        )
219    });
220
221    let mut first = 0;
222    while first < nodes.len() {
223        let layer = context.input.plan.graph()[nodes[first]].layer;
224        let end = nodes[first..]
225            .partition_point(|node| context.input.plan.graph()[*node].layer == layer)
226            + first;
227        order_same_layer_flows(&mut nodes[first..end], context, layer)?;
228        first = end;
229    }
230    Ok(())
231}

233/// A same-layer moving patch may produce a boundary still waiting to be
234/// consumed by its neighbor. Replay those consumers first; unrelated nodes
235/// retain emission order.
236#[hotpath::measure]
237fn order_same_layer_flows(
238    nodes: &mut [NodeIndex],
239    context: &BloqLowerContext<'_>,
240    layer: i64,
241) -> Result<(), CompileError> {
242    if nodes.len() < 2 {
243        return Ok(());
244    }
245    let mut consumers = crate::FxMap::<FlowKey<'_>, smallvec::SmallVec<[usize; 1]>>::default();
246    let mut produced = vec![Vec::new(); nodes.len()];
247    for (index, &node) in nodes.iter().enumerate() {
248        let node = context.detector_node_context(node)?;
249        for source in node.sources {
250            for flow in &source.template.program_template.boundary_flows {
251                if !flow.start.is_empty() {
252                    consumers
253                        .entry(FlowKey::try_new(&flow.start, source.offset)?)
254                        .or_default()
255                        .push(index);
256                }
257                if !flow.end.is_empty() {
258                    produced[index].push(FlowKey::try_new(&flow.end, source.offset)?);
259                }
260            }
261        }
262    }
263
264    let mut successors = vec![crate::FxSet::default(); nodes.len()];
265    let mut indegree = vec![0usize; nodes.len()];
266    for (producer, boundaries) in produced.iter().enumerate() {
267        for boundary in boundaries {
268            for &consumer in consumers.get(boundary).into_iter().flatten() {
269                if consumer != producer && successors[consumer].insert(producer) {
270                    indegree[producer] += 1;
271                }
272            }
273        }
274    }
275    let mut ready = indegree
276        .iter()
277        .enumerate()
278        .filter_map(|(index, &count)| (count == 0).then_some(index))
279        .collect::<std::collections::BTreeSet<_>>();
280    let mut order = Vec::with_capacity(nodes.len());
281    while let Some(index) = ready.pop_first() {
282        order.push(nodes[index]);
283        for &next in &successors[index] {
284            indegree[next] -= 1;
285            if indegree[next] == 0 {
286                ready.insert(next);
287            }
288        }
289    }
290    if order.len() != nodes.len() {
291        return Err(CompileError::NodeFlowCompositionFailed {
292            layer,
293            members: nodes
294                .iter()
295                .flat_map(|node| context.input.plan.graph()[*node].block_members())
296                .map(|member| member.pos)
297                .collect(),
298            source: bloq_circuit::FlowError::Composition(
299                "cyclic same-layer boundary-flow dependency".into(),
300            ),
301        });
302    }
303    nodes.copy_from_slice(&order);
304    Ok(())
305}

307/// Compose the already pinned proxy's boundary flows with their allocated ids.
308#[hotpath::measure]
309fn attach_component_detectors(
310    nodes: &[NodeIndex],
311    context: &mut BloqLowerContext<'_>,
312) -> Result<(), CompileError> {
313    let mut state = DetectorLowerState {
314        detectors: Vec::new(),
315        restarts: Vec::new(),
316        engine: DetectorFlowEngine::new(),
317    };
318    for &node in nodes {
319        let node_context = context.detector_node_context(node)?;
320        append_node_boundary_flows(&mut state, &node_context)?;
321    }
322    let remaining = state
323        .engine
324        .finish()
325        .map_err(|error| CompileError::DetectorFlowCompositionFailed { source: error })?;
326    debug_assert!(
327        remaining.is_empty(),
328        "completed detectors are drained into their closing node"
329    );
330    for (node, detectors) in state.detectors {
331        debug_assert!(context.node_detectors[node].is_empty());
332        context.node_detectors[node] = detectors;
333    }
334    for (node, restarts) in state.restarts {
335        debug_assert!(context.node_restarts[node].is_empty());
336        context.node_restarts[node] = restarts;
337    }
338    Ok(())
339}

341/// Append one node's boundary flows to the component engine, one group per
342/// source.
343///
344/// Each source's `boundary_flows` are already fused into mutually independent
345/// chains, so the whole list goes in as a single `TopLevel` group. Sources are
346/// appended independently: a node's members lie in one z-layer and expose
347/// boundary flows only on their own temporal faces, so two members never collide
348/// on a key. There is no loop arm — a top-open loop's residual arrives as a plain
349/// creator flow, resolved by the backend's post-`REPEAT` lookback.
350fn append_node_boundary_flows<'c>(
351    state: &mut DetectorLowerState<'c>,
352    node: &DetectorNodeContext<'c>,
353) -> Result<(), CompileError> {
354    if !node
355        .sources
356        .iter()
357        .any(|source| source.skip_unmatched_inputs)
358    {
359        let sources: Vec<_> = node
360            .sources
361            .iter()
362            .filter(|source| !source.template.program_template.boundary_flows.is_empty())
363            .collect();
364        let parts: Vec<_> = sources
365            .iter()
366            .map(|source| {
367                OffsetFlows::new(
368                    &source.template.program_template.boundary_flows,
369                    source.offset,
370                )
371            })
372            .collect();
373        state
374            .engine
375            .append_group_allowing_stabilizer_transition(&parts, |part, &measurement| {
376                DetectorTerm::Measurement(InstanceMeasurement {
377                    instance: sources[part].instance_id,
378                    measurement,
379                })
380            })
381            .map_err(|error| node_flow_error(node, error))?;
382        state.drain_node_chains(node.plan_node);
383        return Ok(());
384    }
385
386    for source in &node.sources {
387        if source.template.program_template.boundary_flows.is_empty() {
388            continue;
389        }
390        let instance = source.instance_id;
391        let parts = [OffsetFlows::new(
392            &source.template.program_template.boundary_flows,
393            source.offset,
394        )];
395        let term = |_: usize, &measurement: &u32| {
396            DetectorTerm::Measurement(InstanceMeasurement {
397                instance,
398                measurement,
399            })
400        };
401        if source.skip_unmatched_inputs {
402            state
403                .engine
404                .append_group_skipping_unmatched_inputs(&parts, term)
405        } else {
406            state.engine.append_group_allowing_dangling(&parts, term)
407        }
408        .map_err(|error| node_flow_error(node, error))?;
409    }
410    // Classification (Discard drop / Restart -> `NodeRestart` / else
411    // `NodeDetector`) is shared with the U11 edit recompose so the marker
412    // discipline cannot drift between compile time and post-hoc seam edits.
413    state.drain_node_chains(node.plan_node);
414    Ok(())
415}

417/// Compose the seam between template instances of one `RepeatUntilSuccess`
418/// body: sources in emission order — cultivation, then escape. All
419/// completed chains here are the cultivation template's restart-flagged open
420/// Steane stabilizers closed by the escape template's first merge round, so
421/// they land as `NodeRestart`s (plus any plain detectors a future body shape
422/// might close). Deliberately no `engine.finish()`: the escape template's
423/// end-face creators legitimately stay open — the +Z neighbour closes them at
424/// top level (with the T source in skip mode dropping the Steane consumers
425/// this pass already resolved).
426pub(super) fn compose_instance_seam(
427    sources: &[(&LoweringTemplate, IVec2, TemplateInstanceId)],
428) -> Result<(Vec<NodeDetector>, Vec<NodeRestart>), CompileError> {
429    let mut state = DetectorLowerState {
430        detectors: Vec::new(),
431        restarts: Vec::new(),
432        engine: DetectorFlowEngine::new(),
433    };
434    for &(template, offset, instance_id) in sources {
435        let node = DetectorNodeContext {
436            plan_node: 0,
437            layer: 0,
438            members: &[],
439            sources: vec![MemberFlowSource {
440                template,
441                offset,
442                instance_id,
443                skip_unmatched_inputs: false,
444            }],
445        };
446        append_node_boundary_flows(&mut state, &node)?;
447    }
448    Ok((
449        state
450            .detectors
451            .into_iter()
452            .flat_map(|(_, detectors)| detectors)
453            .collect(),
454        state
455            .restarts
456            .into_iter()
457            .flat_map(|(_, restarts)| restarts)
458            .collect(),
459    ))
460}

462fn node_flow_error(node: &DetectorNodeContext<'_>, error: bloq_circuit::FlowError) -> CompileError {
463    CompileError::NodeFlowCompositionFailed {
464        layer: node.layer,
465        members: node.members.iter().map(|member| member.pos).collect(),
466        source: error,
467    }

471mod tests {
472    use bloq_circuit::{
473        Chunk, ChunkOrLoop, CoordCircuit, DetectorParity, Flow, FlowMarker, Pauli, PauliBasis,
474        PauliMap,

476    use glam::ivec2;

481    /// One-measurement chunk carrying `flow`.
482    fn chunk_with_flow(coord: glam::IVec2, flow: Flow) -> ChunkOrLoop {
483        let mut circuit = CoordCircuit::new();
484        circuit.measure(PauliBasis::Z, [coord]);
485        ChunkOrLoop::Single(Box::new(Chunk {
486            circuit,
487            flows: vec![flow],
488        }))
489    }
490
491    fn template_with_flow(coord: glam::IVec2, flow: Flow) -> LoweringTemplate {
492        LoweringTemplate::from_chunks(vec![chunk_with_flow(coord, flow)], ObservableGateway::new())
493            .expect("single-flow template compiles")
494    }
495
496    /// The cross-template restart composition: a restart-flagged chain left
497    /// open by one template instance and closed by the next fuses at the seam
498    /// into a `NodeRestart` carrying both instances' measurements — never a
499    /// `NodeDetector`.
500    #[test]
501    fn cross_instance_restart_chain_becomes_node_restart() {
502        let seam = PauliMap::from_iter([(ivec2(0, 0), Pauli::Z)]);
503        let lower_template = template_with_flow(
504            ivec2(0, 0),
505            Flow::new(PauliMap::empty(), seam.clone())
506                .with_measurements([0])
507                .with_marker(FlowMarker::Restart),
508        );
509        let upper_template = template_with_flow(
510            ivec2(0, 0),
511            Flow::new(seam, PauliMap::empty()).with_measurements([0]),
512        );
513
514        let node_context = |template, plan_node, instance| DetectorNodeContext {
515            plan_node,
516            layer: 0,
517            members: &[],
518            sources: vec![MemberFlowSource {
519                template,
520                offset: IVec2::ZERO,
521                instance_id: TemplateInstanceId(instance),
522                skip_unmatched_inputs: false,
523            }],
524        };
525        let mut state = DetectorLowerState {
526            detectors: Vec::new(),
527            restarts: Vec::new(),
528            engine: DetectorFlowEngine::new(),
529        };
530        append_node_boundary_flows(&mut state, &node_context(&lower_template, 0, 0))
531            .expect("lower node appends");
532        append_node_boundary_flows(&mut state, &node_context(&upper_template, 1, 1))
533            .expect("upper node closes the seam");
534        assert!(
535            state.engine.finish().expect("no open flows").is_empty(),
536            "the fused chain drains at the closing node"
537        );
538
539        assert!(
540            state.detectors.is_empty(),
541            "no detector for a restart chain"
542        );
543        assert_eq!(state.restarts.len(), 1);
544        // The chain completes at the upper node, so the restart attaches there.
545        let (plan_node, restarts) = &state.restarts[0];
546        assert_eq!(*plan_node, 1);
547        assert_eq!(restarts.len(), 1);
548        assert_eq!(
549            restarts[0].parity,
550            DetectorParity::from_measurements([
551                InstanceMeasurement {
552                    instance: TemplateInstanceId(0),
553                    measurement: 0,
554                },
555                InstanceMeasurement {
556                    instance: TemplateInstanceId(1),
557                    measurement: 0,
558                },
559            ])
560        );
561    }

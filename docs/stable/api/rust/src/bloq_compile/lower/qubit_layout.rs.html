13/// Validate same-layer physical-qubit ownership in `bloq`.
14///
15/// Two nodes may overlap where source adjacency permits it or their predicates
16/// are mutually exclusive. Validation runs at every graph level, so nodes inside
17/// RUS retry bodies are covered. Each
18/// level is checked in isolation: separate `SubGraph`s never cross-check, which
19/// is exactly the intended semantics — a region reports its whole footprint as
20/// one node at the parent level, and each body has its own execution scope.
21/// Hierarchical source graphs are explicitly projected to their flat topology
22/// for this audit; native compilation retains its definition ownership.
23///
24/// # Errors
25///
26/// Returns [`CompileError::OverlappingNodeQubit`] for the first disallowed overlap.
27pub fn validate_bloq_qubit_layout_for_source(
28    bloq: &Bloq,
29    source: &BlockGraph,
30) -> Result<(), CompileError> {
31    let flat = source
32        .has_module_structure()
33        .then(|| source.flatten())
34        .transpose()?;
35    validate_bloq_qubit_layout_with_limits(
36        bloq,
37        flat.as_ref().unwrap_or(source),
38        bloq_utils::boolean::BooleanLimits::DEFAULT,
39    )
40}

42pub(crate) fn validate_bloq_qubit_layout_with_limits(
43    bloq: &Bloq,
44    source: &BlockGraph,
45    limits: bloq_utils::boolean::BooleanLimits,
46) -> Result<(), CompileError> {
47    let mut template_runs = crate::FxMap::default();
48    for (path, level) in bloq.levels() {
49        // Inside a region body, same-layer nodes may legitimately share a
50        // physical footprint when they are temporally *sequential* rather than
51        // concurrent — a T block's cultivation and escape stages sit at the same
52        // layer and qubits, joined by a temporal seam edge. So bodies skip pairs
53        // with a directed path between them and flag only genuinely concurrent
54        // (no path either way) same-layer overlaps. Top-level Region nodes use
55        // the same ordered sharing with adjacent lowering nodes.
56        let skip_sequential = !path.segments().is_empty();
57        validate_level_qubit_layout(
58            bloq,
59            source,
60            level,
61            skip_sequential,
62            limits,
63            &mut template_runs,
64        )?;
65    }
66    Ok(())
67}

69/// Runs the same-layer overlap check within a single graph level. When
70/// `skip_sequential` is set (region bodies), pairs joined by a directed path are
71/// exempt because they occupy the shared footprint at different times.
72#[hotpath::measure]
73fn validate_level_qubit_layout(
74    bloq: &Bloq,
75    source: &BlockGraph,
76    level: &SubGraph,
77    skip_sequential: bool,
78    limits: bloq_utils::boolean::BooleanLimits,
79    template_runs: &mut crate::FxMap<TemplateId, Vec<QubitRun>>,
80) -> Result<(), CompileError> {
81    let mut layers = BTreeMap::<_, Vec<_>>::new();
82    let mut guarded = false;
83    for (id, node) in level.nodes() {
84        if node.try_quantum().is_none() && node.try_region().is_none() {
85            continue;
86        }
87        guarded |= node.activation.is_some()
88            || node
89                .try_quantum()
90                .is_some_and(|quantum| !quantum.guards.is_empty());
91        layers.entry(node.layer()).or_default().push((id, node));
92    }
93    if guarded && !level.is_acyclic() {
94        return Err(BloqValidationError::CyclicGraph.into());
95    }
96    let mut predicates = PredicateAnalysis::with_limits(level, limits);
97    let mut allowed_pairs = crate::FxMap::default();
98    let mut scratch = level.path_scratch();
99    for nodes in layers.values() {
100        // One interval index per row and x parity. A stride-2 run stays one
101        // interval regardless of code distance.
102        let mut occupied = crate::FxMap::<(i32, i32), BTreeMap<i64, OccupiedRun>>::default();
103        for &(right_id, right) in nodes {
104            let footprints = guarded_footprints(bloq, level, right_id, right, template_runs)?;
105            let mut conflict = None;
106            for footprint in &footprints {
107                for run in &footprint.runs {
108                    let Some(row) = occupied.get(&(run.y, run.parity)) else {
109                        continue;
110                    };
111                    let candidates = row
112                        .range(..=run.start)
113                        .next_back()
114                        .into_iter()
115                        .chain(row.range((Excluded(run.start), Excluded(run.end))));
116                    for (&start, interval) in candidates {
117                        let x_index = run.start.max(start);
118                        if x_index >= interval.end {
119                            continue;
120                        }
121                        let x = (2 * x_index + i64::from(run.parity)) as i32;
122                        for &(left_id, left_guard) in &interval.users {
123                            let allowed =
124                                *allowed_pairs.entry((left_id, right_id)).or_insert_with(|| {
125                                    let left = &level[left_id];
126                                    let region_pair =
127                                        left.try_region().is_some() || right.try_region().is_some();
128                                    allows_same_layer_qubit_overlap(source, left, right)
129                                        || ((skip_sequential || region_pair)
130                                            && (level.has_path_with_scratch(
131                                                left_id,
132                                                right_id,
133                                                &mut scratch,
134                                            ) || level.has_path_with_scratch(
135                                                right_id,
136                                                left_id,
137                                                &mut scratch,
138                                            )))
139                                });
140                            if allowed
141                                || !predicates
142                                    .overlap_values(left_guard, footprint.guard)
143                                    .map_err(|source| match source {
144                                        NodeTemplateInstanceMergeError::BooleanResource(
145                                            resource,
146                                        ) => CompileError::from(resource),
147                                        source => {
148                                            BloqValidationError::InvalidInstanceMergeStructure {
149                                                node: right_id,
150                                                source,
151                                            }
152                                            .into()
153                                        }
154                                    })?
155                            {
156                                continue;
157                            }
158                            let key = (left_id.0, x, run.y);
159                            if conflict.is_none_or(|previous| key < previous) {
160                                conflict = Some(key);
161                            }
162                        }
163                    }
164                }
165            }
166            if let Some((left, x, y)) = conflict {
167                return Err(CompileError::OverlappingNodeQubit {
168                    first: BloqNodeId(left),
169                    second: right_id,
170                    qubit: IVec2::new(x, y),
171                });
172            }
173            for footprint in footprints {
174                for run in footprint.runs {
175                    insert_occupied(
176                        occupied.entry((run.y, run.parity)).or_default(),
177                        run,
178                        (right_id, footprint.guard),
179                    );
180                }
181            }
182        }
183    }
184    Ok(())
185}

200impl QubitRun {
201    pub(super) fn translated(
202        self,
203        offset: IVec2,
204    ) -> Result<Self, bloq_circuit::CoordinateOverflowError> {
205        let first = IVec2::new((2 * self.start + i64::from(self.parity)) as i32, self.y);
206        let last = IVec2::new((2 * (self.end - 1) + i64::from(self.parity)) as i32, self.y);
207        let first = bloq_circuit::checked_translate_coordinate(first, offset)?;
208        let last = bloq_circuit::checked_translate_coordinate(last, offset)?;
209        Ok(Self {
210            y: first.y,
211            parity: first.x.rem_euclid(2),
212            start: i64::from(first.x.div_euclid(2)),
213            end: i64::from(last.x.div_euclid(2)) + 1,
214        })
215    }
216}

223fn merge_runs(mut runs: Vec<QubitRun>) -> Vec<QubitRun> {
224    runs.sort_unstable_by_key(|run| (run.y, run.parity, run.start, run.end));
225    let mut merged: Vec<QubitRun> = Vec::with_capacity(runs.len());
226    for run in runs {
227        if let Some(last) = merged.last_mut()
228            && last.y == run.y
229            && last.parity == run.parity
230            && run.start <= last.end
231        {
232            last.end = last.end.max(run.end);
233        } else {
234            merged.push(run);
235        }
236    }
237    merged
238}

240pub(super) fn template_qubit_runs(qubits: &[IVec2]) -> Vec<QubitRun> {
241    merge_runs(
242        qubits
243            .iter()
244            .map(|qubit| QubitRun {
245                y: qubit.y,
246                parity: qubit.x.rem_euclid(2),
247                start: i64::from(qubit.x.div_euclid(2)),
248                end: i64::from(qubit.x.div_euclid(2)) + 1,
249            })
250            .collect(),
251    )
252}

254fn append_instance_runs(
255    bloq: &Bloq,
256    node: &BloqNode,
257    instance: &TemplateInstance,
258    template_runs: &mut crate::FxMap<TemplateId, Vec<QubitRun>>,
259    runs: &mut Vec<QubitRun>,
260) -> Result<(), CompileError> {
261    let Some(template) = bloq.templates().get(instance.template_id) else {
262        return Ok(());
263    };
264    let local = template_runs
265        .entry(instance.template_id)
266        .or_insert_with(|| template_qubit_runs(template.qubits()));
267    for run in local {
268        let run = match run.translated(instance.offset) {
269            Ok(run) => run,
270            Err(_) => {
271                // Template qubits are stored in arbitrary order. Re-scan that
272                // order to keep the old first-overflow diagnostic exactly.
273                return Err(bloq
274                    .node_qubits(node)
275                    .expect_err("a run endpoint overflowed")
276                    .into());
277            }
278        };
279        runs.push(run);
280    }
281    Ok(())
282}

284fn append_node_runs(
285    bloq: &Bloq,
286    node: &BloqNode,
287    template_runs: &mut crate::FxMap<TemplateId, Vec<QubitRun>>,
288    runs: &mut Vec<QubitRun>,
289) -> Result<(), CompileError> {
290    if let Some(region) = node.try_region() {
291        for (_, body) in region.bodies() {
292            for (_, body_node) in body.nodes() {
293                append_node_runs(bloq, body_node, template_runs, runs)?;
294            }
295        }
296    } else if let Some(quantum) = node.try_quantum() {
297        let mut seen = crate::FxSet::default();
298        for instance in &quantum.instances {
299            if seen.insert((instance.template_id, instance.offset)) {
300                append_instance_runs(bloq, node, instance, template_runs, runs)?;
301            }
302        }
303    }
304    Ok(())
305}

307fn split_occupied_at(row: &mut BTreeMap<i64, OccupiedRun>, x: i64) {
308    if let Some((&start, interval)) = row.range(..x).next_back()
309        && x < interval.end
310    {
311        let end = interval.end;
312        let users = interval.users.clone();
313        row.get_mut(&start).expect("occupied interval exists").end = x;
314        row.insert(x, OccupiedRun { end, users });
315    }
316}
317
318fn insert_occupied(
319    row: &mut BTreeMap<i64, OccupiedRun>,
320    run: QubitRun,
321    user: (BloqNodeId, Option<ValueRef>),
322) {
323    split_occupied_at(row, run.start);
324    split_occupied_at(row, run.end);
325    let mut gaps = Vec::new();
326    let mut cursor = run.start;
327    for (&start, interval) in row.range_mut(run.start..run.end) {
328        if cursor < start {
329            gaps.push((cursor, start));
330        }
331        interval.users.push(user);
332        cursor = interval.end;
333    }
334    if cursor < run.end {
335        gaps.push((cursor, run.end));
336    }
337    for (start, end) in gaps {
338        row.insert(
339            start,
340            OccupiedRun {
341                end,
342                users: vec![user],
343            },
344        );
345    }
346}

348fn guarded_footprints(
349    bloq: &Bloq,
350    level: &SubGraph,
351    id: BloqNodeId,
352    node: &BloqNode,
353    template_runs: &mut crate::FxMap<TemplateId, Vec<QubitRun>>,
354) -> Result<Vec<GuardedFootprint>, CompileError> {
355    let Some(quantum) = node
356        .try_quantum()
357        .filter(|quantum| !quantum.guards.is_empty())
358    else {
359        let guard = node
360            .activation
361            .map(|slot| {
362                level
363                    .value_inputs(id)
364                    .find(|input| input.slot == slot)
365                    .map(|input| input.value_ref().expect("value input"))
366                    .ok_or(BloqValidationError::MissingClassicalInput { node: id, slot })
367            })
368            .transpose()?;
369        let mut runs = Vec::new();
370        append_node_runs(bloq, node, template_runs, &mut runs)?;
371        return Ok(vec![GuardedFootprint {
372            guard,
373            runs: merge_runs(runs),
374        }]);
375    };
376    // Keep the first producer for each slot, as the previous per-guard search did.
377    let mut producers = crate::FxMap::default();
378    for input in level.value_inputs(id) {
379        producers
380            .entry(input.slot)
381            .or_insert(input.value_ref().expect("value input"));
382    }
383    let producer = |slot| {
384        producers
385            .get(&slot)
386            .copied()
387            .ok_or(BloqValidationError::MissingClassicalInput { node: id, slot })
388    };
389    let instance_ids = quantum
390        .instances
391        .iter()
392        .map(|instance| instance.id)
393        .collect::<crate::FxSet<_>>();
394    let mut guards = crate::FxMap::default();
395    for guard in &quantum.guards {
396        let predicate = producer(guard.input)?;
397        for &instance in &guard.instances {
398            if !instance_ids.contains(&instance) || guards.insert(instance, predicate).is_some() {
399                return Err(BloqValidationError::InvalidInstanceMergeStructure {
400                    node: id,
401                    source: NodeTemplateInstanceMergeError::InvalidMembership(format!(
402                        "instance i{} is missing or registered more than once",
403                        instance.0,
404                    )),
405                }
406                .into());
407            }
408        }
409    }
410    let mut groups = crate::FxMap::<Option<ValueRef>, Vec<QubitRun>>::default();
411    let mut seen = crate::FxSet::default();
412    for instance in &quantum.instances {
413        let guard = guards.get(&instance.id).copied();
414        let runs = groups.entry(guard).or_default();
415        if seen.insert((guard, instance.template_id, instance.offset)) {
416            append_instance_runs(bloq, node, instance, template_runs, runs)?;
417        }
418    }
419    Ok(groups
420        .into_iter()
421        .map(|(guard, runs)| GuardedFootprint {
422            guard,
423            runs: merge_runs(runs),
424        })
425        .collect())
426}

428fn allows_same_layer_qubit_overlap(source: &BlockGraph, left: &BloqNode, right: &BloqNode) -> bool {
429    let NodeProvenance::BlockComponent {
430        members: left_members,
431    } = &left.provenance
432    else {
433        return false;
434    };
435    let NodeProvenance::BlockComponent {
436        members: right_members,
437    } = &right.provenance
438    else {
439        return false;
440    };
441    if left_members.is_empty() || right_members.is_empty() {
442        return false;
443    }
444
445    left_members.iter().all(|left| {
446        right_members
447            .iter()
448            .all(|right| walking_soft_corridor_overlap_allowed(source, left.pos, right.pos))
449    })
450}
451
452fn walking_soft_corridor_overlap_allowed(source: &BlockGraph, left: IVec3, right: IVec3) -> bool {
453    if left == right {
454        return false;
455    }
456    let Some(left_block) = source.get_block(left) else {
457        return false;
458    };
459    let Some(right_block) = source.get_block(right) else {
460        return false;
461    };
462    let left_kind = left_block.kind();
463    let right_kind = right_block.kind();
464    let right_offsets = right_kind.reserved_offsets();
465    let mut has_allowed_overlap = false;
466
467    for left_offset in left_kind.reserved_offsets() {
468        let overlap_pos = left + left_offset;
469        if right_offsets
470            .iter()
471            .any(|&right_offset| right + right_offset == overlap_pos)
472        {
473            if !left_kind.allows_reserved_overlap(left, right_kind, right, overlap_pos) {
474                return false;
475            }
476            has_allowed_overlap = true;
477        }
478    }
479
480    has_allowed_overlap
481}

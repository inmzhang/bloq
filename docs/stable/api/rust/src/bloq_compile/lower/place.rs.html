2use glam::IVec3;
3use petgraph::graph::NodeIndex;

5use crate::{BlockLayout, CompileError};

7use super::classify::NodeLowering;
8use super::{ChunkSiteSource, LoweredTemplateInstance, PlacedInstance, TemplateInstanceAllocator};

10pub(super) struct PlacedInstances {
11    /// Lowered instances per plan node, indexed by `NodeIndex::index()`.
12    pub(super) node_instances: Vec<Vec<LoweredTemplateInstance>>,
13    /// T block position → its cultivation instance id. The cultivation instance
14    /// lives inside the RUS body (built at node materialization), but its id is
15    /// allocated here — before the escape's — so instance ids follow emission
16    /// order, and the intra-body seam pass can label its flows.
17    pub(super) t_cultivation_instances: crate::FxMap<IVec3, TemplateInstanceId>,
18}
19
20/// Allocate and place one `TemplateInstance` per node member (or one per
21/// temporal-pipe node) in emission order. **No circuit is materialized.**
22pub(super) fn place_instances(
23    classified: &[NodeLowering],
24    lower_order: &[NodeIndex],
25    layout: BlockLayout,
26    instance_allocator: &mut TemplateInstanceAllocator,
27) -> Result<PlacedInstances, CompileError> {
28    let mut node_instances = vec![Vec::new(); classified.len()];
29    let mut t_cultivation_instances = crate::FxMap::default();
30    for &plan_index in lower_order {
31        let instances = &mut node_instances[plan_index.index()];
32        match &classified[plan_index.index()] {
33            // Selective arms are allocated together at node materialization.
34            NodeLowering::Selective { .. } => {}
35            // A T block lowers to a `RepeatUntilSuccess` region. Both stage
36            // instances live inside its body, but the escape instance doubles as
37            // the block's temporal face (U8's source-only residual): registering
38            // it in `node_instances` lets the top-level detector pass close the
39            // +Z neighbour's seam against it and the observable pass resolve the
40            // site.
41            NodeLowering::TRegion { pos, escape, .. } => {
42                t_cultivation_instances.insert(*pos, instance_allocator.allocate());
43                instances.push(LoweredTemplateInstance {
44                    site_source: ChunkSiteSource::Block(*pos),
45                    instance: PlacedInstance {
46                        id: instance_allocator.allocate(),
47                        template: *escape,
48                        offset: layout.offset(*pos)?,
49                    },
50                });
51            }
52            NodeLowering::TemporalPipe {
53                pipe,
54                template,
55                origin,
56            } => {
57                instances.push(LoweredTemplateInstance {
58                    site_source: ChunkSiteSource::TemporalPipe(*pipe),
59                    instance: PlacedInstance {
60                        id: instance_allocator.allocate(),
61                        template: *template,
62                        offset: layout.offset(*origin)?,
63                    },
64                });
65            }
66            NodeLowering::SpatialPort { source, template } => {
67                instances.push(LoweredTemplateInstance {
68                    site_source: ChunkSiteSource::SpatialPort(*source),
69                    instance: PlacedInstance {
70                        id: instance_allocator.allocate(),
71                        template: *template,
72                        offset: layout.offset(*source)?,
73                    },
74                });
75            }
76            NodeLowering::Fixed { members, walls } => {
77                for &(member, template) in members {
78                    instances.push(LoweredTemplateInstance {
79                        site_source: ChunkSiteSource::Block(member.pos),
80                        instance: PlacedInstance {
81                            id: instance_allocator.allocate(),
82                            template,
83                            offset: layout.offset(member.pos)?,
84                        },
85                    });
86                }
87                // Walls come after the members so a shared seam qubit's physical
88                // reset/readout is owned by its cube; the wall's duplicate
89                // aliases onto it when the node's instances merge.
90                for wall in walls {
91                    instances.push(LoweredTemplateInstance {
92                        site_source: ChunkSiteSource::SpatialPipe(wall.pipe),
93                        instance: PlacedInstance {
94                            id: instance_allocator.allocate(),
95                            template: wall.template,
96                            offset: layout.offset(wall.pipe.src)?,
97                        },
98                    });
99                }
100            }
101        }
102    }
103    Ok(PlacedInstances {
104        node_instances,
105        t_cultivation_instances,
106    })
107}

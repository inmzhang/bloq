9fn render_table() -> String {
10    let entries = GalleryItem::iter().collect::<Vec<_>>();
11    let id_width = entries
12        .iter()
13        .map(|entry| entry.id().len())
14        .chain(std::iter::once("Id".len()))
15        .max()
16        .unwrap_or(2);
17    let category_width = entries
18        .iter()
19        .map(|entry| format_entry_categories(*entry))
20        .map(|categories| categories.len())
21        .chain(std::iter::once("Categories".len()))
22        .max()
23        .unwrap_or(10);
24    let description_width = entries
25        .iter()
26        .map(|entry| entry.description().len())
27        .chain(std::iter::once("Description".len()))
28        .max()
29        .unwrap_or(11);
30
31    let mut out = String::new();
32    out.push_str(&terminal::heading("Gallery Catalog"));
33    out.push('\n');
34    out.push_str(&terminal::table_muted(
35        "Built-in block graphs available from `bloq --gallery <id>`.",
36    ));
37    out.push_str("\n\n");
38
39    out.push_str(&table_rule(
40        '┌',
41        '┬',
42        '┐',
43        id_width,
44        category_width,
45        description_width,
46    ));
47    out.push('\n');
48    out.push_str(&table_row(
49        &terminal::heading(&format!("{:<id_width$}", "Id")),
50        &terminal::heading(&format!("{:<category_width$}", "Categories")),
51        &terminal::heading(&format!("{:<description_width$}", "Description")),
52    ));
53    out.push('\n');
54    out.push_str(&table_rule(
55        '├',
56        '┼',
57        '┤',
58        id_width,
59        category_width,
60        description_width,
61    ));
62    out.push('\n');
63
64    for (index, entry) in entries.iter().enumerate() {
65        let id = entry.id();
66        let categories = format_entry_categories(*entry);
67        out.push_str(&table_row(
68            &terminal::table_accent(&format!("{:<id_width$}", id)),
69            &format!("{:<category_width$}", categories),
70            &format!("{:<description_width$}", entry.description()),
71        ));
72        out.push('\n');
73        let last = index + 1 == entries.len();
74        out.push_str(&table_rule(
75            if last { '└' } else { '├' },
76            if last { '┴' } else { '┼' },
77            if last { '┘' } else { '┤' },
78            id_width,
79            category_width,
80            description_width,
81        ));
82        if !last {
83            out.push('\n');
84        }
85    }
86
87    out
88}

90fn table_rule(
91    left: char,
92    mid: char,
93    right: char,
94    id_width: usize,
95    category_width: usize,
96    description_width: usize,
97) -> String {
98    terminal::table_border(&format!(
99        "{left}{}{mid}{}{mid}{}{right}",
100        "─".repeat(id_width + 2),
101        "─".repeat(category_width + 2),
102        "─".repeat(description_width + 2)
103    ))
104}
105
106fn table_row(id: &str, categories: &str, description: &str) -> String {
107    format!("│ {id} │ {categories} │ {description} │")
108}
109
110fn format_entry_categories(entry: GalleryItem) -> String {
111    entry
112        .categories()
113        .iter()
114        .map(ToString::to_string)
115        .collect::<Vec<_>>()
116        .join(", ")
117}

18/// Dense GF(2) coefficient vector with a bit-granular logical length.
19///
20/// Deliberately hand-rolled rather than wrapping [`binar::BitVec`], the
21/// workspace's bit-vector (already used by `bloq_utils::PauliString`). `binar`
22/// allocates and operates in 512-bit `BitBlock`s, sized for the wide rows
23/// `PauliString` carries. Coefficient vectors are one bit per input generator —
24/// usually tens — so every vector would round up to a 64-byte block and every
25/// `xor_assign` would XOR eight words instead of one. Measured on
26/// `GalleryItem::iter()` stabilizer computation, the `binar`-backed version cost
27/// ~36% (0.080s → 0.108s, best of 10); word-level `bit` reads recovered almost
28/// none of it, since the cost is in the block-granular XOR, and `binar` exposes
29/// no mutable word access to route around it.
30///
31/// Swap to `binar` if these vectors ever get wide (thousands of generators),
32/// where block-at-a-time SIMD would start to pay for the padding.
33#[derive(Debug, Clone, PartialEq, Eq)]
34pub(crate) struct CoeffVec {
35    words: Vec<u64>,
36    len: usize,
37}
38
39impl CoeffVec {
40    pub(crate) fn zeros(len: usize) -> Self {
41        Self {
42            words: vec![0; len.div_ceil(64)],
43            len,
44        }
45    }
46
47    pub(crate) fn singleton(index: usize, len: usize) -> Self {
48        let mut value = Self::zeros(len);
49        value.set_bit(index, true);
50        value
51    }
52
53    pub(crate) fn bit(&self, index: usize) -> bool {
54        debug_assert!(index < self.len);
55        let word = index / 64;
56        let bit = index % 64;
57        (self.words[word] >> bit) & 1 == 1
58    }
59
60    pub(crate) fn set_bit(&mut self, index: usize, value: bool) {
61        debug_assert!(index < self.len);
62        let word = index / 64;
63        let bit = index % 64;
64        if value {
65            self.words[word] |= 1_u64 << bit;
66        } else {
67            self.words[word] &= !(1_u64 << bit);
68        }
69    }
70
71    pub(crate) fn xor_assign(&mut self, other: &Self) {
72        debug_assert_eq!(self.len, other.len);
73        for (target, source) in self.words.iter_mut().zip(&other.words) {
74            *target ^= *source;
75        }
76    }
77
78    pub(crate) fn first_one(&self) -> Option<usize> {
79        self.words
80            .iter()
81            .enumerate()
82            .find_map(|(word_index, &word)| {
83                (word != 0).then(|| word_index * 64 + word.trailing_zeros() as usize)
84            })
85    }
86
87    pub(crate) fn is_zero(&self) -> bool {
88        self.words.iter().all(|&word| word == 0)
89    }
90
91    pub(crate) fn iter_ones(&self) -> impl Iterator<Item = usize> + '_ {
92        self.words
93            .iter()
94            .enumerate()
95            .flat_map(move |(word_index, &word)| {
96                let mut word = word;
97                std::iter::from_fn(move || {
98                    if word == 0 {
99                        return None;
100                    }
101                    let bit = word.trailing_zeros() as usize;
102                    word &= word - 1;
103                    let index = word_index * 64 + bit;
104                    (index < self.len).then_some(index)
105                })
106            })
107    }
108
109    pub(crate) fn to_indices(&self) -> Vec<usize> {
110        self.iter_ones().collect()
111    }
112
113    pub(crate) fn len(&self) -> usize {
114        self.len
115    }
116
117    pub(super) fn resize(&mut self, new_len: usize) {
118        if self.len == new_len {
119            return;
120        }
121
122        self.words.resize(new_len.div_ceil(64), 0);
123        self.len = new_len;
124        if let Some(last) = self.words.last_mut() {
125            let trailing = new_len % 64;
126            if trailing != 0 {
127                *last &= (1_u64 << trailing) - 1;
128            }
129        }
130    }
131}

134pub(super) struct CoeffSpan {
135    basis: Vec<CoeffVec>,
136    pivots: Vec<usize>,
137    width: usize,
138}
139
140impl CoeffSpan {
141    pub(super) fn new(width: usize) -> Self {
142        Self {
143            basis: Vec::new(),
144            pivots: Vec::new(),
145            width,
146        }
147    }
148
149    pub(super) fn from_coeffs<'a>(
150        width: usize,
151        coeffs: impl IntoIterator<Item = &'a CoeffVec>,
152    ) -> Self {
153        let mut span = Self::new(width);
154        for coeff in coeffs {
155            span.insert_if_independent(coeff.clone());
156        }
157        span
158    }
159
160    pub(super) fn reduce(&self, value: &CoeffVec) -> CoeffVec {
161        debug_assert_eq!(value.len(), self.width);
162        let mut reduced = value.clone();
163        for (basis, &pivot) in self.basis.iter().zip(&self.pivots) {
164            if reduced.bit(pivot) {
165                reduced.xor_assign(basis);
166            }
167        }
168        reduced
169    }
170
171    #[cfg(test)]
172    pub(super) fn contains(&self, value: &CoeffVec) -> bool {
173        self.reduce(value).is_zero()
174    }
175
176    pub(super) fn insert_if_independent(&mut self, value: CoeffVec) -> bool {
177        let reduced = self.reduce(&value);
178        let Some(pivot) = reduced.first_one() else {
179            return false;
180        };
181        self.basis.push(reduced);
182        self.pivots.push(pivot);
183        true
184    }
185
186    #[cfg(test)]
187    pub(super) fn rank(&self) -> usize {
188        self.basis.len()
189    }
190}

199impl PauliSpan {

204            width,
205        }
206    }

208    pub(super) fn from_rows(rows: &[PauliString], width: usize) -> Self {
209        let mut span = Self::new(width);
210        for row in rows {
211            span.insert_if_independent(row.clone());
212        }

216    pub(super) fn reduce(&self, value: &PauliString) -> PauliString {
217        debug_assert_eq!(value.len(), self.width);
218        let mut reduced = value.clone();
219        for (basis, &(col, axis)) in self.basis.iter().zip(&self.pivots) {
220            if reduced.get(col) & axis {
221                reduced ^= basis;
222            }
223        }
224        reduced
225    }

227    pub(super) fn insert_if_independent(&mut self, value: PauliString) -> bool {
228        let reduced = self.reduce(&value);
229        let Some(pivot) = self.first_pivot(&reduced) else {
230            return false;
231        };
232        self.basis.push(reduced);
233        self.pivots.push(pivot);
234        true
235    }
236
237    #[cfg(test)]
238    pub(super) fn contains(&self, value: &PauliString) -> bool {
239        self.reduce(value).is_identity()
240    }
241
242    pub(super) fn rank(&self) -> usize {
243        self.basis.len()
244    }
245
246    fn first_pivot(&self, value: &PauliString) -> Option<(usize, Pauli)> {
247        debug_assert_eq!(value.len(), self.width);
248        first_pauli_pivot(value)
249    }

252fn first_pauli_pivot(value: &PauliString) -> Option<(usize, Pauli)> {
253    let first = |words: &[u64]| {
254        words.iter().enumerate().find_map(|(word, &bits)| {
255            (bits != 0).then(|| word * 64 + bits.trailing_zeros() as usize)
256        })
257    };
258    match (first(value.x_words()), first(value.z_words())) {
259        (Some(x), Some(z)) => Some(if x <= z { (x, Pauli::X) } else { (z, Pauli::Z) }),
260        (Some(x), None) => Some((x, Pauli::X)),
261        (None, Some(z)) => Some((z, Pauli::Z)),
262        (None, None) => None,
263    }
264}

273/// Factors a fixed row space once, retaining coordinates in the original rows.
274#[derive(Debug)]
275pub(super) struct PauliCombinationBasis {
276    basis: Vec<PauliString>,
277    witnesses: Vec<CoeffVec>,
278    pivots: Vec<(usize, Pauli)>,
279    source_count: usize,
280}
281
282impl PauliCombinationBasis {
283    pub(super) fn new(rows: &[PauliString]) -> Self {
284        let mut basis = Vec::new();
285        let mut witnesses = Vec::new();
286        let mut pivots = Vec::new();
287
288        for (index, row) in rows.iter().enumerate() {
289            let mut reduced = row.clone();
290            let mut witness = CoeffVec::singleton(index, rows.len());
291            for ((basis_row, basis_witness), &(col, axis)) in
292                basis.iter().zip(&witnesses).zip(&pivots)
293            {
294                if reduced.get(col) & axis {
295                    reduced ^= basis_row;
296                    witness.xor_assign(basis_witness);
297                }
298            }
299            let Some(pivot) = first_pauli_pivot(&reduced) else {
300                continue;
301            };
302            basis.push(reduced);
303            witnesses.push(witness);
304            pivots.push(pivot);
305        }
306
307        Self {
308            basis,
309            witnesses,
310            pivots,
311            source_count: rows.len(),
312        }
313    }
314
315    pub(super) fn source_count(&self) -> usize {
316        self.source_count
317    }
318
319    pub(super) fn solve(&self, target: &PauliString) -> Option<CoeffVec> {
320        let mut reduced = target.clone();
321        let mut witness = CoeffVec::zeros(self.source_count);
322        for ((basis_row, basis_witness), &(col, axis)) in
323            self.basis.iter().zip(&self.witnesses).zip(&self.pivots)
324        {
325            if reduced.get(col) & axis {
326                reduced ^= basis_row;
327                witness.xor_assign(basis_witness);
328            }
329        }
330        reduced.is_identity().then_some(witness)
331    }
332}

352    pub(super) rows: Vec<PauliString>,
353    pub(super) coeffs: Vec<CoeffVec>,
354}

356impl TrackedBasis {
357    pub(super) fn new(rows: Vec<PauliString>) -> Self {
358        let width = rows.len();
359        let coeffs = (0..rows.len())
360            .map(|index| CoeffVec::singleton(index, width))
361            .collect::<Vec<_>>();
362        Self { rows, coeffs }
363    }
364
365    pub(super) fn from_parts(rows: Vec<PauliString>, coeffs: Vec<CoeffVec>) -> Self {
366        debug_assert_eq!(rows.len(), coeffs.len());
367        let mut coeffs = coeffs;
368        let width = coeffs.iter().map(CoeffVec::len).max().unwrap_or(0);
369        resize_coeffs(&mut coeffs, width);
370        Self { rows, coeffs }
371    }
372
373    pub(super) fn push(&mut self, row: PauliString, mut coeff: CoeffVec) {
374        if coeff.len() > self.coeff_width() {
375            self.resize_coeff_width(coeff.len());
376        } else {
377            coeff.resize(self.coeff_width());
378        }
379
380        self.rows.push(row);
381        self.coeffs.push(coeff);
382    }
383
384    pub(super) fn replace_suffix(
385        &mut self,
386        start: usize,
387        rows: &[PauliString],
388        coeffs: &[CoeffVec],
389    ) {
390        debug_assert_eq!(self.rows.len().saturating_sub(start), rows.len());
391        debug_assert_eq!(rows.len(), coeffs.len());
392        if let Some(width) = coeffs.iter().map(CoeffVec::len).max()
393            && width > self.coeff_width()
394        {
395            self.resize_coeff_width(width);
396        }
397        let coeff_width = self.coeff_width();
398        self.rows[start..].clone_from_slice(rows);
399        for (target, source) in self.coeffs[start..].iter_mut().zip(coeffs) {
400            *target = source.clone();
401            target.resize(coeff_width);
402        }
403    }
404
405    fn coeff_width(&self) -> usize {
406        self.coeffs.first().map(CoeffVec::len).unwrap_or(0)
407    }
408
409    fn resize_coeff_width(&mut self, width: usize) {
410        resize_coeffs(&mut self.coeffs, width);
411    }
412}
413
414fn resize_coeffs(coeffs: &mut [CoeffVec], width: usize) {
415    for coeff in coeffs {
416        coeff.resize(width);
417    }
418}

435fn choose_measurement_free_coset_representative(
436    solution: &AffineTrackedSolution,
437    basis_rows: &[PauliString],
438    width: usize,
439    prefix_span: &CoeffSpan,
440    seed_reduced: &CoeffVec,
441) -> Option<(PauliString, CoeffVec)> {
442    // Work in the quotient by the frozen measurement prefix: the returned
443    // coefficient must reduce to the same coset as the seed row.
444    let mut target_kernel_reduction = prefix_span.reduce(&solution.particular_coeff);
445    target_kernel_reduction.xor_assign(seed_reduced);
446
447    let reduced_kernel_coeffs = solution
448        .kernel_coeffs
449        .iter()
450        .map(|kernel_coeff| prefix_span.reduce(kernel_coeff))
451        .collect::<Vec<_>>();
452    let kernel_combo = solve_coeff_combination(&reduced_kernel_coeffs, &target_kernel_reduction)?;
453
454    let mut coeff = solution.particular_coeff.clone();
455    for index in kernel_combo.iter_ones() {
456        coeff.xor_assign(&solution.kernel_coeffs[index]);
457    }
458
459    let row = materialize_tracked_row(&coeff, basis_rows, width);
460    Some((row, coeff))
461}

463/// Column-major full GF(2) elimination over the first `col_count` columns of
464/// `rows`, swapping and xoring `witnesses` in lockstep (pass an empty slice to
465/// skip witness tracking). Returns the pivot column of each solved row.
466pub(super) fn echelonize_with_witnesses(
467    rows: &mut [CoeffVec],
468    witnesses: &mut [CoeffVec],
469    col_count: usize,
470) -> Vec<usize> {
471    debug_assert!(witnesses.is_empty() || witnesses.len() == rows.len());
472    let mut pivot_cols = Vec::new();
473    let mut next_row = 0;
474
475    for col in 0..col_count {
476        let Some(pivot_index) = (next_row..rows.len()).find(|&row| rows[row].bit(col)) else {
477            continue;
478        };
479
480        if pivot_index != next_row {
481            rows.swap(pivot_index, next_row);
482            if !witnesses.is_empty() {
483                witnesses.swap(pivot_index, next_row);
484            }
485        }
486
487        let pivot_row = rows[next_row].clone();
488        let pivot_witness = witnesses.get(next_row).cloned();
489        for row_index in 0..rows.len() {
490            if row_index == next_row || !rows[row_index].bit(col) {
491                continue;
492            }
493            rows[row_index].xor_assign(&pivot_row);
494            if let Some(pivot_witness) = &pivot_witness {
495                witnesses[row_index].xor_assign(pivot_witness);
496            }
497        }
498
499        pivot_cols.push(col);
500        next_row += 1;
501    }
502
503    pivot_cols
504}

506// Deliberately not expressed via `echelonize_with_witnesses`: the incremental
507// forward elimination here picks a different (though equally valid) witness
508// than full RREF when `vectors` are linearly dependent, and callers rely on
509// the exact witness for deterministic canonical output.
510pub(crate) fn solve_coeff_combination(vectors: &[CoeffVec], target: &CoeffVec) -> Option<CoeffVec> {
511    let mut basis = Vec::new();
512    let mut witnesses = Vec::new();
513    let mut pivots = Vec::new();
514
515    for (index, vector) in vectors.iter().enumerate() {
516        let mut reduced = vector.clone();
517        let mut witness = CoeffVec::singleton(index, vectors.len());
518
519        for ((basis_vector, basis_witness), &pivot) in basis.iter().zip(&witnesses).zip(&pivots) {
520            if !reduced.bit(pivot) {
521                continue;
522            }
523
524            reduced.xor_assign(basis_vector);
525            witness.xor_assign(basis_witness);
526        }
527
528        let Some(pivot) = reduced.first_one() else {
529            continue;
530        };
531        basis.push(reduced);
532        witnesses.push(witness);
533        pivots.push(pivot);
534    }
535
536    let mut reduced_target = target.clone();
537    let mut witness = CoeffVec::zeros(vectors.len());
538    for ((basis_vector, basis_witness), &pivot) in basis.iter().zip(&witnesses).zip(&pivots) {
539        if !reduced_target.bit(pivot) {
540            continue;
541        }
542
543        reduced_target.xor_assign(basis_vector);
544        witness.xor_assign(basis_witness);
545    }
546
547    reduced_target.is_zero().then_some(witness)
548}
549pub(super) fn complete_from_initial_basis_fast(
550    final_basis: &mut TrackedBasis,
551    basis_rows: &[PauliString],
552    frozen_measurement_rows: &[PauliString],
553    frozen_measurement_coeffs: &[CoeffVec],
554    measurement_cols: &[usize],
555    width: usize,
556) {
557    if final_basis.coeff_width() < basis_rows.len() {
558        final_basis.resize_coeff_width(basis_rows.len());
559    }
560
561    let prefix_span = CoeffSpan::from_coeffs(basis_rows.len(), frozen_measurement_coeffs);
562    let mut coeff_span = CoeffSpan::from_coeffs(basis_rows.len(), &final_basis.coeffs);
563
564    let exact_constraints = measurement_cols
565        .iter()
566        .map(|&col| ExactPauliConstraint {
567            col,
568            pauli: Pauli::I,
569        })
570        .collect::<Vec<_>>();
571
572    // The frozen measurement suffix is invariant across iterations — only slot 0
573    // (the seed row / coeff) varies. Build the solve buffers once and overwrite
574    // slot 0 each iteration rather than reallocating the constant suffix via a
575    // fresh `concat` every time.
576    let mut solve_rows = Vec::with_capacity(1 + frozen_measurement_rows.len());
577    solve_rows.push(PauliString::new(width));
578    solve_rows.extend_from_slice(frozen_measurement_rows);
579    let mut solve_coeffs = Vec::with_capacity(1 + frozen_measurement_coeffs.len());
580    solve_coeffs.push(CoeffVec::zeros(basis_rows.len()));
581    solve_coeffs.extend_from_slice(frozen_measurement_coeffs);
582
583    for (index, row) in basis_rows.iter().enumerate() {
584        let seed_coeff = CoeffVec::singleton(index, basis_rows.len());
585        if !coeff_span.insert_if_independent(seed_coeff.clone()) {
586            continue;
587        }
588
589        let seed_reduced = prefix_span.reduce(&seed_coeff);
590        debug_assert!(
591            !seed_reduced.is_zero(),
592            "independent seed should stay non-zero modulo frozen measurement prefix",
593        );
594
595        solve_rows[0] = row.clone();
596        solve_coeffs[0] = seed_coeff.clone();
597        let representative = solve_affine_tracked(&solve_rows, &solve_coeffs, &exact_constraints)
598            .and_then(|solution| {
599                choose_measurement_free_coset_representative(
600                    &solution,
601                    basis_rows,
602                    width,
603                    &prefix_span,
604                    &seed_reduced,
605                )
606            });
607
608        if let Some((representative_row, representative_coeff)) = representative {
609            final_basis.push(representative_row, representative_coeff);
610        } else {
611            debug_assert_eq!(row.len(), width);
612            final_basis.push(row.clone(), seed_coeff);
613        }
614    }
615
616    debug_assert_eq!(final_basis.rows.len(), final_basis.coeffs.len());
617}

619/// Column-pivot GF(2) elimination over `rows`. Pass a `combinations` slice the
620/// same length as `rows` to track generator provenance in lockstep, or an empty
621/// slice to skip tracking (the plain elimination).
622pub(crate) fn gaussian_elimination_with_tracking<T, F>(
623    rows: &mut [PauliString],
624    combinations: &mut [CoeffVec],
625    pivot_constraints: impl IntoIterator<Item = T>,
626    predicate: F,
627    num_pivotable_rows: Option<usize>,
628) -> usize
629where
630    F: Fn(&PauliString, &T) -> bool,
631{
632    let track = !combinations.is_empty();
633    debug_assert!(!track || rows.len() == combinations.len());
634
635    let num_pivotable_rows = num_pivotable_rows.unwrap_or(rows.len()).min(rows.len());
636    let mut num_solved = 0;
637
638    for constraint in pivot_constraints {
639        if num_solved == num_pivotable_rows {
640            break;
641        }
642
643        let Some(pivot_index) =
644            (num_solved..num_pivotable_rows).find(|&index| predicate(&rows[index], &constraint))
645        else {
646            continue;
647        };
648
649        if pivot_index != num_solved {
650            rows.swap(pivot_index, num_solved);
651            if track {
652                combinations.swap(pivot_index, num_solved);
653            }
654        }
655
656        let (before, pivot_and_after) = rows.split_at_mut(num_solved);
657        let (pivot_row, after) = pivot_and_after
658            .split_first_mut()
659            .expect("solved pivot is in bounds");
660        if track {
661            let (before_coeffs, pivot_and_after_coeffs) = combinations.split_at_mut(num_solved);
662            let (pivot_coeff, after_coeffs) = pivot_and_after_coeffs
663                .split_first_mut()
664                .expect("tracked pivot is in bounds");
665            for (row, coeff) in before
666                .iter_mut()
667                .zip(before_coeffs)
668                .chain(after.iter_mut().zip(after_coeffs))
669            {
670                if predicate(row, &constraint) {
671                    *row ^= &*pivot_row;
672                    coeff.xor_assign(pivot_coeff);
673                }
674            }
675        } else {
676            for row in before.iter_mut().chain(after) {
677                if predicate(row, &constraint) {
678                    *row ^= &*pivot_row;
679                }
680            }
681        }
682
683        num_solved += 1;
684    }
685
686    num_solved
687}

689pub(crate) fn deadline_projection_with_initial_coeffs(
690    generators: &[PauliString],
691    coeffs: &[CoeffVec],
692    zx: &ZXGraph,
693    deadline: i64,
694) -> DeadlineProjection {
695    debug_assert_eq!(generators.len(), coeffs.len());
696    let mut forbidden = Vec::new();
697    // A selective cap gates at z, even after its fill changes NodeKind to
698    // X/Y/Z. Preserve that source identity when projecting a resolved basis.
699    let selective_nodes = zx
700        .action_graph()
701        .ordered_nodes()
702        .filter_map(|node| match node.action {
703            crate::Action::Resolve { target, .. } => zx.node_at(target).map(|node| node.id),
704            _ => None,
705        })
706        .collect::<std::collections::HashSet<_>>();
707    let is_late = |node: &super::super::ZXNode| {
708        let z = i64::from(node.pos.z);
709        if matches!(node.kind, crate::NodeKind::Selective(_)) || selective_nodes.contains(&node.id)
710        {
711            z > deadline
712        } else {
713            z >= deadline
714        }
715    };
716
717    for node in &zx.nodes {
718        if is_late(node) {
719            forbidden.push(node.id);
720        }
721    }
722
723    for edge in &zx.edges {
724        if is_late(&zx.nodes[edge.n1]) || is_late(&zx.nodes[edge.n2]) {
725            forbidden.push(edge.id);
726        }
727    }
728
729    forbidden.sort_unstable();
730    forbidden.dedup();
731    let pivot_constraints = axis_pivot_constraints(forbidden.iter().copied());
732
733    let mut rows = generators.to_vec();
734    let mut coeffs = coeffs.to_vec();
735    let consumed = gaussian_elimination_with_tracking(
736        &mut rows,
737        &mut coeffs,
738        pivot_constraints,
739        |row, &(col, basis)| row.get(col) & basis,
740        None,
741    );
742
743    DeadlineProjection {
744        consumed,
745        rows,
746        coeffs,
747    }
748}

761        let mut coeff = CoeffVec::zeros(width);

769        coeff: &CoeffVec,
770        raw_external: &[PauliString],
771        width: usize,
772    ) -> PauliString {
773        let mut row = PauliString::new(width);
774        for index in coeff.to_indices() {
775            row ^= &raw_external[index];
776        }

12/// A structural well-formedness violation found while validating a
13/// [`BlockGraph`](crate::BlockGraph)'s blocks, pipes, and lattice-surgery
14/// geometry.
15#[derive(Error, Debug, Clone)]
16#[non_exhaustive]
17pub enum InvalidBlockGraphError {
18    /// A block kind requiring one pipe has a different degree.
19    #[error("expected exactly one pipe connected to the block at {0}")]
20    NotExactlySinglePipe(IVec3),
21    /// A block kind that accepts only temporal pipes has a spatial pipe.
22    #[error("spatial pipe connected to the block at {0}")]
23    SpatialPipeNotAllowed(IVec3),
24    /// A spatial port lacks an explicit input, output, or multiplex role.
25    #[error("spatial Port at {0} must declare role=input, role=output, or role=multiplex")]
26    SpatialPortRoleRequired(IVec3),
27    /// A temporal port's declared role disagrees with its pipe direction.
28    #[error("temporal Port at {pos} declares role {declared}, but its pipe makes it {inferred}")]
29    TemporalPortRoleMismatch {
30        /// Port position.
31        pos: IVec3,
32        /// Explicitly declared role.
33        declared: crate::PortRole,
34        /// Role inferred from the temporal pipe.
35        inferred: crate::PortRole,
36    },
37    /// A block accepts only a pipe extending into its past.
38    #[error("only time-like pipe connected to past is allowed for block at {0}")]
39    FuturePipeNotAllowed(IVec3),
40    /// A block accepts only a pipe extending into its future.
41    #[error("only time-like pipe connected to future is allowed for block at {0}")]
42    PastPipeNotAllowed(IVec3),
43    /// Adjacent block faces imply incompatible lattice-surgery orientations.
44    #[error(
45        "cannot resolve the lattice surgery orientation for pipe between blocks at {0} and {1}"
46    )]
47    OrientationContradiction(IVec3, IVec3),
48    /// A multi-pipe junction does not lie in one plane.
49    #[error("block at {0} is not a planar junction")]
50    JunctionNotPlanar(IVec3),
51    /// A pipe face has the same basis on both perpendicular axes.
52    #[error(
53        "degenerate pipe face: block at {0} has the same basis on both axes perpendicular to pipe toward {1}"

55    DegeneratePipeFace(IVec3, IVec3),
56    /// Spatially connected cubes have different symbolic heights.
57    #[error(
58        "spatially connected cubes at {pos} and {neighbor} have different heights (height={height} vs height={neighbor_height})"
59    )]
60    CubeHeightSpatialMismatch {
61        /// First cube position.
62        pos: IVec3,
63        /// Neighboring cube position.

67        /// Neighboring cube height.
68        neighbor_height: crate::CubeHeight,

70    /// A pipe attaches to an invalid cell of a multi-cell cube.
71    #[error(
72        "multi-cell cube at {pos} has an invalid pipe endpoint {endpoint}; spatial pipes attach at the anchor, temporal pipes attach at bottom or top"
73    )]
74    CubeInvalidPipeEndpoint {
75        /// Cube anchor position.
76        pos: IVec3,
77        /// Invalid pipe endpoint.
78        endpoint: IVec3,
79    },
80    /// Two block footprints reserve the same lattice cell.
81    #[error("block footprints overlap at occupied position {pos}")]
82    BlockFootprintOverlap {
83        /// Multiply occupied lattice cell.
84        pos: IVec3,
85    },
86    /// A walking block endpoint has a spatial pipe.
87    #[error("Walking block at {pos} has a non-temporal pipe attached at endpoint {endpoint}")]
88    WalkingPipeNotTemporal {
89        /// Walking-block anchor position.
90        pos: IVec3,
91        /// Endpoint with the invalid pipe.
92        endpoint: IVec3,
93    },
94    /// A walking block's start endpoint has a non-past pipe.
95    #[error("Walking block at {pos} start endpoint only allows past temporal pipes")]
96    WalkingStartPipeNotPast {
97        /// Walking-block anchor position.
98        pos: IVec3,
99    },
100    /// A walking block's end endpoint has a non-future pipe.
101    #[error("Walking block at {pos} end endpoint only allows future temporal pipes")]
102    WalkingEndPipeNotFuture {
103        /// Walking-block anchor position.
104        pos: IVec3,
105    },
106    /// A patch-rotation endpoint does not have exactly one temporal pipe.
107    #[error(
108        "Patch rotation block at {pos} endpoint {endpoint} requires exactly one temporal pipe, got {count}"
109    )]
110    PatchRotationEndpointPipeCount {
111        /// Patch-rotation anchor position.
112        pos: IVec3,
113        /// Endpoint being checked.
114        endpoint: IVec3,
115        /// Number of attached temporal pipes.
116        count: usize,
117    },
118    /// A patch-rotation endpoint has a spatial pipe.
119    #[error(
120        "Patch rotation block at {pos} has a non-temporal pipe attached at endpoint {endpoint}"

122    PatchRotationPipeNotTemporal {
123        /// Patch-rotation anchor position.
124        pos: IVec3,

126        endpoint: IVec3,

142        endpoint: IVec3,

268/// Validates graph structure and action semantics, then — once the cached DAG
269/// is stale — the stabilizer-derived measurement surfaces and dependency edges.
270pub(crate) fn validate(graph: &BlockGraph) -> Result<(), BlockGraphError> {
271    validate_with_limits(graph, crate::ModuleCertificationLimits::DEFAULT)
272}

274pub(crate) fn validate_with_limits(
275    graph: &BlockGraph,
276    limits: crate::ModuleCertificationLimits,
277) -> Result<(), BlockGraphError> {
278    let mut dag = validate_source(graph)?;
279    let regions = graph.branch_regions()?;
280    if graph.has_continuing_branches() {
281        return graph.validate_guarded_actions_with_limits(limits);
282    }
283    if (!regions.is_empty() || graph.has_actions()) && !graph.action_graph().is_analyzed() {
284        let stabilizers = if regions.is_empty() {
285            graph.stabilizers_with_limits(limits)?
286        } else {
287            graph.stabilizers_for_action_analysis_with_limits(limits)?
288        };
289        stabilizers.validate_selective_decoupling()?;
290        dag.attach_readout_dependencies(&stabilizers.generators, Some(&stabilizers.zx_graph))?;
291    }
292    Ok(())
293}

295/// Structure and action semantics only, returning the DAG built along the way.
296///
297/// This is the precondition for building a [`ZXGraph`](crate::ZXGraph), so it
298/// deliberately stops short of the physical analysis in [`validate`]: that
299/// analysis needs a ZX graph to derive the rows it would check.
300pub(crate) fn validate_source(graph: &BlockGraph) -> Result<ActionDag, BlockGraphError> {
301    validate_structure(graph)?;
302    let mut dag = graph.build_action_graph(&graph.actions())?;
303    if !graph.has_actions() {
304        // `build_action_graph` short-circuits on an empty list, so its
305        // graph-constraint pass — every selective block needs a `resolve` —
306        // has not run yet.
307        dag.validate(Some(graph))?;
308    }
309    Ok(dag)
310}

319/// Collects every structural violation instead of stopping at the first.
320///
321/// Editing tools diff this set across a candidate edit so they can reject only
322/// what the edit introduces and leave defects the graph already carried alone;
323/// a fail-fast check cannot tell those apart.
324///
325/// The phases stay ordered and gated exactly as the fail-fast path had them: a
326/// later phase runs only once the earlier ones are clean, because its geometry
327/// inference assumes resolvable pipe endpoints and non-overlapping footprints.
328/// Errors are therefore complete *within* a phase, not across phases.
329pub(crate) fn structural_errors(graph: &BlockGraph) -> Vec<BlockGraphError> {
330    let mut errors = Vec::new();
331
332    collect_block_footprint_errors(graph, &mut errors);
333    if !errors.is_empty() {
334        return errors;
335    }
336    errors.extend(graph.pipes().filter_map(|pipe| pipe.try_endpoints().err()));
337    if !errors.is_empty() {
338        return errors;
339    }
340    for pipe in graph.pipes() {
341        for endpoint in [pipe.src(), pipe.dst()] {
342            if graph.get_endpoint_block(endpoint).is_none() {
343                errors.push(BlockGraphError::BlockNotFound(endpoint));
344            }
345        }
346    }
347    if !errors.is_empty() {
348        return errors;
349    }
350    collect_lattice_surgery_errors(graph, &mut errors);
351    if !errors.is_empty() {
352        return errors;
353    }
354    collect_block_constraint_errors(graph, &mut errors);
355    errors
356}

358fn collect_block_footprint_errors(graph: &BlockGraph, errors: &mut Vec<BlockGraphError>) {
359    let mut occupied = HashMap::<IVec3, Vec<(IVec3, BlockKind)>>::new();
360    for block in graph.blocks() {
361        let reserved = match block.checked_reserved_positions() {
362            Ok(reserved) => reserved,
363            Err(err) => {
364                errors.push(err);
365                continue;
366            }
367        };
368        for pos in reserved {
369            for (other_pos, other_kind) in occupied.get(&pos).into_iter().flatten() {
370                if !block
371                    .kind
372                    .allows_reserved_overlap(block.pos(), *other_kind, *other_pos, pos)
373                {
374                    errors.push(InvalidBlockGraphError::BlockFootprintOverlap { pos }.into());
375                }
376            }
377            occupied
378                .entry(pos)
379                .or_default()
380                .push((block.pos(), block.kind));
381        }
382    }
383}

385fn collect_lattice_surgery_errors(graph: &BlockGraph, errors: &mut Vec<BlockGraphError>) {
386    for (u_pos, v_pos, _, _, pipe) in graph.pipe_endpoints_with_blocks() {
387        debug_assert_eq!(pipe.src(), u_pos);
388        let u_bases = graph.infer_pipe_endpoint_face_bases(pipe, u_pos);
389        let v_bases = graph.infer_pipe_endpoint_face_bases(pipe, v_pos);
390        let delta = u_pos - v_pos;
391        for ((d, u_basis), v_basis) in delta.to_array().into_iter().zip(u_bases).zip(v_bases) {
392            if d == 0
393                && let (Some(u_basis), Some(v_basis)) = (u_basis, v_basis)
394                && (u_basis != v_basis) ^ pipe.hadamard
395            {
396                errors.push(InvalidBlockGraphError::OrientationContradiction(u_pos, v_pos).into());
397                break;
398            }
399        }
400    }
401}

407fn collect_block_constraint_errors(graph: &BlockGraph, errors: &mut Vec<BlockGraphError>) {
408    // Index pipes by owner block once, keeping validation O(V+E) and preserving
409    // pipe order within each block.
410    let mut incidence: HashMap<IVec3, Vec<PipeIncidence>> = HashMap::new();
411    for tuple in graph.pipe_endpoints_with_blocks() {
412        let (_, _, u, v, _) = tuple;
413        incidence.entry(u.pos()).or_default().push(tuple);
414        if v.pos() != u.pos() {
415            incidence.entry(v.pos()).or_default().push(tuple);
416        }
417    }
418
419    // A failing block stops its own checks and the scan moves on to the next
420    // block, rather than abandoning the rest of the graph.
421    for block in graph.blocks() {
422        let neighbors = graph.neighbor_positions(block.pos());
423        let incident: &[PipeIncidence] = match incidence.get(&block.pos()) {
424            Some(pipes) => pipes,
425            None => &[],
426        };
427        match block.kind {
428            BlockKind::Port
429            | BlockKind::Y
430            | BlockKind::Measurement(_)
431            | BlockKind::T
432            | BlockKind::Selective(_) => {
433                if neighbors.len() != 1 {
434                    errors.push(InvalidBlockGraphError::NotExactlySinglePipe(block.pos()).into());
435                    continue;
436                }
437                let has_spatial_pipe = neighbors
438                    .iter()
439                    .any(|&pos| (pos - block.pos()).abs() != IVec3::Z);
440                if block.kind == BlockKind::Port {
441                    let role = block.port_role().expect("Port blocks expose a role");
442                    if has_spatial_pipe && role == crate::PortRole::Auto {
443                        errors.push(
444                            InvalidBlockGraphError::SpatialPortRoleRequired(block.pos()).into(),
445                        );
446                    }
447                    if !has_spatial_pipe && role != crate::PortRole::Auto {
448                        let inferred = if neighbors[0].z > block.pos().z {
449                            crate::PortRole::Input
450                        } else {
451                            crate::PortRole::Output
452                        };
453                        if role != inferred {
454                            errors.push(
455                                InvalidBlockGraphError::TemporalPortRoleMismatch {
456                                    pos: block.pos(),
457                                    declared: role,
458                                    inferred,
459                                }
460                                .into(),
461                            );
462                            continue;
463                        }
464                    }
465                }
466                let spatial_allowed = block.kind == BlockKind::Port;
467                if has_spatial_pipe && !spatial_allowed {
468                    errors.push(InvalidBlockGraphError::SpatialPipeNotAllowed(block.pos()).into());
469                    continue;
470                }
471                if matches!(block.kind, BlockKind::T)
472                    && neighbors.iter().any(|&pos| pos - block.pos() != IVec3::Z)
473                {
474                    // A T block injects into the future, so its single pipe must
475                    // point at +Z; a past-facing pipe is the violation here.
476                    errors.push(InvalidBlockGraphError::PastPipeNotAllowed(block.pos()).into());
477                    continue;
478                }
479                if matches!(
480                    block.kind,
481                    BlockKind::Measurement(_) | BlockKind::Selective(_)
482                ) && neighbors
483                    .iter()
484                    .any(|&pos| pos - block.pos() != IVec3::NEG_Z)
485                {
486                    errors.push(InvalidBlockGraphError::FuturePipeNotAllowed(block.pos()).into());
487                    continue;
488                }
489            }
490            BlockKind::Cube(ck) => {
491                let height_errors = errors.len();
492                collect_cube_height_errors(block, incident, errors);
493                if errors.len() > height_errors {
494                    continue;
495                }
496                let cube_neighbors = cube_endpoint_neighbors(block, incident);
497                // A pipe face is defined by the two axes perpendicular to the pipe
498                // direction. If both perpendicular axes have the same basis, the
499                // lattice surgery orientation is degenerate (e.g. XZZ cube with a
500                // pipe along +X has Z on both Y and Z faces).
501                for (endpoint, delta) in &cube_neighbors {
502                    let diff = (*delta).abs();
503                    let pipe_axis = diff
504                        .to_array()
505                        .iter()
506                        .position(|&d| d != 0)
507                        .expect("neighbor differs on at least one axis");
508                    let perp: [usize; 2] = match pipe_axis {
509                        0 => [1, 2],
510                        1 => [0, 2],
511                        _ => [0, 1],
512                    };
513                    // Each face is independent, so report them all: a cube that
514                    // already has one broken face must still report a second one
515                    // an edit adds, or a before/after diff would not see it.
516                    if ck.bases()[perp[0]] == ck.bases()[perp[1]] {
517                        errors.push(
518                            InvalidBlockGraphError::DegeneratePipeFace(
519                                block.pos(),
520                                *endpoint + *delta,
521                            )
522                            .into(),
523                        );
524                    }
525                }
526
527                if cube_neighbors
528                    .into_iter()
529                    .map(|(_, delta)| delta.abs())
530                    .collect::<HashSet<_>>()
531                    .len()
532                    == 3
533                {
534                    errors.push(InvalidBlockGraphError::JunctionNotPlanar(block.pos()).into());
535                }
536            }
537            BlockKind::Walking(kind) => {
538                let start = block.pos();
539                let end = kind.end_position(start);
540                for &(u_pos, v_pos, u, v, _) in incident {
541                    let (endpoint, other_pos) = if u.pos() == block.pos() {
542                        (u_pos, v_pos)
543                    } else if v.pos() == block.pos() {
544                        (v_pos, u_pos)
545                    } else {
546                        continue;
547                    };
548                    // One violation per incident pipe, then on to the next pipe:
549                    // an existing bad pipe must not mask a newly added one.
550                    let delta = other_pos - endpoint;
551                    if delta.abs() != IVec3::Z {
552                        errors.push(
553                            InvalidBlockGraphError::WalkingPipeNotTemporal {
554                                pos: start,
555                                endpoint,
556                            }
557                            .into(),
558                        );
559                        continue;
560                    }
561                    if endpoint == start && delta != IVec3::NEG_Z {
562                        errors.push(
563                            InvalidBlockGraphError::WalkingStartPipeNotPast { pos: start }.into(),
564                        );
565                        continue;
566                    }
567                    if endpoint == end && delta != IVec3::Z {
568                        errors.push(
569                            InvalidBlockGraphError::WalkingEndPipeNotFuture { pos: start }.into(),
570                        );
571                        continue;
572                    }
573                    if endpoint != start && endpoint != end {
574                        errors.push(
575                            InvalidBlockGraphError::WalkingPipeNotTemporal {
576                                pos: start,
577                                endpoint,
578                            }
579                            .into(),
580                        );
581                        continue;
582                    }
583                }
584            }
585            BlockKind::PatchRotation(kind) => {
586                collect_patch_rotation_errors(graph, block.pos(), kind, incident, errors);
587            }
588        }
589    }
590}

592fn cube_endpoint_neighbors(block: &Block, incident: &[PipeIncidence]) -> Vec<(IVec3, IVec3)> {
593    incident
594        .iter()
595        .filter_map(|&(u_pos, v_pos, u, v, _)| {
596            if u.pos() == block.pos() {
597                Some((u_pos, v_pos - u_pos))
598            } else if v.pos() == block.pos() {
599                Some((v_pos, u_pos - v_pos))
600            } else {
601                None
602            }
603        })
604        .collect()
605}

607/// Reports each endpoint of the rotation independently, so a half-wired block
608/// names both of its missing pipes rather than only the first: an editor adding
609/// them one at a time needs the untouched endpoint to keep reporting the same
610/// violation it already had.
611fn collect_patch_rotation_errors(
612    graph: &BlockGraph,
613    start: IVec3,
614    kind: PatchRotationKind,
615    incident: &[PipeIncidence],
616    errors: &mut Vec<BlockGraphError>,
617) {
618    let end = kind.end_position(start);
619    for endpoint in [start, end] {
620        let pipes = pipes_at_endpoint(incident, start, endpoint);
621        if pipes.len() != 1 {
622            errors.push(
623                InvalidBlockGraphError::PatchRotationEndpointPipeCount {
624                    pos: start,
625                    endpoint,
626                    count: pipes.len(),
627                }
628                .into(),
629            );
630            continue;
631        }
632        let pipe = pipes[0];
633        let neighbor = pipe_neighbor_at(pipe, endpoint)
634            .expect("pipe selected for endpoint must include that endpoint");
635        if (neighbor - endpoint).abs() != IVec3::Z {
636            errors.push(
637                InvalidBlockGraphError::PatchRotationPipeNotTemporal {
638                    pos: start,
639                    endpoint,
640                }
641                .into(),
642            );
643            continue;
644        }
645        if endpoint == start && pipe.is_hadamard() {
646            errors
647                .push(InvalidBlockGraphError::PatchRotationStartPipeHadamard { pos: start }.into());
648            continue;
649        }
650        let expected_basis = kind
651            .pipe_face_bases_at_endpoint(start, endpoint)
652            .expect("patch rotation endpoint was checked above")[UDirection::X.index()];
653        let actual_basis = pipe_neighbor_basis_at_endpoint(graph, pipe, endpoint);
654        if actual_basis.is_some_and(|actual| (actual != expected_basis) ^ pipe.is_hadamard()) {
655            errors.push(
656                InvalidBlockGraphError::PatchRotationBoundaryMismatch {
657                    pos: start,
658                    endpoint,
659                    expected: expected_basis,
660                    actual: actual_basis,
661                }
662                .into(),
663            );
664        }
665    }
666}

668fn pipe_neighbor_basis_at_endpoint(
669    graph: &BlockGraph,
670    pipe: &Pipe,
671    endpoint: IVec3,
672) -> Option<Basis> {
673    let neighbor_pos = pipe_neighbor_at(pipe, endpoint)?;
674    graph.infer_pipe_endpoint_face_bases(pipe, neighbor_pos)[UDirection::X.index()]
675}

677/// Collects the two rules a cube's height imposes on its pipes.
678///
679/// Height agreement is checked pairwise across spacelike pipes, which is
680/// equivalent to the component-wide statement `parser::lower` propagates:
681/// spatial components are exactly the transitive closure of spacelike pipes, so
682/// pairwise equality along every edge is component-wide equality. Per-edge is
683/// also what lets the message name the offending *pair*.
684fn collect_cube_height_errors(
685    block: &Block,
686    incident: &[PipeIncidence],
687    errors: &mut Vec<BlockGraphError>,
688) {
689    let height = block.height();
690    let cells = block.height_cells();
691    let top = block.pos() + IVec3::new(0, 0, cells as i32 - 1);
692
693    for &(u_pos, v_pos, u, v, _) in incident {
694        let (endpoint, other_endpoint, other_block) = if u.pos() == block.pos() {
695            (u_pos, v_pos, v)
696        } else if v.pos() == block.pos() {
697            (v_pos, u_pos, u)
698        } else {
699            continue;
700        };
701        let delta = other_endpoint - endpoint;
702
703        if delta.z == 0 {
704            if cells > 1 && endpoint != block.pos() {
705                errors.push(
706                    InvalidBlockGraphError::CubeInvalidPipeEndpoint {
707                        pos: block.pos(),
708                        endpoint,
709                    }
710                    .into(),
711                );
712                continue;
713            }
714            if other_block.kind.is_cube() && other_block.height() != height {
715                errors.push(
716                    InvalidBlockGraphError::CubeHeightSpatialMismatch {
717                        pos: block.pos(),
718                        neighbor: other_block.pos(),
719                        height,
720                        neighbor_height: other_block.height(),
721                    }
722                    .into(),
723                );
724                continue;
725            }
726        } else if cells > 1 {
727            let valid_temporal_endpoint = (endpoint == block.pos() && delta == IVec3::NEG_Z)
728                || (endpoint == top && delta == IVec3::Z);
729            if !valid_temporal_endpoint {
730                errors.push(
731                    InvalidBlockGraphError::CubeInvalidPipeEndpoint {
732                        pos: block.pos(),
733                        endpoint,
734                    }
735                    .into(),
736                );
737            }
738        }
739    }
740}

742fn pipes_at_endpoint<'a>(
743    incident: &[PipeIncidence<'a>],
744    owner_pos: IVec3,
745    endpoint: IVec3,
746) -> Vec<&'a Pipe> {
747    incident
748        .iter()
749        .filter_map(|&(u_pos, v_pos, u, v, pipe)| {
750            let touches_endpoint = (u.pos() == owner_pos && u_pos == endpoint)
751                || (v.pos() == owner_pos && v_pos == endpoint);
752            touches_endpoint.then_some(pipe)
753        })
754        .collect()
755}
756
757fn pipe_neighbor_at(pipe: &Pipe, endpoint: IVec3) -> Option<IVec3> {
758    if pipe.src() == endpoint {
759        Some(pipe.dst())
760    } else if pipe.dst() == endpoint {
761        Some(pipe.src())
762    } else {
763        None
764    }

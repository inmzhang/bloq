12pub(crate) fn analyze(
13    source: &BlockGraph,
14    limits: ModuleCertificationLimits,
15) -> Result<ActionDag, BlockGraphError> {
16    // Validate declarations and charge expanded sizes before linking addresses.
17    source
18        .validate_with_limits(limits)
19        .map_err(|error| BlockGraphError::ModuleSource(Arc::new(error)))?;
20
21    // Match compiler definition contracts: a parent consuming an output cannot
22    // repair an unsafe named readout in the reusable child definition.
23    let mut definitions = BTreeSet::new();
24    let mut pending = vec![source.root()];
25    while let Some(module) = pending.pop() {
26        if !definitions.insert(module.name.clone()) {
27            continue;
28        }
29        pending.extend(module.instances.iter().map(|instance| {
30            source
31                .module(&instance.definition)
32                .expect("validated definition")
33        }));
34        if module.name != source.root().name {
35            let linked = crate::flatten_module_definition(source, module, "")?;
36            plan(&linked, limits).map_err(|error| definition_error(&module.name, error))?;
37        }
38    }
39
40    // The linker is shared with compilation. Its site ownership makes the
41    // planner compose complete local relations at module seams, rather than
42    // deriving a new relation from a whole-program flat ZX graph.
43    let linked = crate::flatten_module_definition(source, source.root(), "")?;
44    let mut plan = plan(&linked, limits)?;
45    let mut dag = plan
46        .topology
47        .source
48        .build_action_graph(&plan.topology.source.actions())?;
49    dag.attach_guarded_dependencies(plan.action_dependencies()?)?;
50
51    let mut owners = BTreeMap::new();
52    let mut pending = vec![(source.root(), String::new())];
53    while let Some((module, path)) = pending.pop() {
54        pending.extend(module.instances.iter().map(|instance| {
55            (
56                source
57                    .module(&instance.definition)
58                    .expect("validated definition"),
59                crate::program::qualified_name(&path, &instance.name),
60            )
61        }));
62        owners.insert(
63            path.clone(),
64            ActionOwner {
65                definition: module.name.clone(),
66                instance_path: path,
67            },
68        );
69    }
70    debug_assert_eq!(dag.ordered.len(), linked.action_scopes.len());
71    for (&index, scope) in dag.ordered.iter().zip(&linked.action_scopes) {
72        dag.graph[index].owner = Some(owners[scope].clone());
73    }
74    Ok(dag)
75}

77fn plan(
78    linked: &LinkedModuleDefinition,
79    limits: ModuleCertificationLimits,
80) -> Result<crate::GuardedReadoutPlan, BlockGraphError> {
81    let graph = linked.graph.clone().fix_shadowed_faces();
82    let topology = GuardedTopology::new(&graph, limits)?;
83    GuardedSurfaceSpace::new(topology, &linked.sites, limits)?.plan_readouts()
84}

86fn definition_error(name: &str, source: BlockGraphError) -> BlockGraphError {
87    let error = match source {
88        BlockGraphError::Stabilizer(StabilizerError::ResourceLimited {
89            phase,
90            observed,
91            limit,
92        }) => ModuleCertificationError::ResourceLimited {
93            module: name.to_owned(),
94            phase,
95            observed,
96            limit,
97        },
98        source => ModuleCertificationError::Graph {
99            module: name.to_owned(),
100            source,
101        },
102    };
103    BlockGraphError::ModuleMaterialization(Arc::new(error))
104}

111    #[test]
112    fn nested_repeated_rotated_instances_keep_classical_edges_and_owners() {
113        let stage = include_str!("../../tests/fixtures/module_classical.blog")
114            .replace("module main {", "module Stage {");
115        let source = BlockGraph::from_text(&format!(
116            "{stage}\n{}",
117            r#"
118module main {
119  in a: data = 0
120  in b: data = 1
121  in c: data = 2
122  in d: data = 3
123  lower: Stage @ [0, 0, 0]
124  upper: Stage @ [4, 0, 0] rotate Z 180
125  0: Port [0, 0, 0] role=input
126  1: Port [1, 0, 2] role=input
127  2: Port [4, 0, 0] role=input
128  3: Port [3, 0, 2] role=input
129  0 -> lower.first
130  1 -> lower.second
131  2 -> upper.first
132  3 -> upper.second
133}
134"#
135        ))
136        .unwrap();
137        let before = source.to_blog_text();
138        let dag = source.analyze_action_graph().unwrap();
139        assert!(dag.is_analyzed());
140        assert_eq!(dag.ordered_nodes().count(), 6);
141        assert_eq!(source.to_blog_text(), before);
142        assert_eq!(source.root().instances.len(), 2);
143        for (start, scope, x) in [(0, "lower", 0), (3, "upper", 4)] {
144            let measure = dag.node_by_ordinal(start).unwrap();
145            assert_eq!(
146                measure.owner.as_ref().unwrap(),
147                &ActionOwner {
148                    definition: "MeasureZ".into(),
149                    instance_path: format!("{scope}__read"),
150                }
151            );
152            assert!(
153                matches!(&measure.action, Action::Measure { target: MeasureTarget::Node(pos), name }
154                if *pos == glam::IVec3::new(x, 0, 1) && name == &format!("{scope}__read__mz"))
155            );
156            for ordinal in [start + 1, start + 2] {
157                assert_eq!(
158                    dag.node_by_ordinal(ordinal)
159                        .unwrap()
160                        .owner
161                        .as_ref()
162                        .unwrap(),
163                    &ActionOwner {
164                        definition: "SelectCap".into(),
165                        instance_path: format!("{scope}__cap"),
166                    }
167                );
168            }
169            assert!(
170                dag.dependencies()
171                    .any(|edge| edge == (start, start + 1, ActionDependency::Classical))
172            );
173            assert!(
174                dag.dependencies()
175                    .any(|edge| edge == (start + 1, start + 2, ActionDependency::Classical))
176            );
177        }
178        assert!(
179            matches!(dag.node_by_ordinal(5).unwrap().action, Action::Resolve { target, .. }
180            if target == glam::IVec3::new(3, 0, 4))
181        );
182        let error = source
183            .analyze_action_graph_with_limits(ModuleCertificationLimits {
184                max_expanded_instances: 0,
185                ..ModuleCertificationLimits::DEFAULT
186            })
187            .unwrap_err();
188        assert!(matches!(error, BlockGraphError::ModuleSource(_)));
189    }

191    #[test]
192    fn parent_readout_depends_on_child_feedback_across_a_quantum_seam() {
193        let source = BlockGraph::from_text(
194            r#"BLOG 1.0
195module Wire {
196  in input: data = 0
197  out output: data = 2
198  0: Port [0, 0, -1] role=input
199  1: ZXZ [0, 0, 0]
200  2: Port [0, 0, 1] role=output
201  0 -> +Z
202  1 -> +Z
203  feedback Z 1 -> +Z
204}
205module main {
206  in input: data = 10
207  out result = m
208  wire: Wire @ [0, 0, 0]
209  10: Port [0, 0, -1] role=input
210  11: X [0, 0, 1]
211  10 -> wire.input
212  wire.output -> 11
213  m = measure 11
214}
215"#,
216        )
217        .unwrap();
218        let dag = source.analyze_action_graph().unwrap();
219        assert_eq!(dag.ordered_nodes().count(), 2);
220        assert!(
221            dag.dependencies()
222                .any(|edge| edge == (0, 1, ActionDependency::FeedbackAnticommutation))
223        );
224        assert_eq!(
225            dag.node_by_ordinal(0)
226                .unwrap()
227                .owner
228                .as_ref()
229                .unwrap()
230                .instance_path,
231            "wire"
232        );
233        assert_eq!(
234            dag.node_by_ordinal(1)
235                .unwrap()
236                .owner
237                .as_ref()
238                .unwrap()
239                .instance_path,
240            ""
241        );
242        let svg = dag.to_svg().unwrap();
243        assert!(svg.contains("data-definition=\"Wire\" data-instance=\"wire\""));
244        assert!(svg.contains("data-dependency=\"FeedbackAnticommutation\""));
245    }

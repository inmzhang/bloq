14type Terms = Vec<(usize, DecisionId)>;
15pub(crate) type Seam = (usize, usize, DecisionId);

17/// Column and Boolean-variable names are local to this exact reduction recipe.
18/// Full rows retain interior support; external half-edges are never projected out.
19#[derive(Debug, PartialEq, Eq, Hash)]
20struct RelationKey {
21    nodes: Vec<DecisionNode>,
22    rows: Vec<(DecisionId, Terms)>,
23    seams: Vec<Seam>,
24}
25
26#[derive(Debug)]
27struct ReducedRelation {
28    nodes: Vec<DecisionNode>,
29    rows: Vec<BooleanRow>,
30}
31
32type ReductionSlot = Arc<Mutex<Option<Arc<ReducedRelation>>>>;
33
34#[derive(Debug, Default)]
35pub(crate) struct ModuleRelations {
36    reductions: Mutex<HashMap<Arc<RelationKey>, ReductionSlot>>,
37    eliminated: AtomicUsize,
38    reused: AtomicUsize,
39}
40
41impl ModuleRelations {
42    pub(crate) fn work(&self) -> (usize, usize) {
43        (
44            self.eliminated.load(Ordering::Relaxed),
45            self.reused.load(Ordering::Relaxed),
46        )
47    }
48
49    fn cached_reduction(
50        &self,
51        key: RelationKey,
52        equations: usize,
53        solve: impl FnOnce(&RelationKey) -> Result<ReducedRelation, BlockGraphError>,
54    ) -> Result<Arc<ReducedRelation>, BlockGraphError> {
55        let key = Arc::new(key);
56        let slot = Arc::clone(
57            self.reductions
58                .lock()
59                .unwrap_or_else(PoisonError::into_inner)
60                .entry(Arc::clone(&key))
61                .or_default(),
62        );
63        // One caller solves each key while the others wait for it. A failed or
64        // panicking solve leaves the slot empty for the next caller.
65        let mut slot = slot.lock().unwrap_or_else(PoisonError::into_inner);
66        if let Some(reduced) = slot.as_ref() {
67            self.reused.fetch_add(equations, Ordering::Relaxed);
68            return Ok(Arc::clone(reduced));
69        }
70        let reduced = Arc::new(solve(&key)?);
71        self.eliminated.fetch_add(equations, Ordering::Relaxed);
72        *slot = Some(Arc::clone(&reduced));
73        Ok(reduced)
74    }

76    pub(crate) fn reduce(
77        &self,
78        rows: Vec<BooleanRow>,
79        seams: Vec<Seam>,
80        column_rank: &[usize],
81        diagram: &mut BooleanDecisionDiagram,
82        limits: ModuleCertificationLimits,
83    ) -> Result<Vec<BooleanRow>, BlockGraphError> {
84        if seams.is_empty() {
85            return close_seams(rows, &seams, diagram, limits);
86        }
87        let terms = rows
88            .iter()
89            .map(|row| row.terms().len().saturating_add(1))
90            .fold(0usize, usize::saturating_add)
91            .saturating_add(seams.len().saturating_mul(3));
92        diagram.charge(
93            terms
94                .saturating_mul(terms.max(1).ilog2() as usize + 1)
95                .saturating_mul(3),
96        )?;
97        let used = rows
98            .iter()
99            .flat_map(BooleanRow::terms)
100            .map(|&(column, _)| column / 2)
101            .chain(seams.iter().flat_map(|&(left, right, _)| [left, right]))
102            .collect::<HashSet<_>>();
103        let mut columns = used.into_iter().collect::<Vec<_>>();
104        columns.sort_unstable_by_key(|&column| column_rank[column]);
105        let column_map = columns
106            .iter()
107            .enumerate()
108            .map(|(local, &source)| (source, local))
109            .collect::<HashMap<_, _>>();
110        let roots = rows
111            .iter()
112            .flat_map(|row| {
113                std::iter::once(row.active).chain(row.terms().iter().map(|&(_, root)| root))
114            })
115            .chain(seams.iter().map(|&(_, _, root)| root))
116            .collect::<Vec<_>>();
117        diagram.charge(roots.len().saturating_add(columns.len()))?;
118        let mut imported = HashMap::from_iter([(ZERO, ZERO), (ONE, ONE)]);
119        let mut nodes = Vec::new();
120        let mut variables = HashSet::default();
121        // Traverse roots in recipe order, independent of the source arena's
122        // allocation history. Only reachable nodes become owned cache roots.
123        for root in roots {
124            if imported.contains_key(&root) {
125                continue;
126            }
127            let mut pending = vec![(root, false)];
128            while let Some((root, visited)) = pending.pop() {
129                if imported.contains_key(&root) {
130                    continue;
131                }
132                diagram.charge(1)?;
133                let node = diagram.node(root).expect("live relation coefficient");
134                if visited {
135                    variables.insert(node.variable);
136                    let value = DecisionId(nodes.len() + 2);
137                    nodes.push(DecisionNode {
138                        variable: node.variable,
139                        low: imported[&node.low],
140                        high: imported[&node.high],
141                    });
142                    imported.insert(root, value);
143                } else {
144                    pending.extend([(root, true), (node.high, false), (node.low, false)]);
145                }
146            }
147        }
148        diagram.charge(
149            nodes.len().saturating_add(
150                variables
151                    .len()
152                    .saturating_mul(variables.len().max(1).ilog2() as usize + 1),
153            ),
154        )?;
155        let mut variables = variables.into_iter().collect::<Vec<_>>();
156        variables.sort_unstable();
157        let variable_map = variables
158            .iter()
159            .enumerate()
160            .map(|(local, &source)| (source, local))
161            .collect::<HashMap<_, _>>();
162        for node in &mut nodes {
163            node.variable = variable_map[&node.variable];
164        }
165        let normalize = |terms: &[(usize, DecisionId)]| {
166            let mut result = terms
167                .iter()
168                .map(|&(column, value)| {
169                    (2 * column_map[&(column / 2)] + column % 2, imported[&value])
170                })
171                .collect::<Terms>();
172            result.sort_unstable_by_key(|&(column, _)| column);
173            result
174        };
175        let key = RelationKey {
176            nodes,
177            rows: rows
178                .iter()
179                .map(|row| (imported[&row.active], normalize(row.terms())))
180                .collect(),
181            // Seam and axis order retain the source pivot convention.
182            seams: seams
183                .iter()
184                .map(|&(left, right, value)| {
185                    (column_map[&left], column_map[&right], imported[&value])
186                })
187                .collect(),
188        };
189        let equations = seams.len().saturating_mul(2);
190        let reduced = self.cached_reduction(key, equations, |key| {
191            let mut local_limits = limits.boolean_limits();
192            local_limits.max_nodes = local_limits.max_nodes.min(diagram.limits().max_nodes);
193            local_limits.max_steps = local_limits.max_steps.min(diagram.limits().max_steps);
194            let mut local = BooleanDecisionDiagram::with_limits(local_limits);
195            // Reserve work already spent by the caller before local solving.
196            let start = diagram.steps();
197            local.charge(start)?;
198            for node in &key.nodes {
199                local.charge(1)?;
200                local.make_node(node.variable, node.low, node.high)?;
201            }
202            local.charge(key.rows.iter().fold(0usize, |sum, (_, terms)| {
203                sum.saturating_add(
204                    terms
205                        .len()
206                        .saturating_mul(terms.len().max(1).ilog2() as usize + 1)
207                        .saturating_add(1),
208                )
209            }))?;
210            let mut rows = close_seams(
211                key.rows
212                    .iter()
213                    .map(|(active, terms)| BooleanRow::from_terms(*active, terms.clone()))
214                    .collect(),
215                &key.seams,
216                &mut local,
217                limits,
218            )?;
219            local.charge(
220                local.nodes().len().saturating_mul(2).saturating_add(
221                    rows.iter()
222                        .map(|row| row.terms().len().saturating_add(1))
223                        .fold(0usize, usize::saturating_add),
224                ),
225            )?;
226            local.collect_garbage(rows.iter_mut().flat_map(BooleanRow::decisions_mut));
227            diagram.charge(local.steps() - start)?;
228            Ok(ReducedRelation {
229                nodes: local.nodes().to_vec(),
230                rows,
231            })
232        })?;
233        let mut decisions = vec![ZERO, ONE];
234        for node in &reduced.nodes {
235            // Canonical variables are a sorted subset of caller variables;
236            // this injection preserves ordering even across distinct arenas.
237            diagram.charge(1)?;
238            decisions.push(diagram.make_node(
239                variables[node.variable],
240                decisions[node.low.0],
241                decisions[node.high.0],
242            )?);
243        }
244        let mut result = Vec::with_capacity(reduced.rows.len());
245        for row in &reduced.rows {
246            let count = row.terms().len();
247            diagram.charge(
248                count
249                    .saturating_mul(count.max(1).ilog2() as usize + 1)
250                    .saturating_add(1),
251            )?;
252            let mut terms = row
253                .terms()
254                .iter()
255                .map(|&(column, root)| (2 * columns[column / 2] + column % 2, decisions[root.0]))
256                .collect::<Terms>();
257            terms.sort_unstable_by_key(|&(column, _)| column);
258            result.push(BooleanRow::from_terms(decisions[row.active.0], terms));
259        }
260        if diagram.nodes().len() > limits.max_witness_nodes {
261            return Err(StabilizerError::ResourceLimited {
262                phase: "guarded correlation coefficients",
263                observed: diagram.nodes().len(),
264                limit: limits.max_witness_nodes,
265            }
266            .into());
267        }
268        Ok(result)
269    }
270}

272pub(crate) fn close_seams(
273    mut rows: Vec<BooleanRow>,
274    seams: &[Seam],
275    diagram: &mut BooleanDecisionDiagram,
276    limits: ModuleCertificationLimits,
277) -> Result<Vec<BooleanRow>, BlockGraphError> {
278    if seams.is_empty() {
279        diagram.charge(rows.len())?;
280        rows.retain(|row| row.active != ZERO);
281        return Ok(rows);
282    }
283    let mut rows = BooleanRowSpace::new(rows, diagram)?;
284    for &(left, right, hadamard) in seams {
285        for axis in 0..2 {
286            // First-row pivot priority preserves authored outcome/frame conventions.
287            rows.eliminate_linear(
288                &[
289                    (2 * left + axis, ONE),
290                    (2 * right + axis, ONE),
291                    (2 * right, hadamard),
292                    (2 * right + 1, hadamard),
293                ],
294                diagram,
295            )?;
296            if diagram.nodes().len() > limits.max_witness_nodes {
297                return Err(StabilizerError::ResourceLimited {
298                    phase: "guarded correlation coefficients",
299                    observed: diagram.nodes().len(),
300                    limit: limits.max_witness_nodes,
301                }
302                .into());
303            }
304        }
305    }
306    Ok(rows.into_rows(diagram)?)
307}

314    fn cached_full_support_survives_column_variable_and_arena_remapping() {
315        let cache = ModuleRelations::default();
316        let limits = ModuleCertificationLimits::DEFAULT;
317        for (columns, variables) in [([0, 1, 2], [0, 1]), ([2, 0, 1], [4, 9])] {
318            let mut diagram = BooleanDecisionDiagram::default();
319            let x = diagram.make_node(variables[0], ZERO, ONE).unwrap();
320            let y = diagram.make_node(variables[1], ZERO, ONE).unwrap();
321            let not_x = diagram.negate(x).unwrap();
322            let row = |active, terms: &[(usize, DecisionId)]| {
323                let mut terms = terms
324                    .iter()
325                    .map(|&(column, value)| (2 * columns[column], value))
326                    .collect::<Vec<_>>();
327                terms.sort_unstable_by_key(|&(column, _)| column);
328                BooleanRow::from_terms(active, terms)
329            };
330            let rows = vec![
331                row(x, &[(0, ONE), (1, y)]),
332                row(not_x, &[(0, ONE), (2, ONE)]),
333                row(ONE, &[(1, ONE), (2, ONE)]),
334            ];
335            let seams = vec![(columns[0], columns[1], ZERO)];
336            let mut rank = vec![0; 3];
337            for (index, &column) in columns.iter().enumerate() {
338                rank[column] = index;
339            }
340            let mut expected = close_seams(rows.clone(), &seams, &mut diagram, limits).unwrap();
341            let mut actual = cache
342                .reduce(rows, seams, &rank, &mut diagram, limits)
343                .unwrap();
344            diagram.collect_garbage(
345                expected
346                    .iter_mut()
347                    .chain(&mut actual)
348                    .flat_map(BooleanRow::decisions_mut),
349            );
350            for assignment in 0..4 {
351                let values = |rows: &[BooleanRow]| {
352                    rows.iter()
353                        .map(|row| {
354                            let evaluate = |root| {
355                                diagram.evaluate(root, |variable| {
356                                    let index = variables
357                                        .iter()
358                                        .position(|&candidate| candidate == variable)
359                                        .unwrap();
360                                    assignment & (1 << index) != 0
361                                })
362                            };
363                            (
364                                evaluate(row.active),
365                                row.terms()
366                                    .iter()
367                                    .map(|&(column, value)| (column, evaluate(value)))
368                                    .collect::<Vec<_>>(),
369                            )
370                        })
371                        .collect::<Vec<_>>()
372                };
373                assert_eq!(values(&actual), values(&expected));
374            }
375        }
376        assert_eq!(cache.work(), (2, 2));
377    }

408        let mut diagram = BooleanDecisionDiagram::default();

483    #[test]
484    fn cache_misses_use_only_the_callers_remaining_work() {
485        for (caller_limit, core_limit) in [(400, 400), (400, 800), (800, 400)] {
486            let limits = ModuleCertificationLimits {
487                max_boolean_steps: core_limit,
488                ..ModuleCertificationLimits::DEFAULT
489            };
490            let mut caller_limits = limits.boolean_limits();
491            caller_limits.max_steps = caller_limit;
492            let mut diagram = BooleanDecisionDiagram::with_limits(caller_limits);
493            let x = diagram.make_node(0, ZERO, ONE).unwrap();
494            let absent = diagram.negate(x).unwrap();
495            let y = diagram.make_node(1, ZERO, ONE).unwrap();
496            diagram.charge(200 - diagram.steps()).unwrap();
497            let rows = vec![
498                BooleanRow::from_terms(x, [(0, ONE), (4, y)]),
499                BooleanRow::from_terms(absent, [(0, ONE), (2, ONE)]),
500            ];
501            let cache = ModuleRelations::default();
502            let error = cache
503                .reduce(rows, vec![(0, 1, y)], &[0, 1, 2], &mut diagram, limits)
504                .unwrap_err();
505            let BlockGraphError::Stabilizer(StabilizerError::ResourceLimited {
506                phase: "Boolean work steps",
507                observed,
508                limit,
509            }) = error
510            else {
511                panic!("{error:?}")
512            };
513            assert_eq!(limit, caller_limit.min(core_limit));
514            // Each charge in this small recipe is at most 30 steps. A fresh local
515            // quota instead completes the solve before submitting a much larger bill.
516            assert!(
517                observed <= limit + 30,
518                "late budget enforcement: {observed}"
519            );
520            assert_eq!(cache.work(), (0, 0));
521        }
522    }

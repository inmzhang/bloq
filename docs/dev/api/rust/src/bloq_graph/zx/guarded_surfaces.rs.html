61fn next_readout_collection(retained: usize, limit: usize) -> usize {
62    let growth = (READOUT_COLLECTION_TARGET / 4).max(retained / 4);
63    retained
64        .saturating_add(growth)
65        .max(READOUT_COLLECTION_TARGET)
66        .min(limit)
67}

112impl<'a> ModuleScopes<'a> {
113    fn new() -> Self {
114        Self {
115            nodes: vec![ModuleScope {
116                parent: None,
117                depth: 0,
118                path: "",
119                children: BTreeMap::new(),
120            }],
121        }
122    }
123
124    fn insert(&mut self, path: &'a str) -> usize {
125        let mut owner = 0;
126        if path.is_empty() {
127            return owner;
128        }
129        let mut end = 0;
130        for component in path.split("__") {
131            end += component.len();
132            owner = if let Some(&child) = self.nodes[owner].children.get(component) {
133                child
134            } else {
135                let child = self.nodes.len();
136                let depth = self.nodes[owner].depth + 1;
137                self.nodes.push(ModuleScope {
138                    parent: Some(owner),
139                    depth,
140                    path: &path[..end],
141                    children: BTreeMap::new(),
142                });
143                self.nodes[owner].children.insert(component, child);
144                child
145            };
146            end += 2;
147        }
148        owner
149    }
150
151    fn common_ancestor(&self, mut left: usize, mut right: usize) -> usize {
152        if left == 0 || right == 0 {
153            return 0;
154        }
155        while self.nodes[left].depth > self.nodes[right].depth {
156            left = self.nodes[left]
157                .parent
158                .expect("non-root scope has a parent");
159        }
160        while self.nodes[right].depth > self.nodes[left].depth {
161            right = self.nodes[right]
162                .parent
163                .expect("non-root scope has a parent");
164        }
165        while left != right {
166            left = self.nodes[left]
167                .parent
168                .expect("distinct peers are non-root");
169            right = self.nodes[right]
170                .parent
171                .expect("distinct peers are non-root");
172        }
173        left
174    }
175
176    fn lexical_order(&self) -> Vec<usize> {
177        let mut order = Vec::with_capacity(self.nodes.len());
178        let mut pending = vec![0];
179        while let Some(scope) = pending.pop() {
180            order.push(scope);
181            pending.extend(self.nodes[scope].children.values().rev().copied());
182        }
183        order
184    }
185}

200impl ReadoutTiming {
201    fn decision(
202        &mut self,
203        root: DecisionId,
204        diagram: &mut bloq_utils::boolean::BooleanDecisionDiagram,
205    ) -> Result<i64, BooleanResourceError> {
206        let root = diagram.constrain(root, self.domain)?;
207        let cached = |values: &HashMap<DecisionId, i64>, root: DecisionId| {
208            if root == ZERO || root == ONE {
209                Some(i64::MIN)
210            } else {
211                values.get(&root).copied()
212            }
213        };
214        if let Some(time) = cached(&self.decisions, root) {
215            return Ok(time);
216        }
217        let mut pending = vec![root];
218        while let Some(&current) = pending.last() {
219            diagram.charge(1)?;
220            if cached(&self.decisions, current).is_some() {
221                pending.pop();
222                continue;
223            }
224            let node = diagram.node(current).expect("nonconstant decision exists");
225            if let Some(child) = [node.low, node.high]
226                .into_iter()
227                .find(|&child| cached(&self.decisions, child).is_none())
228            {
229                pending.push(child);
230                continue;
231            }
232            let time = self.variables[node.variable]
233                .max(cached(&self.decisions, node.low).expect("child visited"))
234                .max(cached(&self.decisions, node.high).expect("child visited"));
235            self.decisions.insert(current, time);
236            pending.pop();
237        }
238        Ok(self.decisions[&root])
239    }
240
241    /// Control time of a reachable row and its columns with live coefficients.
242    fn controls(
243        &mut self,
244        row: &BooleanRow,
245        diagram: &mut bloq_utils::boolean::BooleanDecisionDiagram,
246    ) -> Result<Option<(i64, Vec<usize>)>, BooleanResourceError> {
247        let care = diagram.apply(BooleanOp::And, row.active, self.domain)?;
248        if care == ZERO {
249            return Ok(None);
250        }
251        let mut control = self.decision(row.active, diagram)?;
252        let mut columns = Vec::new();
253        diagram.charge(row.terms().len())?;
254        for &(column, value) in row.terms() {
255            let value = diagram.constrain(value, care)?;
256            if value == ZERO {
257                continue;
258            }
259            control = control.max(self.decision(value, diagram)?);
260            columns.push(column);
261        }
262        Ok(Some((control, columns)))
263    }
264
265    fn include(
266        &mut self,
267        row: &BooleanRow,
268        diagram: &mut bloq_utils::boolean::BooleanDecisionDiagram,
269    ) -> Result<(), BooleanResourceError> {
270        let Some((control, columns)) = self.controls(row, diagram)? else {
271            return Ok(());
272        };
273        // Choosing a guarded row requires its coefficient controls even when
274        // another term in that same row has a constant coefficient.
275        diagram.charge(columns.len())?;
276        for column in columns {
277            self.columns[column] = self.columns[column].max(control);
278        }
279        Ok(())
280    }
281
282    fn row(
283        &mut self,
284        row: &BooleanRow,
285        own: usize,
286        diagram: &mut bloq_utils::boolean::BooleanDecisionDiagram,
287    ) -> Result<i64, BooleanResourceError> {
288        let Some((control, columns)) = self.controls(row, diagram)? else {
289            return Ok(i64::MIN);
290        };
291        Ok(columns
292            .into_iter()
293            .filter(|&column| column != own)
294            .fold(control, |time, column| time.max(self.columns[column])))
295    }
296}

298impl GuardedPivots {
299    /// Insert a row already reduced against the existing pivots on their
300    /// activation. New pivot pieces are disjoint from the existing owner.
301    /// Availability pivots require `REDUCE_ABOVE` because their columns are
302    /// later released; accepted readouts only substitute into future rows.
303    fn insert_residual<const REDUCE_ABOVE: bool>(
304        &mut self,
305        mut row: BooleanRow,
306        columns: impl Iterator<Item = usize>,
307        diagram: &mut bloq_utils::boolean::BooleanDecisionDiagram,
308    ) -> Result<(BooleanRow, Vec<(usize, DecisionId)>), BooleanResourceError> {
309        let mut added = Vec::new();
310        for column in columns {
311            diagram.charge(1)?;
312            let selected = diagram.apply(BooleanOp::And, row.active, row.get(column))?;
313            if selected == ZERO {
314                continue;
315            }
316            let mut pivot = BooleanRow::new(selected);
317            pivot.xor_scaled(&row, selected, diagram)?;
318            let owner = self.owners.get(&column).copied();
319            let previous = owner
320                .map(|owner| self.rows.take(owner, diagram))
321                .transpose()?
322                .flatten();
323            if REDUCE_ABOVE {
324                self.rows.reduce_column(column, &pivot, diagram)?;
325            }
326            if let Some(mut previous) = previous {
327                previous.xor_scaled(&pivot, ONE, diagram)?;
328                previous.active = diagram.apply(BooleanOp::Or, previous.active, selected)?;
329                self.rows.replace(
330                    owner.expect("a previous row is taken only from an existing owner"),
331                    previous,
332                    diagram,
333                )?;
334            } else {
335                let owner = self
336                    .rows
337                    .insert_indexed(pivot, diagram)?
338                    .expect("selected pivot is active and has nonzero support");
339                self.owners.insert(column, owner);
340            }
341            added.push((column, selected));
342            let absent = diagram.negate(selected)?;
343            row.active = diagram.apply(BooleanOp::And, row.active, absent)?;
344        }
345        Ok((row, added))
346    }
347
348    fn project(
349        &self,
350        row: &mut BooleanRow,
351        diagram: &mut bloq_utils::boolean::BooleanDecisionDiagram,
352    ) -> Result<(), BooleanResourceError> {
353        diagram.charge(row.terms().len())?;
354        // Forward-echelon rows can contain later pivots. Clearing an earlier
355        // pivot may introduce one that was absent from the original support.
356        // Resume on the changed support, rather than visiting every pivot in
357        // the enclosing graph for each sparse local row.
358        let mut after = None;
359        loop {
360            diagram.charge(row.terms().len().max(1).ilog2() as usize + 1)?;
361            let start = after.map_or(0, |last| {
362                row.terms().partition_point(|&(column, _)| column <= last)
363            });
364            let mut selected = None;
365            for &(column, value) in &row.terms()[start..] {
366                diagram.charge(1)?;
367                if let Some(&owner) = self.owners.get(&column) {
368                    selected = Some((column, value, owner));
369                    break;
370                }
371            }
372            let Some((column, value, owner)) = selected else {
373                break;
374            };
375            let factor = diagram.apply(BooleanOp::And, row.active, value)?;
376            row.xor_scaled(
377                self.rows
378                    .row(owner)
379                    .expect("pivot owners retain their indexed rows"),
380                factor,
381                diagram,
382            )?;
383            after = Some(column);
384        }
385        Ok(())
386    }
387}

389struct GuardedAvailability {
390    pivots: GuardedPivots,
391    groups: BTreeMap<AvailabilityKey, Vec<usize>>,
392    active: BTreeSet<usize>,
393    auxiliary_start: usize,
394}

396impl GuardedAvailability {
397    fn new(
398        mut rows: Vec<BooleanRow>,
399        constraints: &[(AvailabilityKey, Vec<(usize, DecisionId)>)],
400        auxiliary_start: usize,
401        diagram: &mut bloq_utils::boolean::BooleanDecisionDiagram,
402    ) -> Result<(Self, BooleanRowSpace), BooleanResourceError> {
403        let mut groups = BTreeMap::<_, Vec<_>>::new();
404        let mut columns = HashMap::<usize, Vec<_>>::new();
405        diagram.charge(constraints.len())?;
406        for (index, (key, terms)) in constraints.iter().enumerate() {
407            let tag = auxiliary_start + index;
408            groups.entry(*key).or_default().push(tag);
409            diagram.charge(terms.len())?;
410            for &(column, guard) in terms {
411                columns.entry(column).or_default().push((tag, guard));
412            }
413        }
414        for row in &mut rows {
415            diagram.charge(row.terms().len())?;
416            let mut tags = BTreeMap::new();
417            for &(column, coefficient) in row.terms() {
418                if let Some(aliases) = columns.get(&column) {
419                    diagram.charge(aliases.len())?;
420                    for &(tag, guard) in aliases {
421                        let value = tags.entry(tag).or_insert(ZERO);
422                        let term = diagram.apply(BooleanOp::And, guard, coefficient)?;
423                        *value = diagram.apply(BooleanOp::Xor, *value, term)?;
424                    }
425                }
426            }
427            for (tag, coefficient) in tags {
428                row.set(tag, coefficient);
429            }
430        }
431        let mut remaining = BooleanRowSpace::new(rows, diagram)?;
432        let mut pivots = GuardedPivots::default();
433        let active =
434            (auxiliary_start..auxiliary_start + constraints.len()).collect::<BTreeSet<_>>();
435        for &column in &active {
436            let pivot = remaining.eliminate_column_sparse(column, diagram)?;
437            if pivot.active != ZERO {
438                pivots.rows.reduce_column(column, &pivot, diagram)?;
439                let owner = pivots
440                    .rows
441                    .insert_indexed(pivot, diagram)?
442                    .expect("active elimination pivot retains its column");
443                pivots.owners.insert(column, owner);
444            }
445        }
446        let rows = remaining.into_rows(diagram)?;
447        let mut available = BooleanRowSpace::default();
448        for mut row in rows {
449            row.truncate_columns(auxiliary_start);
450            let active = row.active;
451            diagram.charge(row.terms().len())?;
452            row.map_coefficients(|value| diagram.apply(BooleanOp::And, active, value))?;
453            if !row.terms().is_empty() {
454                available.push(row, diagram)?;
455            }
456        }
457        Ok((
458            Self {
459                pivots,
460                groups,
461                active,
462                auxiliary_start,
463            },
464            available,
465        ))
466    }
467
468    fn release(
469        &mut self,
470        pending: &BTreeSet<AvailabilityKey>,
471        readouts: &GuardedPivots,
472        available: &mut BooleanRowSpace,
473        diagram: &mut bloq_utils::boolean::BooleanDecisionDiagram,
474    ) -> Result<(), BooleanResourceError> {
475        diagram.charge(self.groups.len())?;
476        for (_, columns) in self.groups.extract_if(.., |key, _| !pending.contains(key)) {
477            for column in columns {
478                self.active.remove(&column);
479                let row = self
480                    .pivots
481                    .owners
482                    .remove(&column)
483                    .map(|owner| self.pivots.rows.take(owner, diagram))
484                    .transpose()?
485                    .flatten();
486                self.pivots.rows.erase_column(column, diagram)?;
487                let Some(mut row) = row else {
488                    continue;
489                };
490                row.set(column, ZERO);
491                diagram.charge(row.terms().len())?;
492                let columns = row
493                    .terms()
494                    .iter()
495                    .filter_map(|&(column, _)| self.active.contains(&column).then_some(column))
496                    .collect::<Vec<_>>();
497                let (mut row, _) =
498                    self.pivots
499                        .insert_residual::<true>(row, columns.into_iter(), diagram)?;
500                if row.active != ZERO {
501                    row.truncate_columns(self.auxiliary_start);
502                    let active = row.active;
503                    diagram.charge(row.terms().len())?;
504                    row.map_coefficients(|value| diagram.apply(BooleanOp::And, active, value))?;
505                    readouts.project(&mut row, diagram)?;
506                    if !row.terms().is_empty() {
507                        available.push(row, diagram)?;
508                    }
509                }
510            }
511        }
512        Ok(())
513    }
514}

568
569    pub(crate) fn with_relations(
570        mut topology: GuardedTopology,
571        sites: &HashMap<IVec3, MaterializedModuleSite>,
572        limits: ModuleCertificationLimits,
573        relations: &HashMap<String, Arc<crate::summary::relations::ModuleRelations>>,
574    ) -> Result<Self, BlockGraphError> {
575        crate::CancellationToken::checkpoint()?;
576        // Intern components once, including empty ancestors. Numeric parents
577        // avoid cloning and comparing every shared prefix during composition.
578        let mut scopes = ModuleScopes::new();
579        let owners = topology
580            .sites
581            .keys()
582            .map(|&position| {
583                let path = sites
584                    .get(&IVec3::from_array(position))
585                    .map_or("", |site| site.instance_path.as_str());
586                (position, scopes.insert(path))
587            })
588            .collect::<BTreeMap<_, _>>();
589        // A source block has at most six incident directions. Most compilations
590        // fit the limit even under this allocation-free global upper bound.
591        let upper_bound = topology.sites.values().fold(0usize, |sum, variants| {
592            sum.saturating_add(variants.len().saturating_mul(6).saturating_add(1))
593        });
594        if upper_bound > limits.max_local_columns {
595            let mut dimensions = vec![0usize; scopes.nodes.len()];
596            for (&position, variants) in &topology.sites {
597                let neighbors = variants
598                    .iter()
599                    .flat_map(|variant| {
600                        variant
601                            .graph
602                            .neighbors(IVec3::from_array(position))
603                            .into_iter()
604                            .map(|block| block.pos().to_array())
605                    })
606                    .collect::<BTreeSet<_>>();
607                let observed = &mut dimensions[owners[&position]];
608                *observed += neighbors.len()
609                    + usize::from(variants.iter().any(|variant| {
610                        variant
611                            .graph
612                            .neighbors(IVec3::from_array(position))
613                            .is_empty()
614                    }));
615                if *observed > limits.max_local_columns {
616                    return Err(StabilizerError::ResourceLimited {
617                        phase: "local ZX columns",
618                        observed: *observed,
619                        limit: limits.max_local_columns,
620                    }
621                    .into());
622                }
623            }
624        }
625        // Keep authored instance boundaries after consuming the local ZX views.
626        // Instance ids are local to this source plan; they never identify equal
627        // physical owners in different instances.
628        let mut modules = HashMap::new();
629        let module_sites = topology
630            .sites
631            .keys()
632            .map(|&position| {
633                let next = modules.len();
634                (position, *modules.entry(owners[&position]).or_insert(next))
635            })
636            .collect();
637        let mut columns = BTreeMap::new();
638        let mut groups = vec![Vec::<BooleanRow>::new(); scopes.nodes.len()];
639        let mut seams = vec![BTreeMap::new(); scopes.nodes.len()];
640        let mut local = BTreeMap::new();
641        for (&position, variants) in &topology.sites {
642            crate::CancellationToken::checkpoint()?;
643            let owner = owners[&position];
644            let mut tensors = BTreeMap::<Vec<(usize, DecisionId)>, DecisionId>::new();
645            let mut local_variants = Vec::new();
646            let mut local_keys = HashMap::new();
647            for variant in variants {
648                let zx = Arc::new(ZXGraph::from_local_block_neighborhood(
649                    &variant.graph,
650                    IVec3::from_array(position),
651                )?);
652                let node = *zx
653                    .node_at(IVec3::from_array(position))
654                    .expect("local variant includes its block");
655                for terms in zx.local_stabilizer_flow_supports_for_node_kind(node.id, node.kind) {
656                    let mut row = BooleanRow::default();
657                    for (column, mut pauli) in terms {
658                        let key = if column < zx.nodes().len() {
659                            if zx.neighbor_edges(node.id).next().is_some() {
660                                continue;
661                            }
662                            SurfaceColumn::Node(position)
663                        } else {
664                            let edge = &zx.edges()[column - zx.nodes().len()];
665                            debug_assert_eq!(edge.n1, node.id);
666                            if edge.hadamard && edge.n1 > edge.n2 {
667                                pauli = pauli.flip();
668                            }
669                            SurfaceColumn::Edge(position, zx.nodes()[edge.n2].pos.to_array())
670                        };
671                        let next = columns.len();
672                        let column = *columns.entry(key).or_insert(next);
673                        for (index, axis) in [Pauli::X, Pauli::Z].into_iter().enumerate() {
674                            if pauli & axis {
675                                row.set(2 * column + index, ONE);
676                            }
677                        }
678                    }
679                    let active = tensors.entry(row.into_terms()).or_insert(ZERO);
680                    *active = topology
681                        .diagram
682                        .apply(BooleanOp::Or, *active, variant.guard)?;
683                }
684                for (neighbor, edge_id) in zx.neighbor_edges(node.id) {
685                    let other = zx.nodes()[neighbor].pos.to_array();
686                    if position >= other {
687                        continue;
688                    }
689                    let parent = scopes.common_ancestor(owner, owners[&other]);
690                    let hadamard = seams[parent].entry((position, other)).or_insert(ZERO);
691                    if zx.edge_by_id(edge_id).hadamard {
692                        *hadamard =
693                            topology
694                                .diagram
695                                .apply(BooleanOp::Or, *hadamard, variant.guard)?;
696                    }
697                }
698                let mut neighbors = zx
699                    .neighbor_edges(node.id)
700                    .map(|(neighbor, edge)| {
701                        (
702                            zx.nodes()[neighbor].pos.to_array(),
703                            zx.edge_by_id(edge).hadamard,
704                        )
705                    })
706                    .collect::<Vec<_>>();
707                neighbors.sort_unstable();
708                let key = (node.kind, node.role, neighbors);
709                if let Some(&index) = local_keys.get(&key) {
710                    let previous: &mut SurfaceTopology = &mut local_variants[index];
711                    previous.guard =
712                        topology
713                            .diagram
714                            .apply(BooleanOp::Or, previous.guard, variant.guard)?;
715                } else {
716                    local_keys.insert(key, local_variants.len());
717                    local_variants.push(SurfaceTopology {
718                        guard: variant.guard,
719                        zx,
720                        node: node.id,
721                    });
722                }
723            }
724            if local_variants.iter().any(|variant: &SurfaceTopology| {
725                variant.zx.neighbor_edges(variant.node).next().is_none()
726            }) {
727                let next = columns.len();
728                columns.entry(SurfaceColumn::Node(position)).or_insert(next);
729            }
730            groups[owner].extend(
731                tensors
732                    .into_iter()
733                    .map(|(bits, active)| BooleanRow::from_terms(active, bits)),
734            );
735            local.insert(position, local_variants);
736        }
737        // Keep source tensor and seam row priority. Readout reduction sweeps
738        // physical columns latest layer first, so a named readout keeps later
739        // support only when no earlier witness exists: it is not read through a
740        // later unnamed record, such as a repeated merge between the same
741        // patches. Within a layer it sweeps along the longer spatial extent.
742        topology.diagram.charge(upper_bound)?;
743        let mut axes = [0, 1];
744        if let Some((x, y, _)) = topology.source.spans() {
745            let spans = [x, y];
746            axes.sort_by_key(|&axis| {
747                (
748                    std::cmp::Reverse(
749                        i64::from(*spans[axis].end()) - i64::from(*spans[axis].start()),
750                    ),
751                    axis,
752                )
753            });
754        }
755        topology.diagram.charge(
756            columns
757                .len()
758                .saturating_mul(columns.len().max(1).ilog2() as usize + 1),
759        )?;
760        let mut ordered = columns.iter_mut().collect::<Vec<_>>();
761        ordered.sort_unstable_by_key(|(key, old)| {
762            let position = match key {
763                SurfaceColumn::Node(position) | SurfaceColumn::Edge(position, _) => position,
764            };
765            (
766                std::cmp::Reverse(position[2]),
767                axes.map(|axis| position[axis]),
768                **old,
769            )
770        });
771        let mut remap = vec![0; ordered.len()];
772        for (new, (_, old)) in ordered.into_iter().enumerate() {
773            remap[*old] = new;
774            *old = new;
775        }
776        let mut scope_order = scopes.lexical_order();
777        for &scope in &scope_order {
778            for row in &mut groups[scope] {
779                let count = row.terms().len();
780                topology
781                    .diagram
782                    .charge(count.saturating_mul(count.max(1).ilog2() as usize + 1))?;
783                let active = row.active;
784                let mut terms = std::mem::take(row).into_terms();
785                for (column, _) in &mut terms {
786                    *column = 2 * remap[*column / 2] + *column % 2;
787                }
788                terms.sort_unstable_by_key(|&(column, _)| column);
789                *row = BooleanRow::from_terms(active, terms);
790            }
791        }
792        // Stable sorting retains component-wise lexical order among peers.
793        // Whole path strings differ here: "a__z" sorts after "a0__z".
794        scope_order.sort_by_key(|&scope| std::cmp::Reverse(scopes.nodes[scope].depth));
795        let mut column_rank = vec![0; columns.len()];
796        for (rank, &column) in columns.values().enumerate() {
797            column_rank[column] = rank;
798        }
799        let mut root = Vec::new();
800        for owner in scope_order {
801            let rows = std::mem::take(&mut groups[owner]);
802            let mut edges = std::mem::take(&mut seams[owner])
803                .into_iter()
804                .collect::<Vec<_>>();
805            // Close lower layers first, as in the ordinary module projector.
806            edges.sort_by_key(|((a, b), _)| (a[2].max(b[2]), *a, *b));
807            let seams = edges
808                .into_iter()
809                .map(|((left, right), hadamard)| {
810                    let lhs = columns[&SurfaceColumn::Edge(left, right)];
811                    let rhs = columns[&SurfaceColumn::Edge(right, left)];
812                    (lhs, rhs, hadamard)
813                })
814                .collect::<Vec<_>>();
815            let cache = relations.get(scopes.nodes[owner].path);
816            let rows = if let Some(cache) = cache {
817                cache.reduce(rows, seams, &column_rank, &mut topology.diagram, limits)?
818            } else {
819                crate::summary::relations::close_seams(rows, &seams, &mut topology.diagram, limits)?
820            };
821            if let Some(parent) = scopes.nodes[owner].parent {
822                groups[parent].extend(rows);
823            } else {
824                root = rows;
825            }
826        }
827        Ok(Self {
828            topology,
829            columns,
830            local,
831            rows: root,
832            surfaces: Vec::new(),
833            limits,
834            module_sites,
835            next_readout_collection: next_readout_collection(0, limits.max_witness_nodes),
836        })
837    }

840    /// same source-outcome coordinate system. Earlier corrected parities remain
841    /// explicit folds; their physical member rows are never expanded again.
842    /// Consumes the working relation before sharing decision ids with physical
843    /// lowering. The result retains local incidence instead of full ZX graphs.
844    ///
845    /// # Errors
846    ///
847    /// Returns an error if causal readout planning or resource accounting fails.
848    pub fn plan_readouts(mut self) -> Result<GuardedReadoutPlan, BlockGraphError> {
849        self.plan_causal_readouts::<true, true>()?;
850        self.into_readout_plan()
851    }

853    fn into_readout_plan(self) -> Result<GuardedReadoutPlan, BlockGraphError> {
854        let selectors = self
855            .topology
856            .resolves
857            .iter()
858            .copied()
859            .collect::<HashMap<_, _>>();
860        let local = self
861            .local
862            .into_iter()
863            .map(|(position, variants)| {
864                let selector = selectors.get(&IVec3::from_array(position)).copied();
865                let variants = variants
866                    .into_iter()
867                    .map(|variant| {
868                        let arms: Box<[LocalArm]> = variant
869                            .zx
870                            .neighbor_edges(variant.node)
871                            .map(|(neighbor, edge_id)| {
872                                let target = variant.zx.nodes()[neighbor].pos;
873                                LocalArm {
874                                    column: self.columns
875                                        [&SurfaceColumn::Edge(position, target.to_array())],
876                                    target,
877                                    reversed: variant.node > neighbor,
878                                    hadamard: variant.zx.edge_by_id(edge_id).hadamard,
879                                }
880                            })
881                            .collect();
882                        LocalTopology {
883                            guard: variant.guard,
884                            kind: variant.zx.nodes()[variant.node].kind,
885                            center: arms
886                                .is_empty()
887                                .then(|| self.columns[&SurfaceColumn::Node(position)]),
888                            arms,
889                            selector,
890                        }
891                    })
892                    .collect();
893                (position, variants)
894            })
895            .collect();
896        let mut columns = vec![SurfaceColumn::Node([0; 3]); self.columns.len()];
897        for (column, index) in self.columns {
898            columns[index] = column;
899        }
900        let mut plan = GuardedReadoutPlan {
901            topology: self.topology,
902            surfaces: self.surfaces,
903            columns,
904            local,
905            max_witness_nodes: self.limits.max_witness_nodes,
906            query_cache: Default::default(),
907            query_cache_terms: 0,
908            fragments: Vec::new(),
909            fragment_uses: Vec::new(),
910        };
911        plan.share_module_queries(&self.module_sites)?;
912        Ok(plan)
913    }

915    // The unreclaimed instantiation is used only by the differential test.
916    fn plan_causal_readouts<const RECLAIM: bool, const DEFER_WITNESSES: bool>(
917        &mut self,
918    ) -> Result<(), BlockGraphError> {
919        self.plan_causal_readouts_with_fast_path::<RECLAIM, DEFER_WITNESSES, true>()
920    }
921
922    fn plan_causal_readouts_with_fast_path<
923        const RECLAIM: bool,
924        const DEFER_WITNESSES: bool,
925        const FACTORED: bool,
926    >(
927        &mut self,
928    ) -> Result<(), BlockGraphError> {
929        let actions = self.topology.source.actions();
930        let feedback_columns = self.index_feedbacks(&actions)?;
931        let representative = self.topology.project(ONE)?;
932        let zx = ZXGraph::from_block_graph_for_analysis(&representative)?;
933        let mut names = Vec::new();
934        let mut crossing = None;
935        let mut basis = std::mem::take(&mut self.rows);
936        let physical_columns = self.columns.len() * 2;
937        let mut named_coordinates = HashMap::<usize, Vec<(usize, DecisionId)>>::new();
938        for action in zx.action_graph().ordered_nodes() {
939            self.topology.diagram.charge(1)?;
940            let Action::Measure { name, target } = &action.action else {
941                continue;
942            };
943            let axis = match action.measurement {
944                // A Y cap's tensor equates its X and Z bits. Selective fills
945                // likewise restrict support to I or the selected axis: use a
946                // linear coordinate that is one on either permitted axis.
947                Some(MeasurementObservable::Concrete(PauliBasis::X | PauliBasis::Y))
948                | Some(MeasurementObservable::Selective(crate::SelectiveKind::XY)) => Pauli::X,
949                Some(MeasurementObservable::Concrete(PauliBasis::Z))
950                | Some(MeasurementObservable::Selective(crate::SelectiveKind::YZ)) => Pauli::Z,
951                Some(MeasurementObservable::Selective(crate::SelectiveKind::XZ)) => Pauli::Y,
952                None => {
953                    return Err(StabilizerError::MeasurementSurfaceUnavailable {
954                        mvar: name.clone(),
955                    }
956                    .into());
957                }
958            };
959            if let MeasureTarget::Node(pos) = target {
960                let variants = &self.local[&pos.to_array()];
961                self.topology.diagram.charge(variants.len())?;
962                if variants.iter().any(|variant| {
963                    matches!(
964                        variant.zx.nodes()[variant.node].kind,
965                        NodeKind::X | NodeKind::Z
966                    ) && variant.zx.nodes()[variant.node].kind.cross_pauli() == axis
967                }) {
968                    crossing = Some(name.clone());
969                }
970            }
971            let target = self.measurement_column(*target, &zx)?;
972            let coordinate = physical_columns + names.len();
973            for axis in axis.iter_xz() {
974                for (column, weight) in self.component_terms(target, axis) {
975                    named_coordinates
976                        .entry(column)
977                        .or_default()
978                        .push((coordinate, weight));
979                }
980            }
981            names.push((name.clone(), coordinate));
982        }
983        for row in &mut basis {
984            self.topology.diagram.charge(row.terms().len())?;
985            let mut values = BTreeMap::new();
986            for &(column, bit) in row.terms() {
987                if let Some(coordinates) = named_coordinates.get(&column) {
988                    self.topology.diagram.charge(coordinates.len())?;
989                    for &(coordinate, weight) in coordinates {
990                        let value = values.entry(coordinate).or_insert(ZERO);
991                        let bit = self.topology.diagram.apply(BooleanOp::And, bit, weight)?;
992                        *value = self.topology.diagram.apply(BooleanOp::Xor, *value, bit)?;
993                    }
994                }
995            }
996            for (coordinate, value) in values {
997                row.set(coordinate, value);
998            }
999        }
1000        if let Some(name) = crossing {
1001            basis = self.crossing_readout_coordinates(basis, &zx, &names, &name)?;
1002        }
1003        let all_resolved = self
1004            .topology
1005            .resolves
1006            .iter()
1007            .map(|(position, _)| position.to_array())
1008            .collect();
1009        // Output exclusion is permanent. Selective fills accumulate as their
1010        // controls become known; a previously unknown site was required to be
1011        // identity, which already satisfies either future fill. Keep the source
1012        // relation separately for terminal and remaining logical readouts.
1013        let mut causal_basis = if names.is_empty() {
1014            Vec::new()
1015        } else {
1016            self.charge_row_copy(&basis)?;
1017            let mut rows = basis.clone();
1018            self.exclude_outputs(&mut rows, &zx)?;
1019            if RECLAIM {
1020                // An unresolved site is later restricted to identity, which
1021                // already satisfies either selector-dependent fill. Apply all
1022                // fills once; subsequent availability constraints only relax.
1023                self.fill_selective(&mut rows, &all_resolved)?;
1024            }
1025            rows
1026        };
1027        self.topology.diagram.charge(self.columns.len())?;
1028        let mut columns_by_site = HashMap::<[i32; 3], Vec<usize>>::new();
1029        for (&column, &index) in &self.columns {
1030            let position = match column {
1031                SurfaceColumn::Node(pos) | SurfaceColumn::Edge(pos, _) => pos,
1032            };
1033            columns_by_site.entry(position).or_default().push(index);
1034        }
1035        let names_set = names
1036            .iter()
1037            .map(|(name, _)| name.clone())
1038            .collect::<BTreeSet<_>>();
1039        let mut known = self
1040            .topology
1041            .variables
1042            .iter()
1043            .filter_map(|variable| match variable {
1044                crate::GuardedVariable::Outcome(name) if !names_set.contains(name) => {
1045                    Some(name.clone())
1046                }
1047                _ => None,
1048            })
1049            .collect::<BTreeSet<_>>();
1050        let mut availability: Option<GuardedAvailability> = None;
1051        let mut available_values = None;
1052        let mut pending_resolves = Vec::new();
1053        let mut pending_sites = Vec::new();
1054        let mut pending_feedbacks = Vec::new();
1055        let mut available_rows = BooleanRowSpace::default();
1056        let mut readout_pivots = GuardedPivots::default();
1057        let mut readout_times = known
1058            .iter()
1059            .map(|name| (name.clone(), i64::MIN))
1060            .collect::<HashMap<_, _>>();
1061        while names.iter().any(|(name, _)| !known.contains(name)) {
1062            crate::CancellationToken::checkpoint()?;
1063            let mut timing = self.readout_timing(&names, &readout_times, &actions)?;
1064            if available_values.is_none() {
1065                self.topology.diagram.charge(
1066                    self.topology
1067                        .resolves
1068                        .len()
1069                        .saturating_add(self.local.len())
1070                        .saturating_add(actions.len()),
1071                )?;
1072                pending_resolves.extend(0..self.topology.resolves.len());
1073                pending_sites.extend(
1074                    self.local.iter().filter_map(|(&position, variants)| {
1075                        (variants.len() > 1).then_some(position)
1076                    }),
1077                );
1078                pending_feedbacks.extend(actions.iter().enumerate().filter_map(
1079                    |(index, action)| {
1080                        matches!(
1081                            action,
1082                            Action::Feedback {
1083                                condition: Some(_),
1084                                ..
1085                            }
1086                        )
1087                        .then_some(index)
1088                    },
1089                ));
1090                available_values = Some(self.topology.availability(&known)?);
1091            }
1092            let available_values = available_values.as_mut().expect("availability initialized");
1093            self.topology
1094                .refresh_availability(available_values, &known)?;
1095            self.topology
1096                .diagram
1097                .charge(pending_resolves.len() + pending_sites.len() + pending_feedbacks.len())?;
1098            let mut newly_resolved = BTreeSet::new();
1099            let mut pending_count = 0;
1100            for index in 0..pending_resolves.len() {
1101                let ordinal = pending_resolves[index];
1102                let (position, root) = self.topology.resolves[ordinal];
1103                if available_values.available(root, &mut self.topology.diagram)? {
1104                    if !RECLAIM {
1105                        newly_resolved.insert(position.to_array());
1106                    }
1107                } else {
1108                    pending_resolves[pending_count] = ordinal;
1109                    pending_count += 1;
1110                }
1111            }
1112            pending_resolves.truncate(pending_count);
1113            let mut pending_count = 0;
1114            for index in 0..pending_sites.len() {
1115                let position = pending_sites[index];
1116                let mut ready = true;
1117                for variant in &self.local[&position] {
1118                    if !available_values.available(variant.guard, &mut self.topology.diagram)? {
1119                        ready = false;
1120                        break;
1121                    }
1122                }
1123                if !ready {
1124                    pending_sites[pending_count] = position;
1125                    pending_count += 1;
1126                }
1127            }
1128            pending_sites.truncate(pending_count);
1129            let unavailable = pending_sites
1130                .iter()
1131                .copied()
1132                .chain(
1133                    pending_resolves
1134                        .iter()
1135                        .map(|&index| self.topology.resolves[index].0.to_array()),
1136                )
1137                .collect::<BTreeSet<_>>();
1138            let mut pending_count = 0;
1139            for index in 0..pending_feedbacks.len() {
1140                let ordinal = pending_feedbacks[index];
1141                let Action::Feedback {
1142                    condition: Some(condition),
1143                    ..
1144                } = &actions[ordinal]
1145                else {
1146                    unreachable!("pending feedbacks have a condition")
1147                };
1148                let root = self.topology.condition(condition);
1149                if !available_values.available(root, &mut self.topology.diagram)? {
1150                    pending_feedbacks[pending_count] = ordinal;
1151                    pending_count += 1;
1152                }
1153            }
1154            pending_feedbacks.truncate(pending_count);
1155            let feedback_absent = &pending_feedbacks;
1156            if !RECLAIM {
1157                self.fill_selective(&mut causal_basis, &newly_resolved)?;
1158            }
1159            let pending = unavailable
1160                .iter()
1161                .copied()
1162                .map(AvailabilityKey::Site)
1163                .chain(
1164                    feedback_absent
1165                        .iter()
1166                        .copied()
1167                        .map(AvailabilityKey::Feedback),
1168                )
1169                .collect::<BTreeSet<_>>();
1170            let mut constraints = Vec::new();
1171            if !RECLAIM || availability.is_none() {
1172                let mut blocked_columns = Vec::new();
1173                for position in unavailable {
1174                    let columns = &columns_by_site[&position];
1175                    self.topology.diagram.charge(columns.len())?;
1176                    blocked_columns.extend(columns.iter().map(|&index| (index, position)));
1177                }
1178                blocked_columns.sort_unstable_by_key(|&(index, _)| index);
1179                for (index, position) in blocked_columns {
1180                    for axis in 0..2 {
1181                        // Multiplex permits Z on its live output, but its
1182                        // physical arms still require known local geometry.
1183                        constraints.push((
1184                            AvailabilityKey::Site(position),
1185                            vec![(2 * index + axis, ONE)],
1186                        ));
1187                    }
1188                }
1189                for &ordinal in feedback_absent {
1190                    let Action::Feedback { targets, .. } = &actions[ordinal] else {
1191                        unreachable!("selector columns name selective nodes")
1192                    };
1193                    self.topology.diagram.charge(targets.len())?;
1194                    let mut terms = Vec::new();
1195                    for target in targets {
1196                        self.topology
1197                            .diagram
1198                            .charge(self.local[&target.target.to_array()].len())?;
1199                        for variant in &self.local[&target.target.to_array()] {
1200                            for index in self.feedback_support(target, variant)? {
1201                                terms.push((index, variant.guard));
1202                            }
1203                        }
1204                    }
1205                    // A source Pauli product can commute through even target
1206                    // overlaps. Preserve every occurrence in its one linear
1207                    // functional instead of blocking each component alone.
1208                    constraints.push((AvailabilityKey::Feedback(ordinal), terms));
1209                }
1210            }
1211            if RECLAIM {
1212                if let Some(availability) = &mut availability {
1213                    availability.release(
1214                        &pending,
1215                        &readout_pivots,
1216                        &mut available_rows,
1217                        &mut self.topology.diagram,
1218                    )?;
1219                } else {
1220                    let (state, rows) = GuardedAvailability::new(
1221                        std::mem::take(&mut causal_basis),
1222                        &constraints,
1223                        physical_columns + names.len(),
1224                        &mut self.topology.diagram,
1225                    )?;
1226                    availability = Some(state);
1227                    available_rows = rows;
1228                }
1229            }
1230            let unknown = names
1231                .iter()
1232                .filter(|(name, _)| !known.contains(name))
1233                .collect::<Vec<_>>();
1234            let mut unknown_columns = vec![false; names.len()];
1235            for (_, column) in &unknown {
1236                unknown_columns[*column - physical_columns] = true;
1237            }
1238            // The quotient alone fixes an arbitrary physical coset. Restore
1239            // accepted relations before choosing an early equivalent witness.
1240            let restored = if RECLAIM && !readout_pivots.rows.is_empty() {
1241                let rows = available_rows
1242                    .rows()
1243                    .chain(readout_pivots.rows.rows())
1244                    .cloned()
1245                    .collect::<Vec<_>>();
1246                self.charge_row_copy(&rows)?;
1247                Some(BooleanRowSpace::new(rows, &mut self.topology.diagram)?)
1248            } else {
1249                None
1250            };
1251            let source_rows = restored.as_ref().unwrap_or(&available_rows);
1252            let relevant = if RECLAIM && FACTORED {
1253                Some(self.named_component_rows(source_rows, &unknown)?)
1254            } else {
1255                None
1256            };
1257            // Restored relations make the named projection non-injective.
1258            let mut factored = if let Some(ids) = relevant.as_ref().filter(|_| restored.is_none()) {
1259                self.injective_named_candidates(
1260                    &available_rows,
1261                    ids,
1262                    &unknown,
1263                    physical_columns,
1264                    physical_columns + names.len(),
1265                )?
1266            } else {
1267                None
1268            };
1269            let reduced = if factored.is_some() {
1270                BooleanRowSpace::default()
1271            } else {
1272                let allowed = if RECLAIM {
1273                    let rows = if let Some(ids) = &relevant {
1274                        let mut rows = Vec::with_capacity(ids.len());
1275                        for &id in ids {
1276                            let row = source_rows.row(id).expect("selected row exists");
1277                            self.topology.diagram.charge(row.terms().len() + 1)?;
1278                            rows.push(row.clone());
1279                        }
1280                        rows
1281                    } else {
1282                        let rows = source_rows.rows().cloned().collect::<Vec<_>>();
1283                        self.charge_row_copy(&rows)?;
1284                        rows
1285                    };
1286                    for row in &rows {
1287                        timing.include(row, &mut self.topology.diagram)?;
1288                    }
1289                    BooleanRowSpace::new(rows, &mut self.topology.diagram)?
1290                } else {
1291                    self.charge_row_copy(&causal_basis)?;
1292                    let mut allowed =
1293                        BooleanRowSpace::new(causal_basis.clone(), &mut self.topology.diagram)?;
1294                    for (_, terms) in constraints {
1295                        allowed.eliminate_linear(&terms, &mut self.topology.diagram)?;
1296                    }
1297                    for row in allowed.rows() {
1298                        timing.include(row, &mut self.topology.diagram)?;
1299                    }
1300                    allowed
1301                };
1302                // Keep only named pivots. Physical kernel pivots normalize
1303                // those rows, but their own forms are never used here.
1304                self.reduce_named_rows::<RECLAIM>(
1305                    allowed,
1306                    unknown.iter().map(|(_, column)| *column),
1307                    &timing.columns,
1308                )?
1309            };
1310            let mut candidates = Vec::new();
1311            for (index, (name, wanted)) in unknown.iter().enumerate() {
1312                self.topology.diagram.charge(1)?;
1313                let row = if let Some(factored) = &mut factored {
1314                    let Some(row) = factored[index].take() else {
1315                        continue;
1316                    };
1317                    row
1318                } else {
1319                    self.named_row(&reduced, *wanted)?
1320                };
1321                let named =
1322                    Self::named_terms(&row, physical_columns, physical_columns + names.len());
1323                if row.active != ONE
1324                    || named.iter().any(|&(column, _)| {
1325                        column != *wanted && unknown_columns[column - physical_columns]
1326                    })
1327                {
1328                    continue;
1329                }
1330                let mut ready = true;
1331                for &(_, bit) in row.terms() {
1332                    if !available_values.available(bit, &mut self.topology.diagram)? {
1333                        ready = false;
1334                        break;
1335                    }
1336                }
1337                if !ready {
1338                    continue;
1339                }
1340                let folds = named
1341                    .iter()
1342                    .filter(|&&(column, _)| !unknown_columns[column - physical_columns])
1343                    .map(|&(column, bit)| (names[column - physical_columns].0.clone(), bit))
1344                    .collect();
1345                let feedbacks = self.feedbacks::<RECLAIM>(&row, &actions, &feedback_columns)?;
1346                let mut time = timing.row(&row, *wanted, &mut self.topology.diagram)?;
1347                for &(_, guard) in &feedbacks {
1348                    time = time.max(timing.decision(guard, &mut self.topology.diagram)?);
1349                }
1350                candidates.push((
1351                    name.clone(),
1352                    GuardedSurface {
1353                        kind: GuardedSurfaceKind::Readout {
1354                            name: name.clone(),
1355                            folds,
1356                        },
1357                        feedbacks,
1358                        row,
1359                    },
1360                    time,
1361                ));
1362            }
1363            // The final budget check may remap the diagram. Query images and
1364            // unselected local candidate rows cannot retain unlisted roots.
1365            drop((factored, reduced, restored, timing));
1366            if candidates.is_empty() {
1367                return Err(StabilizerError::UnavailableControlParity {
1368                    mvar: unknown[0].0.clone(),
1369                    deadline: i64::MAX,
1370                }
1371                .into());
1372            }
1373            available_values.learn(
1374                candidates.iter().map(|(name, _, _)| name.as_str()),
1375                &mut self.topology.diagram,
1376            )?;
1377            let reclaim = RECLAIM && candidates.len() < unknown.len();
1378            let mut reclaimed = Vec::new();
1379            for (name, surface, time) in candidates {
1380                readout_times.insert(name.clone(), time);
1381                known.insert(name);
1382                if reclaim {
1383                    self.topology.diagram.charge(surface.row.terms().len())?;
1384                    let mut row = surface.row.clone();
1385                    // Candidate coefficients were simplified on the source
1386                    // domain. They represent a source relation only there.
1387                    row.active = self.topology.domain;
1388                    reclaimed.push(row);
1389                }
1390                self.surfaces.push(surface);
1391            }
1392            if reclaim {
1393                self.reclaim_readouts(&mut available_rows, &mut readout_pivots, reclaimed)?;
1394            }
1395            self.check_readout_roots(
1396                basis
1397                    .iter_mut()
1398                    .chain(causal_basis.iter_mut())
1399                    .flat_map(BooleanRow::decisions_mut)
1400                    .chain(available_rows.decisions_mut())
1401                    .chain(readout_pivots.rows.decisions_mut())
1402                    .chain(
1403                        availability
1404                            .iter_mut()
1405                            .flat_map(|state| state.pivots.rows.decisions_mut()),
1406                    ),
1407            )?;
1408        }
1409        drop((
1410            causal_basis,
1411            availability,
1412            available_rows,
1413            readout_pivots,
1414            available_values,
1415        ));
1416        let (input_nodes, resource_nodes) = authored_boundary_columns(&zx);
1417        let mut output_ports = zx.output_ports();
1418        if DEFER_WITNESSES {
1419            // Stable terminal choices inspect only these explicit constraints.
1420            // Reconstruct raw physical support before feedback and final kernel
1421            // reduction; a zero-core private row must remain live until then.
1422            self.topology.diagram.charge(
1423                zx.nodes()
1424                    .len()
1425                    .saturating_add(self.topology.resolves.len()),
1426            )?;
1427            // Output pivots and selective fills never query input or protected
1428            // resource columns. Keep their exact payload on the witness tape
1429            // until input normalization actually needs it.
1430            let positions = output_ports
1431                .iter()
1432                .copied()
1433                .chain(
1434                    zx.nodes()
1435                        .iter()
1436                        .filter(|node| node.role == crate::PortRole::Multiplex)
1437                        .map(|node| node.pos),
1438                )
1439                .chain(self.topology.resolves.iter().map(|&(position, _)| position));
1440            let core = positions
1441                .flat_map(|position| {
1442                    self.node_components(position.to_array())
1443                        .into_iter()
1444                        .flatten()
1445                        .map(|(column, _)| column)
1446                })
1447                .chain(names.iter().map(|(_, column)| *column))
1448                .collect::<Vec<_>>();
1449            self.topology
1450                .diagram
1451                .start_witness_tape(core, self.limits.max_witness_nodes)?;
1452            for row in &mut basis {
1453                row.defer_coefficients(&mut self.topology.diagram)?;
1454            }
1455        }
1456        // Remove the named coordinates before resolving terminal measurements.
1457        // This is the kernel used by RuntimeStabilizerBasis::for_output_frames.
1458        let mut basis = BooleanRowSpace::new(basis, &mut self.topology.diagram)?;
1459        for (_, coordinate) in &names {
1460            basis.eliminate_column(*coordinate, &mut self.topology.diagram)?;
1461        }
1462        self.check_readout_roots(basis.decisions_mut())?;
1463        self.topology.diagram.charge(self.topology.resolves.len())?;
1464        for index in 0..self.topology.resolves.len() {
1465            self.fill_selective_site(&mut basis, index)?;
1466            // Terminal planning owns every live row here. Reclaim temporary
1467            // fill products before the next site can exhaust the arena.
1468            self.check_readout_roots(basis.decisions_mut())?;
1469        }
1470        self.check_readout_roots(basis.decisions_mut())?;
1471        let mut output_rows = BooleanRowSpace::default();
1472        let mut output_axes = Vec::new();
1473        let mut output_gates = HashMap::new();
1474        if RECLAIM {
1475            // Output order may change the chosen free axes under conditional
1476            // rank. Preserve the authored pivot activations using only the
1477            // small output projection, then reorder the physical witnesses.
1478            let mut projections = Vec::new();
1479            for &port in &output_ports {
1480                let axes = if zx
1481                    .node_at(port)
1482                    .expect("output ports come from this ZX graph")
1483                    .role
1484                    == crate::PortRole::Multiplex
1485                {
1486                    1
1487                } else {
1488                    2
1489                };
1490                projections.extend(
1491                    [Pauli::X, Pauli::Z]
1492                        .into_iter()
1493                        .take(axes)
1494                        .map(|axis| (port, axis)),
1495                );
1496            }
1497            let output_columns = projections
1498                .iter()
1499                .flat_map(|&(port, axis)| {
1500                    self.component_terms(SurfaceColumn::Node(port.to_array()), axis)
1501                        .into_iter()
1502                        .map(|(column, _)| column)
1503                })
1504                .collect::<BTreeSet<_>>();
1505            let mut projected = BooleanRowSpace::default();
1506            for row in basis.rows() {
1507                self.topology.diagram.charge(row.terms().len())?;
1508                projected.push(
1509                    BooleanRow::from_terms(
1510                        row.active,
1511                        row.terms()
1512                            .iter()
1513                            .copied()
1514                            .filter(|(column, _)| output_columns.contains(column)),
1515                    ),
1516                    &mut self.topology.diagram,
1517                )?;
1518            }
1519            for (port, axis) in projections {
1520                let terms = self.component_terms(SurfaceColumn::Node(port.to_array()), axis);
1521                let pivot = projected.eliminate_linear(&terms, &mut self.topology.diagram)?;
1522                output_gates.insert((port, axis), pivot.active);
1523                self.check_readout_roots(
1524                    basis
1525                        .decisions_mut()
1526                        .chain(projected.decisions_mut())
1527                        .chain(output_gates.values_mut()),
1528                )?;
1529            }
1530            self.topology.diagram.charge(
1531                output_ports
1532                    .len()
1533                    .saturating_mul(output_ports.len().max(1).ilog2() as usize + 1),
1534            )?;
1535            output_ports.sort_unstable_by_key(|port| {
1536                std::cmp::Reverse(
1537                    self.node_components(port.to_array())
1538                        .into_iter()
1539                        .flatten()
1540                        .map(|(column, _)| column)
1541                        .min(),
1542                )
1543            });
1544        }
1545        for port in output_ports {
1546            for (axis, component) in [Pauli::X, Pauli::Z].into_iter().enumerate() {
1547                if axis == 1
1548                    && zx
1549                        .node_at(port)
1550                        .expect("output ports come from this ZX graph")
1551                        .role
1552                        == crate::PortRole::Multiplex
1553                {
1554                    output_axes.push((port, axis, false));
1555                    continue;
1556                }
1557                let gate = output_gates.get(&(port, component)).copied().unwrap_or(ONE);
1558                let terms = self.component_terms(SurfaceColumn::Node(port.to_array()), component);
1559                let gated = terms
1560                    .iter()
1561                    .map(|&(column, weight)| {
1562                        Ok((
1563                            column,
1564                            self.topology.diagram.apply(BooleanOp::And, weight, gate)?,
1565                        ))
1566                    })
1567                    .collect::<Result<Vec<_>, BooleanResourceError>>()?;
1568                let pivot = basis.eliminate_linear(&gated, &mut self.topology.diagram)?;
1569                // Keep the physical witnesses triangular. Expanding each
1570                // earlier row through every later pivot can make its Boolean
1571                // transport coefficients exponential. The emitted frame DAG
1572                // performs this same back-substitution on decoded residuals.
1573                if !RECLAIM {
1574                    output_rows.reduce_linear(&terms, &pivot, &mut self.topology.diagram)?;
1575                }
1576                output_axes.push((port, axis, pivot.active != ZERO));
1577                output_rows.push(pivot, &mut self.topology.diagram)?;
1578                drop((terms, gated));
1579                self.check_readout_roots(
1580                    basis
1581                        .decisions_mut()
1582                        .chain(output_rows.decisions_mut())
1583                        .chain(output_gates.values_mut()),
1584                )?;
1585            }
1586        }
1587        drop(output_gates);
1588        let (inputs, resources) = (input_nodes, resource_nodes);
1589        let components = |node: usize| {
1590            [Pauli::X, Pauli::Z]
1591                .map(|axis| (SurfaceColumn::Node(zx.nodes()[node].pos.to_array()), axis))
1592        };
1593        let inputs = inputs.into_iter().flat_map(components).collect::<Vec<_>>();
1594        // Source T resources and public Ports cannot belong to optional arms.
1595        // Multiplex Z was deliberately not pivoted as an output above.
1596        let protected = resources
1597            .into_iter()
1598            .flat_map(components)
1599            .chain(
1600                zx.nodes()
1601                    .iter()
1602                    .filter(|node| node.role == crate::PortRole::Multiplex)
1603                    .map(|node| components(node.id)[1]),
1604            )
1605            .collect::<Vec<_>>();
1606        let mut basis = basis.into_rows(&mut self.topology.diagram)?;
1607        if DEFER_WITNESSES && !basis.is_empty() {
1608            // The global kernel needs input and protected-resource projections,
1609            // not every physical coefficient. Keep the same source-ordered solve
1610            // while its row operations compose the remaining passive transfers.
1611            self.topology
1612                .diagram
1613                .charge(inputs.len().saturating_add(protected.len()))?;
1614            let mut front = Vec::new();
1615            for &(column, axis) in inputs.iter().chain(&protected) {
1616                let terms = self.component_terms(column, axis);
1617                self.topology.diagram.charge(terms.len())?;
1618                front.extend(terms.into_iter().map(|(column, _)| column));
1619            }
1620            let mut outputs = output_rows.into_rows(&mut self.topology.diagram)?;
1621            self.topology
1622                .diagram
1623                .charge(basis.len().saturating_add(outputs.len()).saturating_mul(2))?;
1624            let remaining = basis.len();
1625            basis.append(&mut outputs);
1626            self.topology
1627                .diagram
1628                .extend_witness_front(&front, &mut basis)?;
1629            outputs = basis.split_off(remaining);
1630            output_rows = BooleanRowSpace::new(outputs, &mut self.topology.diagram)?;
1631            self.check_readout_roots(
1632                basis
1633                    .iter_mut()
1634                    .flat_map(BooleanRow::decisions_mut)
1635                    .chain(output_rows.decisions_mut()),
1636            )?;
1637        }
1638        self.normalize_output_inputs(&mut basis, &mut output_rows, &inputs, &protected)?;
1639        let mut output_rows = output_rows.into_rows(&mut self.topology.diagram)?;
1640        if DEFER_WITNESSES {
1641            for index in 0..basis.len() {
1642                basis[index].expand_coefficients(&mut self.topology.diagram)?;
1643                self.check_readout_roots(
1644                    basis
1645                        .iter_mut()
1646                        .chain(output_rows.iter_mut())
1647                        .flat_map(BooleanRow::decisions_mut),
1648                )?;
1649            }
1650            for index in 0..output_rows.len() {
1651                output_rows[index].expand_coefficients(&mut self.topology.diagram)?;
1652                self.check_readout_roots(
1653                    basis
1654                        .iter_mut()
1655                        .chain(output_rows.iter_mut())
1656                        .flat_map(BooleanRow::decisions_mut),
1657                )?;
1658            }
1659            self.topology.diagram.finish_witness_tape();
1660            self.check_readout_roots(
1661                basis
1662                    .iter_mut()
1663                    .chain(output_rows.iter_mut())
1664                    .flat_map(BooleanRow::decisions_mut),
1665            )?;
1666        }
1667        let output_columns = if RECLAIM {
1668            self.index_output_axes(&output_axes)?
1669        } else {
1670            BTreeMap::new()
1671        };
1672        let mut output_rows = output_rows.into_iter();
1673        for (index, &(port, axis, active)) in output_axes.iter().enumerate() {
1674            let row = if active {
1675                output_rows
1676                    .next()
1677                    .expect("each active output retains its row")
1678            } else {
1679                BooleanRow::new(ZERO)
1680            };
1681            let folds = if RECLAIM && active {
1682                self.output_folds(&row, index, &output_axes, &output_columns)?
1683            } else {
1684                Vec::new()
1685            };
1686            let feedbacks = self.feedbacks::<RECLAIM>(&row, &actions, &feedback_columns)?;
1687            self.surfaces.push(GuardedSurface {
1688                kind: GuardedSurfaceKind::OutputFrame {
1689                    port,
1690                    basis: if axis == 0 {
1691                        PauliBasis::Z
1692                    } else {
1693                        PauliBasis::X
1694                    },
1695                    folds,
1696                },
1697                feedbacks,
1698                row,
1699            });
1700        }
1701        drop(output_columns);
1702        let input_columns = inputs
1703            .iter()
1704            .flat_map(|&(column, axis)| {
1705                self.component_terms(column, axis)
1706                    .into_iter()
1707                    .map(|(column, _)| column)
1708            })
1709            .collect::<Vec<_>>();
1710        let remaining = reduce_boolean_rows(
1711            basis,
1712            input_columns.into_iter().chain(0..physical_columns),
1713            &mut self.topology.diagram,
1714        )?;
1715        // The kernel also contains closed logical surfaces: memory and
1716        // stability readouts have no input or output Port support.
1717        for row in remaining {
1718            let feedbacks = self.feedbacks::<RECLAIM>(&row, &actions, &feedback_columns)?;
1719            self.surfaces.push(GuardedSurface {
1720                kind: GuardedSurfaceKind::LogicalReadout,
1721                feedbacks,
1722                row,
1723            });
1724        }
1725        // Physical lowering retains decision ids in its own caches. Finish the
1726        // planning stage with only live functions before those ids escape.
1727        self.collect_readout_roots(std::iter::empty())?;
1728        self.check_budget()
1729    }

1731    /// Normalize erased input operators within the selected resource class.
1732    /// Only zero-output relations that preserve prepared T resources and the
1733    /// unpivoted Multiplex Z direction can change the input representative.
1734    /// The scalar planner uses this same restricted input normalization.
1735    fn normalize_output_inputs(
1736        &mut self,
1737        remaining: &mut [BooleanRow],
1738        outputs: &mut BooleanRowSpace,
1739        inputs: &[(SurfaceColumn, Pauli)],
1740        protected: &[(SurfaceColumn, Pauli)],
1741    ) -> Result<(), BlockGraphError> {
1742        if outputs.is_empty() || inputs.is_empty() {
1743            return Ok(());
1744        }
1745        self.topology.diagram.charge(inputs.len())?;
1746        let input_columns = inputs
1747            .iter()
1748            .flat_map(|&(column, axis)| {
1749                self.component_terms(column, axis)
1750                    .into_iter()
1751                    .map(|(column, _)| column)
1752            })
1753            .collect::<BTreeSet<_>>();
1754        let mut input_support = false;
1755        for row in remaining.iter() {
1756            self.topology.diagram.charge(row.terms().len())?;
1757            if row
1758                .terms()
1759                .iter()
1760                .any(|&(column, _)| input_columns.contains(&column))
1761            {
1762                input_support = true;
1763                break;
1764            }
1765        }
1766        if !input_support {
1767            return Ok(());
1768        }
1769        self.charge_row_copy(remaining)?;
1770        let mut kernel = BooleanRowSpace::new(remaining.to_vec(), &mut self.topology.diagram)?;
1771        for &(column, axis) in protected {
1772            let terms = self.component_terms(column, axis);
1773            kernel.eliminate_linear(&terms, &mut self.topology.diagram)?;
1774            self.check_readout_roots(
1775                remaining
1776                    .iter_mut()
1777                    .flat_map(BooleanRow::decisions_mut)
1778                    .chain(outputs.decisions_mut())
1779                    .chain(kernel.decisions_mut()),
1780            )?;
1781        }
1782        for &(column, axis) in inputs {
1783            let terms = self.component_terms(column, axis);
1784            let pivot = kernel.eliminate_linear(&terms, &mut self.topology.diagram)?;
1785            outputs.reduce_linear(&terms, &pivot, &mut self.topology.diagram)?;
1786            self.check_readout_roots(
1787                remaining
1788                    .iter_mut()
1789                    .flat_map(BooleanRow::decisions_mut)
1790                    .chain(outputs.decisions_mut())
1791                    .chain(kernel.decisions_mut()),
1792            )?;
1793        }
1794        Ok(())
1795    }

1830    fn readout_timing(
1831        &mut self,
1832        names: &[(String, usize)],
1833        known: &HashMap<String, i64>,
1834        actions: &[Action],
1835    ) -> Result<ReadoutTiming, BlockGraphError> {
1836        self.topology.diagram.charge(
1837            self.topology
1838                .variables
1839                .len()
1840                .saturating_add(names.len())
1841                .saturating_add(self.columns.len().saturating_mul(2)),
1842        )?;
1843        let outcome_time = |name: &str| known.get(name).copied().unwrap_or(i64::MAX);
1844        let variables = self
1845            .topology
1846            .variables
1847            .iter()
1848            .map(|variable| match variable {
1849                crate::GuardedVariable::Outcome(name) => outcome_time(name),
1850                crate::GuardedVariable::Branch { .. } => i64::MAX,
1851            })
1852            .collect();
1853        let mut timing = ReadoutTiming {
1854            columns: vec![i64::MIN; 2 * self.columns.len() + names.len()],
1855            variables,
1856            domain: self.topology.domain,
1857            decisions: HashMap::new(),
1858        };
1859        // Structural choices require their source conditions, not a clock
1860        // inferred from the branch variable's numeric index.
1861        for (index, variable) in self.topology.variables.iter().enumerate() {
1862            if let crate::GuardedVariable::Branch { condition, .. } = variable {
1863                timing.variables[index] =
1864                    timing.decision(*condition, &mut self.topology.diagram)?;
1865            }
1866        }
1867        for (&column, &index) in &self.columns {
1868            let (left, right) = match column {
1869                SurfaceColumn::Node(position) => (position, position),
1870                SurfaceColumn::Edge(left, right) => (left, right),
1871            };
1872            let mut time = i64::from(left[2].max(right[2]));
1873            for position in [left, right] {
1874                for variant in &self.local[&position] {
1875                    time = time.max(timing.decision(variant.guard, &mut self.topology.diagram)?);
1876                }
1877            }
1878            timing.columns[2 * index] = time;
1879            timing.columns[2 * index + 1] = time;
1880        }
1881        for (name, column) in names {
1882            timing.columns[*column] = outcome_time(name);
1883        }
1884        self.topology.diagram.charge(actions.len())?;
1885        for action in actions {
1886            let Action::Feedback { targets, condition } = action else {
1887                continue;
1888            };
1889            let condition = condition
1890                .as_ref()
1891                .map_or(ONE, |expr| self.topology.condition(expr));
1892            let mut columns = HashMap::<usize, DecisionId>::new();
1893            self.topology.diagram.charge(targets.len())?;
1894            for target in targets {

1897                    .diagram

1901                    self.topology.diagram.charge(support.len())?;
1902                    for column in support {

1904                        *guard =
1905                            self.topology
1906                                .diagram
1907                                .apply(BooleanOp::Or, *guard, variant.guard)?;
1908                    }
1909                }
1910                self.topology.diagram.charge(target_columns.len())?;
1911                for (column, guard) in target_columns {
1912                    let coefficient = columns.entry(column).or_insert(ZERO);
1913                    *coefficient =
1914                        self.topology
1915                            .diagram
1916                            .apply(BooleanOp::Xor, *coefficient, guard)?;
1917                }
1918            }
1919            self.topology.diagram.charge(columns.len())?;
1920            for (column, coefficient) in columns {
1921                let guard = self
1922                    .topology
1923                    .diagram
1924                    .apply(BooleanOp::And, coefficient, condition)?;
1925                timing.columns[column] =
1926                    timing.columns[column].max(timing.decision(guard, &mut self.topology.diagram)?);
1927            }
1928        }

1932    fn reduce_named_rows<const SPARSE: bool>(
1933        &mut self,
1934        mut remaining: BooleanRowSpace,
1935        names: impl Iterator<Item = usize>,
1936        timing: &[i64],
1937    ) -> Result<BooleanRowSpace, BooleanResourceError> {
1938        let mut named = BooleanRowSpace::default();
1939        for column in names {
1940            let pivot = if SPARSE {
1941                remaining.eliminate_column_sparse(column, &mut self.topology.diagram)?
1942            } else {
1943                remaining.eliminate_column(column, &mut self.topology.diagram)?
1944            };
1945            if pivot.active != ZERO {
1946                named.reduce_column(column, &pivot, &mut self.topology.diagram)?;
1947                named.push(pivot, &mut self.topology.diagram)?;
1948            }
1949        }
1950        // For these conservative column weights, eliminating late support
1951        // first favors an earlier witness, including the cost of known folds.
1952        // This does not optimize latency independently in each Boolean branch.
1953        let columns = remaining.columns().len();
1954        self.topology
1955            .diagram
1956            .charge(columns.saturating_mul(columns.max(1).ilog2() as usize + 2))?;
1957        let mut physical_and_known = remaining.columns().collect::<Vec<_>>();
1958        physical_and_known.sort_unstable_by_key(|&column| {
1959            (
1960                std::cmp::Reverse(timing.get(column).copied().unwrap_or(i64::MIN)),
1961                column,
1962            )
1963        });
1964        for column in physical_and_known {
1965            if remaining.is_empty() {
1966                break;
1967            }
1968            let pivot = if SPARSE {
1969                remaining.eliminate_column_sparse(column, &mut self.topology.diagram)?
1970            } else {
1971                remaining.eliminate_column(column, &mut self.topology.diagram)?
1972            };
1973            named.reduce_column(column, &pivot, &mut self.topology.diagram)?;
1974        }
1975        Ok(named)
1976    }

1978    /// Select algebraic components after constraints have coupled their rows.
1979    /// Keep every physical and known-name column, including private kernels;
1980    /// only the temporary named solve omits components with no unknown name.
1981    fn named_component_rows(
1982        &mut self,
1983        available: &BooleanRowSpace,
1984        unknown: &[&(String, usize)],
1985    ) -> Result<Vec<usize>, BooleanResourceError> {
1986        let mut columns = HashSet::new();
1987        let mut rows = HashSet::new();
1988        let mut pending = Vec::new();
1989        // Seed every name first so injective spaces need no physical walk.
1990        for (_, column) in unknown {
1991            self.topology.diagram.charge(1)?;
1992            columns.insert(*column);
1993            for id in available.row_ids_with_column(*column) {
1994                self.topology.diagram.charge(1)?;
1995                if rows.insert(id) {
1996                    pending.push(id);
1997                }
1998            }
1999        }
2000        while rows.len() < available.len() {
2001            let Some(id) = pending.pop() else { break };
2002            let row = available.row(id).expect("indexed row exists");
2003            self.topology.diagram.charge(row.terms().len())?;
2004            for &(column, _) in row.terms() {
2005                if columns.insert(column) {
2006                    // Removed postings have zero support on their activation.
2007                    for id in available.row_ids_with_column(column) {
2008                        self.topology.diagram.charge(1)?;
2009                        if rows.insert(id) {
2010                            pending.push(id);
2011                        }
2012                    }
2013                }
2014            }
2015        }
2016        self.topology.diagram.charge(
2017            rows.len()
2018                .saturating_mul(rows.len().max(1).ilog2() as usize + 1),
2019        )?;
2020        let mut rows = rows.into_iter().collect::<Vec<_>>();
2021        rows.sort_unstable();
2022        Ok(rows)
2023    }

2031    /// Try a named-only reduction. Private tags record each source row's exact
2032    /// combination. An empty residual certifies that named projection is
2033    /// injective, so each full witness is the unique lift of its named row.
2034    /// Restricted source domains retain the full reducer until their
2035    /// off-domain coefficient convention is factored.
2036    fn injective_named_candidates(
2037        &mut self,
2038        available: &BooleanRowSpace,
2039        source_ids: &[usize],
2040        unknown: &[&(String, usize)],
2041        physical_columns: usize,
2042        first_tag: usize,
2043    ) -> Result<Option<Vec<Option<BooleanRow>>>, BooleanResourceError> {
2044        if self.topology.domain != ONE {
2045            return Ok(None);
2046        }
2047        self.topology
2048            .diagram
2049            .charge(source_ids.len() + first_tag - physical_columns)?;
2050        let mut unknown_columns = vec![false; first_tag - physical_columns];
2051        for (_, column) in unknown {
2052            unknown_columns[*column - physical_columns] = true;
2053        }
2054        let mut skeleton = Vec::with_capacity(source_ids.len());
2055        for &id in source_ids {
2056            let source = available.row(id).expect("selected row exists");
2057            let terms = Self::named_terms(source, physical_columns, first_tag);
2058            self.topology.diagram.charge(terms.len() + 1)?;
2059            let mut row = BooleanRow::from_terms(source.active, terms.iter().copied());
2060            row.set(first_tag + id, ONE);
2061            skeleton.push(row);
2062        }
2063        let mut remaining = BooleanRowSpace::new(skeleton, &mut self.topology.diagram)?;
2064        let mut named = BooleanRowSpace::default();
2065        for (_, column) in unknown {
2066            let pivot = remaining.eliminate_column_sparse(*column, &mut self.topology.diagram)?;
2067            if pivot.active != ZERO {
2068                named.reduce_column(*column, &pivot, &mut self.topology.diagram)?;
2069                named.push(pivot, &mut self.topology.diagram)?;
2070            }
2071        }
2072        if !remaining.is_empty() {
2073            return Ok(None);
2074        }
2075        let mut result = Vec::with_capacity(unknown.len());
2076        for (_, wanted) in unknown {
2077            let row = self.named_row(&named, *wanted)?;
2078            if row.active != ONE
2079                || Self::named_terms(&row, physical_columns, first_tag)
2080                    .iter()
2081                    .any(|&(column, _)| {
2082                        column != *wanted && unknown_columns[column - physical_columns]
2083                    })
2084            {
2085                result.push(None);
2086                continue;
2087            }
2088            let diagram = &mut self.topology.diagram;
2089            let mut full = BooleanRow::new(ONE);
2090            diagram.charge(row.terms().len())?;
2091            for &(column, scale) in row.terms() {
2092                if column >= first_tag {
2093                    full.xor_scaled(
2094                        available
2095                            .row(column - first_tag)
2096                            .expect("tagged row exists"),
2097                        scale,
2098                        diagram,
2099                    )?;
2100                }
2101            }
2102            // Exact lifts must retain raw named coefficients, including off-domain values.
2103            let full_named = Self::named_terms(&full, physical_columns, first_tag);
2104            let expected_named = Self::named_terms(&row, physical_columns, first_tag);
2105            diagram.charge(full_named.len() + expected_named.len())?;
2106            if full_named != expected_named {
2107                return Ok(None);
2108            }
2109            result.push(Some(full));
2110        }
2111        Ok(Some(result))
2112    }

2190        for row in &mut rows {

2441    fn output_folds(
2442        &mut self,
2443        row: &BooleanRow,
2444        index: usize,
2445        outputs: &[(IVec3, usize, bool)],
2446        columns: &BTreeMap<usize, Vec<(usize, DecisionId)>>,
2447    ) -> Result<Vec<(IVec3, PauliBasis, DecisionId)>, BooleanResourceError> {
2448        if index + 1 == outputs.len() {
2449            return Ok(Vec::new());
2450        }
2451        // Probe the smaller support, keeping column order on either path.
2452        let scan_row = row.terms().len() <= columns.len();
2453        let probes = if scan_row {
2454            row.terms().len()
2455        } else {
2456            columns.len()
2457        };
2458        let width = row.terms().len().max(columns.len()).max(1);
2459        self.topology
2460            .diagram
2461            .charge(probes.saturating_mul(width.ilog2() as usize + 1))?;
2462        let mut values = BTreeMap::new();
2463        let mut accumulate = |coefficient, candidates: &[(usize, DecisionId)]| {
2464            self.topology.diagram.charge(candidates.len())?;
2465            for &(later, weight) in candidates {
2466                if later <= index {
2467                    continue;
2468                }
2469                let term = self
2470                    .topology
2471                    .diagram
2472                    .apply(BooleanOp::And, weight, coefficient)?;
2473                if term == ZERO {
2474                    continue;
2475                }
2476                self.topology
2477                    .diagram
2478                    .charge(values.len().max(1).ilog2() as usize + 1)?;
2479                let value = values.entry(later).or_insert(ZERO);
2480                *value = self.topology.diagram.apply(BooleanOp::Xor, *value, term)?;
2481            }
2482            Ok::<_, BooleanResourceError>(())
2483        };
2484        if scan_row {
2485            for &(column, coefficient) in row.terms() {
2486                if let Some(candidates) = columns.get(&column) {
2487                    accumulate(coefficient, candidates)?;
2488                }
2489            }
2490        } else {
2491            for (&column, candidates) in columns {

2551    fn fill_selective_site(
2552        &mut self,
2553        indexed: &mut BooleanRowSpace,
2554        index: usize,
2555    ) -> Result<(), BooleanResourceError> {
2556        let (position, selector) = self.topology.resolves[index];

2558            .zx
2559            .node_at(position)
2560            .expect("local selective view retains its center node")
2561            .kind
2562        else {
2563            unreachable!("selective fill positions name selective nodes")
2564        };
2565        let low = Pauli::from(kind.pauli_if_false());
2566        let high = Pauli::from(kind.pauli_if_true());
2567        let mut weights = [ZERO; 2];
2568        for (weight, axis) in weights.iter_mut().zip([Pauli::Z, Pauli::X]) {
2569            *weight = match (low & axis, high & axis) {
2570                (false, false) => ZERO,
2571                (true, true) => ONE,
2572                (false, true) => selector,
2573                (true, false) => self.topology.diagram.negate(selector)?,
2574            };
2575        }
2576        // This also fills the terminal source basis. Preserve its reference
2577        // sector, just as for seam composition; resource rows are not neutral.
2578        let mut terms = Vec::new();
2579        for (components, weight) in self
2580            .node_components(position.to_array())
2581            .into_iter()
2582            .zip(weights)
2583        {
2584            for (column, guard) in components {
2585                terms.push((
2586                    column,
2587                    self.topology.diagram.apply(BooleanOp::And, guard, weight)?,
2588                ));
2589            }
2590        }
2591        indexed.eliminate_linear(&terms, &mut self.topology.diagram)?;
2592        Ok(())
2593    }

2595    fn feedback_coefficient(
2596        &mut self,
2597        row: &BooleanRow,
2598        targets: &[crate::FeedbackTarget],
2599    ) -> Result<DecisionId, BooleanResourceError> {
2600        self.topology.diagram.charge(targets.len())?;
2601        let mut result = ZERO;
2602        for target in targets {
2603            self.topology
2604                .diagram
2605                .charge(self.local[&target.target.to_array()].len())?;
2606            let mut value = ZERO;
2607            for variant in &self.local[&target.target.to_array()] {
2608                let edge = variant
2609                    .zx
2610                    .feedback_edge(target)
2611                    .expect("validated feedback wire");
2612                let column = SurfaceColumn::Edge(
2613                    variant.zx.nodes()[edge.n1].pos.to_array(),
2614                    variant.zx.nodes()[edge.n2].pos.to_array(),
2615                );
2616                let bits = [
2617                    self.component(row, column, Pauli::X),
2618                    self.component(row, column, Pauli::Z),
2619                ];
2620                let pauli = Pauli::from(target.pauli);
2621                let x = if pauli & Pauli::Z { bits[0] } else { ZERO };
2622                let z = if pauli & Pauli::X { bits[1] } else { ZERO };
2623                let term = self.topology.diagram.apply(BooleanOp::Xor, x, z)?;
2624                let term = self
2625                    .topology
2626                    .diagram
2627                    .apply(BooleanOp::And, variant.guard, term)?;
2628                value = self.topology.diagram.apply(BooleanOp::Or, value, term)?;
2629            }
2630            result = self.topology.diagram.apply(BooleanOp::Xor, result, value)?;
2631        }
2632        self.topology
2633            .diagram
2634            .apply(BooleanOp::And, row.active, result)
2635    }

2692        row: &BooleanRow,
2693        actions: &[Action],
2694        columns: &HashMap<usize, Vec<usize>>,
2695    ) -> Result<Vec<(u32, DecisionId)>, BooleanResourceError> {

2779/// A combination of `rows` that reads record `name`, preferring one that
2780/// conflicts with no other record. Each condition is linear in the arm
2781/// components, so the admissible rows are a kernel searched for a reader.
2782fn readable_representative(
2783    zx: &ZXGraph,
2784    rows: &[crate::PauliString],
2785    name: &str,
2786) -> Result<crate::zx::CoeffVec, StabilizerError> {
2787    let unavailable = || StabilizerError::MeasurementSurfaceUnavailable {
2788        mvar: name.to_owned(),
2789    };
2790    let mut exact = Vec::new();
2791    let mut others = Vec::new();
2792    let mut readable = Vec::new();
2793    for node in zx.action_graph().ordered_nodes() {
2794        let Action::Measure {
2795            name: record,
2796            target,
2797        } = &node.action
2798        else {
2799            continue;
2800        };
2801        let column = zx.measurement_column(target).ok_or_else(unavailable)?;
2802        let basis = match node.measurement.ok_or_else(unavailable)? {
2803            MeasurementObservable::Concrete(PauliBasis::X) => Pauli::X,
2804            MeasurementObservable::Concrete(PauliBasis::Z) => Pauli::Z,
2805            MeasurementObservable::Concrete(PauliBasis::Y)
2806            | MeasurementObservable::Selective(_) => Pauli::Y,
2807        };
2808        let components = basis
2809            .iter_xz()
2810            .flat_map(|axis| zx.readout_components(column, axis))
2811            .collect::<Vec<_>>();
2812        if record != name {
2813            others.extend(components);
2814            continue;
2815        }
2816        readable = components;
2817        // A node record reads exactly its measured axis.
2818        if basis != Pauli::Y && matches!(target, MeasureTarget::Node(_)) {
2819            exact.extend(zx.readout_components(column, basis.flip()));
2820        }
2821    }
2822    let isolated = exact.iter().chain(&others).copied().collect::<Vec<_>>();
2823    [isolated, exact]
2824        .into_iter()
2825        .find_map(|constraints| {
2826            let mut kernel = rows.to_vec();
2827            let mut combinations = (0..rows.len())
2828                .map(|index| crate::zx::CoeffVec::singleton(index, rows.len()))
2829                .collect::<Vec<_>>();
2830            let pivots = crate::zx::gaussian_elimination_with_tracking(
2831                &mut kernel,
2832                &mut combinations,
2833                constraints,
2834                |row, &(column, axis)| row.get(column) & axis,
2835                None,
2836            );
2837            kernel[pivots..]
2838                .iter()
2839                .zip(&combinations[pivots..])
2840                .find(|(row, _)| {
2841                    readable
2842                        .iter()
2843                        .any(|&(column, axis)| row.get(column) & axis)
2844                })
2845                .map(|(_, combination)| combination.clone())
2846        })
2847        .ok_or_else(unavailable)
2848}

2850pub(super) fn materialize_row(
2851    columns: impl IntoIterator<Item = (SurfaceColumn, usize)>,
2852    row: &BooleanRow,
2853    zx: &ZXGraph,
2854    evaluate: impl Fn(DecisionId) -> bool,
2855) -> crate::PauliString {
2856    let mut paulis = crate::PauliString::new(zx.total_ids());
2857    for (column, index) in columns {
2858        let value = match (
2859            evaluate(row.get(2 * index)),
2860            evaluate(row.get(2 * index + 1)),
2861        ) {
2862            (false, false) => Pauli::I,
2863            (true, false) => Pauli::X,
2864            (false, true) => Pauli::Z,
2865            (true, true) => Pauli::Y,
2866        };
2867        if value == Pauli::I {
2868            continue;
2869        }
2870        let (id, value) = match column {
2871            SurfaceColumn::Node(position) => (
2872                zx.node_at(IVec3::from_array(position))
2873                    .expect("active surface node belongs to the supplied ZX graph")
2874                    .id,
2875                value,
2876            ),
2877            SurfaceColumn::Edge(source, target) => {
2878                let source = zx
2879                    .node_at(IVec3::from_array(source))
2880                    .expect("active surface edge source belongs to the supplied ZX graph")
2881                    .id;
2882                let target = zx
2883                    .node_at(IVec3::from_array(target))
2884                    .expect("active surface edge target belongs to the supplied ZX graph")
2885                    .id;
2886                let edge = zx
2887                    .edge_between(source, target)
2888                    .expect("active surface edge belongs to the supplied ZX graph");
2889                (
2890                    edge.id,
2891                    if edge.hadamard && source > target {
2892                        value.flip()
2893                    } else {
2894                        value
2895                    },
2896                )
2897            }
2898        };
2899        paulis.set(id, value);
2900    }
2901    zx.reconstruct_raw_centers(&mut paulis);
2902    paulis
2903}

5/// Dense GF(2) coefficient vector with a bit-granular logical length.
6///
7/// Deliberately hand-rolled rather than wrapping [`binar::BitVec`], the
8/// workspace's bit-vector (already used by `bloq_utils::PauliString`). `binar`
9/// allocates and operates in 512-bit `BitBlock`s, sized for the wide rows
10/// `PauliString` carries. Coefficient vectors are one bit per input generator —
11/// usually tens — so every vector would round up to a 64-byte block and every
12/// `xor_assign` would XOR eight words instead of one. Measured on
13/// `GalleryItem::iter()` stabilizer computation, the `binar`-backed version cost
14/// ~36% (0.080s → 0.108s, best of 10); word-level `bit` reads recovered almost
15/// none of it, since the cost is in the block-granular XOR, and `binar` exposes
16/// no mutable word access to route around it.
17///
18/// Swap to `binar` if these vectors ever get wide (thousands of generators),
19/// where block-at-a-time SIMD would start to pay for the padding.
20#[derive(Debug, Clone, PartialEq, Eq)]
21pub(crate) struct CoeffVec {
22    words: Vec<u64>,
23    len: usize,
24}
25
26impl CoeffVec {
27    pub(crate) fn zeros(len: usize) -> Self {
28        Self {
29            words: vec![0; len.div_ceil(64)],
30            len,
31        }
32    }
33
34    pub(crate) fn singleton(index: usize, len: usize) -> Self {
35        let mut value = Self::zeros(len);
36        value.set_bit(index, true);
37        value
38    }
39
40    pub(crate) fn bit(&self, index: usize) -> bool {
41        debug_assert!(index < self.len);
42        let word = index / 64;
43        let bit = index % 64;
44        (self.words[word] >> bit) & 1 == 1
45    }
46
47    pub(crate) fn set_bit(&mut self, index: usize, value: bool) {
48        debug_assert!(index < self.len);
49        let word = index / 64;
50        let bit = index % 64;
51        if value {
52            self.words[word] |= 1_u64 << bit;
53        } else {
54            self.words[word] &= !(1_u64 << bit);
55        }
56    }
57
58    pub(crate) fn xor_assign(&mut self, other: &Self) {
59        debug_assert_eq!(self.len, other.len);
60        for (target, source) in self.words.iter_mut().zip(&other.words) {
61            *target ^= *source;
62        }
63    }
64
65    #[cfg(feature = "verify")]
66    pub(crate) fn first_one(&self) -> Option<usize> {
67        self.words
68            .iter()
69            .enumerate()
70            .find_map(|(word_index, &word)| {
71                (word != 0).then(|| word_index * 64 + word.trailing_zeros() as usize)
72            })
73    }
74
75    #[cfg(feature = "verify")]
76    pub(crate) fn is_zero(&self) -> bool {
77        self.words.iter().all(|&word| word == 0)
78    }
79
80    pub(crate) fn iter_ones(&self) -> impl Iterator<Item = usize> + '_ {
81        self.words
82            .iter()
83            .enumerate()
84            .flat_map(move |(word_index, &word)| {
85                let mut word = word;
86                std::iter::from_fn(move || {
87                    if word == 0 {
88                        return None;
89                    }
90                    let bit = word.trailing_zeros() as usize;
91                    word &= word - 1;
92                    let index = word_index * 64 + bit;
93                    (index < self.len).then_some(index)
94                })
95            })
96    }
97
98    pub(crate) fn to_indices(&self) -> Vec<usize> {
99        self.iter_ones().collect()
100    }
101}

103#[derive(Debug, Clone)]
104pub(crate) struct PauliSpan {
105    basis: Vec<PauliString>,
106    pivots: Vec<(usize, Pauli)>,
107    width: usize,
108}
109
110impl PauliSpan {
111    pub(crate) fn new(width: usize) -> Self {
112        Self {
113            basis: Vec::new(),
114            pivots: Vec::new(),
115            width,
116        }
117    }
118
119    pub(crate) fn reduce(&self, value: &PauliString) -> PauliString {
120        debug_assert_eq!(value.len(), self.width);
121        let mut reduced = value.clone();
122        for (basis, &(col, axis)) in self.basis.iter().zip(&self.pivots) {
123            if reduced.get(col) & axis {
124                reduced ^= basis;
125            }
126        }
127        reduced
128    }
129
130    pub(crate) fn insert_if_independent(&mut self, value: PauliString) -> bool {
131        let reduced = self.reduce(&value);
132        let Some(pivot) = first_pauli_pivot(&reduced) else {
133            return false;
134        };
135        self.basis.push(reduced);
136        self.pivots.push(pivot);
137        true
138    }
139}
140
141fn first_pauli_pivot(value: &PauliString) -> Option<(usize, Pauli)> {
142    let first = |words: &[u64]| {
143        words.iter().enumerate().find_map(|(word, &bits)| {
144            (bits != 0).then(|| word * 64 + bits.trailing_zeros() as usize)
145        })
146    };
147    match (first(value.x_words()), first(value.z_words())) {
148        (Some(x), Some(z)) => Some(if x <= z { (x, Pauli::X) } else { (z, Pauli::Z) }),
149        (Some(x), None) => Some((x, Pauli::X)),
150        (None, Some(z)) => Some((z, Pauli::Z)),
151        (None, None) => None,
152    }
153}

155pub(crate) fn reduce_to_basis(rows: &[PauliString], width: usize) -> Vec<PauliString> {
156    let mut span = PauliSpan::new(width);
157    let mut reduced = Vec::new();
158    for row in rows {
159        if span.insert_if_independent(row.clone()) {
160            reduced.push(row.clone());
161        }
162    }
163    reduced
164}

166pub(crate) fn axis_pivot_constraints(cols: impl IntoIterator<Item = usize>) -> Vec<(usize, Pauli)> {
167    cols.into_iter()
168        .flat_map(|col| [(col, Pauli::X), (col, Pauli::Z)])
169        .collect()
170}

172/// Fixed columns reduced in source order, preserving the first-independent
173/// witness used by the one-shot solver.
174#[cfg(feature = "verify")]
175pub(crate) struct CoeffFactorization<'a> {
176    vectors: &'a [CoeffVec],
177    basis: Vec<CoeffVec>,
178    witnesses: Vec<CoeffVec>,
179    pivots: Vec<usize>,
180    dependent: Vec<usize>,
181}
182
183#[cfg(feature = "verify")]
184impl<'a> CoeffFactorization<'a> {
185    pub(crate) fn new(vectors: &'a [CoeffVec]) -> Self {
186        let mut factorization = Self {
187            vectors,
188            basis: Vec::new(),
189            witnesses: Vec::new(),
190            pivots: Vec::new(),
191            dependent: Vec::new(),
192        };
193        for (index, vector) in vectors.iter().enumerate() {
194            let mut reduced = vector.clone();
195            let mut witness = CoeffVec::singleton(index, vectors.len());
196            factorization.reduce(&mut reduced, &mut witness);
197            if let Some(pivot) = reduced.first_one() {
198                factorization.basis.push(reduced);
199                factorization.witnesses.push(witness);
200                factorization.pivots.push(pivot);
201            } else {
202                factorization.dependent.push(index);
203            }
204        }
205        factorization
206    }
207
208    fn reduce(&self, value: &mut CoeffVec, witness: &mut CoeffVec) {
209        for ((basis, basis_witness), &pivot) in
210            self.basis.iter().zip(&self.witnesses).zip(&self.pivots)
211        {
212            if value.bit(pivot) {
213                value.xor_assign(basis);
214                witness.xor_assign(basis_witness);
215            }
216        }
217    }
218
219    pub(crate) fn solve(&self, target: &CoeffVec) -> Option<CoeffVec> {
220        let mut reduced = target.clone();
221        let mut witness = CoeffVec::zeros(self.vectors.len());
222        self.reduce(&mut reduced, &mut witness);
223        reduced.is_zero().then_some(witness)
224    }
225
226    pub(crate) fn kernel_directions(&self) -> Vec<CoeffVec> {
227        self.dependent
228            .iter()
229            .map(|&index| {
230                // Later independent columns cannot change a prefix's unique witness.
231                let mut direction = self
232                    .solve(&self.vectors[index])
233                    .expect("a dependent column is in the factorized span");
234                direction.set_bit(index, true);
235                direction
236            })
237            .collect()
238    }
239}

241/// Column-pivot GF(2) elimination over `rows`. Pass a `combinations` slice the
242/// same length as `rows` to track generator provenance in lockstep, or an empty
243/// slice to skip tracking (the plain elimination).
244pub(crate) fn gaussian_elimination_with_tracking<T, F>(
245    rows: &mut [PauliString],
246    combinations: &mut [CoeffVec],
247    pivot_constraints: impl IntoIterator<Item = T>,
248    predicate: F,
249    num_pivotable_rows: Option<usize>,
250) -> usize
251where
252    F: Fn(&PauliString, &T) -> bool,
253{
254    let track = !combinations.is_empty();
255    debug_assert!(!track || rows.len() == combinations.len());
256
257    let num_pivotable_rows = num_pivotable_rows.unwrap_or(rows.len()).min(rows.len());
258    let mut num_solved = 0;
259
260    for constraint in pivot_constraints {
261        if num_solved == num_pivotable_rows {
262            break;
263        }
264
265        let Some(pivot_index) =
266            (num_solved..num_pivotable_rows).find(|&index| predicate(&rows[index], &constraint))
267        else {
268            continue;
269        };
270
271        if pivot_index != num_solved {
272            rows.swap(pivot_index, num_solved);
273            if track {
274                combinations.swap(pivot_index, num_solved);
275            }
276        }
277
278        let (before, pivot_and_after) = rows.split_at_mut(num_solved);
279        let (pivot_row, after) = pivot_and_after
280            .split_first_mut()
281            .expect("solved pivot is in bounds");
282        if track {
283            let (before_coeffs, pivot_and_after_coeffs) = combinations.split_at_mut(num_solved);
284            let (pivot_coeff, after_coeffs) = pivot_and_after_coeffs
285                .split_first_mut()
286                .expect("tracked pivot is in bounds");
287            for (row, coeff) in before
288                .iter_mut()
289                .zip(before_coeffs)
290                .chain(after.iter_mut().zip(after_coeffs))
291            {
292                if predicate(row, &constraint) {
293                    *row ^= &*pivot_row;
294                    coeff.xor_assign(pivot_coeff);
295                }
296            }
297        } else {
298            for row in before.iter_mut().chain(after) {
299                if predicate(row, &constraint) {
300                    *row ^= &*pivot_row;
301                }
302            }
303        }
304
305        num_solved += 1;
306    }
307
308    num_solved
309}

357        // Source-ordered elimination selects the smallest input mask: replacing
358        // any dependent column uses only earlier columns and lowers that mask.
359        fn exhaustive_solution(columns: &[CoeffVec], target: &CoeffVec) -> Option<CoeffVec> {
360            (0usize..1 << columns.len()).find_map(|mask| {
361                let mut sum = CoeffVec::zeros(target.len);
362                let selected = (0..columns.len()).filter(|index| mask & (1 << index) != 0);
363                for index in selected.clone() {
364                    sum.xor_assign(&columns[index]);
365                }
366                (sum == *target).then(|| coeff(selected, columns.len()))
367            })
368        }
369
370        fn exhaustive_kernel(columns: &[CoeffVec]) -> Vec<CoeffVec> {
371            columns
372                .iter()
373                .enumerate()
374                .filter_map(|(index, column)| {
375                    let witness = exhaustive_solution(&columns[..index], column)?;
376                    Some(coeff(witness.iter_ones().chain([index]), columns.len()))
377                })
378                .collect()
379        }
380
381        fn coeff(indices: impl IntoIterator<Item = usize>, len: usize) -> CoeffVec {
382            let mut value = CoeffVec::zeros(len);
383            for index in indices {
384                value.set_bit(index, true);
385            }
386            value
387        }

389        #[test]
390        fn exact_solutions_and_ordered_kernels_match_exhaustive_small_matrices() {
391            for width in 0..=3 {
392                for count in 0..=4 {
393                    for matrix in 0_usize..(1 << (width * count)) {
394                        let columns = (0..count)
395                            .map(|col| {
396                                coeff(
397                                    (0..width)
398                                        .filter(|row| matrix & (1 << (col * width + row)) != 0),
399                                    width,
400                                )
401                            })
402                            .collect::<Vec<_>>();
403                        let factorization = CoeffFactorization::new(&columns);
404                        assert_eq!(
405                            factorization.kernel_directions(),
406                            exhaustive_kernel(&columns)
407                        );
408                        for mask in 0_usize..(1 << width) {
409                            let target =
410                                coeff((0..width).filter(|row| mask & (1 << row) != 0), width);
411                            let expected = exhaustive_solution(&columns, &target);
412                            assert_eq!(
413                                factorization.solve(&target),
414                                expected,
415                                "width={width}, columns={count}, matrix={matrix}, target={mask}"

417                        }
418                    }

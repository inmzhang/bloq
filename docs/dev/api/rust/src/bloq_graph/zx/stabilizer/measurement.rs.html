57impl MeasurementMetadata {
58    pub(crate) fn collect(zx: &ZXGraph) -> Self {
59        let mut cols = HashMap::new();
60        let mut observables = HashMap::new();
61        let mut edge_records = HashSet::new();
62
63        for node in zx.action_graph().ordered_nodes() {
64            let crate::Action::Measure { name, target } = &node.action else {
65                continue;
66            };
67            let Some(col) = zx.measurement_column(target) else {
68                continue;
69            };
70            cols.insert(name.clone(), col);
71            if let Some(observable) = node.measurement {
72                observables.insert(name.clone(), observable);
73            }
74            if matches!(target, crate::MeasureTarget::Edge { .. }) {
75                edge_records.insert(name.clone());
76            }
77        }
78
79        Self {
80            cols,
81            observables,
82            edge_records,
83        }
84    }
85
86    pub(super) fn sorted_columns(&self) -> Vec<usize> {
87        let mut cols = self.cols.values().copied().collect::<Vec<_>>();
88        cols.sort_unstable();
89        cols
90    }
91
92    pub(super) fn column(&self, name: &str) -> usize {
93        self.cols[name]
94    }
95
96    fn basis_overlap(&self, name: &str) -> Option<Pauli> {
97        match self.observables.get(name).copied() {
98            Some(MeasurementObservable::Concrete(PauliBasis::X)) => Some(Pauli::X),
99            Some(MeasurementObservable::Concrete(PauliBasis::Z)) => Some(Pauli::Z),
100            Some(MeasurementObservable::Concrete(PauliBasis::Y))
101            | Some(MeasurementObservable::Selective(_)) => Some(Pauli::Y),
102            None => None,
103        }
104    }
105
106    fn support_conflicts(&self, name: &str, pauli: Pauli) -> bool {
107        pauli != Pauli::I && self.basis_overlap(name).is_some_and(|basis| pauli & basis)
108    }
109
110    /// Whether the record's column carries its measured component. On an edge,
111    /// this includes `Y` carrying Z parity plus X transport (§2 of input model).
112    pub(super) fn support_matches(&self, name: &str, pauli: Pauli) -> bool {
113        match self.basis_overlap(name) {
114            Some(Pauli::Y) => pauli != Pauli::I,
115            Some(basis) if self.edge_records.contains(name) => pauli & basis,
116            Some(basis) => pauli == basis,
117            None => false,
118        }
119    }
120
121    /// Exact Pauli values admitted by `support_matches`.
122    pub(crate) fn matching_paulis(&self, name: &str) -> Vec<Pauli> {
123        match self.basis_overlap(name) {
124            Some(Pauli::Y) => vec![Pauli::Y, Pauli::X, Pauli::Z],
125            Some(basis) if self.edge_records.contains(name) => vec![basis, Pauli::Y],
126            Some(basis) => vec![basis],
127            None => Vec::new(),
128        }
129    }
130
131    /// `support_matches` without edge transport.
132    fn support_matches_exactly(&self, name: &str, pauli: Pauli) -> bool {
133        match self.basis_overlap(name) {
134            Some(Pauli::Y) => pauli != Pauli::I,
135            Some(basis) => pauli == basis,
136            None => false,
137        }
138    }
139
140    fn support_is_isolated(&self, name: &str, row: &PauliString) -> bool {
141        self.support_matches(name, row.get(self.column(name)))
142            && self.cols.iter().all(|(other_name, &col)| {
143                !self.support_conflicts(other_name, row.get(col)) || other_name == name
144            })
145    }
146}

148/// Readers that would close a cycle if a candidate preceded them.
149#[derive(Default)]
150struct SelfReaders {
151    /// Per reachable feedback, its targets as `(node column, Pauli)`.
152    feedbacks: Vec<Vec<(usize, Pauli)>>,
153    /// Reachable resolve targets, as cap columns.
154    caps: Vec<usize>,
155    /// Columns owned by each reachable terminal branch region.
156    branches: Vec<Vec<usize>>,
157}
158
159impl SelfReaders {
160    fn collect(
161        zx: &ZXGraph,
162        name: &str,
163        prior_surfaces: &[MeasurementSurfaceRow],
164        selective_constraints: &[SelectiveConstraint],
165        fixings: &[(PauliString, StabilizerRowKind)],
166    ) -> Self {
167        let dag = zx.action_graph();
168        let Some(origin) = dag
169            .ordered_nodes()
170            .find(|node| {
171                matches!(&node.action, crate::Action::Measure { name: measured, .. } if measured == name)
172            })
173            .map(|node| node.ordinal)
174        else {
175            return Self::default();
176        };
177
178        let mut successors = vec![Vec::new(); dag.ordered_nodes().count()];
179        for (from, to, _) in dag.dependencies() {
180            successors[from].push(to);
181        }
182        for surface in prior_surfaces {
183            let Some(measurement) = dag
184                .ordered_nodes()
185                .find(|node| {
186                    matches!(&node.action, crate::Action::Measure { name, .. } if name == &surface.name)
187                })
188                .map(|node| node.ordinal)
189            else {
190                continue;
191            };
192            let (reached, active_fixings) =
193                reachable_fixing_support(&surface.row, zx, selective_constraints, fixings);
194            for node in dag.ordered_nodes() {
195                let precedes = match &node.action {
196                    crate::Action::Feedback { targets, .. } => feedback_perturbs_row(
197                        &surface.row,
198                        &feedback_columns(zx, targets),
199                        fixings,
200                        &active_fixings,
201                    ),
202                    crate::Action::Resolve { target, .. } => zx
203                        .node_at(*target)
204                        .is_some_and(|node| reached.contains(&node.id)),
205                    crate::Action::Branch { target, .. } => {
206                        dag.branch_region(*target).is_some_and(|region| {
207                            row_or_active_fixings_touches_columns(
208                                &surface.row,
209                                &branch_region_columns(zx, region),
210                                fixings,
211                                &active_fixings,
212                            )
213                        })
214                    }
215                    _ => false,
216                };
217                if precedes {
218                    successors[node.ordinal].push(measurement);
219                }
220            }
221        }
222
223        let mut reachable = vec![false; successors.len()];
224        let mut pending = vec![origin];
225        while let Some(from) = pending.pop() {
226            for &reader in &successors[from] {
227                if !reachable[reader] {
228                    reachable[reader] = true;
229                    pending.push(reader);
230                }
231            }
232        }
233
234        let mut readers = Self::default();
235        for node in dag.ordered_nodes().filter(|node| reachable[node.ordinal]) {
236            match &node.action {
237                crate::Action::Feedback { targets, .. } => {
238                    readers.feedbacks.push(feedback_columns(zx, targets));
239                }
240                crate::Action::Resolve { target, .. } => {
241                    if let Some(node) = zx.node_at(*target) {
242                        readers.caps.push(node.id);
243                    }
244                }
245                crate::Action::Branch { target, .. } => {
246                    if let Some(region) = dag.branch_region(*target) {
247                        readers.branches.push(branch_region_columns(zx, region));
248                    }
249                }
250                _ => {}
251            }
252        }
253        readers
254    }
255
256    fn is_empty(&self) -> bool {
257        self.feedbacks.is_empty() && self.caps.is_empty() && self.branches.is_empty()
258    }
259
260    /// Whether `row` — cross centres reconstructed — would add an implicit edge
261    /// from a reachable action back to this record and close a cycle.
262    fn traps(
263        &self,
264        row: &PauliString,
265        zx: &ZXGraph,
266        selective_constraints: &[SelectiveConstraint],
267        fixings: &[(PauliString, StabilizerRowKind)],
268    ) -> bool {
269        let (reached, active_fixings) =
270            reachable_fixing_support(row, zx, selective_constraints, fixings);
271
272        self.caps.iter().any(|col| reached.contains(col))
273            || self.branches.iter().any(|columns| {
274                row_or_active_fixings_touches_columns(row, columns, fixings, &active_fixings)
275            })
276            || self
277                .feedbacks
278                .iter()
279                .any(|targets| feedback_perturbs_row(row, targets, fixings, &active_fixings))
280    }
281}

299fn feedback_perturbs_row(
300    row: &PauliString,
301    targets: &[(usize, Pauli)],
302    fixings: &[(PauliString, StabilizerRowKind)],
303    active_fixings: &[bool],
304) -> bool {
305    odd_anticommutes(row, targets)
306        || fixings
307            .iter()
308            .zip(active_fixings)
309            .any(|((fixing_row, _), &active)| active && odd_anticommutes(fixing_row, targets))

312fn reachable_fixing_support(
313    row: &PauliString,
314    zx: &ZXGraph,
315    selective_constraints: &[SelectiveConstraint],
316    fixings: &[(PauliString, StabilizerRowKind)],
317) -> (HashSet<usize>, Vec<bool>) {
318    let mut reached = selective_constraints
319        .iter()
320        .filter(|constraint| row.get(constraint.col) != Pauli::I)
321        .map(|constraint| constraint.col)
322        .collect::<HashSet<_>>();
323    let mut active_fixings = vec![false; fixings.len()];
324    let mut changed = true;
325    while changed {
326        changed = false;
327        for (index, (fixing_row, kind)) in fixings.iter().enumerate() {
328            if active_fixings[index]
329                || !kind.selective_fixing_targets().iter().any(|target| {
330                    zx.node_at(target.pos)
331                        .is_some_and(|node| reached.contains(&node.id))
332                })
333            {
334                continue;
335            }
336            active_fixings[index] = true;
337            changed = true;
338            reached.extend(
339                selective_constraints
340                    .iter()
341                    .filter(|constraint| fixing_row.get(constraint.col) != Pauli::I)
342                    .map(|constraint| constraint.col),
343            );
344        }
345    }
346    (reached, active_fixings)
347}

349fn branch_region_columns(zx: &ZXGraph, region: &crate::BranchRegion) -> Vec<usize> {
350    let mut columns = zx
351        .nodes
352        .iter()
353        .filter(|node| region.contains_block(node.pos))
354        .map(|node| node.id)
355        .collect::<Vec<_>>();
356    columns.extend(zx.edges.iter().filter_map(|edge| {
357        let first = zx.nodes[edge.n1].pos;
358        let second = zx.nodes[edge.n2].pos;
359        (region.contains_block(first) || region.contains_block(second)).then_some(edge.id)
360    }));
361    columns.sort_unstable();
362    columns.dedup();
363    columns
364}

366fn row_or_active_fixings_touches_columns(
367    row: &PauliString,
368    columns: &[usize],
369    fixings: &[(PauliString, StabilizerRowKind)],
370    active_fixings: &[bool],
371) -> bool {
372    columns.iter().any(|&column| row.get(column) != Pauli::I)
373        || fixings
374            .iter()
375            .zip(active_fixings)
376            .any(|((fixing, _), &active)| {
377                active && columns.iter().any(|&column| fixing.get(column) != Pauli::I)
378            })

420            }

425        lhs.deadline

427            .then_with(|| lhs.name.cmp(&rhs.name))
428    });
429    specs

464/// Extracts measurement rows from one time-ordered factorization.
465///
466/// Temporal pivots are ordered by their decode gate. At each record deadline,
467/// their suffix is exactly the subspace with no future physical support. A
468/// chosen row is consumed from that live suffix, so it may carry later records
469/// without ever extending past its own deadline. Any failed certificate leaves
470/// the exact sequential search to the caller.
471pub(super) fn extract_temporal_measurement_prefix(
472    zx: &ZXGraph,
473    basis_rows: &[PauliString],
474    metadata: &MeasurementMetadata,
475    selective_constraints: &[SelectiveConstraint],
476    specs: &[MeasurementSpec],
477    budget: &mut SearchBudget,
478) -> Result<Option<Vec<MeasurementSurfaceRow>>, StabilizerError> {
479    budget.check_matrix(
480        basis_rows.len().saturating_mul(2),
481        zx.total_ids(),
482        basis_rows.len(),
483        basis_rows.len(),
484    )?;
485    budget.visit("temporal measurement search states")?;
486    let mut basis = TrackedBasis::new(basis_rows.to_vec());
487    let constraints = temporal_axis_constraints(zx, selective_constraints);
488    let mut release_gates = prefactor_temporal_basis(&mut basis, &constraints);
489    let mut surfaces = Vec::with_capacity(specs.len());
490
491    for spec in specs {
492        budget.visit("temporal measurement search states")?;
493        let prefix_len = surfaces.len();
494        let consumed = release_gates[prefix_len..]
495            .iter()
496            .take_while(|&&gate| gate > spec.deadline)
497            .count();
498        let live = prefix_len + consumed;
499        let target_rows = target_support_candidates_in_place(
500            &mut basis.rows[live..],
501            &mut basis.coeffs[live..],
502            metadata,
503            &spec.name,
504            true,
505        );
506        let mut accepted = None;
507        for (candidate, replacement) in target_rows.candidates {
508            let SurfaceCandidate {
509                mut row,
510                mut combination,
511            } = candidate;
512            let needs_repair = selective_constraints
513                .iter()
514                .any(|constraint| row.get(constraint.col) == constraint.forbidden);
515            if needs_repair
516                && !repair_selective_support_in_place(
517                    &mut row,
518                    &mut combination,
519                    &mut basis.rows[live + target_rows.count..],
520                    &mut basis.coeffs[live + target_rows.count..],
521                    selective_constraints,
522                )
523            {
524                continue;
525            }
526            zx.reconstruct_cross_center(std::slice::from_mut(&mut row));
527            if !row_respects_measurement_support_constraints(&row, &[], selective_constraints)
528                || !measurement_surface_row_matches_acceptance(
529                    zx, metadata, &spec.name, &row, false,
530                )
531            {
532                continue;
533            }
534            let deadline = accepted_surface_deadline(zx, selective_constraints, spec, &row);
535            if deadline > spec.deadline {
536                continue;
537            }
538            accepted = Some((row, combination, deadline, replacement));
539            break;
540        }
541        let Some((row, combination, deadline, replacement)) = accepted else {
542            return Ok(None);
543        };
544        let replace_index = consumed + replacement;
545        basis.rows[prefix_len + replace_index] = row.clone();
546        basis.coeffs[prefix_len + replace_index] = combination.clone();
547        release_gates[prefix_len + replace_index] = deadline;
548        if replace_index != 0 {
549            basis.rows[prefix_len..=prefix_len + replace_index].rotate_right(1);
550            basis.coeffs[prefix_len..=prefix_len + replace_index].rotate_right(1);
551            release_gates[prefix_len..=prefix_len + replace_index].rotate_right(1);
552        }
553        isolate_measurement_target_in_suffix(
554            &mut basis.rows[prefix_len..],
555            &mut basis.coeffs[prefix_len..],
556            metadata.column(&spec.name),
557            metadata.basis_overlap(&spec.name).unwrap_or(Pauli::Y),
558        );
559        surfaces.push(MeasurementSurfaceRow {
560            name: spec.name.clone(),
561            deadline,
562            row,
563            combination,
564        });
565    }
566
567    Ok(Some(surfaces))
568}

595fn prefactor_temporal_basis(
596    basis: &mut TrackedBasis,
597    constraints: &[(i64, usize, Pauli)],
598) -> Vec<i64> {
599    let mut solved = 0;
600    let mut release_gates = Vec::with_capacity(basis.rows.len());
601    for &(gate, col, axis) in constraints {
602        let Some(pivot_index) =
603            (solved..basis.rows.len()).find(|&index| basis.rows[index].get(col) & axis)
604        else {
605            continue;
606        };
607        if pivot_index != solved {
608            basis.rows.swap(pivot_index, solved);
609            basis.coeffs.swap(pivot_index, solved);
610        }
611        let (before, pivot_and_after) = basis.rows.split_at_mut(solved);
612        let (pivot_row, after) = pivot_and_after
613            .split_first_mut()
614            .expect("solved pivot is in bounds");
615        let (before_coeffs, pivot_and_after_coeffs) = basis.coeffs.split_at_mut(solved);
616        let (pivot_coeff, after_coeffs) = pivot_and_after_coeffs
617            .split_first_mut()
618            .expect("tracked pivot is in bounds");
619        for (row, coeff) in before
620            .iter_mut()
621            .zip(before_coeffs)
622            .chain(after.iter_mut().zip(after_coeffs))
623        {
624            if row.get(col) & axis {
625                *row ^= &*pivot_row;
626                coeff.xor_assign(pivot_coeff);
627            }
628        }
629        release_gates.push(gate);
630        solved += 1;
631    }
632    release_gates.resize(basis.rows.len(), i64::MIN);
633    release_gates
634}

636fn temporal_axis_constraints(
637    zx: &ZXGraph,
638    selective_constraints: &[SelectiveConstraint],
639) -> Vec<(i64, usize, Pauli)> {
640    let node_gates = temporal_node_gates(zx, selective_constraints);
641    let columns = zx
642        .nodes
643        .iter()
644        .filter_map(|node| node_gates[node.id].map(|gate| (gate, node.id)))
645        .chain(zx.edges.iter().filter_map(|edge| {
646            node_gates[edge.n1]
647                .into_iter()
648                .chain(node_gates[edge.n2])
649                .max()
650                .map(|gate| (gate, edge.id))
651        }));
652    let mut constraints = columns
653        .flat_map(|(gate, col)| [Pauli::X, Pauli::Z].map(move |axis| (gate, col, axis)))
654        .collect::<Vec<_>>();
655    constraints.sort_unstable_by(|lhs, rhs| {
656        rhs.0
657            .cmp(&lhs.0)
658            .then_with(|| lhs.1.cmp(&rhs.1))
659            .then_with(|| u8::from(lhs.2).cmp(&u8::from(rhs.2)))
660    });
661    constraints
662}

687#[expect(
688    clippy::too_many_arguments,
689    reason = "surface construction needs the independent search inputs"
690)]
691fn construct_measurement_surface_row_from_basis(
692    basis: &mut TrackedBasis,
693    prefix_len: usize,
694    zx: &ZXGraph,
695    metadata: &MeasurementMetadata,
696    selective_constraints: &[SelectiveConstraint],
697    self_reader_context: &SelfReaderContext<'_>,
698    spec: &MeasurementSpec,
699    budget: &mut SearchBudget,
700) -> Result<MeasurementSurfaceRow, StabilizerError> {
701    let max_limit_z = zx.nodes.iter().map(|node| node.pos.z).max().unwrap_or(0);
702    // Forbidden node/edge columns change only at an endpoint's z. Keep the
703    // original first attempt and skip empty coordinate gaps between changes.
704    let mut support_limits = zx
705        .nodes
706        .iter()
707        .map(|node| node.pos.z)
708        .filter(|&z| z >= spec.minimum_support_z)
709        .collect::<Vec<_>>();
710    if spec.minimum_support_z <= max_limit_z {
711        support_limits.push(spec.minimum_support_z);
712    }
713    support_limits.sort_unstable();
714    support_limits.dedup();
715    // Depends only on the metadata and this measurement's name, so it is
716    // invariant across the support-limit sweep below.
717    let conflicting_pivots = conflicting_measurement_pivots(metadata, &spec.name);
718    let self_readers = SelfReaders::collect(
719        zx,
720        &spec.name,
721        self_reader_context.prior_surfaces,
722        selective_constraints,
723        self_reader_context.fixings,
724    );
725
726    // Prefer cycle-safe, exact-edge, output-free, isolated, early surfaces, in
727    // that order. The last cycle sweep preserves `DependencyCycle` diagnostics.
728    let self_reader_sweeps = if self_readers.is_empty() { 1 } else { 2 };
729    for avoid_self_readers in [true, false].into_iter().take(self_reader_sweeps) {
730        for component_match in [false, true] {
731            for allow_output_support in [false, true] {
732                for require_isolated in [true, false] {
733                    for &limit_z in &support_limits {
734                        let support_forbidden =
735                            support_forbidden_columns(zx, limit_z, allow_output_support);
736                        let mut attempt = |support_forbidden: &[usize], repair_selective: bool| {
737                            attempt_measurement_surface_row(
738                                basis,
739                                prefix_len,
740                                zx,
741                                &MeasurementAttemptContext {
742                                    metadata,
743                                    self_readers: &self_readers,
744                                    self_reader_fixings: self_reader_context.fixings,
745                                    avoid_self_readers,
746                                    selective_constraints,
747                                    name: &spec.name,
748                                    limit_z,
749                                    support_forbidden,
750                                    conflicting_pivots: &conflicting_pivots,
751                                    require_isolated,
752                                    repair_selective,
753                                    component_match,
754                                },
755                                budget,
756                            )
757                        };
758                        // Try selective support unconstrained, pivoted to I, then repaired.
759                        let mut result = attempt(&support_forbidden, false)?;
760                        if matches!(result, MeasurementConstructionAttempt::Unavailable) {
761                            let mut selective_free = support_forbidden.clone();
762                            selective_free.extend(
763                                selective_constraints
764                                    .iter()
765                                    .map(|constraint| constraint.col),
766                            );
767                            selective_free.sort_unstable();
768                            selective_free.dedup();
769                            if selective_free != support_forbidden {
770                                result = attempt(&selective_free, false)?;
771                            }
772                        }
773                        if matches!(result, MeasurementConstructionAttempt::Unavailable)
774                            && !selective_constraints.is_empty()
775                        {
776                            result = attempt(&support_forbidden, true)?;
777                        }
778                        match result {
779                            MeasurementConstructionAttempt::Accepted {
780                                basis: accepted_basis,
781                                mut surface,
782                            } => {
783                                surface.deadline = accepted_surface_deadline(
784                                    zx,
785                                    selective_constraints,
786                                    spec,
787                                    &surface.row,
788                                );
789                                *basis = accepted_basis;
790                                return Ok(surface);
791                            }
792                            MeasurementConstructionAttempt::Unavailable => {}
793                        }
794                    }
795                }
796            }
797        }
798    }
799
800    Err(StabilizerError::MeasurementSurfaceUnavailable {
801        mvar: spec.name.to_string(),
802    })
803}

805/// Decode deadline from actual support, floored by `spec.deadline`. Ordinary
806/// nodes gate at z+1, selective sites at z, and edges use both endpoints.
807pub(super) fn accepted_surface_deadline(
808    zx: &ZXGraph,
809    selective_constraints: &[SelectiveConstraint],
810    spec: &MeasurementSpec,
811    row: &PauliString,
812) -> i64 {
813    let selective_cols = selective_constraints
814        .iter()
815        .map(|constraint| constraint.col)
816        .collect::<std::collections::HashSet<_>>();
817    // Output anchors become frame constraints, not awaited records.
818    let node_gate = |node_id: usize| {
819        let node = &zx.nodes[node_id];
820        if node.is_output_port(zx) {
821            return None;
822        }
823        let z = i64::from(node.pos.z);
824        Some(if selective_cols.contains(&node_id) {
825            z
826        } else {
827            z + 1
828        })
829    };
830
831    let mut deadline = spec.deadline;
832    for (col, _) in row.iter_support() {
833        if col < zx.nodes.len() {
834            if let Some(gate) = node_gate(col) {
835                deadline = deadline.max(gate);
836            }
837        } else {
838            let edge = zx.edge_by_id(col);
839            for gate in [node_gate(edge.n1), node_gate(edge.n2)]
840                .into_iter()
841                .flatten()
842            {
843                deadline = deadline.max(gate);
844            }
845        }
846    }
847    deadline
848}

850fn attempt_measurement_surface_row(
851    committed_basis: &TrackedBasis,
852    prefix_len: usize,
853    zx: &ZXGraph,
854    context: &MeasurementAttemptContext<'_>,
855    budget: &mut SearchBudget,
856) -> Result<MeasurementConstructionAttempt, StabilizerError> {
857    budget.visit("measurement surface search states")?;
858    budget.check_matrix(
859        committed_basis.rows.len().saturating_mul(4),
860        zx.total_ids(),
861        committed_basis.coeffs.len().saturating_mul(4),
862        committed_basis.coeffs.first().map_or(0, CoeffVec::len),
863    )?;
864    let mut scratch_rows = committed_basis.rows[prefix_len..].to_vec();
865    let mut scratch_coeffs = committed_basis.coeffs[prefix_len..].to_vec();
866    if scratch_coeffs.is_empty() {
867        return Ok(MeasurementConstructionAttempt::Unavailable);
868    }
869
870    let mut pivot_constraints = axis_pivot_constraints(context.support_forbidden.iter().copied());
871    if context.require_isolated {
872        pivot_constraints.extend(context.conflicting_pivots.iter().copied());
873    }
874    let consumed = gaussian_elimination_with_tracking(
875        &mut scratch_rows,
876        &mut scratch_coeffs,
877        pivot_constraints,
878        |row, &(col, basis)| row.get(col) & basis,
879        None,
880    );
881
882    let search = target_support_candidates(
883        scratch_rows[consumed..].to_vec(),
884        scratch_coeffs[consumed..].to_vec(),
885        context.metadata,
886        context.name,
887        context.component_match,
888    );
889    if search.candidates.is_empty() {
890        return Ok(MeasurementConstructionAttempt::Unavailable);
891    }
892
893    for candidate in search.candidates {
894        budget.visit("measurement surface search states")?;
895        let SurfaceCandidate {
896            mut row,
897            mut combination,
898        } = candidate;
899        if context.repair_selective
900            && !repair_selective_support(
901                &mut row,
902                &mut combination,
903                &search.kernel_rows,
904                &search.kernel_coeffs,
905                context.selective_constraints,
906            )
907        {
908            continue;
909        }
910        zx.reconstruct_cross_center(std::slice::from_mut(&mut row));
911        if (!context.avoid_self_readers
912            || !context.self_readers.traps(
913                &row,
914                zx,
915                context.selective_constraints,
916                context.self_reader_fixings,
917            ))
918            && row_respects_measurement_support_constraints(
919                &row,
920                context.support_forbidden,
921                context.selective_constraints,
922            )
923            && measurement_surface_row_matches_acceptance(
924                zx,
925                context.metadata,
926                context.name,
927                &row,
928                context.require_isolated,
929            )
930        {
931            let Some(replace_index) = replace_row_with_candidate_preserving_rank(
932                &mut scratch_rows,
933                &mut scratch_coeffs,
934                consumed,
935                &row,
936                &combination,
937            ) else {
938                continue;
939            };
940
941            if replace_index != 0 {
942                scratch_rows.swap(replace_index, 0);
943                scratch_coeffs.swap(replace_index, 0);
944            }
945
946            isolate_measurement_target_in_suffix(
947                &mut scratch_rows,
948                &mut scratch_coeffs,
949                context.metadata.column(context.name),
950                context
951                    .metadata
952                    .basis_overlap(context.name)
953                    .unwrap_or(Pauli::Y),
954            );
955
956            let mut accepted_basis = committed_basis.clone();
957            accepted_basis.replace_suffix(prefix_len, &scratch_rows, &scratch_coeffs);
958            return Ok(MeasurementConstructionAttempt::Accepted {
959                basis: accepted_basis,
960                surface: MeasurementSurfaceRow {
961                    name: context.name.to_string(),
962                    // Refined from the accepted row by the caller.
963                    deadline: i64::from(context.limit_z) + 1,
964                    row,
965                    combination,
966                },
967            });
968        }
969    }
970
971    Ok(MeasurementConstructionAttempt::Unavailable)
972}

987fn support_forbidden_columns(zx: &ZXGraph, limit_z: i32, allow_output_support: bool) -> Vec<usize> {
988    let mut forbidden = Vec::new();
989
990    for node in &zx.nodes {
991        if node.is_output_port(zx) && !allow_output_support {
992            forbidden.push(node.id);
993            continue;
994        }

1009    forbidden.dedup();

1025    kernel_rows: Vec<PauliString>,

1034fn target_support_candidates_in_place(
1035    rows: &mut [PauliString],
1036    combinations: &mut [CoeffVec],
1037    metadata: &MeasurementMetadata,
1038    name: &str,
1039    component_match: bool,
1040) -> TemporalTargetSupportSearch {
1041    let target_col = metadata.column(name);
1042    let count = gaussian_elimination_with_tracking(
1043        rows,
1044        combinations,
1045        axis_pivot_constraints([target_col]),
1046        |row, &(col, basis)| row.get(col) & basis,
1047        None,
1048    );
1049    debug_assert!(count <= 2, "one Pauli column has at most X/Z rank");
1050
1051    let make_candidate = |row: &PauliString, combination: &CoeffVec| {
1052        target_support_candidate(
1053            metadata,
1054            name,
1055            target_col,
1056            row,
1057            combination,
1058            component_match,
1059        )
1060    };
1061    let mut candidates = Vec::new();
1062    for (replacement, (row, combination)) in
1063        rows[..count].iter().zip(&combinations[..count]).enumerate()
1064    {
1065        if let Some(candidate) = make_candidate(row, combination) {
1066            candidates.push((candidate, replacement));
1067        }
1068    }
1069    if count == 2 {
1070        let mut row = rows[0].clone();
1071        row ^= &rows[1];
1072        let mut combination = combinations[0].clone();
1073        combination.xor_assign(&combinations[1]);
1074        if let Some(candidate) = make_candidate(&row, &combination) {
1075            candidates.push((candidate, 0));
1076        }
1077    }
1078    TemporalTargetSupportSearch { candidates, count }
1079}

1081    rows: Vec<PauliString>,
1082    combinations: Vec<CoeffVec>,
1083    metadata: &MeasurementMetadata,
1084    name: &str,
1085    component_match: bool,
1086) -> TargetSupportSearch {
1087    let mut rows = rows;
1088    let mut combinations = combinations;

1091        &mut combinations,

1093        name,
1094        component_match,

1096    let kernel_rows = rows.split_off(search.count);
1097    let kernel_coeffs = combinations.split_off(search.count);
1098    TargetSupportSearch {

1100            .candidates
1101            .into_iter()
1102            .map(|(candidate, _)| candidate)
1103            .collect(),
1104        kernel_rows,
1105        kernel_coeffs,
1106    }

1109/// Adds kernel rows to `row` until no selective column carries its forbidden
1110/// Pauli, mirroring the additions on `combination`. Returns `false` when a
1111/// column that needs repair has no kernel freedom left.
1112///
1113/// Each pass repairs one violated column and then pivots that column out of the
1114/// remaining pool, so later passes are `I` there and cannot undo it. Columns
1115/// that are already satisfied keep their freedom: locking them up front would
1116/// reject rows that only need a *tolerated* Pauli there. Since every pass locks
1117/// a distinct column, one pass per constraint is enough.
1118///
1119/// Any pivot is non-`I` at its own column, and XOR-ing a non-`I` Pauli onto the
1120/// forbidden one lands on one of the other two, so the first pivot always
1121/// repairs the column and no search over pivot subsets is needed.
1122///
1123/// This is not a complete decision procedure: which pivot a pass picks, and in
1124/// which order columns lock, can still strand a satisfiable row. It subsumes
1125/// the two earlier stages, which is what the caller's ladder needs.
1126fn repair_selective_support(
1127    row: &mut PauliString,
1128    combination: &mut CoeffVec,
1129    kernel_rows: &[PauliString],
1130    kernel_coeffs: &[CoeffVec],
1131    constraints: &[SelectiveConstraint],
1132) -> bool {
1133    let mut rows = kernel_rows.to_vec();
1134    let mut coeffs = kernel_coeffs.to_vec();
1135    repair_selective_support_in_place(row, combination, &mut rows, &mut coeffs, constraints)
1136}
1137
1138fn repair_selective_support_in_place(
1139    row: &mut PauliString,
1140    combination: &mut CoeffVec,
1141    rows: &mut [PauliString],
1142    coeffs: &mut [CoeffVec],
1143    constraints: &[SelectiveConstraint],
1144) -> bool {
1145    let mut free = 0;
1146
1147    for _ in 0..constraints.len() {
1148        let Some(col) = constraints
1149            .iter()
1150            .find(|constraint| row.get(constraint.col) == constraint.forbidden)
1151            .map(|constraint| constraint.col)
1152        else {
1153            return true;
1154        };
1155
1156        let solved = gaussian_elimination_with_tracking(
1157            &mut rows[free..],
1158            &mut coeffs[free..],
1159            axis_pivot_constraints([col]),
1160            |kernel_row, &(pivot_col, basis)| kernel_row.get(pivot_col) & basis,
1161            None,
1162        );
1163        if solved == 0 {
1164            return false;
1165        }
1166
1167        *row ^= &rows[free];
1168        combination.xor_assign(&coeffs[free]);
1169        free += solved;
1170    }
1171
1172    constraints
1173        .iter()
1174        .all(|constraint| row.get(constraint.col) != constraint.forbidden)
1175}

1200/// Finds the first witness slot (from `witness_start`) that `candidate` may
1201/// replace while preserving both the coefficient rank and the Pauli-row rank,
1202/// installing it there.
1203fn replace_row_with_candidate_preserving_rank(
1204    rows: &mut [PauliString],
1205    combinations: &mut [CoeffVec],
1206    witness_start: usize,
1207    candidate: &PauliString,
1208    candidate_combination: &CoeffVec,
1209) -> Option<usize> {
1210    let index = replacement_index_preserving_rank(
1211        rows,
1212        combinations,
1213        witness_start,
1214        candidate,
1215        candidate_combination,
1216    )?;
1217    rows[index] = candidate.clone();
1218    combinations[index] = candidate_combination.clone();
1219    Some(index)
1220}

1222fn replacement_index_preserving_rank(
1223    rows: &[PauliString],
1224    combinations: &[CoeffVec],
1225    witness_start: usize,
1226    candidate: &PauliString,
1227    candidate_combination: &CoeffVec,
1228) -> Option<usize> {
1229    debug_assert_eq!(rows.len(), combinations.len());
1230    // `combinations` track an independent stabilizer basis. If the candidate
1231    // lies in that span, its unique coordinates name exactly the members it
1232    // may replace; a candidate outside the span may replace any member.
1233    let coefficient_coordinates = solve_coeff_combination(combinations, candidate_combination);
1234    let pauli_coordinates = solve_pauli_combination(rows, candidate);
1235    for index in witness_start..rows.len() {
1236        if coefficient_coordinates
1237            .as_ref()
1238            .is_some_and(|coordinates| !coordinates.bit(index))
1239            || pauli_coordinates
1240                .as_ref()
1241                .is_some_and(|coordinates| !coordinates.bit(index))
1242        {
1243            continue;
1244        }
1245
1246        return Some(index);
1247    }
1248    None
1249}

1251fn isolate_measurement_target_in_suffix(
1252    rows: &mut [PauliString],
1253    combinations: &mut [CoeffVec],
1254    target_col: usize,
1255    target_basis: Pauli,
1256) {
1257    if rows.is_empty() {
1258        return;
1259    }
1260
1261    let (pivot_row, rows) = rows.split_first_mut().expect("nonempty suffix");
1262    let (pivot_combination, combinations) = combinations
1263        .split_first_mut()
1264        .expect("tracked suffix has matching length");
1265    for (row, combination) in rows.iter_mut().zip(combinations) {
1266        if row.get(target_col) & target_basis {
1267            *row ^= &*pivot_row;
1268            combination.xor_assign(pivot_combination);
1269        }
1270    }
1271}

1273fn conflicting_measurement_pivots(
1274    metadata: &MeasurementMetadata,
1275    name: &str,
1276) -> Vec<(usize, Pauli)> {
1277    let mut pivots = Vec::new();
1278
1279    for (other_name, &col) in &metadata.cols {
1280        if other_name.as_str() == name {
1281            continue;
1282        }
1283
1284        let Some(basis) = metadata.basis_overlap(other_name.as_str()) else {
1285            continue;
1286        };
1287
1288        pivots.extend(basis.iter_xz().map(|axis| (col, axis)));

1290
1291    pivots.sort_unstable_by_key(|&(col, basis)| (col, u8::from(basis)));
1292    pivots

1310/// Diagonalizes every named measurement row when possible, leaving the basis
1311/// unchanged when the rows cannot be isolated without invalidating a selective
1312/// fixing row.
1313pub(super) fn diagonalize_measurement_columns(
1314    final_basis: &mut TrackedBasis,
1315    measurement_names: &[String],
1316    metadata: &MeasurementMetadata,
1317    selective_constraints: &[SelectiveConstraint],
1318    fixing_kinds: &[StabilizerRowKind],
1319) {
1320    if measurement_names.is_empty() {
1321        return;
1322    }
1323    let original = final_basis.clone();
1324    for (index, name) in measurement_names.iter().enumerate() {
1325        let Some(target_basis) = metadata.basis_overlap(name) else {
1326            continue;
1327        };
1328        let target_col = metadata.column(name);
1329        let (before, pivot_and_after) = final_basis.rows.split_at_mut(index);
1330        let (pivot_row, after) = pivot_and_after
1331            .split_first_mut()
1332            .expect("measurement pivot is in bounds");
1333        let (before_coeffs, pivot_and_after_coeffs) = final_basis.coeffs.split_at_mut(index);
1334        let (pivot_coeff, after_coeffs) = pivot_and_after_coeffs
1335            .split_first_mut()
1336            .expect("tracked measurement pivot is in bounds");
1337        for (row, coeff) in before
1338            .iter_mut()
1339            .zip(before_coeffs)
1340            .chain(after.iter_mut().zip(after_coeffs))
1341        {
1342            if row.get(target_col) & target_basis {
1343                *row ^= &*pivot_row;
1344                coeff.xor_assign(pivot_coeff);
1345            }
1346        }
1347    }
1348
1349    let measurements_are_isolated = measurement_names
1350        .iter()
1351        .enumerate()
1352        .all(|(index, name)| metadata.support_is_isolated(name, &final_basis.rows[index]));
1353    // Each fixing row must still own the forbidden Pauli at every site it was
1354    // tagged for. Indexed by push order rather than by constraint, since one
1355    // joint row can fix several sites.
1356    let fixing_rows_are_valid = fixing_kinds.iter().enumerate().all(|(offset, kind)| {
1357        let row = &final_basis.rows[measurement_names.len() + offset];
1358        kind.selective_fixing_targets().iter().all(|target| {
1359            selective_constraints
1360                .iter()
1361                .find(|constraint| constraint.pos == target.pos)
1362                .is_some_and(|constraint| row.get(constraint.col) == constraint.forbidden)
1363        })
1364    });
1365    if !measurements_are_isolated || !fixing_rows_are_valid {
1366        *final_basis = original;
1367    }
1368}
1369pub(crate) fn materialize_measurement_first_basis(
1370    zx: &ZXGraph,
1371    rows: Vec<PauliString>,
1372    row_kinds: Vec<StabilizerRowKind>,
1373    public_len: usize,
1374    row_phases: Vec<u8>,
1375    specs: &[MeasurementSpec],
1376) -> Result<StabilizerGenerators, StabilizerError> {
1377    assert_eq!(
1378        rows.len(),
1379        row_kinds.len(),
1380        "every stabilizer row has one role"
1381    );
1382
1383    // Accepted deadline per named row: the layer its surface promised to close
1384    // by (`accepted_surface_deadline`, written back into the specs by
1385    // canonical table construction. Used to check the materialized surfaces
1386    // still honour that promise.
1387    let deadlines = specs
1388        .iter()
1389        .map(|spec| (spec.name.as_str(), spec.deadline))
1390        .collect::<HashMap<_, _>>();
1391    // Selective-fixing sites are exempt from the extent bound: their arm is
1392    // pinned by their own `resolve` and the surface is re-derived against the
1393    // filled basis at runtime, so a surface routing through one stays legal at
1394    // its own deadline (mirrors `accepted_surface_deadline`'s `node_gate`).
1395    let selective_positions = collect_selective_constraints(zx)
1396        .iter()
1397        .map(|constraint| constraint.pos)
1398        .collect::<std::collections::HashSet<_>>();
1399
1400    assert!(public_len <= rows.len(), "public rows form a valid prefix");
1401    assert_eq!(rows.len(), row_phases.len(), "every row has one phase");
1402    let mut generators = Vec::with_capacity(public_len);
1403    let mut auxiliary_rows = Vec::new();
1404    for (index, (row, kind)) in rows.into_iter().zip(row_kinds).enumerate() {
1405        if index >= public_len {
1406            auxiliary_rows.push(row);
1407            continue;
1408        }
1409        let stabilizer = zx.pauli_string_to_stabilizer_with_row_phase(row, row_phases[index]);
1410        if let StabilizerRowKind::Measurement { name } = &kind {
1411            let deadline = *deadlines
1412                .get(name.as_str())
1413                .expect("every materialized measurement name has an accepted deadline");
1414            check_measurement_extent(name, &stabilizer, deadline, &selective_positions)?;
1415        }
1416        generators.push(StabilizerGenerator::new(stabilizer, kind));
1417    }
1418
1419    Ok(StabilizerGenerators::from_parts(
1420        zx.clone(),
1421        generators,
1422        auxiliary_rows,
1423    ))
1424}

1426/// Checks a materialized measurement surface still closes by the deadline its
1427/// construction accepted: interior support at layer z makes the parity readable
1428/// only from layer z + 1 on, so non-exempt support must stay strictly below the
1429/// deadline. The transformations that run after acceptance (fixing rows, basis
1430/// completion, diagonalization) are supposed to leave the frozen measurement
1431/// prefix alone; this catches it when one does not.
1432///
1433/// Support past the *measure* layer is fine — the deadline moved with it.
1434/// Consumers are dataflow-scheduled after the resulting decode, independent of
1435/// their graph z coordinate. Output-port anchors are unbounded in graph
1436/// analysis and are not interior nodes; compiler preflight rejects them.
1437/// Selective-fixing sites are excluded (see `selective_positions` above).
1438fn check_measurement_extent(
1439    name: &str,
1440    stabilizer: &Stabilizer,
1441    deadline: i64,
1442    selective_positions: &std::collections::HashSet<glam::IVec3>,
1443) -> Result<(), StabilizerError> {
1444    let Some(max_z) = stabilizer
1445        .interior_nodes
1446        .keys()
1447        .filter(|pos| {
1448            !selective_positions.contains(*pos) && !stabilizer.port_stabilizer.contains_key(*pos)
1449        })
1450        .map(|pos| pos.z)
1451        .max()
1452    else {
1453        return Ok(());
1454    };
1455    if i64::from(max_z) >= deadline {
1456        return Err(StabilizerError::MeasurementSurfaceExtendsPastDeadline {
1457            name: name.to_string(),
1458            max_z,
1459            deadline,
1460        });
1461    }
1462    Ok(())
1463}

1918    /// The kernel repair must not undo a column it already fixed.
1919    ///
1920    /// Only `XX` moves `c1` off `Z`, and it flips `c2` onto `Z` on the way;
1921    /// `c2` then has to be cleaned up by a row that leaves `c1` alone. Drawing
1922    /// the second pivot from the whole kernel picks `XX` again and puts `c1`
1923    /// back on its forbidden Pauli — the sum `XI` is the only repair.
1924    #[test]
1925    fn selective_repair_keeps_earlier_columns_clear() {
1926        let (c1, c2) = (0, 1);
1927        let constraints = [
1928            SelectiveConstraint::new(IVec3::ZERO, SelectiveKind::XY, c1),
1929            SelectiveConstraint::new(IVec3::ONE, SelectiveKind::XY, c2),
1930        ];
1931        let kernel = [
1932            PauliString::from_terms(2, [(c1, Pauli::X), (c2, Pauli::X)]),
1933            PauliString::from_terms(2, [(c1, Pauli::X)]),
1934        ];

1936

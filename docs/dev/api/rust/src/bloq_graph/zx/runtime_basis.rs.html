17/// An error from deriving or updating a [`RuntimeStabilizerBasis`].
18#[derive(Debug, Clone, Error)]
19#[non_exhaustive]
20pub enum RuntimeBasisError {
21    /// A selective site is absent from the live basis.
22    #[error("selective node at {0} is not part of the live basis")]
23    UnknownSelective(IVec3),
24    /// A selective site occurs twice in one assignment.
25    #[error("selective node at {0} appears more than once in one assignment")]
26    DuplicateSelective(IVec3),
27    /// A chosen basis is not allowed at the selective site.
28    #[error("basis {chosen} is not allowed at selective {pos} of kind {kind}")]
29    InvalidSelectiveBasis {
30        /// Selective position.
31        pos: IVec3,
32        /// Selective kind.
33        kind: SelectiveKind,
34        /// Chosen basis.
35        chosen: PauliBasis,
36    },
37    /// A measurement variable is absent from the live graph.
38    #[error("measurement variable {0:?} is not present in the live ZX graph")]
39    UnknownMeasurement(String),
40    /// Filling non-Clifford computation is unsupported.
41    #[error("selective fill is only supported for Clifford-only computation")]
42    NonCliffordFillUnsupported,
43    /// An output is absent from the live graph.
44    #[error("output port {0} is not present in the live ZX graph")]
45    UnknownOutput(IVec3),
46    /// ZX graph update failed.
47    #[error("{0}")]
48    ZX(#[from] ZXError),
49    /// Stabilizer-basis derivation failed.
50    #[error("{0}")]
51    Stabilizer(#[from] StabilizerError),
52}

54/// The live stabilizer basis of a ZX graph, tracking the generators and the
55/// selective nodes whose runtime basis choice can still update it.
56///
57/// This is the state that dynamic (selective) fills mutate as each selective is
58/// resolved, and the source of output-correction surfaces.
59#[derive(Debug, Clone)]
60pub struct RuntimeStabilizerBasis {
61    zx_graph: ZXGraph,
62    // Filling treats native T states as ports, but does not turn prepared
63    // resources into arbitrary inputs of the authored map.
64    input_columns: Vec<usize>,
65    prepared_columns: Vec<usize>,
66    rows: Vec<PauliString>,
67    /// Readable-value provenance of every live row, expressed in the original
68    /// public generator ordinals. Rank-completion directions carry the zero
69    /// vector. Row operations update these coefficients with Pauli support.
70    readout_coeffs: Vec<CoeffVec>,
71    selective_kinds: HashMap<IVec3, SelectiveKind>,
72}
73
74impl RuntimeStabilizerBasis {
75    /// The live ZX graph backing this basis.
76    pub fn zx_graph(&self) -> &ZXGraph {
77        &self.zx_graph
78    }
79
80    /// The Pauli-string generators of this basis, in row order.
81    pub fn generators(&self) -> &[PauliString] {
82        &self.rows
83    }
84
85    /// Demotes every [`T`](NodeKind::T) node to a plain [`Port`](NodeKind::Port).
86    /// Used when selective filling treats prepared T states as open inputs.
87    pub fn with_t_nodes_as_ports(mut self) -> Self {
88        self.zx_graph.demote_t_nodes_to_ports();
89        self
90    }

92    /// A complete basis of the graph's correlation relation, with every row
93    /// normalized to permitted selective support.
94    ///
95    /// # Errors
96    ///
97    /// Returns a [`RuntimeBasisError`] if a selective site admits no permitted
98    /// support.
99    pub fn from_zx_graph(zx_graph: ZXGraph) -> Result<Self, RuntimeBasisError> {
100        let mut rows = reduce_to_basis(&zx_graph.to_external_generator_table(), zx_graph.total_ids);
101        // Normalization adds rows to rows, so they remain an independent basis.
102        normalize_selective_support_in_place(
103            &mut rows,
104            None,
105            &collect_selective_constraints(&zx_graph),
106        )?;
107        let public_len = rows.len();
108        Ok(Self::new(zx_graph, rows, public_len))
109    }
110
111    /// Builds the runtime basis from a [`StabilizerGenerators`] presentation,
112    /// reusing its public rows and then its rank completion verbatim.
113    ///
114    /// The *n*th runtime generator is the *n*th input generator, so the row
115    /// ordinals downstream code keys on stay identity-stable.
116    pub fn from_generators(stabilizers: &StabilizerGenerators) -> Self {
117        let rows = stabilizers
118            .generators
119            .iter()
120            .map(|generator| generator.stabilizer.paulis.clone())
121            .chain(stabilizers.auxiliary_rows().iter().cloned())
122            .collect();
123        Self::new(
124            stabilizers.zx_graph.clone(),
125            rows,
126            stabilizers.generators.len(),
127        )
128    }
129
130    fn new(zx_graph: ZXGraph, rows: Vec<PauliString>, public_len: usize) -> Self {
131        let width = rows.len();
132        let readout_coeffs = (0..width)
133            .map(|index| {
134                if index < public_len {
135                    CoeffVec::singleton(index, width)
136                } else {
137                    CoeffVec::zeros(width)
138                }
139            })
140            .collect();
141        let (input_columns, prepared_columns) = authored_boundary_columns(&zx_graph);
142        Self {
143            input_columns,
144            prepared_columns,
145            selective_kinds: collect_selective_constraints(&zx_graph)
146                .into_iter()
147                .map(|constraint| (constraint.pos, constraint.kind))
148                .collect(),
149            zx_graph,
150            rows,
151            readout_coeffs,
152        }
153    }

155    /// A complete basis whose readable coordinates are the source outcomes:
156    /// coordinate `i` of a row is its measured component at the `i`th named
157    /// measurement in action order. T nodes become ports. Crossing-node and Y
158    /// readouts have no linear coordinate.
159    ///
160    /// # Errors
161    ///
162    /// Returns an error if a named source measurement lacks a concrete linear
163    /// coordinate, or a selective site admits no permitted support.
164    pub fn for_source_readouts(zx_graph: ZXGraph) -> Result<Self, RuntimeBasisError> {
165        let mut basis = Self::from_zx_graph(zx_graph)?.with_t_nodes_as_ports();
166        let measurements = basis
167            .zx_graph
168            .action_graph()
169            .ordered_nodes()
170            .filter(|node| matches!(node.action, crate::Action::Measure { .. }))
171            .collect::<Vec<_>>();
172        let mut readout_coeffs = vec![CoeffVec::zeros(measurements.len()); basis.rows.len()];
173        for (ordinal, node) in measurements.into_iter().enumerate() {
174            let crate::Action::Measure { name, target } = &node.action else {
175                unreachable!("filtered to measurements")
176            };
177            let unavailable =
178                || StabilizerError::MeasurementSurfaceUnavailable { mvar: name.clone() };
179            let axis = match node.measurement {
180                Some(crate::MeasurementObservable::Concrete(PauliBasis::X)) => Pauli::X,
181                Some(crate::MeasurementObservable::Concrete(PauliBasis::Z)) => Pauli::Z,
182                _ => return Err(unavailable().into()),
183            };
184            let col = basis
185                .zx_graph
186                .measurement_column(target)
187                .ok_or_else(|| RuntimeBasisError::UnknownMeasurement(name.clone()))?;
188            if basis.zx_graph.nodes().get(col).is_some_and(|node| {
189                matches!(node.kind, NodeKind::X | NodeKind::Z) && node.kind.cross_pauli() == axis
190            }) {
191                return Err(unavailable().into());
192            }
193            for (row, coeff) in basis.rows.iter().zip(&mut readout_coeffs) {
194                coeff.set_bit(ordinal, row.get(col) & axis);
195            }
196        }
197        basis.readout_coeffs = readout_coeffs;
198        Ok(basis)
199    }

201    /// The relation restricted to zero source named-outcome coordinates, with
202    /// T nodes as ports. Closed named readouts can remove those coordinates
203    /// without changing any output operator, so this kernel retains the full
204    /// terminal frame space: the oracle that cross-checks the guarded planner.
205    #[cfg(test)]
206    pub(super) fn for_output_frames(zx: ZXGraph) -> Result<Self, RuntimeBasisError> {
207        let mut basis = Self::from_zx_graph(zx)?.with_t_nodes_as_ports();
208        let constraints = basis
209            .zx_graph
210            .action_graph()
211            .ordered_nodes()
212            .filter_map(|node| {
213                let crate::Action::Measure { target, .. } = &node.action else {
214                    return None;
215                };
216                let Some(crate::MeasurementObservable::Concrete(axis)) = node.measurement else {
217                    unreachable!("source readout measurements have concrete observables")
218                };
219                Some((
220                    basis
221                        .zx_graph
222                        .measurement_column(target)
223                        .expect("named coordinate"),
224                    Pauli::from(axis),
225                ))
226            })
227            .collect::<Vec<_>>();
228        basis.restrict_to_zero(&constraints);
229        Ok(basis)
230    }

232    #[cfg(test)]
233    fn restrict_to_zero(&mut self, constraints: &[(usize, Pauli)]) {
234        let rank = gaussian_elimination_with_tracking(
235            &mut self.rows,
236            &mut self.readout_coeffs,
237            constraints.iter().copied(),
238            |row, &(col, axis)| row.get(col) & axis,
239            None,
240        );
241        self.rows.drain(..rank);
242        self.readout_coeffs.drain(..rank);
243    }

245    /// Resolves the selective node at `pos` to the `chosen` basis and returns
246    /// the evolved basis. Readable provenance stays internal to that basis.
247    ///
248    /// # Errors
249    ///
250    /// Returns a [`RuntimeBasisError`] if `pos` is not a live selective, the
251    /// basis is not allowed for its kind, or the computation is not Clifford.
252    pub fn apply_selective_fill(
253        &self,
254        pos: IVec3,
255        chosen: PauliBasis,
256    ) -> Result<RuntimeStabilizerBasis, RuntimeBasisError> {
257        self.apply_selective_fills(&[(pos, chosen)])
258    }
259
260    /// Resolves one known assignment in order, filling the graph once. Each
261    /// fill eliminates the rows that disagree with its chosen axis, then
262    /// re-normalizes the remaining selective support.
263    ///
264    /// # Errors
265    ///
266    /// Returns [`RuntimeBasisError`] when a requested fill is invalid.
267    ///
268    /// # Panics
269    ///
270    /// Panics if validated selective metadata is internally inconsistent.
271    pub fn apply_selective_fills(
272        &self,
273        fills: &[(IVec3, PauliBasis)],
274    ) -> Result<RuntimeStabilizerBasis, RuntimeBasisError> {
275        if fills.is_empty() {
276            return Ok(self.clone());
277        }
278        if !self.zx_graph.is_clifford_computation() {
279            return Err(RuntimeBasisError::NonCliffordFillUnsupported);
280        }
281
282        let mut port_map = HashMap::with_capacity(fills.len());
283        for &(pos, chosen) in fills {
284            let kind = self
285                .selective_kinds
286                .get(&pos)
287                .copied()
288                .ok_or(RuntimeBasisError::UnknownSelective(pos))?;
289            insert_selective_fill(&mut port_map, pos, kind, chosen)?;
290        }
291
292        let mut rows = self.rows.clone();
293        let mut readout_coeffs = self.readout_coeffs.clone();
294        let mut remaining = collect_selective_constraints(&self.zx_graph);
295        for &(pos, chosen) in fills {
296            let col = self
297                .zx_graph
298                .node_at(pos)
299                .expect("validated selective has a ZX node")
300                .id;
301            remaining.retain(|constraint| constraint.pos != pos);
302            let pivots = gaussian_elimination_with_tracking(
303                &mut rows,
304                &mut readout_coeffs,
305                [(col, Pauli::from(chosen))],
306                |row, &(col, chosen)| {
307                    let pauli = row.get(col);
308                    pauli != Pauli::I && pauli != chosen
309                },
310                None,
311            );
312            rows.drain(..pivots);
313            readout_coeffs.drain(..pivots);
314            normalize_selective_support_in_place(&mut rows, Some(&mut readout_coeffs), &remaining)?;
315        }
316
317        Ok(RuntimeStabilizerBasis {
318            input_columns: self.input_columns.clone(),
319            prepared_columns: self.prepared_columns.clone(),
320            zx_graph: self.zx_graph.fill_ports(&port_map)?,
321            rows,
322            readout_coeffs,
323            selective_kinds: remaining
324                .into_iter()
325                .map(|constraint| (constraint.pos, constraint.kind))
326                .collect(),
327        })
328    }

330    /// Derives compact output rows together with their full stabilizer
331    /// surfaces. Consumers that must account for effects outside readable
332    /// generator provenance, such as feedback on auxiliary directions, use
333    /// the materialized surface.
334    ///
335    /// # Errors
336    ///
337    /// Returns an error if an output is unknown or cannot be supported.
338    pub fn derive_output_correction_surfaces(
339        &self,
340        outputs: &[IVec3],
341    ) -> Result<Vec<(OutputCorrectionRow, Stabilizer)>, RuntimeBasisError> {
342        self.clone().into_output_correction_surfaces(outputs)
343    }
344
345    /// Closed Clifford surfaces constrain internal outcome assignments without
346    /// depending on the state at any open input or output (including T ports).
347    #[cfg(feature = "verify")]
348    pub(crate) fn closed_stabilizer_surfaces(&self) -> Vec<Stabilizer> {
349        let mut rows = self.rows.clone();
350        let constraints = self
351            .zx_graph
352            .nodes()
353            .iter()
354            .filter(|node| matches!(node.kind, NodeKind::Port | NodeKind::T))
355            .flat_map(|node| [(node.id, Pauli::X), (node.id, Pauli::Z)]);
356        let consumed = gaussian_elimination_with_tracking(
357            &mut rows,
358            &mut [],
359            constraints,
360            |row, &(column, axis)| row.get(column) & axis,
361            None,
362        );
363        rows.into_iter()
364            .skip(consumed)
365            .map(|row| self.zx_graph.materialize_stabilizer(row))
366            .collect()
367    }
368
369    /// Consumes this basis while deriving output corrections, moving its dense
370    /// rows and provenance into the solve instead of cloning them.
371    ///
372    /// # Errors
373    ///
374    /// Returns an error if an output is unknown or cannot be supported.
375    pub fn into_output_correction_surfaces(
376        self,
377        outputs: &[IVec3],
378    ) -> Result<Vec<(OutputCorrectionRow, Stabilizer)>, RuntimeBasisError> {
379        let output_cols = requested_output_columns(&self.zx_graph, outputs)?;
380        let protected =
381            protected_boundary_columns(&self.zx_graph, &self.prepared_columns, &output_cols);
382        Ok(pivot_output_correction_rows(
383            self.rows,
384            self.readout_coeffs,
385            outputs,
386            &output_cols,
387            &self.input_columns,
388            &protected,
389        )
390        .into_iter()
391        .map(|(correction, row)| (correction, self.zx_graph.materialize_stabilizer(row)))
392        .collect())
393    }
394}

396/// Capture boundary ownership before native resources are relaxed for fills.
397pub(super) fn authored_boundary_columns(zx: &ZXGraph) -> (Vec<usize>, Vec<usize>) {
398    let mut nodes = zx
399        .nodes()
400        .iter()
401        .filter(|node| node.kind == NodeKind::T || node.is_input_port(zx))
402        .collect::<Vec<_>>();
403    nodes.sort_unstable_by_key(|node| node.pos.to_array());
404    let inputs = nodes
405        .iter()
406        .filter(|node| node.is_input_port(zx))
407        .map(|node| node.id)
408        .collect();
409    let prepared = nodes
410        .iter()
411        .filter(|node| node.kind == NodeKind::T)
412        .map(|node| node.id)
413        .collect();
414    (inputs, prepared)
415}
416
417/// A partial-output query must also preserve every other live output.
418fn protected_boundary_columns(zx: &ZXGraph, prepared: &[usize], outputs: &[usize]) -> Vec<usize> {
419    let mut columns = prepared
420        .iter()
421        .copied()
422        .chain(
423            zx.nodes()
424                .iter()
425                .filter(|node| node.is_output_port(zx) && !outputs.contains(&node.id))
426                .map(|node| node.id),
427        )
428        .collect::<Vec<_>>();
429    columns.sort_unstable_by_key(|&column| zx.nodes()[column].pos.to_array());
430    columns.dedup();
431    columns
432}
433
434fn requested_output_columns(
435    zx_graph: &ZXGraph,
436    outputs: &[IVec3],
437) -> Result<Vec<usize>, RuntimeBasisError> {
438    outputs
439        .iter()
440        .copied()
441        .map(|output| {
442            let node_id = zx_graph
443                .node_at(output)
444                .filter(|node| node.is_output_port(zx_graph))
445                .map(|node| node.id)
446                .ok_or(RuntimeBasisError::UnknownOutput(output))?;
447            Ok(node_id)
448        })
449        .collect()
450}

2//!
3//! Planning owns the working relation and may collect Boolean decisions.

5//! now hold decision ids. Dense ZX rows are materialized only for diagnostics.

27    /// Derive source action prerequisites from the frozen causal recipes.
28    /// Local cases preserve support and endpoint frames without listing joint
29    /// branch assignments. This diagnostic consumes the same bounded queries as
30    /// physical binding and never collects while its decision ids are live.
31    pub(crate) fn action_dependencies(
32        &mut self,
33    ) -> Result<Vec<(usize, usize, crate::ActionDependency)>, BlockGraphError> {
34        use crate::{Action, ActionDependency, GuardedSurfaceKind, GuardedVariable};
35
36        let actions = self.topology.source.actions();
37        let names = actions
38            .iter()
39            .enumerate()
40            .filter_map(|(ordinal, action)| match action {
41                Action::Measure { name, .. } | Action::Let { name, .. } => {
42                    Some((name.clone(), ordinal))
43                }
44                _ => None,
45            })
46            .collect::<BTreeMap<_, _>>();
47        let resolves = actions
48            .iter()
49            .enumerate()
50            .filter_map(|(ordinal, action)| match action {
51                Action::Resolve { target, .. } => Some((target.to_array(), ordinal)),
52                _ => None,
53            })
54            .collect::<BTreeMap<_, _>>();
55        let branches = self.topology.branches.iter().filter_map(|(name, target, _)| {
56            actions.iter().position(|action| matches!(action, Action::Branch { target: site, .. } if site == target))
57                .map(|ordinal| (name.clone(), ordinal))
58        }).collect::<BTreeMap<_, _>>();
59        let mut owners = BTreeMap::new();
60        for region in self.topology.source.branch_regions()? {
61            for block in region.on_true().blocks().chain(region.on_false().blocks()) {
62                owners.insert(block.pos().to_array(), branches[&region.name]);
63            }
64        }
65        let mut edges = Vec::new();
66        for index in 0..self.surfaces.len() {
67            let GuardedSurfaceKind::Readout { name, folds } = self.surfaces[index].kind.clone()
68            else {
69                continue;
70            };
71            let consumer = names[&name];
72            let mut roots = BTreeSet::from([self.surfaces[index].row.active]);
73            self.topology
74                .diagram
75                .charge(self.surfaces[index].row.terms().len())?;
76            roots.extend(
77                self.surfaces[index]
78                    .row
79                    .terms()
80                    .iter()
81                    .map(|&(_, root)| root),
82            );
83            for (predecessor, guard) in folds {
84                if self
85                    .topology
86                    .diagram
87                    .constrain(guard, self.topology.domain)?
88                    != ZERO
89                {
90                    edges.push((
91                        names[&predecessor],
92                        consumer,
93                        ActionDependency::ReadoutParity,
94                    ));
95                    roots.insert(guard);
96                }
97            }
98            for (ordinal, guard) in self.surfaces[index].feedbacks.clone() {
99                if self
100                    .topology
101                    .diagram
102                    .constrain(guard, self.topology.domain)?
103                    != ZERO
104                {
105                    edges.push((
106                        ordinal as usize,
107                        consumer,
108                        ActionDependency::FeedbackAnticommutation,
109                    ));
110                    roots.insert(guard);
111                }
112            }
113            for position in self.support_sites(index) {
114                for case in self.local_cases(index, IVec3::from_array(position), ONE)? {
115                    roots.insert(case.guard);
116                    // Even a constant selector must choose the physical owner
117                    // before a record in that branch can be read.
118                    if let Some(&predecessor) = owners.get(&position) {
119                        edges.push((predecessor, consumer, ActionDependency::BranchSupport));
120                    }
121                    if case.selective.is_some()
122                        && let Some(&predecessor) = resolves.get(&position)
123                    {
124                        edges.push((predecessor, consumer, ActionDependency::SelectiveSupport));
125                    }
126                }
127            }
128            for root in roots {
129                self.topology.diagram.charge(1)?;
130                if root == ZERO || root == ONE {
131                    continue;
132                }
133                let root = self
134                    .topology
135                    .diagram
136                    .constrain(root, self.topology.domain)?;
137                if root == ZERO || root == ONE {
138                    continue;
139                }
140                self.topology
141                    .diagram
142                    .charge(self.topology.diagram.nodes().len())?;
143                for variable in self.topology.diagram.variables(root) {
144                    let (predecessor, reason) = match &self.topology.variables[variable] {
145                        GuardedVariable::Branch { name, .. } => {
146                            (branches.get(name), ActionDependency::BranchSupport)
147                        }
148                        GuardedVariable::Outcome(name) => {
149                            (names.get(name), ActionDependency::ReadoutParity)
150                        }
151                    };
152                    if let Some(&predecessor) = predecessor {
153                        edges.push((predecessor, consumer, reason));
154                    }
155                }
156            }
157        }
158        edges.sort_by_key(|&(from, to, reason)| (from, to, reason as u8));
159        edges.dedup();
160        Ok(edges)
161    }
162}

164/// Position-based Pauli support consumed by physical gateways. Edge values use
165/// the frame of their first endpoint; a reverse lookup is deliberately distinct.
166/// Each directed endpoint pair occurs once.
167/// Neither a dense ZX id space nor a global stabilizer sign is required here.
168#[doc(hidden)]
169pub trait SurfaceSupport {
170    fn node_pauli(&self, position: IVec3) -> Pauli;
171    fn port_pauli(&self, position: IVec3) -> Pauli;
172    fn edge_pauli(&self, source: IVec3, target: IVec3) -> Option<Pauli>;
173    fn edges(&self) -> impl Iterator<Item = ((IVec3, IVec3), Pauli)>;
174
175    /// Directed edges owned by sorted source positions, or every edge if absent.
176    fn owned_edges(
177        &self,
178        positions: Option<&[IVec3]>,
179    ) -> impl Iterator<Item = ((IVec3, IVec3), Pauli)> {
180        self.edges().filter(move |((source, _), _)| {
181            positions.is_none_or(|positions| {
182                positions
183                    .binary_search_by_key(&source.to_array(), IVec3::to_array)
184                    .is_ok()
185            })
186        })
187    }
188}

190impl SurfaceSupport for Stabilizer {
191    fn node_pauli(&self, position: IVec3) -> Pauli {
192        self.interior_nodes
193            .get(&position)
194            .copied()
195            .unwrap_or(Pauli::I)
196    }
197
198    fn port_pauli(&self, position: IVec3) -> Pauli {
199        self.port_stabilizer
200            .get(&position)
201            .copied()
202            .unwrap_or(Pauli::I)
203    }
204
205    fn edge_pauli(&self, source: IVec3, target: IVec3) -> Option<Pauli> {
206        self.interior_edges.get(&(source, target)).copied()
207    }
208
209    fn edges(&self) -> impl Iterator<Item = ((IVec3, IVec3), Pauli)> {
210        self.interior_edges
211            .iter()
212            .map(|(&edge, &pauli)| (edge, pauli))
213    }

216/// One center and its incident edges, with no allocation proportional to the
217/// enclosing graph. The center's nonlinear crossing component is reconstructed
218/// from these arms after evaluating the final symbolic row.
219#[doc(hidden)]
220#[derive(Debug)]
221pub struct LocalPauliSurface {
222    position: IVec3,
223    center: Pauli,
224    port: bool,
225    edges: Vec<((IVec3, IVec3), Pauli)>,
226}

228impl SurfaceSupport for LocalPauliSurface {
229    fn node_pauli(&self, position: IVec3) -> Pauli {
230        if position == self.position {
231            self.center
232        } else {
233            Pauli::I
234        }
235    }
236
237    fn port_pauli(&self, position: IVec3) -> Pauli {
238        if self.port {
239            self.node_pauli(position)
240        } else {
241            Pauli::I
242        }
243    }
244
245    fn edge_pauli(&self, source: IVec3, target: IVec3) -> Option<Pauli> {
246        self.edges
247            .iter()
248            .find(|(edge, _)| *edge == (source, target))
249            .map(|(_, pauli)| *pauli)
250    }
251
252    fn edges(&self) -> impl Iterator<Item = ((IVec3, IVec3), Pauli)> {
253        self.edges.iter().copied()
254    }
255}

310impl GuardedReadoutPlan {
311    pub(super) fn share_module_queries(
312        &mut self,
313        module_sites: &BTreeMap<[i32; 3], usize>,
314    ) -> Result<(), BlockGraphError> {
315        self.topology.diagram.charge(self.columns.len())?;
316        let column_modules = self
317            .columns
318            .iter()
319            .map(|column| {
320                let (SurfaceColumn::Node(position) | SurfaceColumn::Edge(position, _)) = column;
321                module_sites[position]
322            })
323            .collect::<Vec<_>>();
324        let mut shared = crate::FxHashMap::default();
325        let mut retained = 0usize;
326        for (surface, query) in self.surfaces.iter().enumerate() {
327            self.topology.diagram.charge(query.row.terms().len())?;
328            let mut modules = crate::FxHashMap::<usize, Vec<_>>::default();
329            if query.row.active != ZERO {
330                for &(column, coefficient) in query.row.terms() {
331                    let Some(&owner) = column_modules.get(column / 2) else {
332                        continue;
333                    };
334                    modules
335                        .entry(owner)
336                        .or_default()
337                        .push((column, coefficient));
338                }
339            }
340            let mut uses = Vec::new();
341            let mut modules = modules.into_iter().collect::<Vec<_>>();
342            modules.sort_unstable_by_key(|(owner, _)| *owner);
343            for (_, terms) in modules {
344                // Include activation and gateway role, all native half-edge
345                // coefficients, and the actual instance's columns. Missing
346                // coefficients are zero. Never equate output-only summaries.
347                let key = (
348                    query.row.active,
349                    matches!(query.kind, super::GuardedSurfaceKind::Readout { .. }),
350                    terms,
351                );
352                self.topology.diagram.charge(key.2.len())?;
353                let id = if let Some(&id) = shared.get(&key) {
354                    id
355                } else {
356                    let positions =
357                        key.2
358                            .iter()
359                            .map(|&(column, _)| match self.columns[column / 2] {
360                                SurfaceColumn::Node(position)
361                                | SurfaceColumn::Edge(position, _) => position,
362                            })
363                            .collect::<BTreeSet<_>>();
364                    self.topology.diagram.charge(positions.len())?;
365                    let id = self.fragments.len();
366                    self.fragments
367                        .push((surface, positions.into_iter().collect()));
368                    // ponytail: cap optional interning; uncommon oversized
369                    // fragments still bind exactly, without cache admission.
370                    const MAX_SHARED_TERMS: usize = 262_144;
371                    let terms = key.2.len().saturating_add(1);
372                    if terms <= MAX_SHARED_TERMS.saturating_sub(retained) {
373                        retained += terms;
374                        shared.insert(key, id);
375                    }
376                    id
377                };
378                uses.push(id);
379            }
380            self.fragment_uses.push(uses);
381        }
382        Ok(())
383    }

422    /// Charged collection work for this plan's arena, tape and owned handles.
423    /// Caller-owned handles must be charged separately before collection.
424    pub fn collection_work(&mut self) -> usize {
425        self.topology
426            .diagram
427            .nodes()
428            .len()
429            .saturating_add(self.topology.diagram.witness_collection_work())
430            .saturating_add(self.topology.decision_root_count())
431            .saturating_add(
432                self.surfaces
433                    .iter_mut()
434                    .flat_map(GuardedSurface::decisions_mut)
435                    .count(),
436            )
437            .saturating_add(
438                self.local
439                    .values()
440                    .flatten()
441                    .map(|local| 1 + usize::from(local.selector.is_some()))
442                    .sum::<usize>(),
443            )
444            .saturating_add(self.query_cache_terms)
445    }

447    /// Remap frozen recipes together with every caller-owned decision id.
448    /// Call only between local queries, after their returned cases are consumed.
449    /// Optional query images are discarded rather than retaining stale cache keys.
450    pub fn collect_garbage<'a>(
451        &'a mut self,
452        additional: impl IntoIterator<Item = &'a mut DecisionId>,
453    ) -> usize {
454        self.query_cache.clear();
455        self.query_cache_terms = 0;
456        let mut roots = Vec::new();
457        for surface in &mut self.surfaces {
458            roots.extend(surface.decisions_mut());
459        }
460        for local in self.local.values_mut().flatten() {
461            roots.push(&mut local.guard);
462            roots.extend(local.selector.iter_mut());
463        }
464        self.topology
465            .collect_garbage(roots.into_iter().chain(additional))
466    }

468    /// Only the requested site's Pauli pattern is materialized. Guards may
469    /// still depend on any number of source outcomes. Uniform raw support
470    /// excludes its identity branch before partitioning.
471    ///
472    /// # Errors
473    ///
474    /// Returns an error if Boolean evaluation or local materialization exceeds its limits.
475    ///
476    /// # Panics
477    ///
478    /// Panics if `surface` or `position` is not part of this plan.
479    pub fn local_cases(
480        &mut self,
481        surface: usize,
482        position: IVec3,
483        enabled: DecisionId,
484    ) -> Result<Vec<GuardedLocalSurface>, BlockGraphError> {
485        let row = &self.surfaces[surface].row;
486        let mut output = Vec::new();
487        for variant in &self.local[&position.to_array()] {
488            let mut gate = self
489                .topology
490                .diagram
491                .apply(BooleanOp::And, enabled, variant.guard)?;
492            gate = self
493                .topology
494                .diagram
495                .apply(BooleanOp::And, gate, row.active)?;
496            if gate == ZERO {
497                continue;
498            }
499            let column_count = usize::from(variant.center.is_some()) + variant.arms.len();
500            self.topology
501                .diagram
502                .charge(2 * column_count + usize::from(variant.selector.is_some()))?;
503            let mut support = ZERO;
504            let mut roots = Vec::new();
505            let mut coordinates =
506                Vec::with_capacity(2 * column_count + usize::from(variant.selector.is_some()));
507            let mut push_root = |root| {
508                let bit = match root {
509                    ZERO => LocalBit::Constant(false),
510                    ONE => LocalBit::Constant(true),
511                    root => {
512                        let index = roots
513                            .iter()
514                            .position(|&existing| existing == root)
515                            .unwrap_or_else(|| {
516                                roots.push(root);
517                                roots.len() - 1
518                            });
519                        LocalBit::Query(index)
520                    }
521                };
522                coordinates.push(bit);
523            };
524            for column in variant.columns() {
525                for root in [row.get(2 * column), row.get(2 * column + 1)] {
526                    if root != ZERO && support != ONE {
527                        support = if support == ZERO || support == root {
528                            root
529                        } else {
530                            ONE
531                        };
532                    }
533                    push_root(root);
534                }
535            }
536            // An all-I local surface contributes no physical records, boundary
537            // operators, or transport sign. A shared raw root gates every
538            // nonzero Pauli bit; mixed roots keep their original query.
539            if support == ZERO {
540                continue;
541            }
542            if support != ONE {
543                gate = self.topology.diagram.apply(BooleanOp::And, gate, support)?;
544                if gate == ZERO {
545                    continue;
546                }
547            }
548            if let Some(selector) = variant.selector {
549                push_root(selector);
550            }
551            let key = (self.topology.domain, gate, roots);
552            let cases = if let Some(cases) = self.query_cache.get(&key) {
553                Arc::clone(cases)
554            } else {
555                let cases: Arc<[(Vec<bool>, DecisionId)]> =
556                    self.topology.cases(&key.2, gate)?.into();
557                let terms = key.2.len()
558                    + cases
559                        .iter()
560                        .map(|(values, _)| values.len() + 1)
561                        .sum::<usize>();
562                // ponytail: bound retained coordinates; clear-on-full can be
563                // replaced by eviction if measured query churn warrants it.
564                const MAX_CACHED_TERMS: usize = 262_144;
565                if terms <= MAX_CACHED_TERMS {
566                    if self.query_cache_terms + terms > MAX_CACHED_TERMS {
567                        self.query_cache.clear();
568                        self.query_cache_terms = 0;
569                    }
570                    self.query_cache.insert(key, Arc::clone(&cases));
571                    self.query_cache_terms += terms;
572                }
573                cases
574            };
575            self.topology.diagram.charge(
576                cases
577                    .len()
578                    .saturating_mul(coordinates.len().saturating_add(1)),
579            )?;
580            for (values, guard) in cases.iter() {
581                let bit = |index| match coordinates[index] {
582                    LocalBit::Constant(value) => value,
583                    LocalBit::Query(index) => values[index],
584                };
585                // The first arm determines the linear center component. An
586                // isolated variant instead queries its independent coordinate.
587                let mut center = pauli(bit(0), bit(1));
588                let paulis = (usize::from(variant.center.is_some())..column_count)
589                    .map(|index| pauli(bit(2 * index), bit(2 * index + 1)));
590                let cross = match variant.kind {
591                    NodeKind::X | NodeKind::Z => Some(variant.kind.cross_pauli()),
592                    _ => None,
593                };
594                let mut crosses = false;
595                let mut edges = Vec::with_capacity(variant.arms.len());
596                for (arm, mut value) in variant.arms.iter().zip(paulis) {
597                    crosses |= cross.is_some_and(|cross| value & cross);
598                    if value == Pauli::I {
599                        continue;
600                    }
601                    let endpoints = if arm.reversed {
602                        if arm.hadamard {
603                            value = value.flip();
604                        }
605                        (arm.target, position)
606                    } else {
607                        (position, arm.target)
608                    };
609                    // Parallel adjacency repeats the same resolved edge and
610                    // coefficient. Match the full stabilizer's unique edge map.
611                    if !edges.iter().any(|(edge, _)| *edge == endpoints) {
612                        edges.push((endpoints, value));
613                    }
614                }
615                if let Some(cross) = cross {
616                    // Match exact ZX reconstruction: clear stale crossing
617                    // centers even when the last supported arm cancelled.
618                    if center & cross {
619                        center = center ^ cross;
620                    }
621                    if crosses {
622                        center = center | cross;
623                    }
624                }
625                output.push(GuardedLocalSurface {
626                    guard: *guard,
627                    surface: LocalPauliSurface {
628                        position,
629                        center,
630                        port: variant.kind.is_port(),
631                        edges,
632                    },
633                    selective: variant.selector.map(|_| bit(2 * column_count)),
634                });
635            }
636        }
637        let observed = self.topology.diagram.nodes().len();
638        if observed > self.max_witness_nodes {
639            return Err(StabilizerError::ResourceLimited {
640                phase: "guarded readout coefficients",
641                observed,
642                limit: self.max_witness_nodes,
643            }
644            .into());
645        }
646        Ok(output)
647    }

649    /// Visit nonzero row coefficients instead of scanning the global column
650    /// table. Auxiliary named-readout coordinates are not physical support.
651    pub fn support_sites(&self, surface: usize) -> BTreeSet<[i32; 3]> {
652        let row = &self.surfaces[surface].row;
653        if row.active == ZERO {
654            return BTreeSet::new();
655        }
656        row.terms()
657            .iter()
658            .filter_map(|&(column, _)| {
659                self.columns.get(column / 2).map(|column| match *column {
660                    SurfaceColumn::Node(position) | SurfaceColumn::Edge(position, _) => position,
661                })
662            })
663            .collect()
664    }

666    /// Materialize a frozen recipe for an observed source branch. Diagnostic
667    /// callers supply the independent ZX graph and its signed relation.
668    ///
669    /// # Panics
670    ///
671    /// Panics if `surface` is out of range or `zx` does not match this plan.
672    pub fn materialize_surface(
673        &self,
674        surface: usize,
675        zx: &ZXGraph,
676        outcome: impl Fn(&str) -> bool,
677    ) -> Option<Stabilizer> {
678        let row = &self.surfaces[surface].row;
679        let evaluate = |bit| self.topology.evaluate_source(bit, &outcome);
680        evaluate(row.active).then(|| {
681            let paulis = super::guarded_surfaces::materialize_row(
682                self.columns
683                    .iter()
684                    .copied()
685                    .enumerate()
686                    .map(|(index, column)| (column, index)),
687                row,
688                zx,
689                evaluate,
690            );
691            assert!(
692                zx.contains_stabilizer_support(&paulis),
693                "frozen surface {surface} must belong to the signed branch relation"
694            );
695            zx.materialize_stabilizer(paulis)
696        })
697    }

700fn pauli(x: bool, z: bool) -> Pauli {
701    match (x, z) {
702        (false, false) => Pauli::I,
703        (true, false) => Pauli::X,
704        (false, true) => Pauli::Z,
705        (true, true) => Pauli::Y,
706    }
707}

759        let mut pruned = 0;

761        let mut constants = 0;

764            let linked = crate::flatten_module_definition(&program, program.root(), "").unwrap();

772            .plan_readouts()
773            .unwrap();
774            for surface in 0..plan.surfaces.len() {
775                for position in plan
776                    .support_sites(surface)
777                    .into_iter()
778                    .map(IVec3::from_array)
779                {
780                    let row = &plan.surfaces[surface].row;
781                    let mut expected = Vec::new();
782                    for variant in &plan.local[&position.to_array()] {
783                        let gate = plan
784                            .topology
785                            .diagram
786                            .apply(BooleanOp::And, ONE, variant.guard)
787                            .and_then(|gate| {
788                                plan.topology
789                                    .diagram
790                                    .apply(BooleanOp::And, gate, row.active)
791                            })
792                            .unwrap();
793                        if gate == ZERO {
794                            continue;
795                        }
796                        let mut roots = variant
797                            .columns()
798                            .flat_map(|column| [row.get(2 * column), row.get(2 * column + 1)])
799                            .collect::<Vec<_>>();
800                        let raw_len = roots.len();
801                        repeated += usize::from(roots.iter().enumerate().any(|(index, root)| {
802                            *root != ZERO && *root != ONE && roots[..index].contains(root)
803                        }));
804                        constants += usize::from(roots.contains(&ZERO) || roots.contains(&ONE));
805                        let support = roots.iter().copied().find(|&root| root != ZERO);
806                        let uniform = support.is_none_or(|support| {
807                            roots.iter().all(|&root| root == ZERO || root == support)
808                        });
809                        roots.extend(variant.selector);
810                        for (values, guard) in plan.topology.cases(&roots, gate).unwrap() {
811                            if uniform && !values[..raw_len].contains(&true) {
812                                pruned += 1;
813                            } else {
814                                expected.push(guard);
815                            }
816                        }
817                    }
818                    expected.sort_unstable();
819                    let mut actual = plan
820                        .local_cases(surface, position, ONE)
821                        .unwrap()
822                        .into_iter()
823                        .map(|case| case.guard)
824                        .collect::<Vec<_>>();
825                    actual.sort_unstable();
826                    assert_eq!(actual, expected);
827                }
828            }
829        }
830        assert!(pruned > 0);
831        assert!(repeated > 0);
832        assert!(constants > 0);

902        let conditional = crate::lower_blog_graph_ast_deferred(
903            &crate::parse_blog_program_to_ast(include_str!(
904                "../../../docs/fixtures/conditional_cz_strip.blog"

914            GalleryItem::THTH,
915            GalleryItem::GHZSlideThenGlide,
916            GalleryItem::GHZPatchRotations,
917            GalleryItem::CCZInjectedAnd,
918            GalleryItem::Stability,
919        ]
920        .into_iter()
921        .map(GalleryItem::build)
922        .chain([conditional]);
923        let limits = ModuleCertificationLimits::DEFAULT;
924        for program in programs {
925            let linked = crate::flatten_module_definition(&program, program.root(), "").unwrap();
926            let graph = linked.graph.fix_shadowed_faces();
927            let relation = GuardedSurfaceSpace::new(
928                GuardedTopology::new(&graph, limits).unwrap(),
929                &linked.sites,
930                limits,
931            )
932            .unwrap();
933            let columns = relation.columns.clone();
934            let local = relation.local.clone();
935            let mut plan = relation.plan_readouts().unwrap();
936            // CSR adjacency may repeat an endpoint; every occurrence resolves
937            // to that pair's last edge. The visible surface still has one edge.
938            for topology in plan.local.values_mut().flatten() {
939                topology.arms = topology
940                    .arms
941                    .iter()
942                    .chain(topology.arms.first())
943                    .copied()
944                    .collect();
945            }
946            for surface in 0..plan.surfaces.len() {
947                for position in plan
948                    .support_sites(surface)
949                    .into_iter()
950                    .map(IVec3::from_array)
951                {
952                    for case in plan.local_cases(surface, position, ONE).unwrap() {
953                        let witness = plan.topology.witness(case.guard).unwrap().unwrap();
954                        let evaluate = |root| {
955                            plan.topology.diagram.evaluate(root, |variable| {
956                                witness.get(&variable).copied().unwrap_or(false)
957                            })
958                        };
959                        let variant = local[&position.to_array()]
960                            .iter()
961                            .find(|variant| evaluate(variant.guard))
962                            .unwrap();
963                        let zx = &variant.zx;
964                        let row = &plan.surfaces[surface].row;
965                        let mut dense = PauliString::new(zx.total_ids());
966                        for (&column, &index) in &columns {
967                            let value = pauli(
968                                evaluate(row.get(2 * index)),
969                                evaluate(row.get(2 * index + 1)),
970                            );
971                            match column {
972                                SurfaceColumn::Node(at) if at == position.to_array() => {
973                                    dense.set(variant.node, value)
974                                }
975                                SurfaceColumn::Edge(source, target)
976                                    if source == position.to_array() =>
977                                {
978                                    // A column can belong to another guarded local topology.
979                                    let Some(target) = zx.node_at(IVec3::from_array(target)) else {
980                                        continue;
981                                    };
982                                    let target = target.id;
983                                    let Some(edge) = zx.edge_between(variant.node, target) else {
984                                        continue;
985                                    };
986                                    let value = if edge.hadamard && variant.node > target {
987                                        value.flip()
988                                    } else {
989                                        value
990                                    };
991                                    dense.set(edge.id, value);
992                                    dense.set(zx.edge_id(target, variant.node).unwrap(), value);
993                                }
994                                _ => {}
995                            }
996                        }
997                        zx.reconstruct_raw_centers(&mut dense);
998                        let expected = zx.materialize_stabilizer_with_sign(dense, false);
999                        assert_eq!(
1000                            case.surface.node_pauli(position),
1001                            expected.node_pauli(position)
1002                        );
1003                        assert_eq!(
1004                            case.surface.port_pauli(position),
1005                            expected.port_pauli(position)
1006                        );
1007                        assert_eq!(case.surface.edges.len(), expected.interior_edges.len());
1008                        assert_eq!(
1009                            case.surface.edges().collect::<crate::FxHashMap<_, _>>(),
1010                            expected.interior_edges
1011                        );
1012                        let local_variant = plan.local[&position.to_array()]
1013                            .iter()
1014                            .find(|variant| evaluate(variant.guard))
1015                            .unwrap();
1016                        assert_eq!(case.selective, local_variant.selector.map(evaluate));
1017                        assert!(case.surface.edges.len() <= 6);
1018                    }
1019                }
1020            }
1021        }
1022    }

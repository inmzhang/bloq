51/// Availability for one fixed set of known outcomes. Decision nodes may be
52/// appended while querying, but the query must end before collection renumbers
53/// them or another outcome becomes known.
54pub(crate) struct AvailabilityQuery {
55    // None is an unknown outcome. A known outcome needs no further input;
56    // a structural variable instead requires its source condition.
57    requirements: Vec<Option<DecisionId>>,
58    values: crate::FxHashMap<DecisionId, bool>,
59    pending: Vec<DecisionId>,
60}
61
62impl AvailabilityQuery {
63    fn cached(&self, root: DecisionId) -> Option<bool> {
64        if root.0 < 2 {
65            Some(true)
66        } else {
67            self.values.get(&root).copied()
68        }
69    }
70
71    pub(crate) fn available(
72        &mut self,
73        root: DecisionId,
74        diagram: &mut BooleanDecisionDiagram,
75    ) -> Result<bool, BooleanResourceError> {
76        if let Some(value) = self.cached(root) {
77            return Ok(value);
78        }
79        self.pending.clear();
80        self.pending.push(root);
81        while let Some(&root) = self.pending.last() {
82            diagram.charge(1)?;
83            if self.cached(root).is_some() {
84                self.pending.pop();
85                continue;
86            }
87            let available = if let Some(node) = diagram.node(root) {
88                if let Some(condition) = self.requirements[node.variable] {
89                    let inputs = [condition, node.low, node.high];
90                    if inputs
91                        .iter()
92                        .any(|&input| self.cached(input) == Some(false))
93                    {
94                        false
95                    } else if let Some(input) = inputs
96                        .into_iter()
97                        .find(|&input| self.cached(input).is_none())
98                    {
99                        self.pending.push(input);
100                        continue;
101                    } else {
102                        true
103                    }
104                } else {
105                    false
106                }
107            } else {
108                true
109            };
110            self.values.insert(root, available);
111            self.pending.pop();
112        }
113        Ok(self
114            .cached(root)
115            .expect("the requested function was visited"))
116    }
117}

119/// Symbolic image of source Boolean functions. Disjoint source-variable
120/// supports factor, even when their output coordinates are interleaved.
121fn boolean_image(
122    source: &BooleanDecisionDiagram,
123    output: &mut BooleanDecisionDiagram,
124    roots: Vec<(usize, DecisionId)>,
125    limit: usize,
126) -> Result<DecisionId, BlockGraphError> {
127    // One traversal of the shared source DAG: a repeated node or source
128    // variable joins the roots that use it. Constant roots stay isolated.
129    output.charge(roots.len())?;
130    let mut components = petgraph::unionfind::UnionFind::new(roots.len());
131    let mut node_owner = HashMap::new();
132    let mut variable_owner = HashMap::new();
133    let mut pending = Vec::new();
134    for (root_index, &(_, root)) in roots.iter().enumerate() {
135        pending.push(root);
136        while let Some(id) = pending.pop() {
137            output.charge(1)?;
138            let Some(node) = source.node(id) else {
139                continue;
140            };
141            if let Some(&prior) = node_owner.get(&id) {
142                output.charge(1)?;
143                components.union(root_index, prior);
144                continue;
145            }
146            node_owner.insert(id, root_index);
147            if let Some(&prior) = variable_owner.get(&node.variable) {
148                output.charge(1)?;
149                components.union(root_index, prior);
150            } else {
151                variable_owner.insert(node.variable, root_index);
152            }
153            pending.extend([node.low, node.high]);
154        }
155    }
156
157    output.charge(roots.len())?;
158    let mut groups = Vec::<Vec<(usize, DecisionId)>>::new();
159    let mut group_by_component = HashMap::new();
160    for (root_index, root) in roots.into_iter().enumerate() {
161        let component = components.find(root_index);
162        let group = *group_by_component.entry(component).or_insert_with(|| {
163            groups.push(Vec::new());
164            groups.len() - 1
165        });
166        groups[group].push(root);
167    }
168
169    let mut visited = 0;
170    let mut image = DECISION_TRUE;
171    for group in groups {
172        output.charge(1)?;
173        let factor = match group.as_slice() {
174            [(index, root)] if *root == DECISION_TRUE => {
175                output.make_node(*index, DECISION_FALSE, DECISION_TRUE)?
176            }
177            [(index, root)] if *root == DECISION_FALSE => {
178                output.make_node(*index, DECISION_TRUE, DECISION_FALSE)?
179            }
180            [(_, _)] => {
181                // A nonconstant reduced BDD reaches both truth values. One
182                // output bit alone therefore has an unconstrained image.
183                visit_image_state(&mut visited, limit)?;
184                DECISION_TRUE
185            }
186            _ => boolean_image_component(source, output, group, &mut visited, limit)?,
187        };
188        image = output.apply(BooleanOp::And, image, factor)?;
189    }
190    Ok(image)
191}

193fn visit_image_state(visited: &mut usize, limit: usize) -> Result<(), BlockGraphError> {
194    let next = visited.checked_add(1);
195    *visited = next.unwrap_or(usize::MAX);
196    if next.is_none() || *visited > limit {
197        return Err(StabilizerError::ResourceLimited {
198            phase: "structural selector image",
199            observed: *visited,
200            limit,
201        }
202        .into());
203    }
204    Ok(())
205}

207/// Existing residual-tuple search, scoped to one correlated support component.
208/// `visited` is shared across all components, so partitioning cannot bypass
209/// the structural witness budget.
210fn boolean_image_component(
211    source: &BooleanDecisionDiagram,
212    output: &mut BooleanDecisionDiagram,
213    roots: Vec<(usize, DecisionId)>,
214    visited: &mut usize,
215    limit: usize,
216) -> Result<DecisionId, BlockGraphError> {
217    enum Step {
218        Visit(Vec<(usize, DecisionId)>),
219        Join {
220            remaining: Vec<(usize, DecisionId)>,
221            fixed: DecisionId,
222        },
223    }
224    let mut pending = vec![Step::Visit(roots)];
225    let mut values = Vec::new();
226    let mut memo = HashMap::new();
227    while let Some(step) = pending.pop() {
228        let roots = match step {
229            Step::Visit(roots) => roots,
230            Step::Join { remaining, fixed } => {
231                let high = values.pop().expect("visited high image");
232                let low = values.pop().expect("visited low image");
233                let image = output.apply(BooleanOp::Or, low, high)?;
234                memo.insert(remaining, image);
235                values.push(output.apply(BooleanOp::And, fixed, image)?);
236                continue;
237            }
238        };
239        output.charge(roots.len())?;
240        let mut fixed = DECISION_TRUE;
241        let mut remaining = Vec::new();
242        for (index, root) in roots {
243            if root.0 < 2 {
244                let value = if root == DECISION_TRUE {
245                    output.make_node(index, DECISION_FALSE, DECISION_TRUE)?
246                } else {
247                    output.make_node(index, DECISION_TRUE, DECISION_FALSE)?
248                };
249                fixed = output.apply(BooleanOp::And, fixed, value)?;
250            } else {
251                remaining.push((index, root));
252            }
253        }
254        if remaining.is_empty() {
255            values.push(fixed);
256            continue;
257        }
258        if let Some(&image) = memo.get(&remaining) {
259            values.push(output.apply(BooleanOp::And, fixed, image)?);
260            continue;
261        }
262        visit_image_state(visited, limit)?;
263        let variable = remaining
264            .iter()
265            .map(|(_, root)| {
266                source
267                    .node(*root)
268                    .expect("nonterminal root belongs to the source diagram")
269                    .variable
270            })
271            .min()
272            .expect("remaining source roots are nonempty");
273        output.charge(remaining.len().saturating_mul(2))?;
274        let branch = |high| {
275            remaining
276                .iter()
277                .map(|&(index, root)| {
278                    let node = source
279                        .node(root)
280                        .expect("nonterminal root belongs to the source diagram");
281                    (
282                        index,
283                        if node.variable != variable {
284                            root
285                        } else if high {
286                            node.high
287                        } else {
288                            node.low
289                        },
290                    )
291                })
292                .collect()
293        };
294        let low = branch(false);
295        let high = branch(true);
296        pending.extend([
297            Step::Join { remaining, fixed },
298            Step::Visit(high),
299            Step::Visit(low),
300        ]);
301    }
302    Ok(values.pop().expect("the source image produces one value"))
303}

305fn block_footprint_size<'a>(blocks: impl Iterator<Item = &'a crate::Block>) -> [usize; 2] {
306    blocks.fold([0usize; 2], |[count, cells], block| {
307        let footprint = if block.kind().is_cube() {
308            block.height_cells() as usize
309        } else {
310            block.kind().reserved_offsets().len()
311        };
312        [count.saturating_add(1), cells.saturating_add(footprint)]
313    })

325    fn matches_reference(&self, reference: &BlockGraph) -> bool {
326        self.primary_incidence.iter().all(|&(position, count)| {
327            let block = self
328                .blocks
329                .binary_search_by_key(&position.to_array(), |block| block.pos().to_array())
330                .map(|index| &self.blocks[index])
331                .expect("local primary block is retained");
332            reference.get_block(position) == Some(block)
333                && reference.pipes_at(position).count() == count
334        }) && self
335            .pipes
336            .iter()
337            .all(|pipe| reference.get_pipe(pipe.src(), pipe.dst()) == Some(pipe))
338    }
339}

354/// One source-wide index. Local materialization only visits incident records;
355/// distant branch choices are irrelevant to the center and its neighbors.
356#[derive(Debug)]
357struct AuthoredGeometry {
358    blocks: Vec<Authored<Block>>,
359    pipes: Vec<Authored<Pipe>>,
360    blocks_at: BTreeMap<[i32; 3], Vec<usize>>,
361    endpoint_owners: crate::FxHashMap<IVec3, Vec<usize>>,
362    incidence: crate::FxHashMap<IVec3, Vec<usize>>,
363    owners: crate::FxHashMap<IVec3, BTreeSet<DecisionId>>,
364    neighbors: crate::FxHashMap<IVec3, BTreeSet<[i32; 3]>>,
365}
366
367impl AuthoredGeometry {
368    fn new(
369        source: &BlockGraph,
370        regions: &[BranchRegion],
371        shared_pipes: &[Pipe],
372        branch_roots: &HashMap<IVec3, DecisionId>,
373    ) -> Result<Self, BlockGraphError> {
374        let shown = regions
375            .iter()
376            .flat_map(|region| region.shown_arm().blocks().map(Block::pos))
377            .collect::<std::collections::HashSet<_>>();
378        let mut blocks = source
379            .blocks()
380            .filter(|block| !shown.contains(&block.pos()))
381            .cloned()
382            .map(|value| Authored {
383                choice: None,
384                value,
385            })
386            .collect::<Vec<_>>();
387        let mut pipes = Vec::new();
388        for region in regions {
389            for value in [false, true] {
390                let choice = Some((branch_roots[&region.target], value));
391                blocks.extend(
392                    region
393                        .arm(value)
394                        .blocks()
395                        .cloned()
396                        .map(|value| Authored { choice, value }),
397                );
398                pipes.extend(
399                    region
400                        .arm(value)
401                        .pipes()
402                        .cloned()
403                        .map(|value| Authored { choice, value }),
404                );
405            }
406        }
407        // A selected arm pipe takes precedence over an identical common seam,
408        // matching materialize_projection's add-arms-then-shared-pipes order.
409        pipes.extend(shared_pipes.iter().cloned().map(|value| Authored {
410            choice: None,
411            value,
412        }));
413        let mut index = Self {
414            blocks,
415            pipes,
416            blocks_at: BTreeMap::new(),
417            endpoint_owners: crate::FxHashMap::default(),
418            incidence: crate::FxHashMap::default(),
419            owners: crate::FxHashMap::default(),
420            neighbors: crate::FxHashMap::default(),
421        };
422        for (id, record) in index.blocks.iter().enumerate() {
423            let position = record.value.pos();
424            index
425                .blocks_at
426                .entry(position.to_array())
427                .or_default()
428                .push(id);
429            if let Some((root, _)) = record.choice {
430                index.owners.entry(position).or_default().insert(root);
431            }
432            for offset in record.value.connectable_offsets() {
433                let endpoint = crate::checked_add_position(position, offset)?;
434                index.endpoint_owners.entry(endpoint).or_default().push(id);
435            }
436        }
437        for (id, record) in index.pipes.iter().enumerate() {
438            let (left, right) = record.value.try_endpoints()?;
439            let left = index
440                .endpoint_owners
441                .get(&left)
442                .into_iter()
443                .flatten()
444                .map(|&block| index.blocks[block].value.pos())
445                .collect::<std::collections::HashSet<_>>();
446            let right = index
447                .endpoint_owners
448                .get(&right)
449                .into_iter()
450                .flatten()
451                .map(|&block| index.blocks[block].value.pos())
452                .collect::<std::collections::HashSet<_>>();
453            for &position in left.iter().chain(&right) {
454                index.incidence.entry(position).or_default().push(id);
455                if let Some((root, _)) = record.choice {
456                    index.owners.entry(position).or_default().insert(root);
457                }
458            }
459            for &left in &left {
460                for &right in &right {
461                    index
462                        .neighbors
463                        .entry(left)
464                        .or_default()
465                        .insert(right.to_array());
466                    index
467                        .neighbors
468                        .entry(right)
469                        .or_default()
470                        .insert(left.to_array());
471                }
472            }
473        }
474        for records in index.incidence.values_mut() {
475            records.sort_unstable();
476            records.dedup();
477        }
478        Ok(index)
479    }

481    fn block_at(
482        &self,
483        position: IVec3,
484        choices: &LocalChoices,
485    ) -> Result<Option<&Block>, BlockGraphError> {
486        let mut selected = self
487            .blocks_at
488            .get(&position.to_array())
489            .into_iter()
490            .flatten()
491            .map(|&id| &self.blocks[id])
492            .filter(|record| record.selected(choices));
493        let block = selected.next().map(|record| &record.value);
494        if selected.next().is_some() {
495            return Err(BlockGraphError::BlockExists(position));
496        }
497        Ok(block)
498    }
499
500    fn endpoint_owner(
501        &self,
502        endpoint: IVec3,
503        choices: &LocalChoices,
504    ) -> Result<&Block, BlockGraphError> {
505        self.endpoint_owners
506            .get(&endpoint)
507            .into_iter()
508            .flatten()
509            .map(|&id| &self.blocks[id])
510            .find(|record| record.selected(choices))
511            .map(|record| &record.value)
512            .ok_or(BlockGraphError::BlockNotFound(endpoint))
513    }
514
515    fn pipes_at(
516        &self,
517        position: IVec3,
518        choices: &LocalChoices,
519    ) -> Result<Vec<&Pipe>, BlockGraphError> {
520        let mut selected = BTreeMap::<([i32; 3], [i32; 3]), usize>::new();
521        for &id in self.incidence.get(&position).into_iter().flatten() {
522            let record = &self.pipes[id];
523            if !record.selected(choices) {
524                continue;
525            }
526            let (left, right) = record.value.endpoints();
527            if self.endpoint_owner(left, choices)?.pos() != position
528                && self.endpoint_owner(right, choices)?.pos() != position
529            {
530                continue;
531            }
532            let (left, right) = crate::branch::pipe_key(&record.value);
533            let key = (left.to_array(), right.to_array());
534            if let Some(&previous) = selected.get(&key) {
535                if record.choice.is_some() && self.pipes[previous].choice.is_some() {
536                    return Err(BlockGraphError::PipeExists(left, right));
537                }
538            } else {
539                selected.insert(key, id);
540            }
541        }
542        Ok(selected
543            .into_values()
544            .map(|id| &self.pipes[id].value)
545            .collect())
546    }
547
548    fn roots(&self, position: IVec3) -> Vec<DecisionId> {
549        std::iter::once(position.to_array())
550            .chain(self.neighbors.get(&position).into_iter().flatten().copied())
551            .flat_map(|position| {
552                self.owners
553                    .get(&IVec3::from_array(position))
554                    .into_iter()
555                    .flatten()
556                    .copied()
557            })
558            .collect::<BTreeSet<_>>()
559            .into_iter()
560            .collect()
561    }
562
563    fn local(
564        &self,
565        center: IVec3,
566        choices: &LocalChoices,
567    ) -> Result<Option<LocalGeometry>, BlockGraphError> {
568        if self.block_at(center, choices)?.is_none() {
569            return Ok(None);
570        }
571        let mut primary = BTreeSet::from([center.to_array()]);
572        for pipe in self.pipes_at(center, choices)? {
573            for endpoint in [pipe.src(), pipe.dst()] {
574                primary.insert(self.endpoint_owner(endpoint, choices)?.pos().to_array());
575            }
576        }
577        let mut blocks = BTreeMap::new();
578        let mut pipes = BTreeMap::new();
579        let mut primary_incidence = Vec::with_capacity(primary.len());
580        for &position in &primary {
581            let position = IVec3::from_array(position);
582            let mut block = self
583                .block_at(position, choices)?
584                .expect("selected neighbor exists")
585                .clone();
586            let mut counts = [0usize; 3];
587            for pipe in self.pipes_at(position, choices)? {
588                counts[pipe.dir().as_udirection().index()] += 1;
589                for endpoint in [pipe.src(), pipe.dst()] {
590                    // Unmentioned controls only choose lookup-only halo blocks;
591                    // primary kinds/pipes depend entirely on the local case roots.
592                    let neighbor = self.endpoint_owner(endpoint, choices)?;
593                    blocks
594                        .entry(neighbor.pos().to_array())
595                        .or_insert_with(|| neighbor.clone());
596                }
597                let (left, right) = crate::branch::pipe_key(pipe);
598                pipes.insert((left.to_array(), right.to_array()), pipe.clone());
599            }
600            if let BlockKind::Cube(kind) = block.kind() {
601                block.kind = BlockKind::Cube(crate::graph::shadowed_cube_kind(kind, counts));
602            }
603            primary_incidence.push((position, counts.iter().sum()));
604            blocks.insert(position.to_array(), block);
605        }
606        Ok(Some(LocalGeometry {
607            blocks: blocks.into_values().collect(),
608            pipes: pipes.into_values().collect(),
609            primary_incidence,
610        }))
611    }
612}

642    pub(crate) fn decision_root_count(&self) -> usize {
643        1 + self.conditions.len()
644            + self.branches.len()
645            + self.resolves.len()
646            + self.sites.values().map(Vec::len).sum::<usize>()
647            + self
648                .variables
649                .iter()
650                .filter(|variable| matches!(variable, GuardedVariable::Branch { .. }))
651                .count()
652    }

654    /// Collect only while no external stage retains unlisted decision ids.
655    pub(crate) fn collect_garbage<'a>(
656        &'a mut self,
657        additional: impl IntoIterator<Item = &'a mut DecisionId>,
658    ) -> usize {
659        assert!(
660            !self.deferred_sites,
661            "deferred authored-geometry roots must be consumed before Boolean collection"
662        );
663        let roots = std::iter::once(&mut self.domain)
664            .chain(self.conditions.iter_mut().map(|(_, root)| root))
665            .chain(self.branches.iter_mut().map(|(_, _, root)| root))
666            .chain(self.resolves.iter_mut().map(|(_, root)| root))
667            .chain(
668                self.sites
669                    .values_mut()
670                    .flatten()
671                    .map(|site| &mut site.guard),
672            )
673            .chain(
674                self.variables
675                    .iter_mut()
676                    .filter_map(|variable| match variable {
677                        GuardedVariable::Branch { condition, .. } => Some(condition),
678                        GuardedVariable::Outcome(_) => None,
679                    }),
680            );
681        self.diagram.collect_garbage(roots.chain(additional))
682    }

684    /// Builds the factored guarded topology for a source graph.
685    ///
686    /// # Errors
687    ///
688    /// Returns an error if source validation, projection, or resource accounting fails.
689    ///

702    fn new_with_geometry(
703        source: &BlockGraph,
704        limits: ModuleCertificationLimits,
705    ) -> Result<(Self, AuthoredGeometry), BlockGraphError> {
706        source.require_flat_hierarchy("guarded topology")?;
707        source.validate_resource_limits(limits)?;
708        crate::validate::validate_source(source)?;
709        let actions = source.actions();
710        // validate_source already checked every arm and its action binding.
711        // Recover that resolve order without rebuilding candidate arm graphs.
712        let by_target = source
713            .branch_definitions()
714            .iter()
715            .map(|region| (region.target, region))
716            .collect::<HashMap<_, _>>();
717        let regions = actions
718            .iter()
719            .filter_map(|action| match action {
720                Action::Branch { target, .. } => Some((*by_target[target]).clone()),
721                _ => None,
722            })
723            .collect::<Vec<_>>();
724        let shown = regions
725            .iter()
726            .flat_map(|region| {
727                region
728                    .shown_arm()
729                    .blocks()
730                    .map(|block| block.pos().to_array())
731            })
732            .collect::<BTreeSet<_>>();
733        let common_projection_size = block_footprint_size(
734            source
735                .blocks()
736                .filter(|block| !shown.contains(&block.pos().to_array())),
737        );
738        let expressions = actions
739            .iter()
740            .filter_map(|action| match action {
741                Action::Branch { condition, .. }
742                | Action::Resolve { condition, .. }
743                | Action::Feedback {
744                    condition: Some(condition),
745                    ..
746                }
747                | Action::DiscardIf(condition) => Some(condition.clone()),
748                Action::Let { expr, .. } => Some(expr.clone()),
749                Action::Measure { name, .. } => Some(Expr::Var(name.clone())),
750                _ => None,
751            })
752            .map(Arc::new)
753            .collect::<Vec<_>>();
754        // Slots stay fixed when roots are remapped. Keep the first exact
755        // occurrence, matching the ordered condition search.
756        let mut condition_slots = HashMap::new();
757        for (slot, expression) in expressions.iter().enumerate() {
758            condition_slots
759                .entry(Arc::clone(expression))
760                .or_insert(slot);
761        }
762        let aliases = actions
763            .iter()
764            .filter_map(|action| match action {
765                Action::Let { name, expr } => Some((name.as_str(), expr)),
766                _ => None,
767            })
768            .collect();
769        let decisions = crate::action_dag::ResolveDecisionDiagram::build(
770            &expressions.iter().map(Arc::as_ref).collect::<Vec<_>>(),
771            &aliases,
772            limits.boolean_limits(),
773        )?
774        .expect("validated source expressions");
775        let mut diagram = BooleanDecisionDiagram::with_limits(limits.boolean_limits());
776        diagram.charge(decisions.diagram.steps())?;
777        let mut outcomes = vec![String::new(); decisions.variable_indices.len()];
778        for (name, index) in decisions.variable_indices {
779            outcomes[index] = name;
780        }
781        let named_source_conditions = expressions
782            .iter()
783            .zip(&decisions.roots)
784            .filter_map(|(expression, root)| match expression.as_ref() {
785                Expr::Var(name) => Some((name.as_str(), *root)),
786                _ => None,
787            })
788            .collect::<HashMap<_, _>>();
789        let source_condition = |expr: &Expr| {
790            if let Expr::Var(name) = expr {
791                return named_source_conditions[name.as_str()];
792            }
793            decisions.roots[*condition_slots
794                .get(expr)
795                .expect("source conditions were indexed before lowering")]
796        };
797        let branch_sources = actions
798            .iter()
799            .filter_map(|action| match action {
800                Action::Branch { target, condition } => Some((
801                    by_target[target].name.clone(),
802                    *target,
803                    source_condition(condition),
804                )),
805                _ => None,
806            })
807            .collect::<Vec<_>>();
808        // Grouping every branch before every outcome makes a sum of local
809        // branch/outcome products exponential in an ordered decision diagram.
810        // Preserve source outcome order, placing each branch next to its last
811        // condition dependency. Source-node remapping then stays monotone.
812        diagram.charge(
813            decisions
814                .diagram
815                .nodes()
816                .len()
817                .saturating_add(outcomes.len())
818                .saturating_add(branch_sources.len()),
819        )?;
820        let mut last_dependency = vec![None, None];
821        for node in decisions.diagram.nodes() {
822            let last = std::iter::once(node.variable)
823                .chain(last_dependency[node.low.0])
824                .chain(last_dependency[node.high.0])
825                .max();
826            last_dependency.push(last);
827        }
828        let mut after = vec![Vec::new(); outcomes.len() + 1];
829        for (branch, (_, _, condition)) in branch_sources.iter().enumerate() {
830            after[last_dependency[condition.0].map_or(0, |index| index + 1)].push(branch);
831        }
832        let mut outcome_indices = vec![0; outcomes.len()];
833        let mut branch_indices = vec![0; branch_sources.len()];
834        let mut variables = Vec::with_capacity(outcomes.len() + branch_sources.len());
835        for (slot, outcome) in outcomes.into_iter().map(Some).chain([None]).enumerate() {
836            for &branch in &after[slot] {
837                branch_indices[branch] = variables.len();
838                let (name, _, condition) = &branch_sources[branch];
839                variables.push(GuardedVariable::Branch {
840                    name: name.clone(),
841                    condition: *condition,
842                });
843            }
844            if let Some(name) = outcome {
845                outcome_indices[slot] = variables.len();
846                variables.push(GuardedVariable::Outcome(name));
847            }
848        }
849        let mut remap = vec![DECISION_FALSE, DECISION_TRUE];
850        for node in decisions.diagram.nodes() {
851            remap.push(diagram.make_node(
852                outcome_indices[node.variable],
853                remap[node.low.0],
854                remap[node.high.0],
855            )?);
856        }
857        for variable in &mut variables {
858            if let GuardedVariable::Branch { condition, .. } = variable {
859                *condition = remap[condition.0];
860            }
861        }
862        let conditions = expressions
863            .iter()
864            .cloned()
865            .zip(decisions.roots.iter().map(|root| remap[root.0]))
866            .collect::<Vec<_>>();
867        let named_conditions = conditions
868            .iter()
869            .enumerate()
870            .filter_map(|(index, (expression, _))| match expression.as_ref() {
871                Expr::Var(name) => Some((name.clone(), index)),
872                _ => None,
873            })
874            .collect();
875        let mut branch_functions = Vec::new();
876        let mut branches = Vec::new();
877        for ((name, target, condition), index) in branch_sources.into_iter().zip(branch_indices) {
878            let variable = diagram.make_node(index, DECISION_FALSE, DECISION_TRUE)?;
879            branch_functions.push((index, condition));
880            branches.push((name, target, variable));
881        }
882        let mut resolves = Vec::new();
883        for action in &actions {
884            if let Action::Resolve { target, condition } = action {
885                resolves.push((*target, remap[source_condition(condition).0]));
886            }
887        }
888        let domain = boolean_image(
889            &decisions.diagram,
890            &mut diagram,
891            branch_functions,
892            limits.max_witness_nodes,
893        )?;
894        drop(decisions.diagram);
895        let branch_root = branches
896            .iter()
897            .map(|(_, target, root)| (*target, *root))
898            .collect::<HashMap<_, _>>();
899        let shared_pipes = source.branch_shared_pipes();
900        let geometry = AuthoredGeometry::new(source, &regions, &shared_pipes, &branch_root)?;
901        let mut topology = Self {
902            source: source.clone(),
903            diagram,
904            variables,
905            domain,
906            branches,
907            resolves,
908            sites: BTreeMap::new(),
909            projections: HashMap::new(),
910            regions,
911            shared_pipes,
912            conditions,
913            condition_slots,
914            named_conditions,
915            limits,
916            common_projection_size,
917            retained_projection_size: [0; 2],
918            deferred_sites: true,
919        };
920        // The reference retains authored action order and output roles.
921        topology.project(DECISION_TRUE)?;
922        Ok((topology, geometry))
923    }

925    /// Materialize the guarded site views from authored local incidence.
926    ///
927    /// # Errors
928    /// Returns an error if local geometry or its configured budget fails.
929    fn populate_sites(&mut self, geometry: AuthoredGeometry) -> Result<(), BlockGraphError> {
930        // Global planning needs this one reference anyway. Unchanged local
931        // contexts borrow it instead of retaining another copy of every halo.
932        let reference = self.project(DECISION_TRUE)?;
933        for &position in geometry.blocks_at.keys() {
934            let pos = IVec3::from_array(position);
935            let variants = self.site_variants(&geometry, pos, &reference)?;
936            self.sites.insert(position, variants);
937        }
938        self.deferred_sites = false;
939        Ok(())
940    }

942    fn site_variants(
943        &mut self,
944        geometry: &AuthoredGeometry,
945        position: IVec3,
946        reference: &Arc<BlockGraph>,
947    ) -> Result<Vec<GuardedProjection>, BlockGraphError> {
948        let roots = geometry.roots(position);
949        let cases = self.cases(&roots, DECISION_TRUE)?;
950        let mut variants = Vec::new();
951        for (values, guard) in cases {
952            let choices = roots.iter().copied().zip(values).collect::<LocalChoices>();
953            let Some(local) = geometry.local(position, &choices)? else {
954                continue;
955            };
956            let graph = if local.matches_reference(reference) {
957                Arc::clone(reference)
958            } else {
959                self.reserve_geometry(block_footprint_size(local.blocks.iter()))?;
960                let mut graph = BlockGraph::new();
961                for block in local.blocks {
962                    graph.try_add_block(block)?;
963                }
964                for pipe in local.pipes {
965                    graph.try_add_pipe(pipe)?;
966                }
967                Arc::new(graph)
968            };
969            variants.push(GuardedProjection { guard, graph });
970        }
971        Ok(variants)
972    }

986    /// Distinct values of these local functions, retaining each exact guard.
987    ///
988    /// # Errors
989    ///
990    /// Returns an error if Boolean work or the guarded domain exceeds its limit.
991    pub fn cases(
992        &mut self,
993        roots: &[DecisionId],
994        enabled: DecisionId,
995    ) -> Result<Vec<(Vec<bool>, DecisionId)>, BlockGraphError> {
996        let limit = self.limits.max_guarded_domain_size;
997        self.diagram.charge(roots.len().saturating_add(1))?;
998        let enabled = self.diagram.apply(BooleanOp::And, enabled, self.domain)?;
999        if enabled == DECISION_FALSE {
1000            return Ok(Vec::new());
1001        }
1002        // Refine the exact guards directly. Enumerating a joint decision
1003        // image first revisits residual tuples, then rebuilds these same guards.
1004        let mut cases = vec![(Vec::with_capacity(roots.len()), enabled)];
1005        for &root in roots {
1006            let mut next = Vec::new();
1007            for (mut values, guard) in cases {
1008                self.diagram.charge(1)?;
1009                let high = self.diagram.apply(BooleanOp::And, guard, root)?;
1010                let low = self.diagram.apply(BooleanOp::Xor, guard, high)?;
1011                let count =
1012                    usize::from(low != DECISION_FALSE) + usize::from(high != DECISION_FALSE);
1013                if next.len().saturating_add(count) > limit {
1014                    return Err(StabilizerError::ResourceLimited {
1015                        phase: "local structural patterns",
1016                        observed: next.len().saturating_add(count),
1017                        limit,
1018                    }
1019                    .into());
1020                }
1021                if high == DECISION_FALSE {
1022                    values.push(false);
1023                    next.push((values, low));
1024                } else if low == DECISION_FALSE {
1025                    values.push(true);
1026                    next.push((values, high));
1027                } else {
1028                    self.diagram.charge(values.len().saturating_add(1))?;
1029                    let mut lower = values.clone();
1030                    lower.push(false);
1031                    next.push((lower, low));
1032                    values.push(true);
1033                    next.push((values, high));
1034                }
1035            }
1036            cases = next;
1037        }
1038        for (_, guard) in &mut cases {
1039            self.diagram.charge(1)?;
1040            *guard = self.diagram.constrain(*guard, self.domain)?;
1041        }
1042        if cases.len() > limit {
1043            return Err(StabilizerError::ResourceLimited {
1044                phase: "local structural patterns",
1045                observed: cases.len(),
1046                limit,
1047            }
1048            .into());
1049        }
1050        Ok(cases)
1051    }

1126    fn reserve_geometry(&mut self, size: [usize; 2]) -> Result<(), BlockGraphError> {
1127        let mut retained = self.retained_projection_size;
1128        for (index, (phase, limit)) in [
1129            (
1130                "retained projection blocks",
1131                self.limits.max_expanded_blocks,
1132            ),
1133            (
1134                "retained projection footprint cells",
1135                self.limits.max_occupied_cells,
1136            ),
1137        ]
1138        .into_iter()
1139        .enumerate()
1140        {
1141            let next = retained[index].checked_add(size[index]);
1142            if next.is_none_or(|next| next > limit) {
1143                return Err(StabilizerError::ResourceLimited {
1144                    phase,
1145                    observed: next.unwrap_or(usize::MAX),
1146                    limit,
1147                }
1148                .into());
1149            }
1150            retained[index] = next.expect("checked retained projection size");
1151        }
1152        self.retained_projection_size = retained;
1153        Ok(())
1154    }

1156    /// Materializes and caches the branch projection selected by `guard`.
1157    ///
1158    /// # Errors
1159    ///
1160    /// Returns an error if witness search, resource accounting, or projection fails.
1161    ///
1162    /// # Panics
1163    ///
1164    /// Panics if `guard` does not select any reachable local case.
1165    pub fn project(&mut self, guard: DecisionId) -> Result<Arc<BlockGraph>, BlockGraphError> {
1166        let assignment = self.assignment(guard)?.expect("a local case is reachable");
1167        let key = assignment
1168            .iter()
1169            .map(|(_, value)| *value)
1170            .collect::<Vec<_>>();
1171        if let Some(graph) = self.projections.get(&key) {
1172            return Ok(Arc::clone(graph));
1173        }
1174        let mut size = self.common_projection_size;
1175        for (region, &value) in self.regions.iter().zip(&key) {
1176            for (count, amount) in size
1177                .iter_mut()
1178                .zip(block_footprint_size(region.arm(value).blocks()))
1179            {
1180                *count = count
1181                    .checked_add(amount)
1182                    .expect("a projection is bounded by validated source geometry");
1183            }
1184        }
1185        self.reserve_geometry(size)?;
1186        let graph = Arc::new(self.source.project_validated_branches_deferred(
1187            &self.regions,
1188            &self.shared_pipes,
1189            &key,
1190        )?);
1191        self.projections.insert(key, Arc::clone(&graph));
1192        Ok(graph)
1193    }
1194}

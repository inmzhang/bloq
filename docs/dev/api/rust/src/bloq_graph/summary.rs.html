94    /// Finds a definition summary already built as part of this root summary.

96    pub fn definition(&self, name: &str) -> Option<&ModuleSummary> {
97        let mut pending = vec![self];
98        let mut seen = FxHashSet::default();
99        while let Some(summary) = pending.pop() {
100            if summary.name == name {
101                return Some(summary);
102            }
103            if seen.insert(summary.name.as_str()) {
104                pending.extend(summary.dependencies.values().map(Arc::as_ref));
105            }
106        }
107        None
108    }

132    /// Materialize one requested structural projection. Projection topology is
133    /// still selected by the caller, but its stabilizer row space comes from
134    /// bottom-up module composition rather than the flattened graph.
135    #[doc(hidden)]
136    pub fn materialize_projection_stabilizers(
137        &self,
138        zx: &ZXGraph,
139        offset: IVec3,
140        assignments: &[(IVec3, bool)],
141    ) -> Result<StabilizerGenerators, ModuleCertificationError> {
142        let mut key = assignments
143            .iter()
144            .filter(|(_, value)| !value)
145            .map(|&(position, value)| {
146                checked_add_position(position, -offset)
147                    .map(|position| (position, value))
148                    .map_err(|source| ModuleCertificationError::Graph {
149                        module: self.name.clone(),
150                        source,
151                    })
152            })
153            .collect::<Result<Vec<_>, _>>()?;
154        key.sort_unstable_by_key(|(position, value)| (position.to_array(), *value));
155        if key.is_empty() {
156            return self.materialize_stabilizers_from(zx, ModuleOrientation::IDENTITY, offset);
157        }
158        if self.projection.is_some() {
159            let absent = key.iter().map(|&(position, _)| position).collect();
160            return self
161                .select_projection(&absent)?
162                .materialize_stabilizers_from(zx, ModuleOrientation::IDENTITY, offset);
163        }
164        let variant = self
165            .branch_variants
166            .iter()
167            .find(|variant| variant.assignments == key)
168            .ok_or_else(|| {
169                invalid(
170                    &self.name,
171                    "reachable structural projection has no composed module certificate",
172                )
173            })?;
174        variant
175            .summary
176            .materialize_stabilizers_from(zx, ModuleOrientation::IDENTITY, offset)
177    }

179    fn select_projection(
180        &self,
181        absent: &FxHashSet<IVec3>,
182    ) -> Result<ModuleSummary, ModuleCertificationError> {
183        let Some(projection) = &self.projection else {
184            return Ok(self.clone());
185        };
186        if absent
187            .iter()
188            .any(|target| !projection.targets.contains(target))
189        {
190            return Err(invalid(&self.name, "projection selects an unknown branch"));
191        }
192        let module = &projection.definition;
193        let graph_error = |source| ModuleCertificationError::Graph {
194            module: module.name.clone(),
195            source,
196        };
197        let mut selected = module.clone_local_definition();
198        selected.replace_local_body(
199            module
200                .local_body()
201                .project_branches_in_definition(
202                    module
203                        .local_body()
204                        .branch_regions()
205                        .map_err(graph_error)?
206                        .iter()
207                        .map(|region| (region.target, !absent.contains(&region.target))),
208                )
209                .map_err(graph_error)?,
210        );
211        if module.instances.is_empty() {
212            return summarize_leaf_body(module, selected.local_body(), projection.limits);
213        }
214        let children = module
215            .instances
216            .iter()
217            .map(|instance| {
218                let base = &self.dependencies[&instance.definition];
219                let mut key = FxHashSet::default();
220                if let Some(child) = &base.projection {
221                    for &target in &child.targets {
222                        if absent.contains(
223                            &instance
224                                .try_transform_position(target)
225                                .map_err(graph_error)?,
226                        ) {
227                            key.insert(target);
228                        }
229                    }
230                }
231                let child = if key.is_empty() {
232                    Arc::clone(base)
233                } else {
234                    Arc::new(base.select_projection(&key)?)
235                };
236                Ok((instance.name.as_str(), (instance, child)))
237            })
238            .collect::<Result<HashMap<_, _>, ModuleCertificationError>>()?;
239        summarize_composite(&selected, module, &children, projection.limits)
240    }

242    fn materialize_stabilizers_from(
243        &self,
244        zx: &ZXGraph,
245        orientation: ModuleOrientation,
246        offset: IVec3,
247    ) -> Result<StabilizerGenerators, ModuleCertificationError> {
248        // PauliString holds two BitVecs, each rounded to a 512-bit block.
249        let words = [
250            self.certificate_rows.len(),
251            self.measurements.len(),
252            self.discharged_measurements.len(),
253            self.logical_rows.len(),
254            self.adjustments.len(),
255        ]
256        .into_iter()
257        .try_fold(0usize, usize::checked_add)
258        .and_then(|rows| rows.checked_mul(zx.total_ids().div_ceil(512)))
259        .and_then(|blocks| blocks.checked_mul(16));
260        if words.is_none_or(|words| words > self.limits.max_matrix_words) {
261            return Err(resource_limit(
262                &self.name,
263                "dense matrix words",
264                words.unwrap_or(usize::MAX),
265                self.limits.max_matrix_words,
266            ));
267        }
268        let external_basis = self
269            .certificate_rows
270            .iter()
271            .map(|row| {
272                let stabilizer = materialize_summary_row(
273                    &row.signed,
274                    &row.support,
275                    &self.ports,
276                    zx,
277                    orientation,
278                    offset,
279                    &self.name,
280                )?;
281                let mut paulis = stabilizer.paulis;
282                zx.clear_cross_centers(std::slice::from_mut(&mut paulis));
283                let witness = row
284                    .witness
285                    .transformed(orientation, offset)
286                    .map_err(|source| ModuleCertificationError::Graph {
287                        module: self.name.clone(),
288                        source,
289                    })?;
290                let witnessed =
291                    zx.materialize_flow_witness(&witness, &paulis)
292                        .ok_or_else(|| {
293                            invalid(
294                                &self.name,
295                                "composed certificate witness does not fit the linked graph",
296                            )
297                        })?;
298                if witnessed.paulis != paulis {
299                    return Err(invalid(
300                        &self.name,
301                        "composed certificate witness does not reproduce its positioned support",
302                    ));
303                }
304                Ok(PhasedPauliString::new(paulis, witnessed.phase()))
305            })
306            .collect::<Result<Vec<_>, ModuleCertificationError>>()?;
307        let cached_prefix = self
308            .measurements
309            .iter()
310            .chain(&self.discharged_measurements)
311            .map(|row| {
312                materialize_summary_row(
313                    &row.signed,
314                    &row.support,
315                    &self.ports,
316                    zx,
317                    orientation,
318                    offset,
319                    &self.name,
320                )
321                .and_then(|stabilizer| {
322                    Ok((
323                        transformed_row_kind(&row.kind, orientation, offset, &self.name)?,
324                        stabilizer.paulis,
325                    ))
326                })
327            })
328            .chain(self.logical_rows.iter().map(|row| {
329                materialize_summary_row(
330                    &row.signed,
331                    &row.support,
332                    &self.ports,
333                    zx,
334                    orientation,
335                    offset,
336                    &self.name,
337                )
338                .map(|stabilizer| (StabilizerRowKind::Logical, stabilizer.paulis))
339            }))
340            .collect::<Result<Vec<_>, ModuleCertificationError>>()?;
341        let adjustment_basis = self
342            .adjustments
343            .iter()
344            .map(|row| {
345                let stabilizer = materialize_summary_row(
346                    &row.signed,
347                    &row.support,
348                    &self.ports,
349                    zx,
350                    orientation,
351                    offset,
352                    &self.name,
353                )?;
354                let mut paulis = stabilizer.paulis;
355                zx.clear_cross_centers(std::slice::from_mut(&mut paulis));
356                Ok(paulis)
357            })
358            .collect::<Result<Vec<_>, ModuleCertificationError>>()?;
359        zx.stabilizers_from_composed_basis(
360            &external_basis,
361            &adjustment_basis,
362            &cached_prefix,
363            self.limits,
364        )
365        .map_err(crate::RuntimeBasisError::from)
366        .and_then(|stabilizers| {
367            // C0 / LIM-027: leaf certification alone does not establish
368            // closure after composition. Check the presented readouts,
369            // including selective fills, before using them for decoding.
370            stabilizers.validate_measurements_close_before_outputs_with_limits(self.limits)?;
371            Ok(stabilizers)
372        })
373        .map_err(|source| match source {
374            crate::RuntimeBasisError::Stabilizer(crate::StabilizerError::ResourceLimited {
375                phase,
376                observed,
377                limit,
378            }) => resource_limit(&self.name, phase, observed, limit),
379            source => ModuleCertificationError::Runtime {
380                module: self.name.clone(),
381                source,
382            },
383        })
384    }

393    orientation: ModuleOrientation,
394    translation: IVec3,
395    module: &str,
396) -> Result<StabilizerRowKind, ModuleCertificationError> {
397    let mut kind = kind.clone();
398    if let StabilizerRowKind::SelectiveFixing { targets } = &mut kind {
399        for target in targets {
400            target.pos = orientation
401                .try_transform_position(target.pos, translation)
402                .map_err(|source| ModuleCertificationError::Graph {
403                    module: module.to_string(),
404                    source,
405                })?;
406        }
407    }
408    Ok(kind)

411fn materialize_summary_row(
412    signed: &PhasedPauliString,
413    support: &PositionedSupport,
414    ports: &[QuantumPort],
415    zx: &ZXGraph,
416    orientation: ModuleOrientation,
417    translation: IVec3,
418    module: &str,
419) -> Result<Stabilizer, ModuleCertificationError> {
420    let mut support = support
421        .transformed(orientation, translation)
422        .map_err(|source| ModuleCertificationError::Graph {
423            module: module.to_string(),
424            source,
425        })?;
426    for (column, port) in ports.iter().enumerate() {
427        let pauli = signed.paulis.get(column);
428        if pauli == Pauli::I {
429            continue;
430        }
431        let position = orientation
432            .try_transform_position(port.position, translation)
433            .map_err(|source| ModuleCertificationError::Graph {
434                module: module.to_string(),
435                source,
436            })?;
437        if let Some(existing) = support.nodes.insert(position, pauli)
438            && existing != pauli
439        {
440            return Err(invalid(
441                module,
442                &format!(
443                    "certificate support repeats public Port {position}: interior {existing}, boundary {pauli}"
444                ),
445            ));
446        }
447    }
448    support.materialize(zx, signed.phase()).ok_or_else(|| {
449        invalid(
450            module,
451            "composed certificate support does not fit the linked graph",
452        )
453    })
454}

481impl ChildRelation {
482    fn width(&self) -> usize {
483        self.parent_ports.len() + self.endpoints.len()
484    }
485
486    fn append(&mut self, other: Self, mapping: &[usize], width: usize) {
487        for (target, rows) in [
488            (&mut self.rows, other.rows),
489            (&mut self.logical_rows, other.logical_rows),
490            (&mut self.certificate_rows, other.certificate_rows),
491            (&mut self.adjustments, other.adjustments),
492        ] {
493            target.extend(rows.into_iter().map(|mut row| {
494                row.signed = embed_signed_row(&row.signed, mapping, width);
495                row
496            }));
497        }
498        for (target, rows) in [
499            (&mut self.measurements, other.measurements),
500            (
501                &mut self.discharged_measurements,
502                other.discharged_measurements,
503            ),
504        ] {
505            target.extend(rows.into_iter().map(|mut row| {
506                row.signed = embed_signed_row(&row.signed, mapping, width);
507                row
508            }));
509        }
510        self.parent_ports.extend(other.parent_ports);
511        self.endpoints.extend(other.endpoints);
512        self.protected.extend(other.protected);
513        self.frontier.extend(other.frontier);
514        self.protected_groups.extend(other.protected_groups);
515    }
516
517    fn retire_completed(
518        &mut self,
519        completed: &mut Self,
520        obligations: &FxHashSet<SupportTarget>,
521        module: &str,
522    ) -> Result<(), ModuleCertificationError> {
523        // Boundary-zero certificate rows complete the final physical rank, but
524        // cannot change a later seam equation. Keep their witnesses off the live table.
525        completed.certificate_rows.extend(
526            self.certificate_rows
527                .extract_if(.., |row| row.signed.paulis.is_identity())
528                .map(|mut row| {
529                    row.signed.paulis = PauliString::new(0);
530                    row
531                }),
532        );
533        // A closed adjustment that is zero on every pending obligation cannot
534        // change a future seam solve. Retain it only for final presentation.
535        completed.adjustments.extend(
536            self.adjustments
537                .extract_if(.., |row| {
538                    row.signed.paulis.is_identity()
539                        && !row
540                            .support
541                            .nodes
542                            .keys()
543                            .any(|&position| obligations.contains(&SupportTarget::Node(position)))
544                        && !row.support.edges.keys().any(|&(left, right)| {
545                            obligations.contains(&SupportTarget::Edge(left, right))
546                                || obligations.contains(&SupportTarget::Edge(right, left))
547                        })
548                })
549                .map(|mut row| {
550                    row.signed.paulis = PauliString::new(0);
551                    row
552                }),
553        );
554        resize_discharged_boundaries(&mut self.discharged_measurements, 0, module)?;
555        completed
556            .discharged_measurements
557            .append(&mut self.discharged_measurements);
558        Ok(())
559    }
560}

604fn semantic_protected_supports(module: &BlockGraph, body: &BlockGraph) -> Vec<ProtectedSupport> {
605    let mut protected = HashMap::<SupportTarget, Option<Pauli>>::new();
606    let mut insert = |target, forbidden| {
607        protected
608            .entry(target)
609            .and_modify(|current| {
610                if current.is_none() {
611                    *current = forbidden;
612                }
613            })
614            .or_insert(forbidden);
615    };
616    for block in body.blocks() {
617        if let BlockKind::Selective(kind) = block.kind() {
618            let forbidden = match kind {
619                crate::SelectiveKind::XY => Pauli::Z,
620                crate::SelectiveKind::XZ => Pauli::Y,
621                crate::SelectiveKind::YZ => Pauli::X,
622            };
623            insert(SupportTarget::Node(block.pos()), Some(forbidden));
624        }
625    }
626    for region in module.local_body().branch_definitions() {
627        insert(SupportTarget::Node(region.target), None);
628        for cut in region
629            .incoming_for(false)
630            .iter()
631            .chain(region.incoming_for(true))
632        {
633            insert(SupportTarget::Edge(cut.past, cut.inside), None);
634        }
635    }
636    let mut protected = protected
637        .into_iter()
638        .map(|(target, forbidden)| ProtectedSupport { target, forbidden })
639        .collect::<Vec<_>>();
640    protected.sort_unstable_by_key(|constraint| support_target_key(&constraint.target));
641    protected
642}

644/// Positioned support that must survive certificate reduction, but does not
645/// constrain which seam adjustments are legal.
646fn semantic_frontier_supports(
647    module: &BlockGraph,
648    body: &BlockGraph,
649) -> Result<Vec<SupportTarget>, ModuleCertificationError> {
650    let mut frontier = body
651        .blocks()
652        .filter(|block| block.kind().is_t())
653        .map(|block| SupportTarget::Node(block.pos()))
654        .collect::<FxHashSet<_>>();
655    for action in body.actions() {
656        match action {
657            Action::Measure { target, .. } => {
658                let target = match target {
659                    crate::MeasureTarget::Node(position) => SupportTarget::Node(position),
660                    crate::MeasureTarget::Edge { src, dir } => SupportTarget::Edge(
661                        src,
662                        checked_add_position(src, dir.to_ivec3()).map_err(|source| {
663                            ModuleCertificationError::Graph {
664                                module: module.name.clone(),
665                                source,
666                            }
667                        })?,
668                    ),
669                };
670                frontier.insert(target);
671            }
672            Action::Resolve { target, .. } | Action::Branch { target, .. } => {
673                frontier.insert(SupportTarget::Node(target));
674            }
675            Action::Feedback { targets, .. } => {
676                for target in targets {
677                    frontier.insert(match target.direction {
678                        Some(dir) => SupportTarget::Edge(
679                            target.target,
680                            checked_add_position(target.target, dir.to_ivec3()).map_err(
681                                |source| ModuleCertificationError::Graph {
682                                    module: module.name.clone(),
683                                    source,
684                                },
685                            )?,
686                        ),
687                        None => SupportTarget::Node(target.target),
688                    });
689                }
690            }
691            Action::Let { .. } | Action::DiscardIf(_) => {}
692        }
693    }
694    for region in module.local_body().branch_definitions() {
695        frontier.insert(SupportTarget::Node(region.target));
696        frontier.extend(
697            region
698                .incoming_for(false)
699                .iter()
700                .chain(region.incoming_for(true))
701                .map(|cut| SupportTarget::Edge(cut.past, cut.inside)),
702        );
703    }
704    let mut frontier = frontier.into_iter().collect::<Vec<_>>();
705    frontier.sort_unstable_by_key(support_target_key);
706    Ok(frontier)
707}

713    limits: ModuleCertificationLimits,
714    module: &str,
715) -> Result<Vec<ProtectedGroup>, ModuleCertificationError> {
716    measurements
717        .iter()
718        .filter_map(|measurement| {
719            let targets = measurement.kind.selective_fixing_targets();
720            (targets.len() > 1).then_some(targets)
721        })
722        .map(|targets| {
723            let positions = targets.iter().map(|target| target.pos).collect::<Vec<_>>();
724            let kinds = positions
725                .iter()
726                .map(|&position| {
727                    let Some(BlockKind::Selective(kind)) =
728                        body.get_block(position).map(Block::kind)
729                    else {
730                        return Err(invalid(module, "selective fixing target is not selective"));
731                    };
732                    Ok(kind)
733                })
734                .collect::<Result<Vec<_>, ModuleCertificationError>>()?;
735            let domain = zx
736                .action_graph()
737                .resolve_value_domain_bounded_with_limits(
738                    &positions,
739                    limits.max_guarded_domain_size,
740                    limits.boolean_limits(),
741                )
742                .map_err(|error| {
743                    match error
744                        .into_stabilizer("selective value domain", limits.max_guarded_domain_size)
745                    {
746                        crate::StabilizerError::ResourceLimited {
747                            phase,
748                            observed,
749                            limit,
750                        } => resource_limit(module, phase, observed, limit),
751                        source => ModuleCertificationError::Graph {
752                            module: module.to_string(),
753                            source: source.into(),
754                        },
755                    }
756                })?;
757            let mut permitted_support = domain
758                .values()
759                .iter()
760                .map(|values| {
761                    kinds
762                        .iter()
763                        .zip(values)
764                        .map(|(kind, &value)| {
765                            Pauli::from(if value {
766                                kind.pauli_if_true()
767                            } else {
768                                kind.pauli_if_false()
769                            })
770                        })
771                        .collect::<Vec<_>>()
772                })
773                .collect::<Vec<_>>();
774            let reachable = permitted_support.iter().cloned().collect::<FxHashSet<_>>();
775            permitted_support.retain(|support| {
776                reachable.contains(
777                    &support
778                        .iter()
779                        .zip(targets)
780                        .map(|(&pauli, target)| pauli ^ target.forbidden)
781                        .collect::<Vec<_>>(),
782                )
783            });
784            Ok(ProtectedGroup {
785                targets: positions.into_iter().map(SupportTarget::Node).collect(),
786                permitted_support,
787            })
788        })
789        .collect()
790}

834impl PositionedSupport {
835    fn from_stabilizer(stabilizer: &Stabilizer, zx: &ZXGraph) -> Self {
836        // Cross centers are ORs of incident support, not independent GF(2) bits.
837        // Keep raw support through composition and reconstruct at materialization.
838        let mut raw = stabilizer.paulis.clone();
839        zx.clear_cross_centers(std::slice::from_mut(&mut raw));
840        Self {
841            nodes: stabilizer
842                .interior_nodes
843                .iter()
844                .filter(|(position, _)| !stabilizer.port_stabilizer.contains_key(position))
845                .filter_map(|(&p, _)| {
846                    let pauli = raw.get(zx.node_at(p).expect("surface node exists").id);
847                    (pauli != Pauli::I).then_some((p, pauli))
848                })
849                .collect(),
850            edges: stabilizer
851                .interior_edges
852                .iter()
853                .map(|(&edge, &pauli)| (edge, pauli))
854                .collect(),
855        }
856    }
857
858    fn multiply_assign(&mut self, other: &Self) {
859        debug_assert!(
860            self.edges.keys().all(|&(left, right)| {
861                !other.edges.contains_key(&(right, left)) || left == right
862            }),
863            "positioned support edge orientation mismatch"
864        );
865        xor_pauli_maps(&mut self.nodes, &other.nodes);
866        xor_pauli_maps(&mut self.edges, &other.edges);
867    }
868
869    fn get(&self, target: SupportTarget) -> Pauli {
870        match target {
871            SupportTarget::Node(position) => self.nodes.get(&position).copied().unwrap_or(Pauli::I),
872            SupportTarget::Edge(left, right) => self
873                .edges
874                .get(&(left, right))
875                .or_else(|| self.edges.get(&(right, left)))
876                .copied()
877                .unwrap_or(Pauli::I),
878        }
879    }
880
881    fn weight(&self) -> usize {
882        self.nodes.len() + self.edges.len()
883    }
884
885    fn strip_ports(&mut self, ports: &FxHashSet<IVec3>, retained: &[SupportTarget]) {
886        self.nodes.retain(|position, _| !ports.contains(position));
887        self.edges.retain(|&(left, right), _| {
888            (!ports.contains(&left) && !ports.contains(&right))
889                || retained.contains(&SupportTarget::Edge(left, right))
890                || retained.contains(&SupportTarget::Edge(right, left))
891        });
892    }
893
894    fn transformed(
895        &self,
896        orientation: ModuleOrientation,
897        translation: IVec3,
898    ) -> Result<Self, crate::BlockGraphError> {
899        let transform = |position| orientation.try_transform_position(position, translation);
900        Ok(Self {
901            nodes: self
902                .nodes
903                .iter()
904                .map(|(&position, &pauli)| Ok((transform(position)?, pauli)))
905                .collect::<Result<_, crate::BlockGraphError>>()?,
906            edges: self
907                .edges
908                .iter()
909                .map(|(&(left, right), &pauli)| Ok(((transform(left)?, transform(right)?), pauli)))
910                .collect::<Result<_, crate::BlockGraphError>>()?,
911        })
912    }
913
914    fn add_seam(&mut self, seam: SupportSeam, mut pauli: Pauli) -> u8 {
915        if pauli == Pauli::I {
916            return 0;
917        }
918        let phase = 2 * u8::from(seam.boundary_to_edge_hadamard && pauli == Pauli::Y);
919        if seam.boundary_to_edge_hadamard {
920            pauli = pauli.flip();
921        }
922        self.edges.remove(&(seam.edge.1, seam.edge.0));
923        self.edges.insert(seam.edge, pauli);
924        phase
925    }
926
927    fn materialize(&self, zx: &ZXGraph, mut phase: u8) -> Option<Stabilizer> {
928        let mut row = PauliString::new(zx.total_ids());
929        for (&position, &pauli) in &self.nodes {
930            row.set(zx.node_at(position)?.id, pauli);
931        }
932        for (&(source, target), &pauli_at_source) in &self.edges {
933            let source_id = zx.node_at(source)?.id;
934            let target_id = zx.node_at(target)?.id;
935            let forward = zx.edge_id(source_id, target_id)?;
936            let reverse = zx.edge_id(target_id, source_id)?;
937            let edge = &zx.edges()[forward - zx.nodes().len()];
938            let mut stored = pauli_at_source;
939            if source_id > target_id && edge.hadamard {
940                if stored == Pauli::Y {
941                    phase = (phase + 2) % 4;
942                }
943                stored = stored.flip();
944            }
945            row.set(forward, stored);
946            row.set(reverse, stored);
947        }
948        // A spatial rotation can recolor a degree-two identity spider, changing
949        // which component was stored as raw node support. Recover its native
950        // component from the broadcast on all incident arms before rebuilding
951        // crossing centers. Isolated nodes retain their explicit raw support.
952        for node in zx
953            .nodes()
954            .iter()
955            .filter(|node| matches!(node.kind, crate::NodeKind::X | crate::NodeKind::Z))
956        {
957            let Some(neighbors) = zx
958                .neighbors(node.id)
959                .filter(|neighbors| !neighbors.is_empty())
960            else {
961                continue;
962            };
963            let native = node.kind.cross_pauli().flip();
964            let broadcast = neighbors.iter().all(|&neighbor| {
965                let edge = zx.edge_between(node.id, neighbor).expect("incident edge");
966                let axis = if edge.hadamard && node.id > neighbor {
967                    native.flip()
968                } else {
969                    native
970                };
971                row.get(edge.id) & axis
972            });
973            let current = row.get(node.id);
974            if (current & native) != broadcast {
975                row.set(node.id, current ^ native);
976            }
977        }
978        zx.reconstruct_cross_center(std::slice::from_mut(&mut row));
979        Some(zx.pauli_string_to_stabilizer_with_row_phase(row, phase))
980    }
981}

1176fn add_expansion_size(
1177    size: &mut [usize; 3],
1178    amount: [usize; 3],
1179    limits: ModuleCertificationLimits,
1180) -> Result<(), crate::StabilizerError> {
1181    for ((current, amount), (phase, limit)) in size.iter_mut().zip(amount).zip([
1182        ("expanded blocks", limits.max_expanded_blocks),
1183        ("occupied footprint cells", limits.max_occupied_cells),
1184        ("expanded module instances", limits.max_expanded_instances),
1185    ]) {
1186        let next = current.checked_add(amount);
1187        if next.is_none_or(|next| next > limit) {
1188            return Err(crate::StabilizerError::ResourceLimited {
1189                phase,
1190                observed: next.unwrap_or(usize::MAX),
1191                limit,
1192            });
1193        }
1194        *current = next.expect("checked size fits");
1195    }
1196    Ok(())
1197}

1199fn source_expansion_size(
1200    source: &BlockGraph,
1201    limits: ModuleCertificationLimits,
1202) -> Result<[usize; 3], crate::StabilizerError> {
1203    let mut size = [0usize; 3];
1204    // The shown arm already lives in the body. Count its authored blocks only
1205    // when they differ from that body, so even an unvalidated source is bounded.
1206    let authored = source.branch_definitions().iter().flat_map(|region| {
1207        region.arm(!region.shown_true()).blocks().chain(
1208            region
1209                .shown_arm()
1210                .blocks()
1211                .filter(|block| source.get_block(block.pos()) != Some(*block)),
1212        )
1213    });
1214    for block in source.blocks().chain(authored) {
1215        let cells = match block.kind() {
1216            BlockKind::Walking(kind) => Some(kind.movement_3d()),
1217            BlockKind::PatchRotation(kind) => Some(kind.movement_3d()),
1218            _ => None,
1219        }
1220        .map_or_else(
1221            || block.height_cells() as usize,
1222            |movement| {
1223                // Moving blocks reserve their axis-aligned start/end bounding box.
1224                movement
1225                    .to_array()
1226                    .into_iter()
1227                    .map(|delta| delta.unsigned_abs() as usize + 1)
1228                    .product()
1229            },
1230        );
1231        add_expansion_size(&mut size, [1, cells, 0], limits)?;
1232    }
1233    Ok(size)
1234}

1236/// Count each definition once, then add its cached size for each instance edge.
1237/// Both branch arms and Ports consumed by seams count toward the allocation bound.
1238pub(crate) fn preflight_program_expansion(
1239    program: &BlockGraph,
1240    limits: ModuleCertificationLimits,
1241) -> Result<(), ModuleError> {
1242    let mut sizes = HashMap::<&str, [usize; 3]>::new();
1243    for module in module_postorder(program) {
1244        let error = |source| {
1245            let crate::StabilizerError::ResourceLimited {
1246                phase,
1247                observed,
1248                limit,
1249            } = source
1250            else {
1251                unreachable!("size counting only reports resource limits")
1252            };
1253            program_geometry_error(
1254                &module.name,
1255                resource_limit(&module.name, phase, observed, limit),
1256            )
1257        };
1258        let mut size = source_expansion_size(module.local_body(), limits).map_err(error)?;
1259        add_expansion_size(&mut size, [0, 0, 1], limits).map_err(error)?;
1260        for instance in &module.instances {
1261            add_expansion_size(&mut size, sizes[instance.definition.as_str()], limits)
1262                .map_err(error)?;
1263        }
1264        sizes.insert(&module.name, size);
1265    }
1266    Ok(())

1291fn module_postorder_from<'a>(
1292    program: &'a BlockGraph,
1293    roots: impl IntoIterator<Item = &'a str>,
1294) -> Vec<&'a BlockGraph> {
1295    let definitions = program
1296        .modules()
1297        .map(|module| (module.name.as_str(), module))
1298        .collect::<HashMap<_, _>>();
1299    let mut seen = FxHashSet::default();
1300    let mut ordered = Vec::new();
1301    for root in roots {
1302        let mut pending = vec![(root, false)];
1303        while let Some((name, exiting)) = pending.pop() {
1304            let module = definitions[name];
1305            if exiting {
1306                ordered.push(module);
1307            } else if seen.insert(name) {
1308                pending.push((name, true));
1309                pending.extend(
1310                    module
1311                        .instances
1312                        .iter()
1313                        .rev()
1314                        .map(|instance| (instance.definition.as_str(), false)),
1315                );
1316            }
1317        }
1318    }
1319    ordered
1320}

1322fn module_geometry(
1323    program: &BlockGraph,
1324    name: &str,
1325    memo: &mut HashMap<String, ModuleGeometry>,
1326) -> Result<ModuleGeometry, ModuleError> {
1327    if let Some(geometry) = memo.get(name) {
1328        return Ok(geometry.clone());
1329    }
1330    for module in module_postorder_from(program, [name]) {
1331        if memo.contains_key(&module.name) {
1332            continue;
1333        }
1334        let name = module.name.as_str();
1335        let geometry = if module.instances.is_empty() {
1336            let body = module.copy_local_geometry().fix_shadowed_faces();
1337            let signatures = module
1338                .interface
1339                .quantum_ports
1340                .iter()
1341                .map(|port| port_signature(name, &body, port.position))
1342                .collect::<Result<Vec<_>, _>>()
1343                .map_err(|error| program_geometry_error(name, error))?;
1344            ModuleGeometry {
1345                ports: module.interface.quantum_ports.clone(),
1346                signatures,
1347            }
1348        } else {
1349            let mut children = HashMap::new();
1350            for instance in &module.instances {
1351                let geometry = memo[&instance.definition].clone();
1352                children.insert(instance.name.as_str(), (instance, geometry));
1353            }
1354
1355            let mut binds = Vec::new();
1356            for connection in &module.quantum_connections {
1357                match connection {
1358                    QuantumConnection::Input {
1359                        block,
1360                        input,
1361                        hadamard,
1362                    } => {
1363                        binds.push(geometry_bind_seam(
1364                            input, *block, *hadamard, true, &children,
1365                        ));
1366                    }
1367                    QuantumConnection::Output {
1368                        output,
1369                        block,
1370                        hadamard,
1371                    } => {
1372                        binds.push(geometry_bind_seam(
1373                            output, *block, *hadamard, false, &children,
1374                        ));
1375                    }
1376                    QuantumConnection::Pipe {
1377                        output,
1378                        input,
1379                        hadamard,
1380                    } => {
1381                        geometry_direct_seam(output, input, *hadamard, &children, name)
1382                            .map_err(|error| program_geometry_error(name, error))?;
1383                    }
1384                }
1385            }
1386            validate_bind_faces(module, &binds)
1387                .map_err(|error| program_geometry_error(name, error))?;
1388            let extra_pipes = binds
1389                .iter()
1390                .map(|bind| (bind.block, bind.signature.direction.as_udirection()))
1391                .collect::<Vec<_>>();
1392            let body = module
1393                .copy_local_geometry()
1394                .fix_shadowed_faces_with(&extra_pipes);
1395            validate_bind_bases(module, &body, &binds)
1396                .map_err(|error| program_geometry_error(name, error))?;
1397            ModuleGeometry {
1398                ports: module.interface.quantum_ports.clone(),
1399                signatures: parent_signatures(module, &body, &binds)
1400                    .map_err(|error| program_geometry_error(name, error))?,
1401            }
1402        };
1403        memo.insert(name.to_string(), geometry);
1404    }
1405    Ok(memo[name].clone())
1406}

1416/// Default worker count for summarizing or compiling a module hierarchy:
1417/// the machine's parallelism capped at 4, and always 1 on wasm.
1418pub fn default_module_jobs() -> NonZeroUsize {
1419    #[cfg(not(target_arch = "wasm32"))]
1420    {
1421        const MAX_JOBS: NonZeroUsize = NonZeroUsize::new(4).expect("4 is nonzero");
1422        std::thread::available_parallelism().map_or(NonZeroUsize::MIN, |jobs| jobs.min(MAX_JOBS))
1423    }
1424    #[cfg(target_arch = "wasm32")]
1425    {
1426        NonZeroUsize::MIN
1427    }
1428}

1430/// Maps `f` over `items` across at most `jobs` scoped worker threads, keeping
1431/// the results in input order.
1432///
1433/// Chunked rather than work-stealing: the callers
1434/// map over module definitions, whose per-item cost is uneven but whose counts
1435/// are small.
1436///
1437/// wasm has no threads, so it always runs serially.
1438///
1439/// # Panics
1440///
1441/// Panics if `f` panics in a worker thread.
1442///
1443// ponytail: scoped workers are enough; add a pool only if spawn profiles hot.
1444pub fn map_jobs<T: Sync, R: Send>(
1445    items: &[T],
1446    jobs: NonZeroUsize,
1447    f: impl Fn(&T) -> R + Sync,
1448) -> Vec<R> {
1449    #[cfg(target_arch = "wasm32")]
1450    let _ = jobs;
1451    #[cfg(not(target_arch = "wasm32"))]
1452    if items.len() > 1 && jobs.get() > 1 {
1453        let worker_count = jobs.get().min(items.len());
1454        let f = &f;
1455        let cancellation = crate::CancellationToken::current();
1456        return std::thread::scope(|scope| {
1457            let workers = (0..worker_count)
1458                .map(|worker| {
1459                    let start = items.len() * worker / worker_count;
1460                    let end = items.len() * (worker + 1) / worker_count;
1461                    let chunk = &items[start..end];
1462                    let cancellation = cancellation.clone();
1463                    scope.spawn(move || {
1464                        let run = || chunk.iter().map(f).collect::<Vec<_>>();
1465                        match cancellation {
1466                            Some(token) => token.scope(run),
1467                            None => run(),
1468                        }
1469                    })
1470                })
1471                .collect::<Vec<_>>();
1472            workers
1473                .into_iter()
1474                .flat_map(|worker| worker.join().expect("map_jobs worker does not panic"))
1475                .collect()
1476        });
1477    }
1478    items.iter().map(f).collect()
1479}

1481fn summarize_program(
1482    program: &BlockGraph,
1483    limits: ModuleCertificationLimits,
1484    jobs: NonZeroUsize,
1485) -> Result<Arc<ModuleSummary>, ModuleCertificationError> {
1486    preflight_program_expansion(program, limits).map_err(|error| match error {
1487        ModuleError::InvalidGeometry { source, .. } => source,
1488        _ => unreachable!("size preflight only reports resource limits"),
1489    })?;
1490    let mut depths = HashMap::new();
1491    let mut levels: Vec<Vec<&BlockGraph>> = Vec::new();
1492    let mut guarded = false;
1493    for module in module_postorder_from(program, [program.root().name.as_str()]) {
1494        guarded |= module.local_body().has_continuing_branches();
1495        let depth = module
1496            .instances
1497            .iter()
1498            .map(|instance| depths[instance.definition.as_str()] + 1)
1499            .max()
1500            .unwrap_or(0);
1501        levels.resize_with(levels.len().max(depth + 1), Vec::new);
1502        levels[depth].push(module);
1503        depths.insert(module.name.as_str(), depth);
1504    }
1505
1506    let mut summaries = HashMap::new();
1507    for level in levels {
1508        let results = map_jobs(&level, jobs, |&module| {
1509            summarize_definition(program, module, limits, &summaries, guarded)
1510        });
1511        for (module, summary) in level.iter().zip(results) {
1512            summaries.insert(module.name.clone(), Arc::new(summary?));
1513        }
1514    }
1515    Ok(Arc::clone(
1516        summaries
1517            .get(&program.root().name)
1518            .expect("dependency order includes the root module"),
1519    ))
1520}

1522fn summarize_definition(
1523    program: &BlockGraph,
1524    module: &BlockGraph,
1525    limits: ModuleCertificationLimits,
1526    summaries: &HashMap<String, Arc<ModuleSummary>>,
1527    guarded: bool,
1528) -> Result<ModuleSummary, ModuleCertificationError> {
1529    let projection = if guarded {
1530        let linked = crate::flatten_module_definition(program, module, "").map_err(|source| {
1531            ModuleCertificationError::Graph {
1532                module: module.name.clone(),
1533                source,
1534            }
1535        })?;
1536        let targets = linked
1537            .graph
1538            .branch_regions()
1539            .map_err(|source| ModuleCertificationError::Graph {
1540                module: module.name.clone(),
1541                source,
1542            })?
1543            .into_iter()
1544            .map(|region| region.target)
1545            .collect::<Vec<_>>();
1546        if !targets.is_empty() {
1547            let certify = || -> Result<(), crate::BlockGraphError> {
1548                let topology = crate::GuardedTopology::new(&linked.graph, limits)?;
1549                crate::GuardedSurfaceSpace::new(topology, &linked.sites, limits)?
1550                    .plan_readouts()
1551                    .map(drop)
1552            };
1553            certify().map_err(|source| ModuleCertificationError::Graph {
1554                module: module.name.clone(),
1555                source,
1556            })?;
1557            Some(Box::new(ModuleProjection {
1558                definition: module.clone_local_definition(),
1559                targets,
1560                limits,
1561            }))
1562        } else {
1563            None
1564        }
1565    } else {
1566        None
1567    };
1568    if module.instances.is_empty() {
1569        let mut summary = summarize_leaf(module, limits)?;
1570        if guarded {
1571            summary.projection = projection;
1572            return Ok(summary);
1573        }
1574        summary.branch_variants = branch_assignments(module, limits)?
1575            .into_iter()
1576            .map(|assignments| {
1577                let mut key = assignments
1578                    .iter()
1579                    .copied()
1580                    .filter(|(_, value)| !value)
1581                    .collect::<Vec<_>>();
1582                key.sort_unstable_by_key(|(position, value)| (position.to_array(), *value));
1583                let body = module
1584                    .local_body()
1585                    .project_branches_deferred(assignments.iter().copied())
1586                    .map_err(|source| ModuleCertificationError::Graph {
1587                        module: module.name.clone(),
1588                        source,
1589                    })?;
1590                Ok(ModuleBranchVariant {
1591                    assignments: key,
1592                    summary: Box::new(summarize_leaf_body(module, &body, limits)?),
1593                })
1594            })
1595            .collect::<Result<_, ModuleCertificationError>>()?;
1596        return Ok(summary);
1597    }
1598    let children = module
1599        .instances
1600        .iter()
1601        .map(|instance| {
1602            let summary = summaries
1603                .get(&instance.definition)
1604                .expect("dependency summary is ready");
1605            (instance.name.as_str(), (instance, Arc::clone(summary)))
1606        })
1607        .collect();
1608    let mut canonical = module.clone_local_definition();
1609    canonical.replace_local_body(module.local_body().canonical_true_branch_view().map_err(
1610        |source| ModuleCertificationError::Graph {
1611            module: module.name.clone(),
1612            source,
1613        },
1614    )?);
1615    let mut summary = summarize_composite(&canonical, module, &children, limits)?;
1616    if guarded {
1617        summary.projection = projection;
1618    } else {
1619        summary.branch_variants = summarize_composite_variants(module, summaries, limits)?;
1620    }
1621    Ok(summary)
1622}

1624fn branch_assignments(
1625    module: &BlockGraph,
1626    limits: ModuleCertificationLimits,
1627) -> Result<Vec<Vec<(IVec3, bool)>>, ModuleCertificationError> {
1628    let targets = module
1629        .local_body()
1630        .branch_regions()
1631        .map_err(|source| ModuleCertificationError::Graph {
1632            module: module.name.clone(),
1633            source,
1634        })?
1635        .into_iter()
1636        .map(|region| region.target)
1637        .collect::<Vec<_>>();
1638    if targets.is_empty() {
1639        return Ok(Vec::new());
1640    }
1641    if targets.len() > limits.max_guarded_domain_size {
1642        return Err(resource_limit(
1643            &module.name,
1644            "structural module alternatives",
1645            targets.len(),
1646            limits.max_guarded_domain_size,
1647        ));
1648    }
1649    Ok((0..targets.len())
1650        .map(|alternative| {
1651            targets
1652                .iter()
1653                .enumerate()
1654                .map(|(index, &target)| (target, index != alternative))
1655                .collect()
1656        })
1657        .collect())
1658}

1660fn summarize_composite_variants(
1661    module: &BlockGraph,
1662    summaries: &HashMap<String, Arc<ModuleSummary>>,
1663    limits: ModuleCertificationLimits,
1664) -> Result<Vec<ModuleBranchVariant>, ModuleCertificationError> {
1665    let local_assignments = branch_assignments(module, limits)?;
1666    let variant_count = local_assignments.len().saturating_add(
1667        module
1668            .instances
1669            .iter()
1670            .map(|instance| summaries[&instance.definition].branch_variants.len())
1671            .sum::<usize>(),
1672    );
1673    if variant_count > limits.max_guarded_domain_size {
1674        return Err(resource_limit(
1675            &module.name,
1676            "structural module alternatives",
1677            variant_count,
1678            limits.max_guarded_domain_size,
1679        ));
1680    }
1681    let mut variants = Vec::with_capacity(variant_count);
1682    let canonical_children = module
1683        .instances
1684        .iter()
1685        .map(|instance| {
1686            (
1687                instance.name.as_str(),
1688                (instance, Arc::clone(&summaries[&instance.definition])),
1689            )
1690        })
1691        .collect::<HashMap<_, _>>();
1692    for assignments in local_assignments {
1693        let body = module
1694            .local_body()
1695            .project_branches_in_definition(assignments.iter().copied())
1696            .map_err(|source| ModuleCertificationError::Graph {
1697                module: module.name.clone(),
1698                source,
1699            })?;
1700        let mut projected = module.clone_local_definition();
1701        projected.replace_local_body(body);
1702        variants.push(ModuleBranchVariant {
1703            assignments: assignments
1704                .into_iter()
1705                .filter(|(_, value)| !value)
1706                .collect(),
1707            summary: Box::new(summarize_composite(
1708                &projected,
1709                module,
1710                &canonical_children,
1711                limits,
1712            )?),
1713        });
1714    }
1715
1716    let mut canonical = module.clone_local_definition();
1717    canonical.replace_local_body(module.local_body().canonical_true_branch_view().map_err(
1718        |source| ModuleCertificationError::Graph {
1719            module: module.name.clone(),
1720            source,
1721        },
1722    )?);
1723    for (selected, instance) in module.instances.iter().enumerate() {
1724        let child = &summaries[&instance.definition];
1725        for alternative in &child.branch_variants {
1726            let children = module
1727                .instances
1728                .iter()
1729                .enumerate()
1730                .map(|(index, instance)| {
1731                    let summary = if index == selected {
1732                        Arc::new(alternative.summary.as_ref().clone())
1733                    } else {
1734                        Arc::clone(&summaries[&instance.definition])
1735                    };
1736                    (instance.name.as_str(), (instance, summary))
1737                })
1738                .collect::<HashMap<_, _>>();
1739            let mut assignments = alternative
1740                .assignments
1741                .iter()
1742                .map(|&(target, value)| {
1743                    instance
1744                        .try_transform_position(target)
1745                        .map(|target| (target, value))
1746                        .map_err(|source| ModuleCertificationError::Graph {
1747                            module: module.name.clone(),
1748                            source,
1749                        })
1750                })
1751                .collect::<Result<Vec<_>, _>>()?;
1752            assignments.sort_unstable_by_key(|(position, value)| (position.to_array(), *value));
1753            variants.push(ModuleBranchVariant {
1754                assignments,
1755                summary: Box::new(summarize_composite(&canonical, module, &children, limits)?),
1756            });
1757        }
1758    }
1759    Ok(variants)
1760}

1790fn summarize_readouts(
1791    body: &BlockGraph,
1792    zx: &ZXGraph,
1793    stabilizers: &StabilizerGenerators,
1794    table: &crate::zx::ProjectedExternalTable,
1795    port_columns: &[usize],
1796) -> Vec<SummaryMeasurement> {
1797    let measurement_columns = zx.measurement_columns();
1798    let measurement_observables = zx
1799        .action_graph()
1800        .ordered_nodes()
1801        .filter_map(|node| {
1802            let Action::Measure { name, .. } = &node.action else {
1803                return None;
1804            };
1805            let observable = match node.measurement? {
1806                MeasurementObservable::Concrete(PauliBasis::X) => Pauli::X,
1807                MeasurementObservable::Concrete(PauliBasis::Z) => Pauli::Z,
1808                MeasurementObservable::Concrete(PauliBasis::Y)
1809                | MeasurementObservable::Selective(_) => Pauli::Y,
1810            };
1811            Some((name.as_str(), observable))
1812        })
1813        .collect::<HashMap<_, _>>();
1814    let owned_positions = body
1815        .blocks()
1816        .filter(|block| !block.kind().is_port())
1817        .map(Block::pos)
1818        .collect::<FxHashSet<_>>();
1819    // One basis per certificate, reduced against every generator below.
1820    let flow_basis = table.flow_witness_basis();
1821    stabilizers
1822        .generators
1823        .iter()
1824        .enumerate()
1825        .filter_map(|(index, generator)| {
1826            let (name, target, observable, self_readers) = match &generator.kind {
1827                StabilizerRowKind::Measurement { name } => (
1828                    name.clone(),
1829                    support_target(zx, measurement_columns[name.as_str()]),
1830                    measurement_observables[name.as_str()],
1831                    self_reader_targets(zx, name),
1832                ),
1833                StabilizerRowKind::SelectiveFixing { targets } => (
1834                    format!("#fix{index}"),
1835                    SupportTarget::Node(targets[0].pos),
1836                    Pauli::I,
1837                    Vec::new(),
1838                ),
1839                StabilizerRowKind::Logical => return None,
1840            };
1841            let support = PositionedSupport::from_stabilizer(&generator.stabilizer, zx);
1842            let mut raw = generator.stabilizer.paulis.clone();
1843            zx.clear_cross_centers(std::slice::from_mut(&mut raw));
1844            let witness = flow_basis
1845                .witness_for_row(&raw)
1846                .expect("certified generator belongs to the projected row space");
1847            Some(SummaryMeasurement {
1848                name,
1849                kind: generator.kind.clone(),
1850                signed: PhasedPauliString::new(
1851                    PauliString::from_terms(
1852                        port_columns.len(),
1853                        port_columns.iter().enumerate().map(|(target, &source)| {
1854                            (target, generator.stabilizer.paulis.get(source))
1855                        }),
1856                    ),
1857                    stabilizer_phase(&generator.stabilizer),
1858                ),
1859                support,
1860                witness,
1861                target,
1862                observable,
1863                self_readers,
1864                owned_positions: owned_positions.clone(),
1865            })
1866        })
1867        .collect::<Vec<_>>()
1868}

1927                        .enumerate()

1934        })
1935        .collect::<Vec<_>>();
1936    let protected_groups =
1937        semantic_protected_groups(&body, &measurements, zx, limits, &module.name)?;
1938    let (discharged_measurements, measurements): (Vec<_>, Vec<_>) = measurements
1939        .into_iter()
1940        .partition(measurement_can_discharge);
1941    let protected = semantic_protected_supports(module, &body);
1942    let frontier = semantic_frontier_supports(module, &body)?;
1943    let adjustments = adjustment_rows(
1944        zx,
1945        &certificate.table,
1946        &rows,
1947        &measurements,
1948        &protected,
1949        &frontier,
1950        module.interface.quantum_ports.len(),
1951    );
1952    let certificate_rows = retained_certificate_rows(
1953        zx,
1954        &certificate.table,
1955        &rows,
1956        module.interface.quantum_ports.len(),
1957    );
1958    Ok(ModuleSummary {
1959        name: module.name.clone(),
1960        limits,
1961        dependencies: HashMap::new(),
1962        ports: module.interface.quantum_ports.clone(),
1963        signatures,
1964        rows,
1965        logical_rows,
1966        certificate_rows,
1967        adjustments,
1968        measurements,
1969        discharged_measurements,
1970        protected,
1971        frontier,
1972        protected_groups,
1973        branch_variants: Vec::new(),
1974        projection: None,
1975    })
1976}

1978fn measurement_can_discharge(measurement: &SummaryMeasurement) -> bool {
1979    matches!(measurement.kind, StabilizerRowKind::Measurement { .. })
1980        && measurement.signed.paulis.is_identity()
1981        && support_target_is_owned(measurement.target, &measurement.owned_positions)
1982        && measurement
1983            .self_readers
1984            .iter()
1985            .all(|&target| support_target_is_owned(target, &measurement.owned_positions))
1986}

1988fn support_target_is_owned(target: SupportTarget, owned: &FxHashSet<IVec3>) -> bool {
1989    match target {
1990        SupportTarget::Node(position) => owned.contains(&position),
1991        SupportTarget::Edge(left, right) => owned.contains(&left) && owned.contains(&right),
1992    }
1993}

1995fn support_target(zx: &ZXGraph, column: usize) -> SupportTarget {
1996    if column < zx.nodes().len() {
1997        SupportTarget::Node(zx.nodes()[column].pos)
1998    } else {
1999        let edge = &zx.edges()[column - zx.nodes().len()];
2000        SupportTarget::Edge(zx.nodes()[edge.n1].pos, zx.nodes()[edge.n2].pos)
2001    }
2002}

2004fn self_reader_targets(zx: &ZXGraph, measurement: &str) -> Vec<SupportTarget> {
2005    let actions = zx.action_graph().ordered_nodes().collect::<Vec<_>>();
2006    let Some(start) = actions.iter().position(
2007        |node| matches!(&node.action, Action::Measure { name, .. } if name == measurement),
2008    ) else {
2009        return Vec::new();
2010    };
2011    let mut successors = vec![Vec::new(); actions.len()];
2012    for (from, to, _) in zx.action_graph().dependencies() {
2013        successors[from].push(to);
2014    }
2015    let mut seen = FxHashSet::default();
2016    let mut stack = successors[start].clone();
2017    while let Some(ordinal) = stack.pop() {
2018        if seen.insert(ordinal) {
2019            stack.extend(successors[ordinal].iter().copied());
2020        }
2021    }
2022    let mut targets = FxHashSet::default();
2023    for ordinal in seen {
2024        match &actions[ordinal].action {
2025            Action::Resolve { target, .. } | Action::Branch { target, .. } => {
2026                targets.insert(SupportTarget::Node(*target));
2027            }
2028            Action::Feedback {

2033                        .node_at(feedback.target)

2036                        Some(dir) => zx
2037                            .node_at(
2038                                checked_add_position(feedback.target, dir.to_ivec3())
2039                                    .expect("validated feedback edge"),
2040                            )

2042                        None if node.kind == crate::NodeKind::Port => zx

2044                            .and_then(|neighbors| (neighbors.len() == 1).then(|| neighbors[0])),

2694                            .expect("validated bind endpoint fits"),
2695                    ),
2696                    boundary_to_edge_hadamard: bind.hadamard,
2697                },
2698            )
2699        })
2700        .collect::<Vec<_>>();
2701    let bound = binds
2702        .iter()
2703        .map(|(bind, _)| &bind.endpoint)
2704        .collect::<HashSet<_>>();
2705    let retained = (0..parent_width)
2706        .chain(
2707            joined
2708                .endpoints
2709                .iter()
2710                .enumerate()
2711                .filter_map(|(index, endpoint)| {
2712                    (!bound.contains(endpoint)).then_some(parent_width + index)
2713                }),
2714        )
2715        .collect::<Vec<_>>();
2716    let mut protected = joined.protected.clone();
2717    merge_protected_supports(&mut protected, parent_protected);
2718    let mut frontier = joined.frontier.clone();
2719    frontier.extend_from_slice(parent_frontier);
2720    for rows in [&mut joined.rows, &mut joined.logical_rows] {
2721        *rows = compose_supported_rows(
2722            std::mem::take(rows),
2723            &mut [],
2724            &[],
2725            &[],
2726            &seam_pairs,
2727            &retained,
2728            &module.name,
2729        )?;
2730    }
2731    joined.certificate_rows =
2732        compose_certificate_rows(joined.certificate_rows, &seam_pairs, &retained);
2733    joined.adjustments = compose_supported_rows(
2734        joined.adjustments,
2735        &mut joined.measurements,
2736        &protected,
2737        &frontier,
2738        &seam_pairs,
2739        &retained,
2740        &module.name,
2741    )?;
2742    resize_discharged_boundaries(
2743        &mut joined.discharged_measurements,
2744        retained.len(),
2745        &module.name,
2746    )?;
2747    joined
2748        .endpoints
2749        .retain(|endpoint| !bound.contains(endpoint));
2750    Ok(joined)
2751}

2915    fn solve(

2919        let limit = self.limits.max_normalization_states;
2920        if *self.visited == limit {

2927        }
2928        *self.visited += 1;
2929        let rows = self.adjustments.len();

2932            columns
2933                .checked_add(rows)?
2934                .checked_mul(columns.min(rows))?
2935                .checked_add(projection)?

2937                .checked_add(rows)
2938        });

2944                words.unwrap_or(usize::MAX),

2982    fn search(

2986    ) -> Result<Option<Vec<usize>>, ModuleCertificationError> {
2987        let mut point = NormalizationPoint::Factor(0);
2988        let mut frames = Vec::<NormalizationFrame<'_>>::new();

2990        loop {

2992            if combination.is_some() {
2993                let next = match point {
2994                    NormalizationPoint::Factor(index) => {
2995                        let Some(factor) = factors.get(index) else {
2996                            return Ok(combination);

2999                            NormalizationFactor::Site(constraint) => {

3001                                let mut values = match constraint.forbidden {
3002                                    Some(forbidden) => [Pauli::I, Pauli::X, Pauli::Z, Pauli::Y]
3003                                        .into_iter()
3004                                        .filter(|&pauli| pauli != forbidden)
3005                                        .collect::<Vec<_>>(),
3006                                    None => vec![current],
3007                                };
3008                                values.sort_by_key(|&pauli| pauli != current);
3009                                NormalizationChoices::Site {
3010                                    target: constraint.target,
3011                                    values: values.into_iter(),
3012                                }
3013                            }
3014                            NormalizationFactor::Group(group) => {
3015                                let current = group
3016                                    .targets
3017                                    .iter()
3018                                    .map(|&target| self.row.support.get(target))
3019                                    .collect::<Vec<_>>();

3021                                    || group
3022                                        .permitted_support
3023                                        .iter()
3024                                        .any(|support| partial_support_matches(&current, support));
3025                                NormalizationChoices::Group {
3026                                    group,
3027                                    current,
3028                                    permitted,
3029                                    next: 0,
3030                                }
3031                            }
3032                        };
3033                        Some((index, choices))
3034                    }
3035                    NormalizationPoint::Partial {
3036                        factor,
3037                        support,
3038                        start,
3039                    } => {
3040                        let NormalizationFactor::Group(group) = factors[factor] else {
3041                            unreachable!("partial choices belong to a group")
3042                        };
3043                        let chosen = &partial[start..];
3044                        let template = &group.permitted_support[support];
3045                        let repeated = group.permitted_support[..support].iter().any(|earlier| {
3046                            partial_support_matches(chosen, earlier)
3047                                && template[chosen.len()..] == earlier[chosen.len()..]

3050                            None
3051                        } else if chosen.len() == template.len() {
3052                            if chosen.iter().any(|&pauli| pauli != Pauli::I)
3053                                && chosen
3054                                    .iter()

3056                                    .any(|(&pauli, &target)| pauli != self.row.support.get(target))
3057                            {
3058                                point = NormalizationPoint::Factor(factor + 1);
3059                                continue;
3060                            }
3061                            None
3062                        } else {
3063                            Some((
3064                                factor,
3065                                NormalizationChoices::Partial {
3066                                    group,
3067                                    support,
3068                                    start,
3069                                    next: 0,
3070                                },
3071                            ))
3072                        }
3073                    }
3074                };
3075                if let Some((factor, choices)) = next {
3076                    frames.push(NormalizationFrame {
3077                        factor,
3078                        constraints: constraints.len(),
3079                        partial: partial.len(),
3080                        choices,
3081                    });
3082                }
3083            }
3084            // Resume the deepest unfinished choice. Both factors and partial
3085            // supports share heap frames, so source width never grows the call stack.
3086            loop {
3087                let Some(frame) = frames.last_mut() else {
3088                    return Ok(None);
3089                };

3091                partial.truncate(frame.partial);
3092                let next = match &mut frame.choices {
3093                    NormalizationChoices::Site { target, values } => values.next().map(|desired| {
3094                        self.constrain(constraints, *target, desired);
3095                        NormalizationPoint::Factor(frame.factor + 1)
3096                    }),
3097                    NormalizationChoices::Group {
3098                        group,
3099                        current,
3100                        permitted,
3101                        next,
3102                    } => {
3103                        let choice = *next;
3104                        *next += 1;
3105                        match choice {
3106                            0 => {
3107                                if !*permitted {
3108                                    continue;
3109                                }
3110                                for (&target, &desired) in group.targets.iter().zip(current.iter())
3111                                {

3114                                Some(NormalizationPoint::Factor(frame.factor + 1))
3115                            }
3116                            1 => {

3118                                    continue;

3120                                for &target in &group.targets {
3121                                    self.constrain(constraints, target, Pauli::I);
3122                                }
3123                                Some(NormalizationPoint::Factor(frame.factor + 1))
3124                            }
3125                            _ => group.permitted_support.get(choice - 2).map(|support| {
3126                                // A permitted cap is I or its basis: one affine
3127                                // equation prunes the entire remaining subtree.
3128                                constraints.extend(group.targets.iter().zip(support).map(
3129                                    |(&target, &basis)| {
3130                                        let axis = basis.flip();
3131                                        (
3132                                            target,
3133                                            axis,
3134                                            pauli_component_parity(
3135                                                self.row.support.get(target),
3136                                                axis,
3137                                            ),
3138                                        )
3139                                    },
3140                                ));
3141                                NormalizationPoint::Partial {
3142                                    factor: frame.factor,
3143                                    support: choice - 2,
3144                                    start: partial.len(),
3145                                }
3146                            }),
3147                        }
3148                    }
3149                    NormalizationChoices::Partial {
3150                        group,
3151                        support,
3152                        start,
3153                        next,
3154                    } => {
3155                        let column = partial.len() - *start;
3156                        let basis = group.permitted_support[*support][column];
3157                        let desired = match *next {
3158                            0 => Some(Pauli::I),
3159                            1 if basis != Pauli::I => Some(basis),
3160                            _ => None,
3161                        };
3162                        *next += 1;
3163                        desired.map(|desired| {
3164                            partial.push(desired);
3165                            self.constrain(constraints, group.targets[column], desired);
3166                            NormalizationPoint::Partial {
3167                                factor: frame.factor,
3168                                support: *support,
3169                                start: *start,
3170                            }
3171                        })
3172                    }
3173                };
3174                if let Some(next) = next {
3175                    point = next;
3176                    break;
3177                }
3178                frames.pop();
3179            }
3180        }
3181    }
3182}

3306fn solve_binary_combination(rows: &[Vec<bool>], target: &[bool]) -> Option<Vec<usize>> {
3307    let mut basis = vec![None::<(Vec<bool>, Vec<bool>)>; target.len()];
3308    for (index, source) in rows.iter().enumerate() {
3309        let mut row = source.clone();
3310        let mut coeff = vec![false; rows.len()];
3311        coeff[index] = true;
3312        for column in 0..target.len() {
3313            if !row[column] {
3314                continue;
3315            }
3316            if let Some((pivot, pivot_coeff)) = &basis[column] {
3317                xor_bits(&mut row, pivot);
3318                xor_bits(&mut coeff, pivot_coeff);
3319            } else {
3320                basis[column] = Some((row, coeff));
3321                break;
3322            }
3323        }
3324    }
3325
3326    let mut residual = target.to_vec();
3327    let mut coeff = vec![false; rows.len()];
3328    for column in 0..target.len() {
3329        if !residual[column] {
3330            continue;
3331        }
3332        let (pivot, pivot_coeff) = basis[column].as_ref()?;
3333        xor_bits(&mut residual, pivot);
3334        xor_bits(&mut coeff, pivot_coeff);
3335    }
3336    residual.iter().all(|&bit| !bit).then(|| {
3337        coeff
3338            .iter()
3339            .enumerate()
3340            .filter_map(|(i, &set)| set.then_some(i))
3341            .collect()
3342    })
3343}

3345fn xor_bits(left: &mut [bool], right: &[bool]) {
3346    for (left, &right) in left.iter_mut().zip(right) {
3347        *left ^= right;
3348    }

4090                // This pivot is discarded. Its final surviving user can keep
4091                // the larger support and witness buffers. These payloads commute;
4092                // signed Pauli multiplication must retain its original order.
4093                if Some(index) == last_use {
4094                    if row.row.support.weight() < pivot.row.support.weight() {
4095                        std::mem::swap(&mut row.row.support, &mut pivot.row.support);
4096                    }
4097                    if row.row.witness.source_count() < pivot.row.witness.source_count() {
4098                        std::mem::swap(&mut row.row.witness, &mut pivot.row.witness);
4099                    }
4100                }
4101                EliminationRow::multiply_assign(&mut row.row, &pivot.row);
4102            }
4103        }
4104        solved += 1;
4105    }
4106    rows.drain(..solved);
4107}

4109/// Contracts a measurement-free table without admitting future rows or
4110/// columns. Rows whose last seam closes keep their support/witness here, then
4111/// rejoin only for the caller's final canonical reduction.
4112fn compose_frontier_rows(
4113    rows: Vec<SupportedRow>,
4114    seams: &[(usize, usize, bool, bool, SupportSeam)],
4115    retained: &[usize],
4116    keep_closed: impl Fn(&SupportedRow) -> bool,
4117) -> Vec<SupportedRow> {
4118    let Some(width) = rows.first().map(|row| row.signed.paulis.len()) else {
4119        return rows;
4120    };
4121    let mut seam_stage = vec![None; width];
4122    for (stage, &(left, right, ..)) in seams.iter().enumerate() {
4123        debug_assert!(seam_stage[left].is_none());
4124        debug_assert!(seam_stage[right].is_none());
4125        seam_stage[left] = Some(stage);
4126        seam_stage[right] = Some(stage);
4127    }
4128
4129    let mut pending = (0..seams.len())
4130        .map(|_| Vec::<FrontierRow>::new())
4131        .collect::<Vec<_>>();
4132    let mut finished = Vec::new();
4133    for (order, row) in rows.into_iter().enumerate() {
4134        let first_stage = row
4135            .signed
4136            .paulis
4137            .iter_support()
4138            .filter_map(|(column, _)| seam_stage[column])
4139            .min();
4140        let mut row = FrontierRow { order, row };
4141        if let Some(stage) = first_stage {
4142            pending[stage].push(row);
4143        } else {
4144            remap_row_columns(&mut row.row, None, retained);
4145            if !row.row.signed.paulis.is_identity() || keep_closed(&row.row) {
4146                finished.push(row);
4147            }
4148        }
4149    }
4150
4151    let mut active = Vec::<FrontierRow>::new();
4152    let mut columns = retained.to_vec();
4153    for (stage, &(left, right, hadamard, transpose, support_seam)) in seams.iter().enumerate() {
4154        let mut next_columns = retained.to_vec();
4155        next_columns.extend(active.iter().flat_map(|row| {
4156            row.row
4157                .signed
4158                .paulis
4159                .iter_support()
4160                .map(|(column, _)| columns[column])
4161        }));
4162        next_columns.extend(pending[stage].iter().flat_map(|row| {
4163            row.row
4164                .signed
4165                .paulis
4166                .iter_support()
4167                .map(|(column, _)| column)
4168        }));
4169        next_columns.extend([left, right]);
4170        next_columns.sort_unstable();
4171        next_columns.dedup();
4172
4173        for row in &mut active {
4174            remap_row_columns(&mut row.row, Some(&columns), &next_columns);
4175        }
4176        for row in &mut pending[stage] {
4177            remap_row_columns(&mut row.row, None, &next_columns);
4178        }
4179        active.append(&mut pending[stage]);
4180        active.sort_unstable_by_key(|row| row.order);
4181        columns = next_columns;
4182
4183        let left = columns
4184            .binary_search(&left)
4185            .expect("live frontier contains the seam's left column");
4186        let right = columns
4187            .binary_search(&right)
4188            .expect("live frontier contains the seam's right column");
4189        if hadamard {
4190            for row in &mut active {
4191                conjugate_supported_h(&mut row.row, right);
4192            }
4193        }
4194        eliminate_frontier_seam(&mut active, left, right);
4195        for row in &mut active {
4196            contract_supported_row(&mut row.row, left, right, transpose, support_seam);
4197        }
4198
4199        let mut next_active = Vec::with_capacity(active.len());
4200        for mut row in active.drain(..) {
4201            let reaches_future = row.row.signed.paulis.iter_support().any(|(column, _)| {
4202                seam_stage[columns[column]].is_some_and(|row_stage| row_stage > stage)
4203            });
4204            if reaches_future {
4205                next_active.push(row);
4206            } else {
4207                remap_row_columns(&mut row.row, Some(&columns), retained);
4208                if !row.row.signed.paulis.is_identity() || keep_closed(&row.row) {
4209                    finished.push(row);
4210                }
4211            }
4212        }
4213        active = next_active;
4214    }
4215    debug_assert!(active.is_empty());
4216    debug_assert!(pending.iter().all(Vec::is_empty));
4217
4218    finished.sort_unstable_by_key(|row| row.order);
4219    finished.into_iter().map(|row| row.row).collect()
4220}

4222fn compose_supported_rows(
4223    mut rows: Vec<SupportedRow>,
4224    measurements: &mut [SummaryMeasurement],
4225    protected: &[ProtectedSupport],
4226    frontier: &[SupportTarget],
4227    seams: &[(usize, usize, bool, bool, SupportSeam)],
4228    retained: &[usize],
4229    module: &str,
4230) -> Result<Vec<SupportedRow>, ModuleCertificationError> {
4231    if measurements.is_empty() && seams.len() > 1 {
4232        rows = compose_frontier_rows(rows, seams, retained, |row| {
4233            protected
4234                .iter()
4235                .any(|constraint| row.support.get(constraint.target) != Pauli::I)
4236                || frontier
4237                    .iter()
4238                    .any(|&target| row.support.get(target) != Pauli::I)
4239        });
4240    } else {
4241        for &(left, right, hadamard, transpose, support_seam) in seams {
4242            if hadamard {
4243                for row in &mut rows {
4244                    conjugate_supported_h(row, right);
4245                }
4246                for measurement in &mut *measurements {
4247                    conjugate_measurement_h(measurement, right);
4248                }
4249            }
4250
4251            if !measurements.is_empty() {
4252                let mut adjustment_basis = AdjustmentBasis::new(&rows, measurements, protected);
4253                for index in 0..measurements.len() {
4254                    let mut candidates = safe_adjustments(
4255                        &mut adjustment_basis,
4256                        measurements,
4257                        index,
4258                        protected,
4259                        module,
4260                    )?;
4261                    let measurement = &mut measurements[index];
4262                    eliminate_supported_seam(
4263                        &mut candidates,
4264                        std::slice::from_mut(measurement),
4265                        left,
4266                        right,
4267                    );
4268                    if measurement.signed.paulis.get(left) != measurement.signed.paulis.get(right) {
4269                        return Err(invalid(
4270                            module,
4271                            &format!(
4272                                "measurement '{}' cannot close across a module seam",
4273                                measurement.name
4274                            ),
4275                        ));
4276                    }
4277                }
4278            }
4279
4280            let eliminated = eliminate_supported_seam(&mut rows, &mut [], left, right);
4281            rows.drain(..eliminated);
4282
4283            for row in &mut rows {
4284                contract_supported_row(row, left, right, transpose, support_seam);
4285            }
4286            rows.retain(|row| {
4287                !row.signed.paulis.is_identity()
4288                    || measurements.iter().any(|measurement| {
4289                        row.support.get(measurement.target) != Pauli::I
4290                            || measurement
4291                                .self_readers
4292                                .iter()
4293                                .any(|&target| row.support.get(target) != Pauli::I)
4294                    })
4295                    || protected
4296                        .iter()
4297                        .any(|constraint| row.support.get(constraint.target) != Pauli::I)
4298                    || frontier
4299                        .iter()
4300                        .any(|&target| row.support.get(target) != Pauli::I)
4301            });
4302            for measurement in &mut *measurements {
4303                contract_measurement(measurement, left, right, transpose, support_seam);
4304            }
4305        }
4306
4307        for row in &mut rows {
4308            remap_row_columns(row, None, retained);
4309        }
4310    }
4311    let mut rank = signed_gaussian_elimination(
4312        &mut rows,
4313        (0..retained.len()).flat_map(|column| [Pauli::X, Pauli::Z].map(move |axis| (column, axis))),
4314        |row, &(column, axis)| row.signed.paulis.get(column) & axis,
4315    );
4316    if rank < rows.len() {
4317        let mut targets = measurements
4318            .iter()
4319            .map(|measurement| measurement.target)
4320            .chain(
4321                measurements
4322                    .iter()
4323                    .flat_map(|measurement| measurement.self_readers.iter().copied()),
4324            )
4325            .chain(protected.iter().map(|constraint| constraint.target))
4326            .chain(frontier.iter().copied())
4327            .collect::<FxHashSet<_>>()
4328            .into_iter()
4329            .collect::<Vec<_>>();
4330        targets.sort_unstable_by_key(support_target_key);
4331        rank = eliminate_support_targets(&mut rows, rank, &targets);
4332    }
4333    rows.truncate(rank);
4334    for measurement in measurements {
4335        measurement.signed.paulis = PauliString::from_terms(
4336            retained.len(),
4337            retained
4338                .iter()
4339                .enumerate()
4340                .map(|(target, &source)| (target, measurement.signed.paulis.get(source))),
4341        );
4342    }
4343    Ok(rows)
4344}

4346/// Compose a complete certificate basis without dropping boundary-zero rows.
4347/// These rows are private rank completion, not public observables.
4348fn compose_certificate_rows(
4349    mut rows: Vec<SupportedRow>,
4350    seams: &[(usize, usize, bool, bool, SupportSeam)],
4351    retained: &[usize],
4352) -> Vec<SupportedRow> {
4353    if seams.len() > 1 {
4354        rows = compose_frontier_rows(rows, seams, retained, |row| row.support.weight() != 0);
4355    } else {
4356        for &(left, right, hadamard, transpose, support_seam) in seams {
4357            if hadamard {
4358                for row in &mut rows {
4359                    conjugate_supported_h(row, right);
4360                }
4361            }
4362            let eliminated = eliminate_supported_seam(&mut rows, &mut [], left, right);
4363            rows.drain(..eliminated);
4364            for row in &mut rows {
4365                contract_supported_row(row, left, right, transpose, support_seam);
4366            }
4367        }
4368        for row in &mut rows {
4369            remap_row_columns(row, None, retained);
4370        }
4371        rows.retain(|row| !row.signed.paulis.is_identity() || row.support.weight() != 0);
4372    }
4373
4374    let boundary_rank = signed_gaussian_elimination(
4375        &mut rows,
4376        (0..retained.len()).flat_map(|column| [Pauli::X, Pauli::Z].map(move |axis| (column, axis))),
4377        |row, &(column, axis)| row.signed.paulis.get(column) & axis,
4378    );
4379    // Live composition needs a basis only on the remaining boundary. Keep all
4380    // residual witnesses for final rank completion; reducing their physical
4381    // support at every connector makes earlier prefixes grow through later rows.
4382    if !seams.is_empty() || boundary_rank == rows.len() {
4383        return rows;
4384    }
4385    let mut targets = rows
4386        .iter()
4387        .flat_map(|row| {
4388            row.support
4389                .nodes
4390                .keys()
4391                .copied()
4392                .map(SupportTarget::Node)
4393                .chain(
4394                    row.support
4395                        .edges
4396                        .keys()
4397                        .copied()
4398                        .map(|(left, right)| SupportTarget::Edge(left, right)),
4399                )
4400        })
4401        .collect::<FxHashSet<_>>()
4402        .into_iter()
4403        .collect::<Vec<_>>();
4404    targets.sort_unstable_by_key(support_target_key);
4405    let rank = eliminate_support_targets(&mut rows, boundary_rank, &targets);
4406    rows.truncate(rank);
4407    rows
4408}

4455/// A seam's row supports stay fixed while each measurement selects its own
4456/// constraints. Project them once; individual reductions clone only packed rows.
4457struct AdjustmentBasis<'a> {
4458    sources: Vec<&'a SupportedRow>,
4459    candidates: Vec<PhasedPauliString>,
4460    columns: FxHashMap<SupportTarget, usize>,
4461    boundary_width: usize,
4462    selector_start: usize,
4463    prefix_rank: usize,
4464    next_measurement: usize,
4465}
4466
4467impl<'a> AdjustmentBasis<'a> {
4468    fn new(
4469        rows: &'a [SupportedRow],
4470        measurements: &[SummaryMeasurement],
4471        protected: &[ProtectedSupport],
4472    ) -> Self {
4473        let mut sources = rows.iter().collect::<Vec<_>>();
4474        sources.sort_by_key(|row| row.support.weight());
4475        let boundary_width = sources.first().map_or(0, |row| row.signed.paulis.len());
4476        let mut targets = measurements
4477            .iter()
4478            .flat_map(|measurement| {
4479                std::iter::once(measurement.target).chain(measurement.self_readers.iter().copied())
4480            })
4481            .chain(protected.iter().map(|constraint| constraint.target))
4482            .collect::<Vec<_>>();
4483        targets.sort_unstable_by_key(support_target_key);
4484        targets.dedup();
4485        // Auxiliary support bits and source selectors use X-only columns, so
4486        // their XORs cannot contribute phases to the signed boundary rows.
4487        let selector_start = boundary_width + 2 * targets.len();
4488        let candidates = sources
4489            .iter()
4490            .enumerate()
4491            .map(|(source, row)| {
4492                let mut paulis = PauliString::from_terms(
4493                    selector_start + sources.len(),
4494                    row.signed.paulis.iter_support(),
4495                );
4496                for (column, &target) in targets.iter().enumerate() {
4497                    let value = row.support.get(target);
4498                    for (offset, axis) in [Pauli::X, Pauli::Z].into_iter().enumerate() {
4499                        if value & axis {
4500                            paulis.set(boundary_width + 2 * column + offset, Pauli::X);
4501                        }
4502                    }
4503                }
4504                paulis.set(selector_start + source, Pauli::X);
4505                PhasedPauliString::new(paulis, row.signed.phase())
4506            })
4507            .collect();
4508        let columns = targets
4509            .into_iter()
4510            .enumerate()
4511            .map(|(column, target)| (target, boundary_width + 2 * column))
4512            .collect();
4513        Self {
4514            sources,
4515            candidates,
4516            columns,
4517            boundary_width,
4518            selector_start,
4519            prefix_rank: 0,
4520            next_measurement: 0,
4521        }
4522    }
4523}

4525fn safe_adjustments(
4526    basis: &mut AdjustmentBasis<'_>,
4527    measurements: &[SummaryMeasurement],
4528    index: usize,
4529    protected: &[ProtectedSupport],
4530    module: &str,
4531) -> Result<Vec<SupportedRow>, ModuleCertificationError> {
4532    debug_assert_eq!(index, basis.next_measurement);
4533    let measurement = &measurements[index];
4534    let mut constraints = measurements
4535        .iter()
4536        .enumerate()
4537        .skip(index)
4538        .flat_map(|(other, candidate)| {
4539            let axes = if other == index {
4540                Pauli::Y
4541            } else {
4542                candidate.observable
4543            };
4544            axes.iter_xz().map(move |axis| (candidate.target, axis))
4545        })
4546        .collect::<Vec<_>>();
4547    constraints.extend(
4548        measurement
4549            .self_readers
4550            .iter()
4551            .copied()
4552            .flat_map(|target| [(target, Pauli::X), (target, Pauli::Z)]),
4553    );
4554    // A joint fixing names several caps; preserving only its first target
4555    // lets seam adjustments change the flip axis on the remaining caps.
4556    constraints.extend(
4557        measurement
4558            .kind
4559            .selective_fixing_targets()
4560            .iter()
4561            .flat_map(|target| {
4562                [Pauli::X, Pauli::Z].map(|axis| (SupportTarget::Node(target.pos), axis))
4563            }),
4564    );
4565    for constraint in protected {
4566        if measurement
4567            .kind
4568            .selective_fixing_targets()
4569            .iter()
4570            .any(|target| SupportTarget::Node(target.pos) == constraint.target)
4571        {
4572            continue;
4573        }
4574        if let Some(forbidden) = constraint.forbidden {
4575            let difference = measurement.support.get(constraint.target) ^ forbidden;
4576            let Some(axis) = difference.iter_xz().next() else {
4577                return Err(invalid(
4578                    module,
4579                    &format!(
4580                        "measurement '{}' has forbidden selective support",
4581                        measurement.name
4582                    ),
4583                ));
4584            };
4585            constraints.push((constraint.target, axis));
4586        } else {
4587            constraints.extend([(constraint.target, Pauli::X), (constraint.target, Pauli::Z)]);
4588        }
4589    }
4590
4591    let sources = &basis.sources;
4592    let boundary_width = basis.boundary_width;
4593    let selector_start = basis.selector_start;
4594    let mut candidates = basis.candidates.clone();
4595    let eliminated = signed_gaussian_elimination_from(
4596        &mut candidates,
4597        basis.prefix_rank,
4598        constraints
4599            .into_iter()
4600            .map(|(target, axis)| basis.columns[&target] + usize::from(axis == Pauli::Z)),
4601        |row, &column| row.paulis.get(column) == Pauli::X,
4602    );
4603    let mut candidates = candidates
4604        .into_iter()
4605        .skip(eliminated)
4606        .map(|row| {
4607            let mut support = PositionedSupport::default();
4608            let mut witness = FlowWitness::default();
4609            for (column, _) in row
4610                .paulis
4611                .iter_support()
4612                .filter(|&(column, _)| column >= selector_start)
4613            {
4614                let source = sources[column - selector_start];
4615                support.multiply_assign(&source.support);
4616                witness.xor_assign(&source.witness);
4617            }
4618            SupportedRow {
4619                signed: PhasedPauliString::new(
4620                    PauliString::from_terms(
4621                        boundary_width,
4622                        (0..boundary_width).map(|column| (column, row.paulis.get(column))),
4623                    ),
4624                    row.phase(),
4625                ),
4626                support,
4627                witness,
4628            }
4629        })
4630        .collect::<Vec<_>>();
4631    candidates.sort_by_key(|row| row.support.weight());
4632    // The next measurement has the same constraints through this observable.
4633    // Retain all prefix rows: later pivots must still update earlier rows' phases.
4634    let column = basis.columns[&measurement.target];
4635    basis.prefix_rank = signed_gaussian_elimination_from(
4636        &mut basis.candidates,
4637        basis.prefix_rank,
4638        measurement
4639            .observable
4640            .iter_xz()
4641            .map(|axis| column + usize::from(axis == Pauli::Z)),
4642        |row, &column| row.paulis.get(column) == Pauli::X,
4643    );
4644    basis.next_measurement += 1;
4645    Ok(candidates)
4646}

4648fn eliminate_supported_seam(
4649    rows: &mut [SupportedRow],
4650    measurements: &mut [SummaryMeasurement],
4651    left: usize,
4652    right: usize,
4653) -> usize {
4654    let mut solved = 0;
4655    for axis in [Pauli::X, Pauli::Z] {
4656        let mismatch = |row: &SupportedRow| {
4657            (row.signed.paulis.get(left) ^ row.signed.paulis.get(right)) & axis
4658        };
4659        let Some(pivot) = (solved..rows.len()).find(|&row| mismatch(&rows[row])) else {
4660            continue;
4661        };
4662        rows.swap(pivot, solved);
4663        let (before, pivot_and_after) = rows.split_at_mut(solved);
4664        let (pivot, after) = pivot_and_after
4665            .split_first_mut()
4666            .expect("solved pivot is in bounds");
4667        for row in before.iter_mut().chain(after) {
4668            if mismatch(row) {
4669                EliminationRow::multiply_assign(row, pivot);
4670            }
4671        }
4672        for measurement in &mut *measurements {
4673            if (measurement.signed.paulis.get(left) ^ measurement.signed.paulis.get(right)) & axis {
4674                multiply_measurement(measurement, pivot);
4675            }
4676        }
4677        solved += 1;
4678    }
4679    solved
4680}

4704fn contract_supported_row(
4705    row: &mut SupportedRow,
4706    left: usize,
4707    right: usize,
4708    transpose: bool,
4709    seam: SupportSeam,
4710) {
4711    let pauli = row.signed.paulis.get(left);
4712    if transpose && pauli == Pauli::Y {
4713        row.signed.shift_phase(2);
4714    }
4715    row.signed.paulis.set(left, Pauli::I);

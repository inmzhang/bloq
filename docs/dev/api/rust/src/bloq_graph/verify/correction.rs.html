24/// The readout of every named record in `basis`, by name: a relation row
25/// that reads the record's key and no other named key, preferring one that
26/// avoids live outputs and then the earliest final layer. Unnamed records may
27/// cross it; sampling compensates them. A crossing centre is the OR of its
28/// arms, so every condition is linear in the arm components.
29pub(super) fn named_readouts(
30    basis: &RuntimeStabilizerBasis,
31    records: &BTreeMap<String, MeasurementKey>,
32) -> Result<BTreeMap<String, crate::Stabilizer>, VerifyLogicalError> {
33    let zx = basis.zx_graph();
34    let components = records
35        .iter()
36        .map(|(name, key)| (name.as_str(), key_components(zx, key)))
37        .collect::<Vec<_>>();
38    let outputs = zx
39        .nodes()
40        .iter()
41        .filter(|node| node.is_output_port(zx))
42        .flat_map(|node| {
43            let axes: &[Pauli] = if node.role == crate::PortRole::Multiplex {
44                &[Pauli::X]
45            } else {
46                &[Pauli::X, Pauli::Z]
47            };
48            axes.iter().map(move |&axis| (node.id, axis))
49        })
50        .collect::<Vec<_>>();
51    // Columns by layer, latest first; an edge belongs to its later endpoint.
52    let mut layers = BTreeMap::<std::cmp::Reverse<i32>, Vec<(usize, Pauli)>>::new();
53    let columns = zx
54        .nodes()
55        .iter()
56        .map(|node| (node.id, node.pos.z))
57        .chain(zx.edges().iter().map(|edge| {
58            let z = zx.nodes()[edge.n1].pos.z.max(zx.nodes()[edge.n2].pos.z);
59            (edge.id, z)
60        }));
61    for (column, z) in columns {
62        layers
63            .entry(std::cmp::Reverse(z))
64            .or_default()
65            .extend([(column, Pauli::X), (column, Pauli::Z)]);
66    }
67    components
68        .iter()
69        .map(|(name, target)| {
70            let others = components
71                .iter()
72                .filter(|(other, _)| other != name)
73                .flat_map(|(_, components)| components.iter().copied())
74                .collect::<Vec<_>>();
75            [true, false]
76                .into_iter()
77                .find_map(|output_free| {
78                    let mut constraints = others.clone();
79                    if output_free {
80                        constraints.extend(outputs.iter().copied());
81                    }
82                    earliest_reading_row(basis.generators(), &constraints, target, &layers)
83                })
84                .map(|row| ((*name).to_owned(), zx.materialize_stabilizer(row)))
85                .ok_or_else(|| VerifyLogicalError::MeasurementPhaseUnavailable((*name).to_owned()))
86        })
87        .collect()
88}

90/// Linear components whose OR is the support a key reads.
91fn key_components(zx: &ZXGraph, key: &MeasurementKey) -> Vec<(usize, Pauli)> {
92    let (column, pauli) = match *key {
93        MeasurementKey::Node { pos, pauli } => {
94            (zx.node_at(pos).expect("record node exists").id, pauli)
95        }
96        MeasurementKey::Edge { src, dst, pauli } => {
97            let src = zx.node_at(src).expect("record edge source exists").id;
98            let dst = zx.node_at(dst).expect("record edge target exists").id;
99            (
100                zx.edge_between(src, dst).expect("record edge exists").id,
101                pauli,
102            )
103        }
104    };
105    pauli
106        .iter_xz()
107        .flat_map(|axis| zx.readout_components(column, axis))
108        .collect()
109}

111/// A row of the span of `rows` that avoids every constraint component and
112/// carries some target component, with no support after the earliest layer
113/// at which such a row exists.
114fn earliest_reading_row(
115    rows: &[crate::PauliString],
116    constraints: &[(usize, Pauli)],
117    target: &[(usize, Pauli)],
118    layers: &BTreeMap<std::cmp::Reverse<i32>, Vec<(usize, Pauli)>>,
119) -> Option<crate::PauliString> {
120    let reads =
121        |row: &crate::PauliString| target.iter().any(|&(column, axis)| row.get(column) & axis);
122    let mut kernel = rows.to_vec();
123    let restrict = |kernel: &mut Vec<crate::PauliString>, constraints: &[(usize, Pauli)]| {
124        let pivots = crate::zx::gaussian_elimination_with_tracking(
125            kernel,
126            &mut [],
127            constraints.iter().copied(),
128            |row, &(column, axis)| row.get(column) & axis,
129            None,
130        );
131        kernel.drain(..pivots);
132    };
133    restrict(&mut kernel, constraints);
134    let mut best = kernel.iter().find(|row| reads(row))?.clone();
135    for columns in layers.values() {
136        restrict(&mut kernel, columns);
137        match kernel.iter().find(|row| reads(row)) {
138            Some(row) => best = row.clone(),
139            None => break,
140        }
141    }
142    Some(best)
143}

159impl InternalCorrection {
160    fn parity_columns(
161        &self,
162        surfaces: &[crate::Stabilizer],
163        phases: &BTreeMap<MeasurementKey, Parity>,
164    ) -> Vec<CoeffVec> {
165        self.sampled_vars()
166            .map(|variable| {
167                let mut column = CoeffVec::zeros(surfaces.len());
168                for (row, surface) in surfaces.iter().enumerate() {
169                    column.set_bit(
170                        row,
171                        phases
172                            .iter()
173                            .filter(|(_, parity)| parity.iter().any(|value| value == variable))
174                            .fold(false, |odd, (key, _)| {
175                                odd ^ stabilizer_supports_measurement(surface, key)
176                            }),
177                    );
178                }
179                column
180            })
181            .collect()
182    }
183    pub(super) fn new(
184        zx: &ZXGraph,
185        named_phases: &BTreeMap<MeasurementKey, Parity>,
186        named_vars: &BTreeMap<String, Var>,
187        records: BTreeMap<String, MeasurementKey>,
188        next_var: &mut usize,
189    ) -> Result<Self, VerifyLogicalError> {
190        let keys = measurement_keys(zx);
191        let mut measurement_vars = named_phases.clone();
192        let mut sampled = Vec::new();
193        for key in keys {
194            if measurement_vars.contains_key(&key)
195                || matches!(&key, MeasurementKey::Node { pos, .. }
196                if measurement_vars.keys().any(|named| matches!(named, MeasurementKey::Node { pos: named, .. } if named == pos)))
197            {
198                continue;
199            }
200            let variable = fresh_var(next_var)?;
201            measurement_vars.insert(key.clone(), Parity::single(variable));
202            sampled.push((key, variable));
203        }
204
205        let initial_basis = RuntimeStabilizerBasis::from_zx_graph(zx.clone())
206            .map_err(|source| VerifyLogicalError::RuntimeBasisInitializationFailed { source })?;
207        let anchors = if sampled.is_empty() {
208            BTreeMap::new()
209        } else {
210            named_readouts(&initial_basis, &records)?
211        };
212        if !sampled.is_empty() && !named_vars.is_empty() {
213            // A free native phase can cross a named row. Compensate through the
214            // same right-inverse that realizes named outcomes, so the action still
215            // reads its supplied outcome after unnamed measurements are sampled.
216            for (name, readout) in &anchors {
217                let named = named_vars[name];
218                for (key, variable) in &sampled {
219                    if !stabilizer_supports_measurement(readout, key) {
220                        continue;
221                    }
222                    for (target, parity) in named_phases {
223                        if parity.iter().any(|value| value == named) {
224                            let target = measurement_vars
225                                .get_mut(target)
226                                .expect("named phase exists");
227                            *target = &*target + &Parity::single(*variable);
228                        }
229                    }
230                }
231            }
232        }
233        Ok(Self {
234            measurement_vars,
235            sampled,
236            initial_basis,
237            named_vars: named_vars.clone(),
238            records,
239        })
240    }

250    fn resolved_measurement_vars(
251        &self,
252        selectives: &FxHashMap<IVec3, MeasurementKey>,
253        rows: &[crate::Stabilizer],
254    ) -> Result<BTreeMap<MeasurementKey, Parity>, VerifyLogicalError> {
255        let mut phases = BTreeMap::<MeasurementKey, Parity>::new();
256        for (key, parity) in &self.measurement_vars {
257            let target = match key {
258                MeasurementKey::Node { pos, .. } => selectives.get(pos).unwrap_or(key),
259                MeasurementKey::Edge { .. } => key,
260            };
261            let entry = phases.entry(target.clone()).or_default();
262            *entry = &*entry + parity;
263        }
264        if self.named_vars.is_empty() {
265            return Ok(phases);
266        }
267        let columns = phases
268            .keys()
269            .map(|key| {
270                let mut column = CoeffVec::zeros(rows.len());
271                for (row, surface) in rows.iter().enumerate() {
272                    column.set_bit(row, stabilizer_supports_measurement(surface, key));
273                }
274                column
275            })
276            .collect::<Vec<_>>();
277        // A fill can move a later named readout onto different native records.
278        // Reparameterize the whole affine fiber, rather than preserving an
279        // obsolete pre-fill compensation or losing an anonymous direction.
280        let sampled = self.sampled_vars().collect::<BTreeSet<_>>();
281        for parity in phases.values_mut() {
282            *parity = Parity::new(
283                parity
284                    .iter()
285                    .filter(|variable| !sampled.contains(variable))
286                    .collect::<Vec<_>>(),
287                phase_constant(parity),
288            );
289        }
290        for (row, (name, &variable)) in self.named_vars.iter().enumerate() {
291            let mut rhs = CoeffVec::singleton(row, rows.len());
292            for (column, parity) in columns.iter().zip(phases.values()) {
293                if parity.iter().any(|value| value == variable) {
294                    rhs.xor_assign(column);
295                }
296            }
297            let adjustment = solve_coeff_combination(&columns, &rhs)
298                .ok_or_else(|| VerifyLogicalError::MeasurementPhaseUnavailable(name.clone()))?;
299            for (index, parity) in phases.values_mut().enumerate() {
300                if adjustment.bit(index) {
301                    *parity = &*parity + &Parity::single(variable);
302                }
303            }
304        }
305        let mut constant = CoeffVec::zeros(rows.len());
306        for (column, parity) in columns.iter().zip(phases.values()) {
307            if phase_constant(parity) {
308                constant.xor_assign(column);
309            }
310        }
311        let adjustment = solve_coeff_combination(&columns, &constant).ok_or_else(|| {
312            VerifyLogicalError::MeasurementPhaseUnavailable("affine readout offset".into())
313        })?;
314        for (index, parity) in phases.values_mut().enumerate() {
315            if adjustment.bit(index) {
316                *parity = parity.negated();
317            }
318        }
319        let directions = kernel_directions(&columns);
320        if directions.len() > self.sampled.len() {
321            return Err(VerifyLogicalError::MeasurementPhaseUnavailable(
322                "resolved anonymous coordinates".into(),
323            ));
324        }
325        for (direction, (_, variable)) in directions.iter().zip(&self.sampled) {
326            for (index, parity) in phases.values_mut().enumerate() {
327                if direction.bit(index) {
328                    *parity = &*parity + &Parity::single(*variable);
329                }
330            }
331        }
332        Ok(phases)
333    }

339    pub(super) fn resolve(
340        &self,
341        actions: &[ActionNode],
342        selectives: &FxHashMap<IVec3, PauliBasis>,
343    ) -> Result<ResolvedCorrection, VerifyLogicalError> {
344        let mut basis = self.initial_basis.clone();
345        let mut keys = FxHashMap::default();
346        let mut resolves = actions
347            .iter()
348            .filter_map(|action| match action.action {
349                Action::Resolve { target, .. } => Some((target.z, action.ordinal, target)),
350                _ => None,
351            })
352            .collect::<Vec<_>>();
353        resolves.sort_unstable_by_key(|&(z, ordinal, _)| (z, ordinal));
354        for (_, _, target) in resolves {
355            let (updated, key) = replay_fill(&basis, target, selectives[&target])?;
356            basis = updated;
357            keys.insert(target, key);
358        }
359        // A fill moves a selective record onto its chosen native key.
360        let records = self
361            .records
362            .iter()
363            .map(|(name, key)| {
364                let key = match key {
365                    MeasurementKey::Node { pos, .. } => keys.get(pos).unwrap_or(key),
366                    MeasurementKey::Edge { .. } => key,
367                };
368                (name.clone(), key.clone())
369            })
370            .collect();
371        let readouts = named_readouts(&basis, &records)?
372            .into_values()
373            .collect::<Vec<_>>();
374        let phases = self.resolved_measurement_vars(&keys, &readouts)?;
375        Ok(ResolvedCorrection { basis, phases })
376    }

378    pub(super) fn apply(
379        &self,
380        graph: &mut QuizxGraph,
381        outputs: &[IVec3],
382        values: &FxHashMap<Var, bool>,
383        resolved: &ResolvedCorrection,
384        actions: &[ActionNode],
385        feedback_values: &FxHashMap<usize, bool>,
386    ) -> Result<(), VerifyLogicalError> {
387        let ResolvedCorrection {
388            basis,
389            phases: measurement_vars,
390        } = resolved;
391        let measured = measurement_vars
392            .iter()
393            .map(|(key, parity)| {
394                (
395                    key.clone(),
396                    phase_value(parity, |variable| values[&variable]),
397                )
398            })
399            .collect::<BTreeMap<_, _>>();
400        let sampled = self.sampled_vars().collect::<BTreeSet<_>>();
401        let reference = measurement_vars
402            .iter()
403            .map(|(key, parity)| {
404                let value = phase_value(parity, |variable| {
405                    !sampled.contains(&variable) && values[&variable]
406                });
407                (key.clone(), value)
408            })
409            .collect::<BTreeMap<_, _>>();
410
411        let Some(reference) =
412            self.reference_outcomes(basis, measurement_vars, reference, actions, feedback_values)
413        else {
414            *graph.scalar_mut() = FScalar::zero();
415            return Ok(());
416        };
417        let (surfaces, correction) = solve_evolved_correction(basis, outputs, |surface| {
418            measured
419                .keys()
420                .any(|key| stabilizer_supports_measurement(surface, key))
421        })?;
422        let flips = surfaces
423            .iter()
424            .map(|surface| {
425                measured.iter().fold(false, |parity, (key, value)| {
426                    parity
427                        ^ ((*value ^ reference[key])
428                            && stabilizer_supports_measurement(surface, key))
429                })
430            })
431            .collect::<Vec<_>>();
432        match correction.evaluate(&flips) {
433            Ok(corrections) => insert_output_corrections(graph, outputs, &corrections),
434            Err(OutputCorrectionError::Inconsistent) => *graph.scalar_mut() = FScalar::zero(),
435            Err(
436                error @ (OutputCorrectionError::RhsLengthMismatch { .. }
437                | OutputCorrectionError::UnknownOutput { .. }),
438            ) => return Err(error.into()),
439        }
440        Ok(())
441    }

443    /// Correct within the authored named branch. Its all-zero unnamed outcome
444    /// assignment can be impossible: repeated merges must agree with the named
445    /// parity. Closed surfaces find a consistent reference with free bits zero.
446    fn reference_outcomes(
447        &self,
448        basis: &RuntimeStabilizerBasis,
449        measurement_vars: &BTreeMap<MeasurementKey, Parity>,
450        reference: BTreeMap<MeasurementKey, bool>,
451        actions: &[ActionNode],
452        feedback_values: &FxHashMap<usize, bool>,
453    ) -> Option<BTreeMap<MeasurementKey, bool>> {
454        let surfaces = basis.closed_stabilizer_surfaces();
455        let mut rhs = CoeffVec::zeros(surfaces.len());
456        for (index, surface) in surfaces.iter().enumerate() {
457            let parity = reference
458                .iter()
459                .fold(surface.sign, |parity, (key, &value)| {
460                    parity ^ (value && stabilizer_supports_measurement(surface, key))
461                })
462                ^ feedback_sign(surface, basis.zx_graph(), actions, feedback_values);
463            rhs.set_bit(index, parity);
464        }
465        // Each sampled variable includes its compensating native phases.
466        // Solving individual phase coordinates would change the named branch.
467        let columns = self.parity_columns(&surfaces, measurement_vars);
468        let solution = solve_coeff_combination(&columns, &rhs)?;
469        Some(self.apply_coordinate(measurement_vars, reference, &solution))
470    }
471
472    fn apply_coordinate(
473        &self,
474        phases: &BTreeMap<MeasurementKey, Parity>,
475        mut outcomes: BTreeMap<MeasurementKey, bool>,
476        coordinate: &CoeffVec,
477    ) -> BTreeMap<MeasurementKey, bool> {
478        for (key, value) in &mut outcomes {
479            let parity = &phases[key];
480            for (index, (_, variable)) in self.sampled.iter().enumerate() {
481                *value ^= coordinate.bit(index) && parity.iter().any(|value| value == *variable);
482            }
483        }
484        outcomes
485    }
486}

488fn kernel_directions(columns: &[CoeffVec]) -> Vec<CoeffVec> {
489    let mut basis = Vec::<CoeffVec>::new();
490    let mut selected = Vec::new();
491    let mut kernel = Vec::new();
492    for (index, column) in columns.iter().enumerate() {
493        if let Some(combination) = solve_coeff_combination(&basis, column) {
494            let mut direction = CoeffVec::singleton(index, columns.len());
495            for bit in combination.iter_ones() {
496                direction.set_bit(selected[bit], true);
497            }
498            kernel.push(direction);
499        } else {
500            selected.push(index);
501            basis.push(column.clone());
502        }
503    }
504    kernel
505}

507fn feedback_sign(
508    surface: &crate::Stabilizer,
509    zx: &ZXGraph,
510    actions: &[ActionNode],
511    values: &FxHashMap<usize, bool>,
512) -> bool {
513    actions
514        .iter()
515        .filter(|action| values.get(&action.ordinal) == Some(&true))
516        .fold(false, |odd, action| {
517            let Action::Feedback { targets, .. } = &action.action else {
518                return odd;
519            };
520            targets.iter().fold(odd, |odd, target| {
521                let Some((column, pauli)) = zx.feedback_column(target) else {
522                    return odd;
523                };
524                odd ^ surface.paulis.get(column).anticommutes(pauli)
525            })
526        })
527}

529pub(super) fn fresh_var(next: &mut usize) -> Result<Var, VerifyLogicalError> {
530    let variable = Var::try_from(*next).map_err(|_| VerifyLogicalError::TooManySymbols)?;
531    *next += 1;
532    Ok(variable)
533}

535fn measurement_keys(zx: &ZXGraph) -> Vec<MeasurementKey> {
536    let mut keys = BTreeSet::new();
537    for node in zx.nodes() {
538        let pauli = match node.kind {
539            NodeKind::X => Pauli::Z,
540            NodeKind::Y => Pauli::Y,
541            NodeKind::Z => Pauli::X,
542            NodeKind::Selective(kind) => Pauli::from(kind.pauli_if_false()),
543            _ => continue,
544        };
545        let continues = zx.neighbors(node.id).is_some_and(|neighbors| {
546            neighbors.iter().any(|&neighbor| {
547                zx.nodes()[neighbor].pos.z > node.pos.z || zx.nodes()[neighbor].is_output_port(zx)
548            })
549        });
550        if !continues {
551            keys.insert(MeasurementKey::Node {
552                pos: node.pos,
553                pauli,
554            });
555        }
556    }
557    for edge in zx.edges().iter().filter(|edge| edge.n1 < edge.n2) {
558        let lhs = zx.nodes()[edge.n1];
559        let rhs = zx.nodes()[edge.n2];
560        if lhs.pos.z != rhs.pos.z
561            || !matches!(lhs.kind, NodeKind::X | NodeKind::Z)
562            || !matches!(rhs.kind, NodeKind::X | NodeKind::Z)
563        {
564            continue;
565        }
566        if (lhs.kind == NodeKind::X) == ((rhs.kind == NodeKind::X) ^ edge.hadamard) {
567            keys.insert(MeasurementKey::Edge {
568                src: lhs.pos,
569                dst: rhs.pos,
570                pauli: if lhs.kind == NodeKind::X {
571                    Pauli::X
572                } else {
573                    Pauli::Z
574                },
575            });
576        }
577    }
578    keys.into_iter().collect()
579}

581fn insert_output_corrections(
582    graph: &mut QuizxGraph,
583    outputs: &[IVec3],
584    corrections: &[(IVec3, Pauli)],
585) {
586    for &(position, pauli) in corrections {
587        let output = outputs
588            .iter()
589            .position(|candidate| *candidate == position)
590            .expect("correction names a requested output");
591        let boundary = graph.outputs()[output];
592        let vertex_types: &[VType] = match pauli {
593            Pauli::I => &[],
594            Pauli::X => &[VType::X],
595            Pauli::Y => &[VType::Z, VType::X],
596            Pauli::Z => &[VType::Z],
597        };
598        for &vertex_type in vertex_types {
599            let (neighbor, edge_type) = graph
600                .incident_edges(boundary)
601                .next()
602                .expect("an output boundary has one wire");
603            graph.remove_edge(boundary, neighbor);
604            let correction = graph.add_vertex_with_phase(vertex_type, Phase::from(1_i64));
605            graph.add_edge(boundary, correction);
606            graph.add_edge_with_type(correction, neighbor, edge_type);
607        }
608    }

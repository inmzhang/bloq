14/// One selective-fill replay step: apply `chosen` at `pos` over `basis` (with
15/// the T-port fallback) and return the evolved basis together with the
16/// measurement key it records.
17pub(crate) fn replay_fill(
18    basis: &RuntimeStabilizerBasis,
19    pos: IVec3,
20    chosen: PauliBasis,
21) -> Result<(RuntimeStabilizerBasis, MeasurementKey), VerifyLogicalError> {
22    let basis = apply_selective_fill_with_fallback(basis, pos, chosen)?;
23    let key = MeasurementKey::Node {
24        pos,
25        pauli: Pauli::from(chosen),
26    };
27    Ok((basis, key))
28}

30/// Applies a selective fill at `pos` in the `chosen` basis, retrying with T
31/// nodes demoted to ports if the fill would otherwise be non-Clifford.
32fn apply_selective_fill_with_fallback(
33    basis: &RuntimeStabilizerBasis,
34    pos: IVec3,
35    chosen: PauliBasis,
36) -> Result<RuntimeStabilizerBasis, VerifyLogicalError> {
37    match basis.apply_selective_fill(pos, chosen) {
38        Ok(transition) => Ok(transition),
39        Err(RuntimeBasisError::NonCliffordFillUnsupported) => basis
40            .clone()
41            .with_t_nodes_as_ports()
42            .apply_selective_fill(pos, chosen)
43            .map_err(|err| VerifyLogicalError::RuntimeBasisUpdateFailed { pos, source: err }),
44        Err(err) => Err(VerifyLogicalError::RuntimeBasisUpdateFailed { pos, source: err }),
45    }
46}

48/// The evolved-basis terminal solve, retaining support on every measured key.
49/// Crossing-node predicates are evaluated after the final row combinations.
50/// Closed reference constraints retain their separate signed materialization.
51pub(crate) fn solve_evolved_correction(
52    basis: &RuntimeStabilizerBasis,
53    outputs: &[IVec3],
54    keys: &[&MeasurementKey],
55) -> Result<(Vec<CoeffVec>, SymbolicOutputCorrection), VerifyLogicalError> {
56    let (zx, rows) = basis
57        .clone()
58        .into_output_correction_rows(outputs)
59        .map_err(|_| VerifyLogicalError::OutputSurfaceUnavailable(outputs.to_vec()))?;
60    let components = keys
61        .iter()
62        .map(|key| key.components(&zx).unwrap_or_default())
63        .collect::<Vec<_>>();
64    let mut constraints = Vec::new();
65    let mut supports = Vec::new();
66    for (constraint, row) in rows {
67        let mut support = CoeffVec::zeros(keys.len());
68        if row.len() == zx.total_ids() && zx.row_has_external_edge_pair_support(&row) {
69            for (index, components) in components.iter().enumerate() {
70                support.set_bit(
71                    index,
72                    components
73                        .iter()
74                        .any(|&(column, axis)| row.get(column) & axis),
75                );
76            }
77        } else {
78            let surface = zx.materialize_stabilizer(row);
79            for (index, key) in keys.iter().enumerate() {
80                support.set_bit(index, stabilizer_supports_measurement(&surface, key));
81            }
82        }
83        if !support.is_zero() {
84            constraints.push(constraint);
85            supports.push(support);
86        }
87    }
88    let symbolic = solve_output_correction_symbolic(outputs, &constraints)?;
89    Ok((supports, symbolic))
90}

92#[cfg(test)]
93pub(crate) fn signed_output_correction(
94    basis: &RuntimeStabilizerBasis,
95    outputs: &[IVec3],
96    keys: &[&MeasurementKey],
97) -> (Vec<CoeffVec>, SymbolicOutputCorrection) {
98    let mut constraints = Vec::new();
99    let mut masks = Vec::new();
100    for (constraint, surface) in basis.derive_output_correction_surfaces(outputs).unwrap() {
101        let mut mask = CoeffVec::zeros(keys.len());
102        for (index, key) in keys.iter().enumerate() {
103            mask.set_bit(index, stabilizer_supports_measurement(&surface, key));
104        }
105        if !mask.is_zero() {
106            constraints.push(constraint);
107            masks.push(mask);
108        }
109    }
110    (
111        masks,
112        solve_output_correction_symbolic(outputs, &constraints).unwrap(),
113    )
114}

120    fn all_keys(zx: &crate::ZXGraph) -> Vec<MeasurementKey> {
121        let mut keys = Vec::new();
122        for node in zx.nodes() {
123            for pauli in [Pauli::X, Pauli::Z, Pauli::Y] {
124                keys.push(MeasurementKey::Node {
125                    pos: node.pos,
126                    pauli,
127                });
128            }
129        }
130        for edge in zx.edges().iter().filter(|edge| edge.n1 < edge.n2) {
131            let left = zx.nodes()[edge.n1].pos;
132            let right = zx.nodes()[edge.n2].pos;
133            for (src, dst) in [(left, right), (right, left)] {
134                for pauli in [Pauli::X, Pauli::Z, Pauli::Y] {
135                    keys.push(MeasurementKey::Edge { src, dst, pauli });
136                }
137            }
138        }
139        let absent = glam::ivec3(i32::MAX, 42, 17);
140        assert!(zx.node_at(absent).is_none());
141        keys.push(MeasurementKey::Node {
142            pos: absent,
143            pauli: Pauli::Y,
144        });
145        keys.push(MeasurementKey::Edge {
146            src: absent,
147            dst: glam::ivec3(i32::MAX - 1, 42, 17),
148            pauli: Pauli::Y,
149        });
150        if let Some(first) = zx.nodes().first()
151            && let Some(other) = zx
152                .nodes()
153                .iter()
154                .find(|other| other.id != first.id && zx.edge_id(first.id, other.id).is_none())
155        {
156            keys.push(MeasurementKey::Edge {
157                src: first.pos,
158                dst: other.pos,
159                pauli: Pauli::X,
160            });
161        }
162        if let Some(first) = keys.first().cloned() {
163            keys.push(first); // The supplied key order may contain duplicates.
164        }
165        keys
166    }

168    fn assert_matches_signed(
169        basis: &RuntimeStabilizerBasis,
170        outputs: &[IVec3],
171        keys: &[MeasurementKey],
172    ) -> usize {
173        let refs = keys.iter().collect::<Vec<_>>();
174        let (expected_masks, expected) = signed_output_correction(basis, outputs, &refs);
175        let (actual_masks, actual) = solve_evolved_correction(basis, outputs, &refs).unwrap();
176        assert_eq!(
177            actual_masks, expected_masks,
178            "ordered physical support masks"
179        );
180        assert_eq!(
181            actual, expected,
182            "exact symbolic frames and consistency equations"
183        );
184        actual_masks.len()
185    }

187    #[test]
188    fn output_queries_match_signed_gallery_surfaces_and_key_order() {
189        for gallery in [
190            crate::GalleryItem::CNOT,
191            crate::GalleryItem::CZSpatialH,
192            crate::GalleryItem::CZTemporalH,
193            crate::GalleryItem::S,
194            crate::GalleryItem::T,
195            crate::GalleryItem::TWithPreparedY,
196            crate::GalleryItem::THTH,
197            crate::GalleryItem::YMemory,
198            crate::GalleryItem::Stability,
199        ] {
200            let graph = gallery.build().materialize_root_graph().unwrap();
201            let zx = crate::ZXGraph::from_block_graph_for_analysis(&graph).unwrap();
202            let basis = RuntimeStabilizerBasis::from_zx_graph(zx)
203                .unwrap()
204                .with_t_nodes_as_ports();
205            let outputs = basis.zx_graph().output_ports();
206            let mut keys = all_keys(basis.zx_graph());
207            assert_matches_signed(&basis, &outputs, &keys);
208            keys.reverse();
209            assert_matches_signed(&basis, &outputs[..outputs.len().min(1)], &keys);
210            assert_eq!(assert_matches_signed(&basis, &outputs, &[]), 0);
211        }
212    }

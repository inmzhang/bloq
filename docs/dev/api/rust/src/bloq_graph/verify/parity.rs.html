10pub(super) fn phase_constant(parity: &Parity) -> bool {
11    // QuiZX exposes the variable iterator but no getter for the affine bit.
12    parity != &Parity::new(parity.iter().collect::<Vec<_>>(), false)
13}
14
15pub(super) fn phase_value(parity: &Parity, mut value: impl FnMut(Var) -> bool) -> bool {
16    parity.iter().fold(phase_constant(parity), |odd, variable| {
17        odd ^ value(variable)
18    })
19}

36/// Identifies a measurement record by the ZX site it reads and its Pauli basis.
37#[derive(Debug, Clone, PartialEq, Eq, Hash)]
38pub(crate) enum MeasurementKey {
39    /// A single-node measurement at `pos` in the `pauli` basis.
40    Node { pos: IVec3, pauli: Pauli },
41    /// An edge measurement between `src` and `dst`; `pauli` is in the smaller
42    /// node-ID endpoint's frame, so an H edge may use its flipped basis at `dst`.
43    Edge {
44        src: IVec3,
45        dst: IVec3,
46        pauli: Pauli,
47    },
48}
49
50impl PartialOrd for MeasurementKey {
51    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
52        Some(self.cmp(other))
53    }
54}
55
56impl Ord for MeasurementKey {
57    fn cmp(&self, other: &Self) -> Ordering {
58        self.sort_key().cmp(&other.sort_key())
59    }
60}
61
62impl MeasurementKey {
63    pub(super) fn for_action(zx: &ZXGraph, action: &ActionNode) -> Option<Self> {
64        let Action::Measure { target, .. } = action.action else {
65            return None;
66        };
67        let pauli = match action.measurement? {
68            MeasurementObservable::Concrete(basis) => Pauli::from(basis),
69            MeasurementObservable::Selective(_) => Pauli::Y,
70        };
71        Some(match target {
72            MeasureTarget::Node(pos) => Self::Node { pos, pauli },
73            MeasureTarget::Edge { src, dir } => {
74                let dst = src + dir.to_ivec3();
75                let (src, dst) = if zx.node_at(src)?.id < zx.node_at(dst)?.id {
76                    (src, dst)
77                } else {
78                    (dst, src)
79                };
80                Self::Edge { src, dst, pauli }
81            }
82        })
83    }
84
85    fn sort_key(&self) -> (u8, [i32; 3], [i32; 3], u8) {
86        match *self {
87            Self::Node { pos, pauli } => (0, pos.to_array(), [0; 3], u8::from(pauli)),
88            Self::Edge { src, dst, pauli } => (1, src.to_array(), dst.to_array(), u8::from(pauli)),
89        }
90    }
91}

93/// Whether a stabilizer's interior support covers a measurement key — the
94/// support predicate used by internal-measurement output correction.
95pub(crate) fn stabilizer_supports_measurement(
96    stabilizer: &Stabilizer,
97    key: &MeasurementKey,
98) -> bool {
99    match key {
100        MeasurementKey::Node { pos, pauli } => stabilizer
101            .interior_nodes
102            .get(pos)
103            .is_some_and(|support| *support & *pauli),
104        MeasurementKey::Edge { src, dst, pauli } => stabilizer
105            .interior_edges
106            .get(&(*src, *dst))
107            .or_else(|| stabilizer.interior_edges.get(&(*dst, *src)))
108            .is_some_and(|support| *support & *pauli),
109    }
110}

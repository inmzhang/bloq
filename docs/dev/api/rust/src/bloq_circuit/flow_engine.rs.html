19/// Measurements carried by an open or completed flow.
20///
21/// Detector flows almost always carry one or two measurements (the current
22/// round's outcome plus the consumed previous round's), so they stay inline
23/// and flow tracking does not touch the heap per flow. The inline capacity
24/// deliberately matches [`DetectorParity`]'s term buffer so completed flows
25/// convert into parities by moving the buffer.
26pub type FlowEngineMeasurements<M> = SmallVec<[M; 2]>;
27
28/// A matched flow ready to emit as a detector.
29#[derive(Debug, Clone, PartialEq, Eq)]
30pub struct CompletedFlow<M> {
31    /// Every measurement fused along the chain.
32    pub measurements: FlowEngineMeasurements<M>,
33    /// XOR of the signs on every fused flow.
34    pub sign: bool,
35    /// The detector's coordinates, if the chain carries one.
36    pub center: Option<IVec2>,
37    /// Fused marker over every flow in this chain. See [`FlowMarker`]: a
38    /// `Discard` chain emits no detector; a `Restart` chain emits a restart
39    /// parity instead.
40    pub marker: FlowMarker,
41}

43/// A completed flow classified for detector synthesis.
44///
45/// Classification uses the fused [`FlowMarker`]. Produced by
46/// [`FlowEngine::drain_composed_chains`]; a
47/// `Discard`-marked or empty-parity chain is dropped and never appears here.
48#[derive(Debug, Clone, PartialEq, Eq)]
49pub enum ComposedChain<M> {
50    /// An ordinary detector at the chain's center coordinates.
51    Detector {
52        /// The detector's parity.
53        parity: DetectorParity<M>,
54        /// The detector's coordinates, if known.
55        center: Option<IVec2>,
56    },
57    /// A `RepeatUntilSuccess` restart syndrome (fused marker `Restart`); no
58    /// coordinates, as restart parities are never decoder food.
59    Restart {
60        /// The restart post-selection syndrome.
61        parity: DetectorParity<M>,
62    },
63}

65/// A boundary flow left open after residual composition.
66///
67/// [`FlowEngine::into_residual_flows`] returns one fused chain spliced from
68/// `start` (its origin boundary, empty for an internally-created chain) to `end`
69/// (the boundary it leaves open, empty if it closed against a boundary input),
70/// carrying every measurement composed along the way.
71#[derive(Debug, Clone, PartialEq, Eq)]
72pub struct ResidualFlow<M> {
73    /// The chain's origin boundary; empty for an internally-created chain.
74    pub start: PauliMap,
75    /// The boundary the chain leaves open; empty if it closed against an input.
76    pub end: PauliMap,
77    /// Every measurement composed along the chain.
78    pub measurements: FlowEngineMeasurements<M>,
79    /// XOR of the signs on every fused flow.
80    pub sign: bool,
81    /// The detector's coordinates, if any.
82    pub center: Option<IVec2>,
83    /// Carries the fused [`FlowMarker`] onto the template's `boundary_flows` so
84    /// the instance-seam pass can suppress detectors (`Discard`) or route the
85    /// cross-instance chain to a restart parity (`Restart`).
86    pub marker: FlowMarker,
87}

125/// A borrowed, translated open-flow boundary key.
126///
127/// The key holds a [`PauliMap`] plus its placement offset. Hashing and equality
128/// act on the translated entries, so equal
129/// absolute boundaries match regardless of which (map, offset) pair
130/// produced them, and matching never materializes a translated map. The
131/// translated hash is computed once at construction: table growth re-hashes
132/// every stored key, so a recomputed hash would re-translate every entry on
133/// each rehash.
134#[derive(Debug, Clone, Copy)]
135pub struct FlowKey<'c> {
136    map: &'c PauliMap,
137    offset: IVec2,
138    hash: u64,
139}
140
141impl<'c> FlowKey<'c> {
142    /// Borrow a boundary at a checked placement without allocating a translated map.
143    ///
144    /// # Errors
145    ///
146    /// Returns [`crate::CoordinateOverflowError`] if a translated coordinate
147    /// exceeds the signed 32-bit lattice.
148    pub fn try_new(
149        map: &'c PauliMap,
150        offset: IVec2,
151    ) -> Result<Self, crate::CoordinateOverflowError> {
152        for (coord, _) in map {
153            checked_translate_coordinate(*coord, offset)?;
154        }
155        Ok(Self::new(map, offset))
156    }
157
158    fn new(map: &'c PauliMap, offset: IVec2) -> Self {
159        use std::hash::Hasher as _;
160        let mut hasher = rustc_hash::FxHasher::default();
161        map.hash_translated(offset, &mut hasher);
162        Self {
163            map,
164            offset,
165            hash: hasher.finish(),
166        }
167    }
168
169    fn translated(&self) -> PauliMap {
170        self.map.translated(self.offset)
171    }
172}

174impl std::hash::Hash for FlowKey<'_> {
175    fn hash<H: std::hash::Hasher>(&self, state: &mut H) {
176        state.write_u64(self.hash);
177    }

180impl PartialEq for FlowKey<'_> {
181    fn eq(&self, other: &Self) -> bool {
182        if self.hash != other.hash {
183            return false;
184        }
185        if self.offset.to_array() == other.offset.to_array() {
186            return self.map == other.map;
187        }
188        self.map.len() == other.map.len()
189            && self.map.iter().zip(other.map.iter()).all(
190                |((coord, pauli), (other_coord, other_pauli))| {
191                    *coord + self.offset == *other_coord + other.offset && pauli == other_pauli
192                },
193            )
194    }
195}

205/// Tracks open flows and completed detectors while chunks are appended.
206///
207/// Open-flow keys borrow the boundary [`PauliMap`]s of the appended chunks
208/// (lifetime `'c`), so matching flows across chunks never clones a key. This
209/// works because every appended flow outlives the engine; the engine is a
210/// per-pass scratch structure, not a long-lived store.
211#[must_use]
212#[derive(Clone)]
213pub struct FlowEngine<'c, M> {
214    open_flows: FxHashMap<FlowKey<'c>, OpenFlow<M>>,
215    // Each group drains these payloads before publishing outputs. Keep the
216    // empty buffer for the next group instead of allocating it every round.
217    matched_inputs: Vec<Option<OpenFlow<M>>>,
218    /// Known stabilizers that commute through a temporary gauge basis but are
219    /// not themselves measured by it. They rejoin the active span when the
220    /// original basis is restored.
221    latent_flows: Vec<(PauliMap, OpenFlow<M>)>,
222    completed: Vec<CompletedFlow<M>>,
223    /// Boundary chains that closed against a boundary *input* (a flow that
224    /// entered at an unmatched `start` and later reached an empty `end`).
225    /// Populated only by [`UnmatchedInputMode::KeepAsResidual`]; empty under the
226    /// other modes, so it costs nothing when residual extraction is unused.
227    residual_completed: Vec<ResidualFlow<M>>,
228}

280    /// Append a boundary group while allowing its stabilizer span to differ
281    /// from the currently-open span. Exact matching remains the fast path. At
282    /// a gauge transition, products in the two spans' intersection emit
283    /// detectors; old-only generators retire and the group's ordinary creator
284    /// flows start its new-only generators.
285    ///
286    /// # Errors
287    ///
288    /// Returns [`FlowError`] if boundaries cannot be translated or composed.
289    pub fn append_group_allowing_stabilizer_transition(
290        &mut self,
291        parts: &[OffsetFlows<'c>],
292        mut meas_mapper: impl FnMut(usize, &u32) -> M,
293    ) -> Result<(), FlowError>
294    where
295        M: Copy,
296    {
297        let mut consumer_count = 0;
298        for part in parts {
299            for flow in part.flows.iter().filter(|flow| !flow.start.is_empty()) {
300                for (coord, _) in &flow.start {
301                    checked_translate_coordinate(*coord, part.offset)?;
302                }
303                consumer_count += 1;
304            }
305        }
306        if !self.latent_flows.is_empty() {
307            return self.append_stabilizer_transition(parts, &mut meas_mapper);
308        }
309
310        // Exact matching is overwhelmingly the common case. Remove each input
311        // once here and pass its payload into fusion, instead of probing it with
312        // `contains_key` and then hashing the same boundary again in
313        // `process_flow`. On a miss, restore the table before taking the
314        // stabilizer-transition path.
315        let consumer_keys = || {
316            parts.iter().flat_map(|part| {
317                part.flows
318                    .iter()
319                    .filter(|flow| !flow.start.is_empty())
320                    .map(|flow| FlowKey::new(&flow.start, part.offset))
321            })
322        };
323        let mut matched_inputs = std::mem::take(&mut self.matched_inputs);
324        matched_inputs.reserve(consumer_count);
325        for target in consumer_keys() {
326            let Some(open) = self.open_flows.remove(&target) else {
327                let duplicate_consumer = consumer_keys()
328                    .take(matched_inputs.len())
329                    .any(|seen| seen == target);
330                for (key, open) in consumer_keys().zip(matched_inputs.into_iter().flatten()) {
331                    self.open_flows.insert(key, open);
332                }
333                if !duplicate_consumer {
334                    return self.append_stabilizer_transition(parts, &mut meas_mapper);
335                }
336                // Let the ordinary path report its usual unmatched-input error.
337                return self.append_group_with_mode(
338                    parts,
339                    meas_mapper,
340                    UnmatchedInputMode::Error,
341                    None,
342                );
343            };
344            matched_inputs.push(Some(open));
345        }
346        self.append_group_with_mode(
347            parts,
348            meas_mapper,
349            UnmatchedInputMode::Error,
350            Some(matched_inputs),
351        )
352    }

371    /// [`Self::append_group_skipping_unmatched_inputs`], but a flow whose `start`

390    fn append_group_with_mode(
391        &mut self,
392        parts: &[OffsetFlows<'c>],
393        mut meas_mapper: impl FnMut(usize, &u32) -> M,
394        unmatched_input_mode: UnmatchedInputMode,
395        matched_inputs: Option<Vec<Option<OpenFlow<M>>>>,
396    ) -> Result<(), FlowError> {
397        let mut inputs = match matched_inputs {
398            Some(inputs) => inputs,
399            None => {
400                let mut inputs = std::mem::take(&mut self.matched_inputs);
401                for part in parts {
402                    for flow in part.flows.iter().filter(|flow| !flow.start.is_empty()) {
403                        for (coord, _) in &flow.start {
404                            checked_translate_coordinate(*coord, part.offset)?;
405                        }
406                        inputs.push(
407                            self.open_flows
408                                .remove(&FlowKey::new(&flow.start, part.offset)),
409                        );
410                    }
411                }
412                inputs
413            }
414        };
415        let mut matched_inputs = inputs.drain(..);
416        // Only non-empty outputs occupy the table. Consumer-only detector
417        // flows can outnumber creators, and need no hash-table capacity.
418        self.open_flows.reserve(
419            parts
420                .iter()
421                .flat_map(|part| part.flows)
422                .filter(|flow| !flow.end.is_empty())
423                .count(),
424        );
425
426        // Consumers must see previous open flows before creators can replace
427        // keys, across all of the chunk's member slices.
428        for (part_index, part) in parts.iter().enumerate() {
429            for flow in part.flows {
430                if !flow.start.is_empty() {
431                    let previous = matched_inputs
432                        .next()
433                        .expect("pre-matched inputs cover every consumer");
434                    self.process_flow(
435                        flow,
436                        part.offset,
437                        part_index,
438                        &mut meas_mapper,
439                        unmatched_input_mode,
440                        previous,
441                    )?;
442                }
443            }
444        }
445        debug_assert!(
446            matched_inputs.as_slice().is_empty(),
447            "every pre-matched input is consumed"
448        );
449        drop(matched_inputs);
450        self.matched_inputs = inputs;
451        for (part_index, part) in parts.iter().enumerate() {
452            for flow in part.flows {
453                if flow.start.is_empty() {
454                    self.process_flow(
455                        flow,
456                        part.offset,
457                        part_index,
458                        &mut meas_mapper,
459                        unmatched_input_mode,
460                        None,
461                    )?;
462                }
463            }
464        }
465
466        Ok(())
467    }

475    /// Consume the engine after a residual composition (chunks appended with
476    /// [`Self::append_residual_group`]) and return every boundary flow as a
477    /// fused chain: the chains still open (`origin_start → open key`) plus those
478    /// that closed against a boundary input (`origin_start → ∅`). Chains that
479    /// closed with an empty origin are internal detectors; they collect in
480    /// `completed` (drainable via [`Self::drain_completed`]) and are excluded here.
481    ///
482    /// The caller owns ordering: `open_flows` iterates in hash order, so a caller
483    /// that needs determinism (e.g. reproducible detector emission) must sort the
484    /// result by a canonical key.
485    pub fn into_residual_flows(self) -> Vec<ResidualFlow<M>> {
486        let mut flows = self.residual_completed;
487        flows.extend(
488            self.open_flows
489                .into_iter()
490                .map(|(key, open)| (key.translated(), open))
491                .chain(self.latent_flows)
492                .map(|(end, open)| ResidualFlow {
493                    start: open
494                        .origin_start
495                        .map_or_else(PauliMap::empty, |start| *start),
496                    end,
497                    measurements: open.measurements,
498                    sign: open.sign,
499                    center: open.center,
500                    marker: open.marker,
501                }),
502        );
503        flows
504    }

506    fn process_flow(
507        &mut self,
508        flow: &'c Flow,
509        offset: IVec2,
510        part_index: usize,
511        meas_mapper: &mut impl FnMut(usize, &u32) -> M,
512        unmatched_input_mode: UnmatchedInputMode,
513        previous: Option<OpenFlow<M>>,
514    ) -> Result<(), FlowError> {
515        // Flow keys translate coordinates inside Hash/Eq. Check before a key
516        // reaches either trait so malformed API/binary input returns an error
517        // instead of debug-panicking or release-wrapping.
518        if offset != IVec2::ZERO {
519            for (coord, _) in flow.start.iter().chain(flow.end.iter()) {
520                checked_translate_coordinate(*coord, offset)?;
521            }
522            if let Some(center) = flow.center {
523                checked_translate_coordinate(center, offset)?;
524            }
525        }
526
527        let mut measurements: FlowEngineMeasurements<M> =
528            SmallVec::with_capacity(flow.measurements.len());
529        for measurement in &flow.measurements {
530            measurements.push(meas_mapper(part_index, measurement));
531        }
532
533        let mut center = flow.center.map(|center| center + offset);
534        let mut sign = flow.sign;
535        let mut marker = flow.marker;
536        // The chain's origin boundary, carried so a residual composition can
537        // report where each open chain began. Empty unless this flow enters at
538        // an unmatched boundary `start` (see `KeepAsResidual` below) or inherits
539        // a non-empty origin from the chain it consumes.
540        let mut origin_start = None;
541
542        if !flow.start.is_empty() {
543            // Inputs were removed before any output of this group was published.
544            match previous {
545                Some(prev) => {
546                    // The chunk's own measurements come first, then the
547                    // carried history.
548                    measurements.extend(prev.measurements);
549                    sign ^= prev.sign;
550                    center = center.or(prev.center);
551                    origin_start = prev.origin_start;
552                    // Two flows fusing into one chain can mix discard and
553                    // restart; downstream routing checks `Discard` first, so a
554                    // fall-through would silently drop the restart parity. This
555                    // is a topology-dependent conflict, so reject rather than panic.
556                    marker = marker.fuse(prev.marker).ok_or_else(|| {
557                        FlowError::Composition(
558                            "fused chain mixes discard and restart components".into(),
559                        )
560                    })?;
561                }
562                None => match unmatched_input_mode {
563                    UnmatchedInputMode::Skip => return Ok(()),
564                    // A boundary input: keep the chain open, recording where it
565                    // entered so the residual can close it against a neighbor.
566                    UnmatchedInputMode::KeepAsResidual => {
567                        origin_start = Some(Box::new(flow.start.translated(offset)));
568                    }
569                    UnmatchedInputMode::Error => {
570                        return Err(FlowError::Composition(format!(
571                            "unmatched flow input {}",
572                            flow.start.translated(offset)
573                        )));
574                    }
575                },
576            }
577        }
578
579        if !flow.end.is_empty() {
580            // Flow continues: store for future matching
581            let end_key = FlowKey::new(&flow.end, offset);
582            let Entry::Vacant(entry) = self.open_flows.entry(end_key) else {
583                return Err(FlowError::Composition(format!(
584                    "multiple flows create boundary {end_key}"
585                )));
586            };
587            entry.insert(OpenFlow {
588                measurements,
589                sign,
590                center,
591                origin_start,
592                marker,
593            });
594        } else if !measurements.is_empty() || sign || marker == FlowMarker::Discard {
595            // A chain reached an empty end. If it began at a boundary input
596            // (non-empty origin) it is a residual flow that closes against a
597            // neighbor; otherwise it closed entirely within this composition (an
598            // internal detector). Non-residual modes never set a non-empty
599            // origin, so they always take the `completed` branch.
600            //
601            // A measurement-free chain is normally dropped (it can never carry a
602            // detector parity). A *discarding* one is kept: it exists precisely to
603            // consume an incoming seam stabilizer with no reconstruction of its
604            // own (a selective block's opposite-basis stabilizer), so it must
605            // survive onto the residual to match that seam and suppress its
606            // detector. `discard` is always false in the non-selective paths.
607            if let Some(origin_start) = origin_start {
608                self.residual_completed.push(ResidualFlow {
609                    start: *origin_start,
610                    end: PauliMap::empty(),
611                    measurements,
612                    sign,
613                    center,
614                    marker,
615                });
616            } else {
617                self.completed.push(CompletedFlow {
618                    measurements,
619                    sign,
620                    center,
621                    marker,
622                });
623            }
624        }
625
626        Ok(())
627    }

629    fn append_stabilizer_transition(
630        &mut self,
631        parts: &[OffsetFlows<'c>],
632        meas_mapper: &mut impl FnMut(usize, &u32) -> M,
633    ) -> Result<(), FlowError>
634    where
635        M: Copy,
636    {
637        let mut incoming = Vec::new();
638        for (part_index, part) in parts.iter().enumerate() {
639            for flow in part.flows.iter().filter(|flow| !flow.start.is_empty()) {
640                if !flow.end.is_empty() {
641                    return Err(FlowError::Composition(
642                        "stabilizer-span transitions require boundary consumer flows".into(),
643                    ));
644                }
645                let mut payload = empty_open_flow();
646                payload.measurements.extend(
647                    flow.measurements
648                        .iter()
649                        .map(|measurement| meas_mapper(part_index, measurement)),
650                );
651                payload.sign = flow.sign;
652                payload.center = flow
653                    .center
654                    .map(|center| checked_translate_coordinate(center, part.offset))
655                    .transpose()?;
656                payload.marker = flow.marker;
657                incoming.push((flow.start.translated(part.offset), payload));
658            }
659        }
660
661        let mut old: Vec<_> = std::mem::take(&mut self.open_flows)
662            .into_iter()
663            .map(|(key, flow)| (key.translated(), flow))
664            .chain(std::mem::take(&mut self.latent_flows))
665            .collect();
666        old.sort_by(|(a, _), (b, _)| pauli_map_cmp(a, b));
667        let old_len = old.len();
668        // Borrow the boundaries: reduction clones each one as it consumes it.
669        let generators: Vec<&PauliMap> = old
670            .iter()
671            .map(|(map, _)| map)
672            .chain(incoming.iter().map(|(start, _)| start))
673            .collect();
674        let dependencies = pauli_dependencies(&generators);
675        for (index, mask) in dependencies {
676            if index < old_len {
677                return Err(FlowError::Composition(
678                    "open flow boundaries are not independent generators".into(),
679                ));
680            }
681            if first_set_bit(&mask).is_none_or(|first| first >= old_len) {
682                return Err(FlowError::Composition(
683                    "incoming flow boundaries are not independent generators".into(),
684                ));
685            }
686            let mut completed = empty_open_flow();
687            let mut product = PauliMap::empty();
688            for (index, (map, flow)) in incoming.iter().enumerate() {
689                if mask[(index + old_len) / 64] & (1 << ((index + old_len) % 64)) != 0 {
690                    multiply_pauli_flow(&mut product, &mut completed, map, flow)?;
691                }
692            }
693            for (index, (map, flow)) in old.iter().enumerate() {
694                if mask[index / 64] & (1 << (index % 64)) != 0 {
695                    multiply_pauli_flow(&mut product, &mut completed, map, flow)?;
696                }
697            }
698            if !completed.measurements.is_empty()
699                || completed.sign
700                || completed.marker == FlowMarker::Discard
701            {
702                self.completed.push(CompletedFlow {
703                    measurements: completed.measurements,
704                    sign: completed.sign,
705                    center: completed.center,
706                    marker: completed.marker,
707                });
708            }
709        }
710
711        // Measuring a new stabilizer destroys one anticommuting old generator;
712        // products of the remaining anticommuting generators with that pivot
713        // survive. This is the usual stabilizer-tableau measurement update.
714        let mut commutations = pauli_commutation_columns(
715            old.iter().map(|(map, _)| map),
716            incoming.iter().map(|(map, _)| map),
717        );
718        for measured in 0..incoming.len() {
719            let mut targets = std::mem::take(&mut commutations[measured]);
720            let Some(pivot) = first_set_bit(&targets) else {
721                continue;
722            };
723            let last = old.len() - 1;
724            let (pivot_map, pivot_flow) = old.swap_remove(pivot);
725            swap_remove_bit(&mut targets, pivot, last);
726            for (word, &bits) in targets.iter().enumerate() {
727                let mut bits = bits;
728                while bits != 0 {
729                    let target = word * 64 + bits.trailing_zeros() as usize;
730                    let (known, flow) = &mut old[target];
731                    multiply_pauli_flow(known, flow, &pivot_map, &pivot_flow)?;
732                    bits &= bits - 1;
733                }
734            }
735            // Commutation is bilinear: multiplying by the pivot toggles exactly
736            // its anticommuting columns. Keep the original swap-remove order.
737            for column in &mut commutations[measured + 1..] {
738                let anticommutes = column[pivot / 64] & (1 << (pivot % 64)) != 0;
739                swap_remove_bit(column, pivot, last);
740                if anticommutes {
741                    xor_words(column, &targets);
742                }
743            }
744        }
745
746        // Keep only surviving old stabilizers outside the newly measured span.
747        // Exact active checks are represented by the creator flows below.
748        let mut basis = FxHashMap::default();
749        for (measured, flow) in &incoming {
750            let mut measured = measured.clone();
751            let mut flow = flow.clone();
752            reduce_pauli_flow(&mut measured, &mut flow, &basis)?;
753            let pivot = pauli_pivot(&measured)
754                .expect("incoming flow boundaries were checked as independent");
755            basis.insert(pivot, (measured, flow));
756        }
757        for (mut known, mut flow) in old {
758            reduce_pauli_flow(&mut known, &mut flow, &basis)?;
759            let Some(pivot) = pauli_pivot(&known) else {
760                continue;
761            };
762            basis.insert(pivot, (known.clone(), flow.clone()));
763            self.latent_flows.push((known, flow));
764        }
765
766        // The incoming consumers have now either contributed to an intersection
767        // detector or initialized a new-only gauge check. In both cases they do
768        // not remain open. The group's creator flows establish the new boundary
769        // basis for the next node.
770        for (part_index, part) in parts.iter().enumerate() {
771            for flow in part.flows.iter().filter(|flow| flow.start.is_empty()) {
772                self.process_flow(
773                    flow,
774                    part.offset,
775                    part_index,
776                    meas_mapper,
777                    UnmatchedInputMode::Error,
778                    None,
779                )?;
780            }
781        }
782        Ok(())
783    }
784}

788fn pauli_pivot(map: &PauliMap) -> Option<PauliPivot> {
789    map.iter().next_back().map(|(coord, pauli)| {
790        let component = if *pauli & crate::Pauli::Z { 2 } else { 1 };
791        (coord.x, coord.y, component)
792    })
793}
794
795// Row XOR can only introduce bits below its pivot. Retained higher bits
796// never need revisiting, so walk present bits instead of scanning every basis row.
797fn pauli_pivot_before(map: &PauliMap, before: PauliPivot) -> Option<PauliPivot> {
798    let maximum = pauli_pivot(map)?;
799    if maximum < before {
800        return Some(maximum);
801    }
802    let (x, y, component) = before;
803    let coord = IVec2::new(x, y);
804    if component == 2
805        && map
806            .get(&coord)
807            .is_some_and(|pauli| *pauli & crate::Pauli::X)
808    {
809        return Some((x, y, 1));
810    }
811    map.entry_before(coord).map(|(coord, pauli)| {
812        (
813            coord.x,
814            coord.y,
815            if pauli & crate::Pauli::Z { 2 } else { 1 },
816        )
817    })
818}

826                crate::Pauli::Z
827            } else {
828                crate::Pauli::X

833/// Columns name incoming measurements; each bit names an old generator.
834/// Construct only shared-coordinate pairs instead of testing every row pair.
835fn pauli_commutation_columns<'a>(
836    known: impl ExactSizeIterator<Item = &'a PauliMap>,
837    measured: impl Iterator<Item = &'a PauliMap>,
838) -> Vec<Vec<u64>> {
839    let words = known.len().div_ceil(64);
840    let mut sites: FxHashMap<IVec2, SmallVec<[(usize, crate::Pauli); 4]>> = FxHashMap::default();
841    for (index, map) in known.enumerate() {
842        for (&coord, &pauli) in map {
843            sites.entry(coord).or_default().push((index, pauli));
844        }
845    }
846    measured
847        .map(|map| {
848            let mut column = vec![0; words];
849            for (coord, &pauli) in map {
850                for &(index, other) in sites.get(coord).into_iter().flatten() {
851                    if pauli.anticommutes(other) {
852                        column[index / 64] ^= 1 << (index % 64);
853                    }
854                }
855            }
856            column
857        })
858        .collect()
859}

861fn swap_remove_bit(words: &mut [u64], index: usize, last: usize) {
862    let value = (words[last / 64] >> (last % 64)) & 1;
863    words[index / 64] = (words[index / 64] & !(1 << (index % 64))) | (value << (index % 64));
864    words[last / 64] &= !(1 << (last % 64));
865}

867fn reduce_pauli_flow<M: Copy>(
868    value: &mut PauliMap,
869    flow: &mut OpenFlow<M>,
870    basis: &FxHashMap<PauliPivot, (PauliMap, OpenFlow<M>)>,
871) -> Result<(), FlowError> {
872    let mut pivot = pauli_pivot(value);
873    while let Some(current) = pivot {
874        if let Some((row, row_flow)) = basis.get(&current) {
875            multiply_pauli_flow(value, flow, row, row_flow)?;
876        }
877        pivot = pauli_pivot_before(value, current);
878    }
879    Ok(())
880}

882fn multiply_pauli_flow<M: Copy>(
883    target: &mut PauliMap,
884    flow: &mut OpenFlow<M>,
885    source: &PauliMap,
886    source_flow: &OpenFlow<M>,
887) -> Result<(), FlowError> {
888    let phase = target.product_phase(source);
889    if !phase.is_multiple_of(2) {
890        return Err(FlowError::Composition(
891            "stabilizer-span products require commuting boundaries".into(),
892        ));
893    }
894    flow.sign ^= phase == 2;
895    *target = &*target ^ source;
896    xor_open_flow(flow, source_flow)
897}

919fn pauli_dependencies(generators: &[&PauliMap]) -> Vec<(usize, Vec<u64>)> {
920    let mut basis: FxHashMap<PauliPivot, (PauliMap, Vec<u64>)> = FxHashMap::default();
921    let mut dependencies = Vec::new();
922    for (index, generator) in generators.iter().enumerate() {
923        let mut value = (*generator).clone();
924        let mut mask = vec![0; generators.len().div_ceil(64)];
925        mask[index / 64] = 1 << (index % 64);
926        let mut pivot = pauli_pivot(&value);
927        while let Some(current) = pivot {
928            if let Some((row, row_mask)) = basis.get(&current) {
929                value = &value ^ row;
930                xor_words(&mut mask, row_mask);
931            }
932            pivot = pauli_pivot_before(&value, current);
933        }
934        if let Some(pivot) = pauli_pivot(&value) {
935            basis.insert(pivot, (value, mask));
936        } else {
937            dependencies.push((index, mask));
938        }
939    }
940    dependencies
941}

943#[cfg(test)]
944fn eager_pauli_dependencies(generators: &[&PauliMap]) -> Vec<(usize, Vec<bool>)> {
945    let mut basis: Vec<(PauliPivot, PauliMap, Vec<bool>)> = Vec::new();
946    let mut dependencies = Vec::new();
947    for (index, generator) in generators.iter().enumerate() {
948        let mut value = (*generator).clone();
949        let mut mask = vec![false; generators.len()];
950        mask[index] = true;
951        let start = pauli_pivot(&value).map_or(basis.len(), |maximum| {
952            basis.partition_point(|(pivot, _, _)| *pivot > maximum)
953        });
954        for (pivot, row, row_mask) in &basis[start..] {
955            if has_pauli_pivot(&value, *pivot) {
956                value = &value ^ row;
957                xor_bits(&mut mask, row_mask);
958                if value.is_empty() {
959                    break;
960                }
961            }
962        }
963        if let Some(pivot) = pauli_pivot(&value) {
964            let index = basis.partition_point(|(existing, _, _)| *existing > pivot);
965            basis.insert(index, (pivot, value, mask));
966        } else {
967            dependencies.push((index, mask));
968        }
969    }
970    dependencies
971}

983fn xor_open_flow<M: Copy>(target: &mut OpenFlow<M>, source: &OpenFlow<M>) -> Result<(), FlowError> {
984    target.measurements.extend_from_slice(&source.measurements);
985    target.sign ^= source.sign;
986    target.center = target.center.or(source.center);
987    target.marker = target.marker.fuse(source.marker).ok_or_else(|| {
988        FlowError::Composition(
989            "stabilizer-span transition mixes discard and restart components".into(),
990        )
991    })?;
992    if let Some(source_origin) = source.origin_start.as_deref() {
993        target.origin_start = match target.origin_start.take() {
994            Some(mut target_origin) => {
995                let phase = target_origin.product_phase(source_origin);
996                if !phase.is_multiple_of(2) {
997                    return Err(FlowError::Composition(
998                        "stabilizer-span products require commuting origin boundaries".into(),
999                    ));
1000                }
1001                target.sign ^= phase == 2;
1002                *target_origin = &*target_origin ^ source_origin;
1003                (!target_origin.is_empty()).then_some(target_origin)
1004            }
1005            None => Some(Box::new(source_origin.clone())),
1006        };
1007    }
1008    Ok(())
1009}

1045/// Loop-carried detector states opened for a loop body, awaiting their `next`
1046/// parity once the body's closing flows are known.
1047///
1048/// Produced by [`FlowEngine::start_loop_detector_states`] and consumed by
1049/// [`FlowEngine::finish_loop_detector_states`].
1050#[derive(Debug, Clone)]
1051pub struct PendingLoopDetectorStates<'c, M = u32> {
1052    loop_states: Vec<(FlowKey<'c>, LoopCarriedDetectorState<M>)>,
1053}
1054
1055impl<'c, M: Copy + Ord> FlowEngine<'c, DetectorTerm<M>> {
1056    /// Assign a loop-carried detector state to every open flow the loop body
1057    /// is about to consume: the open measurements become the state's initial
1058    /// parity and are replaced by a single loop-state term.
1059    ///
1060    /// Open flows are matched against the first body chunk's start keys
1061    /// directly; a valid body never consumes the same key twice (the append
1062    /// would fail), so each open flow gains at most one state.
1063    ///
1064    /// # Errors
1065    ///
1066    /// Returns [`FlowError`] if coordinates or the loop-state id space overflow.
1067    ///
1068    /// # Panics
1069    ///
1070    /// Panics if the preflighted loop-state range is internally inconsistent.
1071    pub fn start_loop_detector_states(
1072        &mut self,
1073        body_first_chunk: &[OffsetFlows<'c>],
1074        next_loop_state: &mut u32,
1075    ) -> Result<PendingLoopDetectorStates<'c, M>, FlowError> {
1076        for part in body_first_chunk
1077            .iter()
1078            .filter(|part| part.offset != IVec2::ZERO)
1079        {
1080            for flow in part.flows.iter().filter(|flow| !flow.start.is_empty()) {
1081                for (coordinate, _) in &flow.start {
1082                    checked_translate_coordinate(*coordinate, part.offset)?;
1083                }
1084            }
1085        }
1086        let possible_states =
1087            loop_state_upper_bound(body_first_chunk.iter().map(|part| part.flows.len()));
1088        if possible_states == usize::MAX || possible_states > (u32::MAX - *next_loop_state) as usize
1089        {
1090            self.check_loop_state_capacity(body_first_chunk, *next_loop_state)?;
1091        }
1092
1093        let mut loop_states = Vec::with_capacity(self.open_flows.len().min(possible_states));
1094        for part in body_first_chunk {
1095            for flow in part.flows {
1096                if flow.start.is_empty() {
1097                    continue;
1098                }
1099                let key = FlowKey::new(&flow.start, part.offset);
1100                let Some(open) = self.open_flows.get_mut(&key) else {
1101                    continue;
1102                };
1103                let state = LoopStateId(*next_loop_state);
1104                *next_loop_state = next_loop_state
1105                    .checked_add(1)
1106                    .expect("loop state range was preflighted");
1107                let initial = DetectorParity::from_term_buf(std::mem::replace(
1108                    &mut open.measurements,
1109                    smallvec::smallvec![DetectorTerm::LoopState(state)],
1110                ))
1111                .with_sign(std::mem::take(&mut open.sign));
1112                loop_states.push((
1113                    key,
1114                    LoopCarriedDetectorState {
1115                        state,
1116                        initial,
1117                        next: DetectorParity::default(),
1118                    },
1119                ));
1120            }
1121        }
1122        Ok(PendingLoopDetectorStates { loop_states })
1123    }
1124
1125    #[cold]
1126    fn check_loop_state_capacity(
1127        &self,
1128        body_first_chunk: &[OffsetFlows<'c>],
1129        next_loop_state: u32,
1130    ) -> Result<(), FlowError> {
1131        body_first_chunk
1132            .iter()
1133            .flat_map(|part| {
1134                part.flows.iter().filter(|flow| {
1135                    !flow.start.is_empty()
1136                        && self
1137                            .open_flows
1138                            .contains_key(&FlowKey::new(&flow.start, part.offset))
1139                })
1140            })
1141            .try_fold(next_loop_state, |next, _| next.checked_add(1))
1142            .ok_or_else(|| {
1143                FlowError::Composition("loop detector state id space exhausted".into())
1144            })?;
1145        Ok(())
1146    }
1147
1148    /// Drain completed chains, classifying each by its fused [`FlowMarker`] into
1149    /// a [`ComposedChain`]: a `Discard`-marked or empty-parity chain produces
1150    /// nothing, a `Restart` chain becomes a restart syndrome, everything else a
1151    /// detector at the chain's center. This is the shared marker discipline for
1152    /// detector synthesis — the compile-time boundary pass
1153    /// (`bloq_compile::lower::detector`) and the seam recompose
1154    /// (`bloq_ir::edit`) both drain through here so the two cannot drift.
1155    pub fn drain_composed_chains(&mut self) -> impl Iterator<Item = ComposedChain<M>> + '_ {
1156        self.completed.drain(..).filter_map(|completed| {
1157            if completed.marker == FlowMarker::Discard {
1158                return None;
1159            }
1160            let parity =
1161                DetectorParity::from_term_buf(completed.measurements).with_sign(completed.sign);
1162            if parity.is_empty() {
1163                return None;
1164            }
1165            Some(if completed.marker == FlowMarker::Restart {
1166                ComposedChain::Restart { parity }
1167            } else {
1168                ComposedChain::Detector {
1169                    parity,
1170                    center: completed.center,
1171                }
1172            })
1173        })
1174    }
1175
1176    /// Completes the pending loop states: each state's `next` parity is read
1177    /// from the flow still open at its key after the body was appended.
1178    pub fn finish_loop_detector_states(
1179        &self,
1180        pending: PendingLoopDetectorStates<'c, M>,
1181    ) -> Vec<LoopCarriedDetectorState<M>> {
1182        pending
1183            .loop_states
1184            .into_iter()
1185            .map(|(key, mut detector_state)| {
1186                if let Some(open) = self.open_flows.get(&key) {
1187                    detector_state.next =
1188                        DetectorParity::from_terms(open.measurements.iter().copied())
1189                            .with_sign(open.sign);
1190                }
1191                detector_state
1192            })
1193            .collect()
1194    }
1195}

1//! Execute a whole compiled [`Bloq`] program without adding noise on the shared
2//! [`Simulator`] engine. This is the *one* walk that owns SEM-ORD ordering,
3//! region control flow, and measurement-record binding; [`run_bloq`],
4//! [`run_bloq_with_hook`], and [`run_bloq_with_io`] are its entry points. (The
5//! single-template runner lives with the verifier in [`crate::verify`].)
6//!
7//! # Whole-program model (`run_bloq`)
8//!
9//! Measurement ids are allocated contiguously across every instance measurement
10//! at every nesting level (`InstanceMeasurement{instance, measurement}` → global
11//! id). Nodes are executed in `deterministic_emit_order` onto ONE shared
12//! [`Simulator`] over the program's global coordinate layout; a quantum
13//! node's circuit comes from `emission_plan` (instances offset-translated), with
14//! its local measurement ids remapped to the global ids.
15//!
16//! ## Region control flow (dynamic, per-shot — the point of a procedural engine)
17//!
18//! Region bodies are nested [`SubGraph`](bloq_ir::SubGraph)s over the same
19//! template pool; the walk recurses into them on the same shared simulator:
20//! * `RepeatUntilSuccess` — execute sandboxed body attempts until both the
21//!   post-selection restart parities and the IR restart predicate accept, up to
22//!   [`DEFAULT_RUS_ATTEMPT_CAP`] attempts. Rejected attempts restore the
23//!   pre-body simulator and records (SEM-ATTEMPT); the accepted attempt alone
24//!   composes upward.
25//!
26//! Quantum membership guards select active instances and their record aliases;
27//! activation slots suppress inactive classical recipes and region bodies.
28//! Classical nodes fold as in a flat program (observable recipes/linear `Compute`;
29//! `Discard` rejects the shot and stops its execution). An `Observable` is a
30//! complete corrected-parity producer, folding local records and bit-carrying
31//! inputs. This no-op executor uses a zero decoder flip; composition
32//! therefore has the same bit here. Boundaries contribute only to reports.
33//! The per-shot `ObservableReport` is measured on the final state
34//! ([`Program::observable_value`]) — the record/feedback fold plus any
35//! composed boundary-Pauli face. Detectors (template-local + node
36//! instance-space, at every level) are verified per shot with the usual
37//! `LoopState`/`RepeatBody` policy.
38//!
39//! # Observable contract
40//!
41//! An `Observable` is its raw record parity plus live `Output`-face boundary
42//! Paulis. Feedforward byproducts therefore remain visible; callers combine the
43//! raw value with the terminal frame sign when they need a deterministic logical
44//! value. `Input` faces are source endpoints already closed by composition and
45//! are not folded. Because output faces are read from the terminal simulator
46//! state, preparation rejects a later quantum node that consumes one.

139#[derive(Debug, Clone)]
140struct InputPatch {
141    input_instance: TemplateInstanceId,
142    prepare_before: Vec<TemplateInstanceId>,
143    port: IVec3,
144    data_qubits: Vec<usize>,
145    stabilizers: Vec<PauliString>,
146    logical_x: PauliString,
147    logical_z: PauliString,
148}
149
150impl InputPatch {
151    fn prepare_injection_scaffold(&self, sim: &mut Simulator) -> Result<usize, ExecError> {
152        let mut intersection = None;
153        for &qubit in &self.data_qubits {
154            let on_x = self.logical_x.xbit(qubit);
155            if on_x {
156                sim.reset_x(qubit)?;
157            } else {
158                sim.reset(qubit)?;
159            }
160            if on_x && self.logical_z.zbit(qubit) && intersection.replace(qubit).is_some() {
161                return Err(ExecError::InvalidInjectionPatch { port: self.port });
162            }
163        }
164        intersection.ok_or(ExecError::InvalidInjectionPatch { port: self.port })
165    }
166
167    fn postselect_injection_patch(&self, sim: &mut Simulator) -> Result<(), ExecError> {
168        for stabilizer in &self.stabilizers {
169            sim.postselect_observable(stabilizer, false)?;
170        }
171        Ok(())
172    }
173}

195impl PreparationContext<'_> {
196    /// Queue an ideal logical CCZ resource on three input seeds. The executor
197    /// applies the gate immediately before the first quantum component that
198    /// consumes any leg, so earlier block execution cannot overwrite it.
199    ///
200    /// # Errors
201    ///
202    /// Returns [`ExecError::InvalidCczPreparation`] when any two ports are equal.
203    pub fn prepare_ccz(&self, ports: [IVec3; 3]) -> Result<(), ExecError> {
204        if ports[0] == ports[1] || ports[0] == ports[2] || ports[1] == ports[2] {
205            return Err(ExecError::InvalidCczPreparation);
206        }
207        self.deferred_ccz.borrow_mut().push(ports);
208        Ok(())
209    }

259enum BoundOperators<'a> {
260    /// Physical reads retain Output faces; Input faces stay in the IR.
261    Owned(Vec<&'a InstanceBoundaryOperator>),
262    Recipe(Vec<&'a InstanceBoundaryOperator>, Box<[BloqNodeId]>),
263}
264
265impl BoundOperators<'_> {
266    fn has_output(&self) -> bool {
267        match self {
268            Self::Owned(operators) => !operators.is_empty(),
269            Self::Recipe(operators, children) => !operators.is_empty() || !children.is_empty(),
270        }
271    }
272}
273
274fn collect_bound_operators<'a>(
275    bindings: &FxHashMap<u32, BoundOperators<'a>>,
276    producer: BloqNodeId,
277    out: &mut Vec<&'a InstanceBoundaryOperator>,
278    expanded: &mut usize,
279) -> Result<(), ExecError> {
280    let mut stack = vec![(producer, false, false)];
281    let mut active = FxHashSet::default();
282    while let Some((producer, exit, nested_recipe)) = stack.pop() {
283        if exit {
284            active.remove(&producer);
285            continue;
286        }
287        if nested_recipe {
288            *expanded = expanded.saturating_add(1);
289            if *expanded > MAX_PHYSICAL_BOUNDARY_BINDINGS {
290                return Err(ExecError::BoundaryBindingExpansionLimit {
291                    limit: MAX_PHYSICAL_BOUNDARY_BINDINGS,
292                });
293            }
294        }
295        if !active.insert(producer) {
296            return Err(ExecError::MalformedGraph("cyclic binding recipe"));
297        }
298        match bindings.get(&producer.0) {
299            Some(BoundOperators::Owned(operators)) => {
300                if nested_recipe {
301                    *expanded = expanded.saturating_add(operators.len());
302                    if *expanded > MAX_PHYSICAL_BOUNDARY_BINDINGS {
303                        return Err(ExecError::BoundaryBindingExpansionLimit {
304                            limit: MAX_PHYSICAL_BOUNDARY_BINDINGS,
305                        });
306                    }
307                }
308                out.extend(operators);
309                active.remove(&producer);
310            }
311            Some(BoundOperators::Recipe(operators, children)) => {
312                *expanded = expanded.saturating_add(operators.len());
313                if *expanded > MAX_PHYSICAL_BOUNDARY_BINDINGS {
314                    return Err(ExecError::BoundaryBindingExpansionLimit {
315                        limit: MAX_PHYSICAL_BOUNDARY_BINDINGS,
316                    });
317                }
318                out.extend(operators);
319                stack.push((producer, true, nested_recipe));
320                stack.extend(children.iter().rev().map(|child| (*child, false, true)));
321            }
322            None => {
323                active.remove(&producer);
324            }
325        }
326    }
327    Ok(())
328}

348impl ShotSnapshot {
349    fn capture(state: &Shot) -> Self {
350        Self {
351            sim: state.sim.clone(),
352            injected_inputs: state.injected_inputs.clone(),
353            deferred_ccz: state.deferred_ccz.clone(),
354            record: state.record.clone(),
355            discarded: state.discarded,
356            excluded_instances: state.excluded_instances.clone(),
357            executed_scopes: state.executed_scopes.clone(),
358            active_membership_inputs: state.active_membership_inputs.clone(),
359            branch_choices: state.branch_choices.clone(),
360            early_faces: state.early_faces.clone(),
361            classical_values: state.classical_values.clone(),
362            saved_outputs: state.saved_outputs.clone(),
363        }
364    }
365
366    fn restore(&self, state: &mut Shot) {
367        state.sim.clone_from(&self.sim);
368        state.injected_inputs.clone_from(&self.injected_inputs);
369        state.deferred_ccz.clone_from(&self.deferred_ccz);
370        state.record.clone_from(&self.record);
371        state.discarded = self.discarded;
372        state
373            .excluded_instances
374            .clone_from(&self.excluded_instances);
375        state.executed_scopes.clone_from(&self.executed_scopes);
376        state
377            .active_membership_inputs
378            .clone_from(&self.active_membership_inputs);
379        state.branch_choices.clone_from(&self.branch_choices);
380        state.early_faces.clone_from(&self.early_faces);
381        state.classical_values.clone_from(&self.classical_values);
382        state.saved_outputs.clone_from(&self.saved_outputs);
383    }
384}

571/// The shared whole-program walk behind [`run_bloq`] and [`run_bloq_with_hook`].
572/// It calls one post-program `hook` per non-discarded shot (out-of-band
573/// analysis); the plain [`run_bloq`] path passes a no-op.
574fn run_bloq_internal<P, F>(
575    bloq: &Bloq,
576    shots: usize,
577    seed: u64,
578    mode: RunMode,
579    prepare: &mut P,
580    hook: &mut F,
581) -> Result<VerifyReport, ExecError>
582where
583    P: FnMut(&mut Simulator, &PreparationContext<'_>) -> Result<(), ExecError>,
584    F: FnMut(&mut Simulator, &ShotContext<'_>) -> Result<(), ExecError>,
585{
586    let program = Program::prepare(bloq, mode)?;
587    let layout = program.report_layout()?;
588
589    let mut detector_shots: Vec<Vec<bool>> = vec![Vec::new(); layout.detectors.len()];
590    let mut observable_shots: Vec<Vec<bool>> = vec![Vec::new(); layout.observable_nodes.len()];
591    let mut frame_x: Vec<Vec<Option<bool>>> = vec![Vec::new(); layout.frames.len()];
592    let mut frame_z: Vec<Vec<Option<bool>>> = vec![Vec::new(); layout.frames.len()];
593    let mut discarded = 0usize;
594    let mut max_rank = 0usize;
595    let mut branch_selectors: Vec<bool> = Vec::new();
596    for shot in 0..shots {
597        let mut completed =
598            program.execute_shot(shot, seed, matches!(mode, RunMode::PrepareInputs), prepare)?;
599        max_rank = max_rank.max(completed.state.peak_rank);
600        branch_selectors.extend(completed.state.branch_choices.iter().copied());
601        if completed.state.discarded {
602            discarded += 1;
603            continue;
604        }
605        if program.core.bloq.logical_outputs().iter().any(|output| {
606            completed
607                .state
608                .excluded_instances
609                .contains(&output.instance)
610        }) {
611            return Err(ExecError::MalformedGraph(
612                "logical output owner is not selected",
613            ));
614        }
615
616        // Per-shot hook on the pristine post-program state (output patches still
617        // live). Runs before the observable/frame folds below so a mutating hook
618        // sees an undisturbed boundary; a mutating hook thus voids this shot's
619        // observable/frame reports but not its record-based detectors.
620        {
621            let shot_frames = layout
622                .frames
623                .iter()
624                .map(|frame| ShotFrame {
625                    port: frame.port,
626                    x: completed.top_values.get(&frame.x.0).copied().flatten(),
627                    z: completed.top_values.get(&frame.z.0).copied().flatten(),
628                })
629                .collect::<Vec<_>>();
630            let saved_outputs = program
631                .early_output_captures
632                .iter()
633                .zip(&completed.state.saved_outputs)
634                .filter_map(|(capture, &qubit)| {
635                    qubit.map(|qubit| SavedOutput {
636                        port: capture.port,
637                        qubit,
638                    })
639                })
640                .collect::<Vec<_>>();
641            let named_branch_selectors = program
642                .core
643                .bloq
644                .nodes()
645                .filter_map(|(id, node)| {
646                    let bloq_ir::NodeProvenance::BranchSelector { name } = &node.provenance else {
647                        return None;
648                    };
649                    let value = completed
650                        .state
651                        .classical_values
652                        .get(&(LevelPath::default(), id))
653                        .copied()
654                        .flatten()?;
655                    Some((name.as_str(), value))
656                })
657                .collect::<Vec<_>>();
658            let ctx = ShotContext {
659                shot,
660                record: &completed.state.record,
661                branch_selectors: &completed.state.branch_choices,
662                named_branch_selectors: &named_branch_selectors,
663                outputs: &layout.outputs,
664                frames: &shot_frames,
665                saved_outputs: &saved_outputs,
666                qubit_count: completed.state.sim.num_qubits(),
667            };
668            hook(&mut completed.state.sim, &ctx)?;
669        }
670
671        // Detectors owned by inactive memberships or regions are skipped.
672        // Any other missing record is a hard error.
673        for ((scope, parity), out) in layout.detectors.iter().zip(&mut detector_shots) {
674            if !parity.applies(scope, &completed.state) {
675                continue;
676            }
677            let value = xor_records(parity, scope, &completed.state)?;
678            out.push(value);
679        }
680        for (node_id, out) in layout.observable_nodes.iter().zip(&mut observable_shots) {
681            if let Some(value) = program.observable_value(
682                *node_id,
683                &completed.top_values,
684                &completed.top_bindings,
685                &mut completed.state,
686            )? {
687                out.push(value);
688            }
689        }
690        for (i, frame) in layout.frames.iter().enumerate() {
691            frame_x[i].push(completed.top_values.get(&frame.x.0).copied().flatten());
692            frame_z[i].push(completed.top_values.get(&frame.z.0).copied().flatten());
693        }
694    }
695
696    Ok(VerifyReport {
697        shots,
698        discarded,
699        detectors: detector_shots.into_iter().map(detector_report).collect(),
700        observables: observable_shots
701            .into_iter()
702            .map(|per_shot| ObservableReport { per_shot })
703            .collect(),
704        frame_pairs: layout
705            .frames
706            .iter()
707            .zip(frame_x)
708            .zip(frame_z)
709            .map(|((frame, x_bits), z_bits)| FramePairReport {
710                port: frame.port,
711                x_bits,
712                z_bits,
713            })
714            .collect(),
715        max_rank,
716        branch_selectors,
717    })
718}

722        let core = PreparedProgramCore::build(bloq)?;
723        let rus_restarts = collect_rus_restarts(&core.bloq, &core.global_meas)?;

725        let output_cuts = super::closure::logical_output_cuts(&core.bloq)?;
726        let mut program = Self {

747    fn validate_early_read_compatibility(&self) -> Result<(), ExecError> {
748        let operators = self
749            .early_reads
750            .iter()
751            .map(|read| {
752                let mut product = PauliString::new(self.core.coord_index.len());
753                for binding in &read.bindings {
754                    product = &product
755                        * self
756                            .core
757                            .boundary_operator(&binding.operator(&self.core.bloq).operator)?;
758                }
759                Ok(product)
760            })
761            .collect::<Result<Vec<_>, ExecError>>()?;
762        for (index, first) in self.early_reads.iter().enumerate() {
763            for (other, second) in self.early_reads[..index].iter().enumerate() {
764                if first.before == second.before
765                    && (first.incomplete || second.incomplete)
766                    && !operators[index].commutes_with(&operators[other])
767                {
768                    return Err(ExecError::IncompatibleEarlyOutputReads {
769                        first: first.observable.0,
770                        second: second.observable.0,
771                        before: first.before.1.0,
772                    });
773                }
774            }
775        }
776        Ok(())
777    }

818    fn execute_shot<P>(
819        &self,
820        shot: usize,
821        seed: u64,
822        stage_inputs: bool,
823        prepare: &mut P,
824    ) -> Result<CompletedShot<'_>, ExecError>
825    where
826        P: FnMut(&mut Simulator, &PreparationContext<'_>) -> Result<(), ExecError>,
827    {
828        let shot_seed = seed ^ shot as u64;
829        let base_qubits = self.core.coord_index.len();
830        let mut state = Shot {
831            sim: Simulator::with_seed(base_qubits, shot_seed),
832            staged_inputs: StagedInputSeeds::default(),
833            injected_inputs: vec![false; self.inputs.len()],
834            deferred_ccz: Vec::new(),
835            record: ShotRecord::with_len(self.core.record_count),
836            discarded: false,
837            peak_rank: 1,
838            excluded_instances: rustc_hash::FxHashSet::default(),
839            executed_scopes: rustc_hash::FxHashSet::default(),
840            active_membership_inputs: rustc_hash::FxHashSet::default(),
841            branch_choices: Vec::new(),
842            early_faces: vec![None; self.early_reads.len()],
843            classical_values: FxHashMap::default(),
844            saved_outputs: vec![None; self.early_output_captures.len()],
845            seed: shot_seed,
846        };
847        if stage_inputs {
848            let mut seeds = Vec::with_capacity(self.inputs.len());
849            for input in &self.inputs {
850                let qubit = state.sim.num_qubits();
851                state.sim.reset_x(qubit)?;
852                seeds.push(InputSeed {
853                    port: input.port,
854                    qubit,
855                });
856            }
857            let deferred_ccz = RefCell::new(Vec::new());
858            prepare(
859                &mut state.sim,
860                &PreparationContext {
861                    shot,
862                    inputs: &seeds,
863                    deferred_ccz: &deferred_ccz,
864                },
865            )?;
866            for ports in deferred_ccz.into_inner() {
867                let mut prepared = DeferredCcz {
868                    seeds: [0; 3],
869                    prepare_before: std::array::from_fn(|_| Vec::new()),
870                    prepared: false,
871                };
872                for (index, port) in ports.into_iter().enumerate() {
873                    let input_index = self
874                        .inputs
875                        .iter()
876                        .position(|input| input.port == port)
877                        .ok_or(ExecError::UnknownPreparationInput { port })?;
878                    prepared.seeds[index] = seeds[input_index].qubit;
879                    prepared.prepare_before[index] =
880                        self.inputs[input_index].prepare_before.clone();
881                }
882                state.deferred_ccz.push(prepared);
883            }
884            for (index, (input, seed)) in self.inputs.iter().zip(seeds).enumerate() {
885                for &target in &input.prepare_before {
886                    state
887                        .staged_inputs
888                        .entry(target)
889                        .or_default()
890                        .push((index, seed.qubit));
891                }
892            }
893        }
894        state.peak_rank = state.peak_rank.max(state.sim.rank());
895        let mut top_values = FxHashMap::default();
896        let mut top_bindings = FxHashMap::default();
897        self.execute_level(
898            self.core.bloq.top(),
899            &LevelPath::default(),
900            &mut state,
901            &mut top_values,
902            &mut top_bindings,
903        )?;
904        Ok(CompletedShot {
905            state,
906            top_values,
907            top_bindings,
908        })
909    }

912        for (owner, parity) in self.rus_restarts.get(scope).into_iter().flatten() {

914                return Ok(true);
915            }
916        }
917        Ok(false)

920    /// Execute one [`SubGraph`] level in emit order on the shared simulator.
921    /// `scope` is the enclosing region chain (empty at the top level); `values`
922    /// holds this level's classical node values.
923    fn execute_level<'a>(
924        &'a self,
925        level: &'a SubGraph,
926        scope: &LevelPath,
927        state: &mut Shot,
928        values: &mut FxHashMap<u32, Option<bool>>,
929        bindings: &mut FxHashMap<u32, BoundOperators<'a>>,
930    ) -> Result<(), ExecError> {
931        state.executed_scopes.insert(scope.clone());
932        for &id in self.core.emit_order(scope)? {
933            let node = level.node(id).expect("emit order yields real ids");
934            if let Some(slot) = node.activation {
935                let inputs = self.slot_inputs(level, id, values);
936                let active = inputs.get(&slot).copied().ok_or(ExecError::MalformedGraph(
937                    "classical activation input is unavailable",
938                ))?;
939                if !active {
940                    if let Some(region) = node.try_region() {
941                        for (_, body) in region.bodies() {
942                            collect_instances(body, &mut state.excluded_instances);
943                        }
944                    } else if let Some(quantum) = node.try_quantum() {
945                        state
946                            .excluded_instances
947                            .extend(quantum.instances.iter().map(|instance| instance.id));
948                    }
949                    values.insert(id.0, Some(false));
950                    state
951                        .classical_values
952                        .insert((scope.clone(), id), Some(false));
953                    continue;
954                }
955            }
956            if !self.early_output_captures.is_empty() {
957                self.capture_early_outputs(scope, id, state)?;
958            } else if !self.early_reads.is_empty() {
959                self.read_early_faces(scope, id, state)?;
960            }
961            match (node.try_classical(), &node.kind) {
962                (_, BloqNodeKind::Quantum(quantum)) => {
963                    if quantum.guards.is_empty() {
964                        self.inject_input_seeds(quantum, state)?;
965                        self.run_quantum_node(id, scope, state)?;
966                    } else {
967                        let inputs = self.slot_inputs(level, id, values);
968                        let selected = self.core.selected_ops(scope, id, node, &inputs)?;
969                        for guard in &quantum.guards {
970                            if inputs[&guard.input] {
971                                state.active_membership_inputs.insert((
972                                    scope.clone(),
973                                    id,
974                                    guard.input,
975                                ));
976                            } else {
977                                state.excluded_instances.extend(&guard.instances);
978                            }
979                        }
980                        state.record.select_aliases(&selected.aliases);
981                        self.inject_input_seeds(&selected.quantum, state)?;
982                        let rank = selected.ops.run(&mut state.sim, &mut state.record)?;
983                        state.peak_rank = state.peak_rank.max(rank);
984                    }
985                }
986                (Some(ClassicalNode::Discard { condition }), _) => {
987                    let inputs = self.slot_inputs(level, id, values);
988                    state.discarded = condition
989                        .eval(&mut |slot| inputs.get(&slot).copied())
990                        .ok_or(ExecError::MalformedGraph(
991                            "discard condition is unavailable",
992                        ))?;
993                }
994                (_, BloqNodeKind::Classical(classical)) => {
995                    let classical = classical.as_ref();
996                    // Frame-content recipes fold over the shot record.
997                    let value = self.classical_value(
998                        level,
999                        id,
1000                        classical,
1001                        values,
1002                        &state.record,
1003                        &state.excluded_instances,
1004                    )?;
1005                    values.insert(id.0, value);
1006                    if matches!(
1007                        node.provenance,
1008                        bloq_ir::NodeProvenance::BranchSelector { .. }
1009                    ) {
1010                        state
1011                            .branch_choices
1012                            .push(value.ok_or(ExecError::MalformedGraph(
1013                                "branch selector has no value",
1014                            ))?);
1015                        state.classical_values.insert((scope.clone(), id), value);
1016                    }
1017                    if let ClassicalNode::Observable { operators, .. } = classical {
1018                        let children = level
1019                            .data_inputs(id)
1020                            .filter(|input| input.output.is_none())
1021                            .map(|input| input.producer)
1022                            .filter(|child| {
1023                                bindings
1024                                    .get(&child.0)
1025                                    .is_some_and(BoundOperators::has_output)
1026                            })
1027                            .collect();
1028                        bindings.insert(
1029                            id.0,
1030                            BoundOperators::Recipe(
1031                                operators
1032                                    .iter()
1033                                    .filter(|operator| {
1034                                        operator.face == bloq_ir::BoundaryFace::Output
1035                                    })
1036                                    .collect(),
1037                                children,
1038                            ),
1039                        );
1040                    }
1041                }
1042                (_, BloqNodeKind::Region(region)) => {
1043                    self.execute_region(level, id, region, scope, state, values, bindings)?
1044                }
1045            }
1046            if !self.early_reads.is_empty()
1047                && let Some(&value) = values.get(&id.0)
1048            {
1049                state.classical_values.insert((scope.clone(), id), value);
1050            }
1051            if state.discarded {
1052                // SEM-DISCARD: no later scheduled node executes in this shot.
1053                break;
1054            }
1055        }
1056        Ok(())
1057    }

1059    /// Region control flow (see the [module docs](self)).
1060    // Threaded walk state (level/scope/state/values) alongside the node params —
1061    // a tree walk, not a public API surface.
1062    #[expect(
1063        clippy::too_many_arguments,
1064        reason = "recursive execution threads one region-walk state"
1065    )]
1066    fn execute_region<'a>(
1067        &'a self,
1068        level: &'a SubGraph,
1069        id: BloqNodeId,
1070        region: &'a RegionNode,
1071        scope: &LevelPath,
1072        state: &mut Shot,
1073        values: &mut FxHashMap<u32, Option<bool>>,
1074        bindings: &mut FxHashMap<u32, BoundOperators<'a>>,
1075    ) -> Result<(), ExecError> {
1076        let RegionNode::RepeatUntilSuccess {
1077            body,
1078            restart_condition,
1079            restart_source,
1080        } = region;
1081        let child = scope.child(id, BodySelector::Body);
1082        let snapshot = ShotSnapshot::capture(state);
1083        let mut attempt = 0;
1084        loop {
1085            if attempt != 0 {
1086                snapshot.restore(state);
1087            }
1088            state
1089                .sim
1090                .reseed_rng(rus_attempt_seed(state.seed, &child, attempt));
1091
1092            let mut body_values = FxHashMap::default();
1093            let mut body_bindings = FxHashMap::default();
1094            self.execute_level(body, &child, state, &mut body_values, &mut body_bindings)?;
1095            // Both mandatory restart signals are evaluated unconditionally
1096            // (no short-circuit): an unevaluable restart condition is an
1097            // error even on an attempt a restart parity already rejected.
1098            let restart_detector = self.restart_signaled(&child, state)?;
1099            let open = restart_source.and_then(|source| {
1100                zero_decoder_output(
1101                    body_values.get(&source.node.0).copied().flatten(),
1102                    Some(source.output),
1103                )
1104            });
1105            let restart_predicate = restart_condition
1106                .eval(&mut |slot| {
1107                    let input = level.value_inputs(id).find(|input| input.slot == slot);
1108                    match input {
1109                        Some(input) => zero_decoder_output(
1110                            values.get(&input.producer.0).copied().flatten(),
1111                            input.output,
1112                        ),
1113                        None => open,
1114                    }
1115                })
1116                .ok_or(ExecError::MalformedGraph(
1117                    "RUS restart condition is not evaluable",
1118                ))?;
1119
1120            if !(restart_detector || restart_predicate) {
1121                // Attempt-local draws must not perturb the parent RNG stream.
1122                state.sim.restore_rng_from(&snapshot.sim);
1123                values.insert(id.0, bound_body_value(body, &body_values));
1124                bindings.insert(
1125                    id.0,
1126                    BoundOperators::Owned(bound_body_bindings(body, &body_bindings)?),
1127                );
1128                break;
1129            }
1130            attempt += 1;
1131            if attempt >= DEFAULT_RUS_ATTEMPT_CAP {
1132                state.discarded = true;
1133                break;
1134            }
1135        }
1136        Ok(())
1137    }

1139    fn inject_input_seeds(
1140        &self,
1141        quantum: &bloq_ir::QuantumNode,
1142        state: &mut Shot,
1143    ) -> Result<(), ExecError> {
1144        for preparation in &mut state.deferred_ccz {
1145            if !preparation.prepared
1146                && preparation
1147                    .prepare_before
1148                    .iter()
1149                    .flatten()
1150                    .any(|cut| quantum.instances.iter().any(|instance| instance.id == *cut))
1151            {
1152                let [a, b, c] = preparation.seeds;
1153                state.sim.ccz(a, b, c)?;
1154                preparation.prepared = true;
1155            }
1156        }
1157        for instance in &quantum.instances {
1158            for &(input_index, seed) in state.staged_inputs.get(&instance.id).into_iter().flatten()
1159            {
1160                if state.injected_inputs[input_index] {
1161                    continue;
1162                }
1163                let input = &self.inputs[input_index];
1164                let corner = input.prepare_injection_scaffold(&mut state.sim)?;
1165                state.sim.swap(seed, corner);
1166                input.postselect_injection_patch(&mut state.sim)?;
1167                state.injected_inputs[input_index] = true;
1168            }
1169        }
1170        Ok(())
1171    }

1188    /// The `Value` output of a record-based classical producer in one shot.
1189    fn classical_value(
1190        &self,
1191        level: &SubGraph,
1192        id: BloqNodeId,
1193        classical: &ClassicalNode,
1194        values: &FxHashMap<u32, Option<bool>>,
1195        record: &ShotRecord,
1196        excluded: &rustc_hash::FxHashSet<TemplateInstanceId>,
1197    ) -> Result<Option<bool>, ExecError> {
1198        let mut record_value = false;
1199        for im in classical.measurements() {
1200            if excluded.contains(&im.instance) {
1201                continue;
1202            }
1203            let gid = *self
1204                .core
1205                .global_meas
1206                .get(im)
1207                .ok_or(ExecError::MalformedGraph(
1208                    "observable names an unknown measurement",
1209                ))?;
1210            record_value ^= record.require(gid)?;
1211        }
1212        match classical {
1213            ClassicalNode::Compute { expr } => {
1214                let inputs = self.slot_inputs(level, id, values);
1215                Ok(expr.eval(&mut |slot| inputs.get(&slot).copied()))
1216            }
1217            // Recipes XOR records and data inputs. Output boundary operators
1218            // fold at report time ([`Self::observable_value`]); an unavailable
1219            // data input keeps the recipe unavailable.
1220            ClassicalNode::Observable { .. } => {
1221                let mut value = record_value;
1222                for input in level.data_inputs(id) {
1223                    match zero_decoder_output(
1224                        values.get(&input.producer.0).copied().flatten(),
1225                        input.output,
1226                    ) {
1227                        Some(bit) => value ^= bit,
1228                        None => return Ok(None),
1229                    }
1230                }
1231                Ok(Some(value))
1232            }
1233            ClassicalNode::Discard { .. } => Ok(None),
1234        }
1235    }

1237    /// This level's `Value` inputs of `id`, resolved to `slot → bit` from
1238    /// already-computed producer values.
1239    fn slot_inputs(
1240        &self,
1241        level: &SubGraph,
1242        id: BloqNodeId,
1243        values: &FxHashMap<u32, Option<bool>>,
1244    ) -> FxHashMap<u32, bool> {
1245        let mut inputs = FxHashMap::default();
1246        for input in level.value_inputs(id) {

1257    /// An `Observable` node's value: the XOR of its record producers plus any
1258    /// live `Output`-face boundary Pauli measured on the final state.
1259    ///
1260    /// `Input` faces contribute **nothing** to the physical fold. An
1261    /// input boundary operator is a *source endpoint* (e.g. a T block's
1262    /// magic-state anchor), closed internally by the cultivation seam when the
1263    /// templates compose. Its MPP-boundary
1264    /// realization is the *static* verifier's perfect-port stand-in; `run_bloq`
1265    /// executes the composed circuit, where that closure is already carried by the
1266    /// folded cultivation records, and the input face no longer
1267    /// exists as a live boundary. Measuring it on the terminal state would read
1268    /// the wrong operator at the wrong time (the patch has since been
1269    /// merged/consumed), so we skip it. Only `Output` faces are live terminal
1270    /// boundaries and are measured here. [`super::closure::plan_early_output_reads`]
1271    /// moves a consumed face's read ahead of the node that reuses its support.
1272    fn observable_value(
1273        &self,
1274        obs: BloqNodeId,
1275        top_values: &FxHashMap<u32, Option<bool>>,
1276        top_bindings: &FxHashMap<u32, BoundOperators<'_>>,
1277        state: &mut Shot,
1278    ) -> Result<Option<bool>, ExecError> {
1279        let guard = self.core.bloq.top()[obs].activation;
1280        if let Some(slot) = guard {
1281            let inputs = self.slot_inputs(self.core.bloq.top(), obs, top_values);
1282            if inputs.get(&slot) != Some(&true) {
1283                return Ok(None);
1284            }
1285        }
1286        let Some(mut value) = top_values.get(&obs.0).copied().flatten() else {
1287            return Ok(None);
1288        };
1289        let mut terminal_operator: Option<&PauliString> = None;
1290        let mut combined = false;
1291        let mut boundary_scratch = self.boundary_scratch.borrow_mut();
1292        let mut output_operators = self.core.bloq.top()[obs]
1293            .try_classical()
1294            .expect("observable is classical")
1295            .operators()
1296            .iter()
1297            .filter(|operator| operator.face == bloq_ir::BoundaryFace::Output)
1298            .collect::<Vec<_>>();
1299        let mut expanded = 0;
1300        for input in self
1301            .core
1302            .bloq
1303            .top()
1304            .data_inputs(obs)
1305            .filter(|input| input.output.is_none())
1306        {
1307            collect_bound_operators(
1308                top_bindings,
1309                input.producer,
1310                &mut output_operators,
1311                &mut expanded,
1312            )?;
1313        }
1314        let mut cached_bindings = Vec::new();
1315        for (read, bit) in self.early_reads.iter().zip(&state.early_faces) {
1316            if let Some(bit) = bit
1317                && read.observable == obs
1318                && read.bindings.iter().all(|binding| {
1319                    output_operators
1320                        .iter()
1321                        .any(|operator| std::ptr::eq(binding.operator(&self.core.bloq), *operator))
1322                })
1323            {
1324                value ^= bit;
1325                cached_bindings.extend(&read.bindings);
1326            }
1327        }
1328        for operator in output_operators {
1329            if !cached_bindings
1330                .iter()
1331                .any(|binding| std::ptr::eq(binding.operator(&self.core.bloq), operator))
1332            {
1333                let prepared = self.core.boundary_operator(&operator.operator)?;
1334                if let Some(first) = terminal_operator {
1335                    if !combined {
1336                        boundary_scratch.clone_from(first);
1337                        combined = true;
1338                    }
1339                    // Keep the exact phase: XX * ZZ = -YY.
1340                    *boundary_scratch = &*boundary_scratch * prepared;
1341                } else {
1342                    terminal_operator = Some(prepared);
1343                }
1344            }
1345        }
1346        if let Some(first) = terminal_operator {
1347            let pauli = if combined { &boundary_scratch } else { first };
1348            value ^= Self::measure_boundary_pauli(pauli, &mut state.sim)?;
1349        }
1350        Ok(Some(value))
1351    }

1354        let captures = self
1355            .core
1356            .bloq
1357            .logical_outputs()
1358            .iter()
1359            .zip(&self.output_cuts)
1360            .filter(|(_, cuts)| !cuts.is_empty())
1361            .map(|(output, cuts)| {

1363                    cuts.clone(),
1364                    output.instance,

1367                    self.joint_pauli(&output.z)?,
1368                ))
1369            })
1370            .collect::<Result<Vec<_>, ExecError>>()?;
1371        let base_width = self.core.coord_index.len();
1372        let variant_count = captures.len();

1376            .into_iter()
1377            .map(
1378                |(before, instance, port, logical_x, logical_z)| EarlyOutputCapture {

1380                    instance,
1381                    port,

1387                                logical_x: super::circuit::widen_pauli(&logical_x, width),

1399    /// Swap a logical output away immediately before another node reuses its
1400    /// patch. The program's Pauli frame remains for the terminal hook to apply.
1401    fn capture_early_outputs(
1402        &self,
1403        scope: &LevelPath,
1404        id: BloqNodeId,
1405        state: &mut Shot,
1406    ) -> Result<(), ExecError> {
1407        for (slot, capture) in self.early_output_captures.iter().enumerate() {
1408            if !capture
1409                .before
1410                .iter()
1411                .any(|(cut_scope, cut_id)| *cut_id == id && cut_scope == scope)
1412                || state.saved_outputs[slot].is_some()
1413                || state.excluded_instances.contains(&capture.instance)
1414            {
1415                continue;
1416            }
1417            let qubit = state.sim.num_qubits();
1418            let operators = capture
1419                .operators
1420                .get(qubit.saturating_sub(self.core.coord_index.len()))
1421                .ok_or(ExecError::MalformedGraph(
1422                    "early-output capture has no prepared width",
1423                ))?;
1424            state.sim.reset(qubit)?;
1425            state
1426                .sim
1427                .controlled_pauli(&operators.logical_z, &operators.ancilla_x)?;
1428            state
1429                .sim
1430                .controlled_pauli(&operators.ancilla_z, &operators.logical_x)?;
1431            state
1432                .sim
1433                .controlled_pauli(&operators.logical_z, &operators.ancilla_x)?;
1434            state.saved_outputs[slot] = Some(qubit);
1435        }
1436        Ok(())
1437    }

1439    /// Resolve pure classical dataflow at a cut, including computations whose
1440    /// inputs are ready but whose nodes have not yet been scheduled.
1441    fn available_value(
1442        &self,
1443        scope: &LevelPath,
1444        id: BloqNodeId,
1445        state: &Shot,
1446        memo: &mut FxHashMap<(LevelPath, BloqNodeId), Option<bool>>,
1447    ) -> Result<Option<bool>, ExecError> {
1448        let key = (scope.clone(), id);
1449        if let Some(&value) = state.classical_values.get(&key).or_else(|| memo.get(&key)) {
1450            return Ok(value);
1451        }
1452        let level = self
1453            .core
1454            .bloq
1455            .level_at(scope)
1456            .expect("binding scope exists");
1457        let Some(classical) = level[id].try_classical() else {
1458            return Ok(None);
1459        };
1460        if let Some(slot) = level[id].activation {
1461            let input = level
1462                .value_inputs(id)
1463                .find(|input| input.slot == slot)
1464                .ok_or(ExecError::MalformedGraph(
1465                    "classical guard has no value input",
1466                ))?;
1467            match zero_decoder_output(
1468                self.available_value(scope, input.producer, state, memo)?,
1469                input.output,
1470            ) {
1471                Some(true) => {}
1472                Some(false) => {
1473                    memo.insert(key, Some(false));
1474                    return Ok(Some(false));
1475                }
1476                None => return Ok(None),
1477            }
1478        }
1479        let mut values = FxHashMap::default();
1480        for input in level.data_inputs(id) {
1481            values.insert(
1482                input.producer.0,
1483                self.available_value(scope, input.producer, state, memo)?,
1484            );
1485        }
1486        let value = match self.classical_value(
1487            level,
1488            id,
1489            classical,
1490            &values,
1491            &state.record,
1492            &state.excluded_instances,
1493        ) {
1494            Err(ExecError::MissingRecord(_)) => Ok(None),
1495            result => result,
1496        }?;
1497        memo.insert(key, value);
1498        Ok(value)
1499    }

1521    /// Take the planned reads of any `Output` face node `id` is about to destroy
1522    /// (see [`super::closure::plan_early_output_reads`]).
1523    fn read_early_faces(
1524        &self,
1525        scope: &LevelPath,
1526        id: BloqNodeId,
1527        state: &mut Shot,
1528    ) -> Result<(), ExecError> {
1529        // LIM-022: decide every gate at this cut before projecting any face.
1530        // Unknown values are cached only here: a later cut may have their records.
1531        let mut memo = FxHashMap::default();
1532        let reads = self
1533            .early_reads
1534            .iter()
1535            .enumerate()
1536            .filter(|(slot, read)| {
1537                read.before == (scope.clone(), id) && state.early_faces[*slot].is_none()
1538            })
1539            .map(|(slot, read)| {
1540                let active = self
1541                    .binding_active(&read.bindings[0], state, &mut memo)?
1542                    .ok_or(ExecError::EarlyOutputGateUnavailable {
1543                        observable: read.observable.0,
1544                        before: id.0,
1545                    })?;
1546                Ok((slot, read, active))
1547            })
1548            .collect::<Result<Vec<_>, ExecError>>()?;
1549        for (slot, read, active) in reads {
1550            if !active {
1551                continue;
1552            }
1553            let operators = read
1554                .bindings
1555                .iter()
1556                .filter(|binding| {
1557                    !self
1558                        .early_reads
1559                        .iter()
1560                        .zip(&state.early_faces)
1561                        .any(|(earlier, bit)| {
1562                            earlier.observable == read.observable
1563                                && bit.is_some()
1564                                && earlier.bindings.contains(binding)
1565                        })
1566                })
1567                .map(|binding| binding.operator(&self.core.bloq))
1568                .collect::<Vec<_>>();
1569            if operators.is_empty() {
1570                continue;
1571            }
1572            let bit = if let [operator] = operators.as_slice() {
1573                self.measure_boundary(operator, &mut state.sim)?
1574            } else {
1575                let mut pauli = PauliString::new(self.core.coord_index.len());
1576                for operator in operators {
1577                    pauli = &pauli * self.core.boundary_operator(&operator.operator)?;
1578                }
1579                Self::measure_boundary_pauli(&pauli, &mut state.sim)?
1580            };
1581            state.early_faces[slot] = Some(bit);
1582        }
1583        Ok(())
1584    }

1586    /// Read a boundary-operator Pauli (a logical face). The operator's
1587    /// coordinates are already instance-global (translated at lowering — see
1588    /// [`bloq_ir::InstanceBoundaryOperator`]), so they resolve directly; the
1589    /// instance lookup only validates that the graph names a known instance.
1590    ///
1591    /// A live logical face is deterministic, so the non-collapsing expectation
1592    /// reads it without disturbing the state — which is what makes an early read
1593    /// at a cut safe, with the rest of the program still to run. A face with no
1594    /// definite value (`⟨P⟩ = 0`, or a fractional Clifford+T expectation) has
1595    /// nothing to read non-destructively, so it is projected, as the terminal
1596    /// read has always done.
1597    fn measure_boundary(
1598        &self,
1599        operator: &bloq_ir::lowering::InstanceBoundaryOperator,
1600        sim: &mut Simulator,
1601    ) -> Result<bool, ExecError> {
1602        if !self.core.instance_offset.contains_key(&operator.instance) {
1603            return Err(ExecError::MalformedGraph(
1604                "boundary operator names an unknown instance",
1605            ));
1606        }
1607        if operator.operator.is_empty() {
1608            return Err(ExecError::EmptyMppProduct);
1609        }
1610        let pauli = self.core.boundary_operator(&operator.operator)?;
1611        Self::measure_boundary_pauli(pauli, sim)
1612    }

1614    fn measure_boundary_pauli(pauli: &PauliString, sim: &mut Simulator) -> Result<bool, ExecError> {
1615        let expectation = sim.peek_observable_expectation(pauli)?;
1616        if (1.0 - expectation.abs()) < DETERMINISTIC_EXPECTATION_TOLERANCE {
1617            return Ok(expectation < 0.0);
1618        }
1619        Ok(sim.measure_observable(pauli)?.outcome)
1620    }

1624    /// translated by their instance offset at lowering (see

1632                offset: IVec2::ZERO,

1634            },

1638    /// Physical patches and logical operators for every compiler-authored
1639    /// program input.
1640    fn input_patches(&self) -> Result<Vec<InputPatch>, ExecError> {
1641        let mut placements = FxHashMap::default();
1642        let mut spatial_cubes: FxHashMap<_, Vec<_>> = FxHashMap::default();
1643        let mut spatial_waits = FxHashMap::default();
1644        self.core.bloq.walk(|cx| {
1645            if let Some(quantum) = cx.node.try_quantum() {
1646                placements.extend(
1647                    quantum
1648                        .instances
1649                        .iter()
1650                        .map(|instance| (instance.id, (instance.template_id, instance.offset))),
1651                );
1652                for instance in &quantum.instances {
1653                    if matches!(
1654                        instance.provenance,
1655                        bloq_ir::InstanceProvenance::SpatialPortSubstitution {
1656                            role: bloq_ir::PortRole::Input,
1657                            part: bloq_ir::SpatialPortPart::TemporalPort,
1658                            ..
1659                        }
1660                    ) {
1661                        // Inject before held rounds; a later reset erases their syndrome history.
1662                        let wait = cx.level.outgoing(cx.id).find_map(|edge| {
1663                            if !matches!(edge.edge, bloq_ir::BloqEdge::Quantum(_)) {
1664                                return None;
1665                            }
1666                            let node = &cx.level[edge.target];
1667                            node.memory_rounds()?;
1668                            node.try_quantum()?
1669                                .instances
1670                                .iter()
1671                                .find(|wait| wait.offset == instance.offset)
1672                        });
1673                        if let Some(wait) = wait {
1674                            spatial_waits.insert(instance.id, wait.id);
1675                        }
1676                    }
1677                    if let bloq_ir::InstanceProvenance::SpatialPortSubstitution {
1678                        source,
1679                        role,
1680                        part: bloq_ir::SpatialPortPart::Cube,
1681                    } = instance.provenance
1682                        && role == bloq_ir::PortRole::Input
1683                    {
1684                        spatial_cubes.entry(source).or_default().push(instance.id);
1685                    }
1686                }
1687            }
1688            bloq_ir::WalkControl::Continue
1689        });
1690        self.core
1691            .bloq
1692            .logical_inputs()
1693            .iter()
1694            .map(|input| {
1695                let &(template_id, offset) =
1696                    placements
1697                        .get(&input.instance)
1698                        .ok_or(ExecError::MalformedGraph(
1699                            "logical input names an unknown instance",
1700                        ))?;
1701                let template = &self.core.bloq.templates()[template_id];
1702                let maps = template
1703                    .boundary_flows
1704                    .iter()
1705                    .filter(|flow| flow.start.is_empty() && !flow.end.is_empty())
1706                    .map(|flow| {
1707                        if flow.sign {
1708                            return Err(ExecError::MalformedGraph(
1709                                "input Port has a signed create stabilizer",
1710                            ));
1711                        }
1712                        Ok(flow.end.try_translated(offset)?)
1713                    })
1714                    .collect::<Result<Vec<_>, ExecError>>()?;
1715                let mut data = rustc_hash::FxHashSet::default();
1716                for map in &maps {
1717                    data.extend(map.iter().map(|(coordinate, _)| *coordinate));
1718                }
1719                let mut data = data.into_iter().collect::<Vec<_>>();
1720                data.sort_by_key(glam::IVec2::to_array);
1721                let data_qubits = data
1722                    .into_iter()
1723                    .map(|coordinate| {
1724                        self.core
1725                            .coord_index
1726                            .get(&coordinate)
1727                            .map(|&index| index as usize)
1728                            .ok_or(ExecError::UnknownCoord(coordinate))
1729                    })
1730                    .collect::<Result<Vec<_>, _>>()?;
1731                let stabilizers = maps
1732                    .iter()
1733                    .map(|map| self.joint_pauli(map))
1734                    .collect::<Result<Vec<_>, _>>()?;
1735                Ok(InputPatch {
1736                    input_instance: input.instance,
1737                    prepare_before: spatial_waits
1738                        .get(&input.instance)
1739                        .copied()
1740                        .map(|instance| vec![instance])
1741                        .or_else(|| spatial_cubes.get(&input.port).cloned())
1742                        .unwrap_or_else(|| vec![input.instance]),
1743                    port: input.port,
1744                    data_qubits,
1745                    stabilizers,
1746                    logical_x: self.joint_pauli(&input.x)?,
1747                    logical_z: self.joint_pauli(&input.z)?,
1748                })
1749            })
1750            .collect()
1751    }

1782    fn visit_regions(
1783        bloq: &Bloq,
1784        level: &SubGraph,
1785        scope: &LevelPath,
1786        global_meas: &FxHashMap<InstanceMeasurement, u32>,
1787        out: &mut FxHashMap<LevelPath, Vec<(LevelPath, ResolvedDetector)>>,
1788    ) -> Result<(), ExecError> {
1789        for (id, node) in level.nodes() {
1790            let Some(region) = node.try_region() else {
1791                continue;
1792            };
1793            for (selector, body) in region.bodies() {
1794                let child = scope.child(id, selector);
1795                if matches!(region, RegionNode::RepeatUntilSuccess { .. }) {
1796                    let mut parities = Vec::new();
1797                    collect_restart_parities(bloq, body, &child, global_meas, &mut parities)?;
1798                    out.insert(child.clone(), parities);
1799                }
1800                visit_regions(bloq, body, &child, global_meas, out)?;
1801            }

1806    fn collect_restart_parities(
1807        bloq: &Bloq,
1808        level: &SubGraph,
1809        scope: &LevelPath,
1810        global_meas: &FxHashMap<InstanceMeasurement, u32>,
1811        out: &mut Vec<(LevelPath, ResolvedDetector)>,
1812    ) -> Result<(), ExecError> {
1813        for (id, node) in level.nodes() {
1814            if let Some(quantum) = node.try_quantum() {
1815                for instance in &quantum.instances {
1816                    let template = bloq.templates().get(instance.template_id).ok_or(
1817                        ExecError::MalformedGraph("instance references a missing template"),
1818                    )?;
1819                    for restart in &template.restarts {
1820                        out.push((
1821                            scope.clone(),
1822                            resolve_template_parity(&restart.parity, instance.id, global_meas)?
1823                                .guarded(
1824                                    id,
1825                                    quantum
1826                                        .guards
1827                                        .iter()
1828                                        .find(|guard| guard.instances.contains(&instance.id))
1829                                        .map(|guard| guard.input),
1830                                ),
1831                        ));
1832                    }
1833                }
1834                for (index, restart) in quantum.restarts.iter().enumerate() {
1835                    let restart_index =
1836                        u32::try_from(index).map_err(|_| ExecError::IdOverflow("restart"))?;
1837                    out.push((
1838                        scope.clone(),
1839                        resolve_node_parity(&restart.parity, global_meas)?
1840                            .with_contributions(
1841                                id,
1842                                quantum.guards.iter().flat_map(|guard| {
1843                                    guard
1844                                        .restart_parities
1845                                        .iter()
1846                                        .filter(move |(row, _)| *row as usize == index)
1847                                        .map(move |(_, parity)| {
1848                                            (guard.input, resolve_node_parity(parity, global_meas))
1849                                        })
1850                                }),
1851                            )?
1852                            .guarded(
1853                                id,
1854                                quantum
1855                                    .guards
1856                                    .iter()
1857                                    .find(|guard| guard.restarts.contains(&restart_index))

1904fn collect_detectors_into(
1905    bloq: &Bloq,
1906    level: &SubGraph,
1907    scope: &LevelPath,
1908    global_meas: &FxHashMap<InstanceMeasurement, u32>,
1909    out: &mut Vec<(LevelPath, ResolvedDetector)>,
1910) -> Result<(), ExecError> {
1911    for (id, node) in level.nodes() {
1912        if let Some(quantum) = node.try_quantum() {
1913            for instance in &quantum.instances {
1914                let template =
1915                    bloq.templates()
1916                        .get(instance.template_id)
1917                        .ok_or(ExecError::MalformedGraph(
1918                            "instance references a missing template",
1919                        ))?;
1920                for detector in &template.detectors {
1921                    out.push((
1922                        scope.clone(),
1923                        resolve_template_parity(&detector.parity, instance.id, global_meas)?
1924                            .guarded(
1925                                id,
1926                                quantum
1927                                    .guards
1928                                    .iter()
1929                                    .find(|guard| guard.instances.contains(&instance.id))
1930                                    .map(|guard| guard.input),
1931                            ),
1932                    ));
1933                }
1934            }
1935            for detector in bloq.node_detectors(quantum)? {
1936                let parity = resolve_node_parity(&detector.parity(), global_meas)?;
1937                let (membership, parity) = match detector.address() {
1938                    NodeDetectorAddress::Inline(index) => {
1939                        let membership = quantum
1940                            .guards
1941                            .iter()
1942                            .find(|guard| guard.detectors.contains(&index))
1943                            .map(|guard| guard.input);
1944                        let parity = parity.with_contributions(
1945                            id,
1946                            quantum.guards.iter().flat_map(|guard| {
1947                                guard
1948                                    .detector_parities
1949                                    .iter()
1950                                    .filter(move |(row, _)| *row == index)
1951                                    .map(move |(_, parity)| {
1952                                        (guard.input, resolve_node_parity(parity, global_meas))
1953                                    })
1954                            }),
1955                        )?;
1956                        (membership, parity)
1957                    }
1958                    NodeDetectorAddress::Bundle { use_index, .. } => {
1959                        let membership = quantum
1960                            .guards
1961                            .iter()
1962                            .find(|guard| guard.detector_bundles.contains(&use_index))
1963                            .map(|guard| guard.input);
1964                        (membership, parity)
1965                    }
1966                };
1967                out.push((scope.clone(), parity.guarded(id, membership)));
1968            }
1969        }
1970        if let Some(region) = node.try_region() {
1971            for (selector, body) in region.bodies() {
1972                let child = scope.child(id, selector);
1973                collect_detectors_into(bloq, body, &child, global_meas, out)?;
1974            }
1975        }
1976    }
1977    Ok(())
1978}

1991/// A detector's sign and `(owning instance, global record id)` terms, so
1992/// per-shot evaluation can normalize its expected value and distinguish
1993/// deselected instances' records from genuine ones.
1994#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord)]
1995pub(super) struct ResolvedDetector {
1996    pub sign: bool,
1997    pub terms: Vec<(TemplateInstanceId, u32)>,
1998    membership: Option<(BloqNodeId, u32)>,
1999    conditioned: Vec<(BloqNodeId, u32, ResolvedDetector)>,
2000}
2001
2002impl ResolvedDetector {
2003    fn with_contributions(
2004        mut self,
2005        node: BloqNodeId,
2006        contributions: impl Iterator<Item = (u32, Result<ResolvedDetector, ExecError>)>,
2007    ) -> Result<Self, ExecError> {
2008        for (input, parity) in contributions {
2009            let parity = parity?;
2010            self.conditioned.push((node, input, parity));
2011        }
2012        Ok(self)
2013    }
2014    fn guarded(mut self, node: BloqNodeId, input: Option<u32>) -> Self {
2015        self.membership = input.map(|slot| (node, slot));
2016        self
2017    }
2018
2019    fn applies(&self, scope: &LevelPath, state: &Shot) -> bool {
2020        state.executed_scopes.contains(scope)
2021            && self.membership.is_none_or(|(node, slot)| {
2022                state
2023                    .active_membership_inputs
2024                    .contains(&(scope.clone(), node, slot))
2025            })
2026    }
2027}

2029/// Resolve a parity's measurement terms to `(owning instance, global record)`.
2030/// `owner` maps a term to the `InstanceMeasurement` that keys `global_meas` —
2031/// template parities synthesise it from their fixed instance, node parities
2032/// carry it — which is the only way the two callers differ.
2033fn resolve_parity<T: Copy + Ord>(
2034    parity: &DetectorParity<T>,
2035    global_meas: &FxHashMap<InstanceMeasurement, u32>,
2036    owner: impl Fn(T) -> InstanceMeasurement,
2037    unknown: &'static str,
2038) -> Result<ResolvedDetector, ExecError> {
2039    let mut ids = Vec::with_capacity(parity.terms().len());
2040    for term in parity.terms() {
2041        match term {
2042            DetectorTerm::Measurement(m) => {
2043                let key = owner(*m);
2044                let global = *global_meas
2045                    .get(&key)
2046                    .ok_or(ExecError::MalformedGraph(unknown))?;
2047                ids.push((key.instance, global));
2048            }
2049            DetectorTerm::LoopState(state) => return Err(ExecError::UnresolvedLoopState(state.0)),
2050        }
2051    }
2052    Ok(ResolvedDetector {
2053        sign: parity.sign(),
2054        terms: ids,
2055        membership: None,
2056        conditioned: Vec::new(),
2057    })
2058}

2060fn resolve_template_parity(
2061    parity: &DetectorParity<u32>,
2062    instance: TemplateInstanceId,
2063    global_meas: &FxHashMap<InstanceMeasurement, u32>,
2064) -> Result<ResolvedDetector, ExecError> {
2065    resolve_parity(
2066        parity,
2067        global_meas,
2068        |measurement| InstanceMeasurement {
2069            instance,
2070            measurement,
2071        },
2072        "template detector names an unknown measurement",
2073    )
2074}

2078    global_meas: &FxHashMap<InstanceMeasurement, u32>,
2079) -> Result<ResolvedDetector, ExecError> {

2081        parity,
2082        global_meas,
2083        |im| im,
2084        "node detector names an unknown measurement",
2085    )

2135/// Evaluate a detector whose owner scope ran this shot. Terms naming instances
2136/// excluded from this shot are dropped. Owner scope and membership guards,
2137/// not record terms, decide detector applicability.
2138/// Any other missing record is a hard [`ExecError::MissingRecord`].
2139fn xor_records(
2140    parity: &ResolvedDetector,
2141    scope: &LevelPath,
2142    state: &Shot,
2143) -> Result<bool, ExecError> {
2144    let mut value = parity.sign;
2145    for &(instance, id) in &parity.terms {
2146        if state.excluded_instances.contains(&instance) {
2147            continue;
2148        }
2149        value ^= state.record.require(id)?;
2150    }
2151    for (node, input, contribution) in &parity.conditioned {
2152        if !state
2153            .active_membership_inputs
2154            .contains(&(scope.clone(), *node, *input))
2155        {
2156            continue;
2157        }
2158        value ^= contribution.sign;
2159        for &(instance, id) in &contribution.terms {
2160            if !state.excluded_instances.contains(&instance) {
2161                value ^= state.record.require(id)?;
2162            }
2163        }
2164    }
2165    Ok(value)
2166}

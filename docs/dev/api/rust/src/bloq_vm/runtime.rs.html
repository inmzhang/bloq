85impl RunResult {
86    /// Read a signed VM Pauli expectation from the untouched terminal state.
87    ///
88    /// # Errors
89    ///
90    /// Returns a fixed-register or backend error.
91    pub fn expectation(&self, operator: &PauliProduct) -> Result<f64, RuntimeError> {
92        let operator = engine_pauli_product(operator, self.simulator.num_qubits())?;
93        Ok(self.simulator.peek_observable_expectation(&operator)?)
94    }
95
96    /// Read the terminal `(X, Y, Z)` Bloch vector with VM frame corrections.
97    ///
98    /// # Errors
99    ///
100    /// Returns [`RuntimeError::MissingBit`] for an unavailable frame or a
101    /// fixed-register/backend error for malformed logical operators.
102    pub fn logical_bloch(&self, output: &LogicalOutput) -> Result<(f64, f64, f64), RuntimeError> {
103        let frame_x = self
104            .artifact
105            .final_bits
106            .get(output.frame_x as usize)
107            .copied()
108            .flatten()
109            .ok_or(RuntimeError::MissingBit(output.frame_x))?;
110        let frame_z = self
111            .artifact
112            .final_bits
113            .get(output.frame_z as usize)
114            .copied()
115            .flatten()
116            .ok_or(RuntimeError::MissingBit(output.frame_z))?;
117        let width = self.simulator.num_qubits();
118        let mut logical_x = engine_pauli_product(&output.x, width)?;
119        let mut logical_z = engine_pauli_product(&output.z, width)?;
120        logical_x.phase_shift(2 * i32::from(frame_z));
121        logical_z.phase_shift(2 * i32::from(frame_x));
122        let mut logical_y = &logical_x * &logical_z;
123        logical_y.phase_shift(1);
124        Ok((
125            self.simulator.peek_observable_expectation(&logical_x)?,
126            self.simulator.peek_observable_expectation(&logical_y)?,
127            self.simulator.peek_observable_expectation(&logical_z)?,
128        ))
129    }
130}

344/// Runtime cause for a dynamically inserted hold interval.
345///
346/// When blockers overlap, classification uses this conservative precedence:
347/// a direct physical sibling or resource is [`Self::Synchronization`], then an
348/// unfinished causal query is [`Self::CausalCut`], and only an issued decoder
349/// solve is [`Self::DecoderLatency`]. The reason is sampled when the whole
350/// interval is issued and does not change partway through a QEC round. Factory
351/// GAP rounds remain identified by [`MemoryKind::FactoryGap`].
352#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
353#[serde(rename_all = "snake_case")]
354pub enum WaitReason {
355    /// A directly joined physical sibling or factory resource is unfinished.
356    Synchronization,
357    /// Measurements or other causal inputs needed to issue a query are unfinished.
358    CausalCut,
359    /// Query inputs are complete and an issued decoder solve is still running.
360    DecoderLatency,
361}

558/// Failures on the *dynamic causal runtime* path.

560/// That path is [`run()`] and the timing, retry, noise, and mock-decoding

619fn validate_program(program: &Program) -> Result<(), RuntimeError> {
620    let tasks = program.tasks.len();
621    if let Some(last_task) = tasks.checked_sub(1) {
622        u32::try_from(last_task).map_err(|_| RuntimeError::IdOverflow("task"))?;
623    }
624    let bits = program.bit_count as usize;
625    let records = program.record_count as usize;
626    let qubits = program.qubit_count as usize;
627    if !program.task_origins.is_empty() && program.task_origins.len() != tasks {
628        return Err(RuntimeError::InvalidProgram("invalid task origin table"));
629    }
630    validate_stream(&program.entry, tasks, bits)?;
631    let mut input_tasks = HashSet::new();
632    for input in &program.inputs {
633        let mut data_qubits = HashSet::new();
634        if input.task as usize >= tasks
635            || program.tasks[input.task as usize].source != Some(SourceRole::LogicalInput)
636            || !input_tasks.insert(input.task)
637            || input
638                .data_qubits
639                .iter()
640                .any(|&qubit| qubit as usize >= qubits || !data_qubits.insert(qubit))
641        {
642            return Err(RuntimeError::InvalidProgram("invalid logical input"));
643        }
644        for stabilizer in &input.stabilizers {
645            validate_product(stabilizer, qubits)?;
646        }
647        validate_product(&input.x, qubits)?;
648        validate_product(&input.z, qubits)?;
649    }
650    for output in &program.outputs {
651        if output.frame_x as usize >= bits || output.frame_z as usize >= bits {
652            return Err(RuntimeError::InvalidProgram("invalid output frame"));
653        }
654        validate_product(&output.x, qubits)?;
655        validate_product(&output.z, qubits)?;
656    }
657    for task in &program.tasks {
658        if !valid_duration(task.release)
659            || !valid_duration(task.duration)
660            || task.dependencies.iter().any(|&id| id as usize >= tasks)
661            || task.activation.is_some_and(|id| id as usize >= bits)
662            || task.output.is_some_and(|id| id as usize >= bits)
663            || task.qubits.iter().any(|&id| id as usize >= qubits)
664        {
665            return Err(RuntimeError::InvalidProgram("invalid task operand"));
666        }
667        validate_instruction(&task.instruction, tasks, bits, records, qubits)?;
668    }
669    for task in &program.tasks {
670        if let Instruction::Rus { .. } = &task.instruction {
671            validate_rus_isolation(program, task)?;
672        }
673    }
674    Ok(())
675}

677fn validate_rus_isolation(
678    program: &Program,
679    task: &crate::instruction::Task,
680) -> Result<(), RuntimeError> {
681    let Instruction::Rus {
682        body,
683        owned_qubits,
684        retry_prepare,
685        decoder_hold,
686        cultivation_exits,
687        ..
688    } = &task.instruction
689    else {
690        return Ok(());
691    };
692    let owned = owned_qubits.iter().copied().collect::<HashSet<_>>();
693    let outside_operation = |operation: &QuantumOp| {
694        quantum_op_qubits(operation)
695            .iter()
696            .any(|qubit| !owned.contains(qubit))
697    };
698    let outside_stream = |stream: &QuantumStream| {
699        stream
700            .moments
701            .iter()
702            .flat_map(|moment| &moment.operations)
703            .any(outside_operation)
704    };
705    let outside_cycle = |cycle: &MemoryCycle| {
706        outside_stream(&cycle.stream)
707            || cycle
708                .boundary_flows
709                .iter()
710                .flat_map(|flow| flow.start.terms.iter().chain(flow.end.terms.iter()))
711                .any(|(qubit, _)| !owned.contains(qubit))
712    };
713    if owned.len() != owned_qubits.len()
714        || decoder_hold.is_empty() != cultivation_exits.is_empty()
715        || (task.source == Some(SourceRole::Factory) && decoder_hold.is_empty())
716    {
717        return Err(RuntimeError::InvalidProgram("invalid RUS isolation"));
718    }
719
720    if outside_stream(retry_prepare) {
721        return Err(RuntimeError::InvalidProgram(
722            "RUS retry preparation touches unowned support",
723        ));
724    }
725
726    let mut members = HashSet::new();
727    for &member in &body.tasks {
728        if !members.insert(member) {
729            continue;
730        }
731        let member_task = &program.tasks[member as usize];
732        let outside_instruction = match &member_task.instruction {
733            Instruction::Quantum(quantum) => quantum
734                .alternatives
735                .iter()
736                .any(|alternative| outside_stream(&alternative.stream)),
737            Instruction::MemoryRounds { stream, .. } => outside_stream(stream),
738            Instruction::Idle { operations, .. } => operations.iter().any(outside_operation),
739            Instruction::WaitFor { memory, .. } => memory.iter().any(outside_cycle),
740            Instruction::Bind(bindings) => bindings
741                .iter()
742                .flat_map(|binding| &binding.operator.terms)
743                .any(|(qubit, _)| !owned.contains(qubit)),
744            Instruction::Rus { .. } => {
745                return Err(RuntimeError::InvalidProgram(
746                    "nested dynamic RUS is unsupported",
747                ));
748            }
749            Instruction::Eval(_)
750            | Instruction::Accumulate(_)
751            | Instruction::ReadoutRecipe { .. }
752            | Instruction::Observable { .. }
753            | Instruction::Decode(_)
754            | Instruction::Discard(_)
755            | Instruction::SignalReady { .. } => false,
756        };
757        if member_task
758            .qubits
759            .iter()
760            .any(|qubit| !owned.contains(qubit))
761            || outside_instruction
762        {
763            return Err(RuntimeError::InvalidProgram(
764                "RUS body touches qubits outside its isolated source support",
765            ));
766        }
767    }
768    if cultivation_exits.iter().any(|exit| !members.contains(exit)) {
769        return Err(RuntimeError::InvalidProgram(
770            "RUS cultivation exit is outside its body",
771        ));
772    }
773    if decoder_hold.iter().any(outside_cycle) {
774        return Err(RuntimeError::InvalidProgram(
775            "RUS decoder hold has invalid data support",
776        ));
777    }
778    Ok(())
779}

781fn validate_stream(stream: &Stream, tasks: usize, bits: usize) -> Result<(), RuntimeError> {
782    if stream.tasks.iter().any(|&id| id as usize >= tasks)
783        || stream.bindings.iter().any(|&id| id as usize >= tasks)
784        || stream.value.is_some_and(|id| id as usize >= bits)
785    {
786        return Err(RuntimeError::InvalidProgram("invalid stream operand"));
787    }
788    Ok(())
789}

791fn validate_instruction(
792    instruction: &Instruction,
793    tasks: usize,
794    bits: usize,
795    records: usize,
796    qubits: usize,
797) -> Result<(), RuntimeError> {
798    match instruction {
799        Instruction::Quantum(quantum) => {
800            for alternative in &quantum.alternatives {
801                if alternative
802                    .when
803                    .iter()
804                    .any(|&(bit, _)| bit as usize >= bits)
805                {
806                    return Err(RuntimeError::InvalidProgram("invalid quantum selector"));
807                }
808                if alternative
809                    .selected_records
810                    .iter()
811                    .chain(alternative.excluded_records.iter())
812                    .any(|&record| record as usize >= records)
813                {
814                    return Err(RuntimeError::InvalidProgram("invalid quantum record"));
815                }
816                validate_quantum_stream(&alternative.stream, records, qubits)?;
817                for parity in alternative
818                    .detectors
819                    .iter()
820                    .chain(alternative.restarts.iter())
821                {
822                    validate_parity(parity, records)?;
823                }
824            }
825        }
826        Instruction::Eval(expression) | Instruction::Discard(expression) => {
827            validate_bool(expression, bits)?;
828        }
829        Instruction::Accumulate(parity) => validate_parity(parity, records)?,
830        Instruction::Bind(bindings) => {
831            for binding in bindings {
832                validate_product(&binding.operator, qubits)?;
833            }
834        }
835        Instruction::ReadoutRecipe {
836            bits: inputs,
837            bindings,
838        }
839        | Instruction::Observable {
840            bits: inputs,
841            bindings,
842            ..
843        } => {
844            if inputs.iter().any(|&id| id as usize >= bits)
845                || bindings.iter().any(|&id| id as usize >= tasks)
846            {
847                return Err(RuntimeError::InvalidProgram("invalid observable operand"));
848            }
849        }
850        Instruction::Decode(request) => {
851            if request.observable as usize >= tasks
852                || request.raw as usize >= bits
853                || !valid_duration(request.round_duration)
854            {
855                return Err(RuntimeError::InvalidProgram("invalid decoder operand"));
856            }
857        }
858        Instruction::Rus {
859            body,
860            restart,
861            owned_qubits,
862            attempt_bits,
863            attempt_records,
864            retry_prepare,
865            decoder_hold,
866            cultivation_exits,
867            ..
868        } => {
869            validate_stream(body, tasks, bits)?;
870            validate_bool(restart, bits)?;
871            if owned_qubits.iter().any(|&id| id as usize >= qubits)
872                || attempt_bits.iter().any(|&id| id as usize >= bits)
873                || attempt_records.iter().any(|&id| id as usize >= records)
874                || cultivation_exits.iter().any(|&id| id as usize >= tasks)
875            {
876                return Err(RuntimeError::InvalidProgram("invalid RUS operand"));
877            }
878            for cycle in decoder_hold {
879                validate_memory_cycle(cycle, records, qubits)?;
880            }
881            validate_quantum_stream(retry_prepare, records, qubits)?;
882        }
883        Instruction::WaitFor { until, memory } => {
884            if until.iter().any(|&id| id as usize >= tasks) {
885                return Err(RuntimeError::InvalidProgram("invalid wait operand"));
886            }
887            for cycle in memory {
888                validate_memory_cycle(cycle, records, qubits)?;
889            }
890        }
891        Instruction::SignalReady { .. } => {}
892        Instruction::MemoryRounds {
893            stream, detectors, ..
894        } => {
895            validate_quantum_stream(stream, records, qubits)?;
896            for parity in detectors {
897                validate_parity(parity, records)?;
898            }
899        }
900        Instruction::Idle {
901            duration,
902            noise_ratio,
903            operations,
904        } => {
905            if !valid_duration(*duration) || !valid_duration(*noise_ratio) {
906                return Err(RuntimeError::InvalidProgram("invalid idle timing"));
907            }
908            for operation in operations {
909                validate_quantum_op(operation, records, qubits)?;
910            }
911        }
912    }
913    Ok(())
914}
915
916fn validate_memory_cycle(
917    cycle: &MemoryCycle,
918    records: usize,
919    qubits: usize,
920) -> Result<(), RuntimeError> {
921    let stream_duration = stream_duration(&cycle.stream)?;
922    if stream_duration <= 0.0
923        || cycle.round_duration <= 0.0
924        || !cycle.round_duration.is_finite()
925        || (stream_duration - cycle.round_duration).abs() > TIME_EPSILON
926    {
927        return Err(RuntimeError::InvalidProgram("invalid memory duration"));
928    }
929    validate_quantum_stream(&cycle.stream, records, qubits)?;
930    for parity in &cycle.detectors {
931        validate_parity(parity, records)?;
932    }
933    for initializer in &cycle.initializers {
934        if initializer.record as usize >= records {
935            return Err(RuntimeError::InvalidProgram("invalid frontier record"));
936        }
937        validate_parity(&initializer.parity, records)?;
938    }
939    for flow in &cycle.boundary_flows {
940        validate_product(&flow.start, qubits)?;
941        validate_product(&flow.end, qubits)?;
942        validate_parity(&flow.parity, records)?;
943        if flow
944            .frontier_record
945            .is_some_and(|record| record as usize >= records)
946        {
947            return Err(RuntimeError::InvalidProgram("invalid frontier record"));
948        }
949    }
950    Ok(())
951}

953fn validate_quantum_stream(
954    stream: &QuantumStream,
955    records: usize,
956    qubits: usize,
957) -> Result<(), RuntimeError> {
958    for moment in &stream.moments {
959        if !valid_duration(moment.duration) {
960            return Err(RuntimeError::InvalidProgram("invalid moment duration"));
961        }
962        for operation in &moment.operations {
963            validate_quantum_op(operation, records, qubits)?;
964        }
965    }
966    Ok(())
967}
968
969fn validate_quantum_op(
970    operation: &QuantumOp,
971    records: usize,
972    qubits: usize,
973) -> Result<(), RuntimeError> {
974    let valid_qubit = |qubit: u32| (qubit as usize) < qubits;
975    let valid_record = |record: u32| (record as usize) < records;
976    let valid = match operation {
977        QuantumOp::Gate1 { qubit, .. }
978        | QuantumOp::Pauli { qubit, .. }
979        | QuantumOp::T { qubit, .. }
980        | QuantumOp::Reset { qubit, .. } => valid_qubit(*qubit),
981        QuantumOp::Gate2 {
982            control, target, ..
983        } => valid_qubit(*control) && valid_qubit(*target) && control != target,
984        QuantumOp::Measure {
985            observable,
986            records: outputs,
987            flip_probability,
988        } => {
989            validate_product(observable, qubits)?;
990            valid_probability(*flip_probability)
991                && !outputs.is_empty()
992                && outputs.iter().all(|&record| valid_record(record))
993        }
994        QuantumOp::ConditionalPauli { qubit, control, .. } => {
995            valid_qubit(*qubit) && valid_record(*control)
996        }
997        QuantumOp::Depolarize1 {
998            probability,
999            qubits: operands,
1000        }
1001        | QuantumOp::PauliError {
1002            probability,
1003            qubits: operands,
1004            ..
1005        } => valid_probability(*probability) && operands.iter().all(|&qubit| valid_qubit(qubit)),
1006        QuantumOp::Depolarize2 { probability, pairs } => {
1007            valid_probability(*probability)
1008                && pairs
1009                    .iter()
1010                    .all(|&(left, right)| left != right && valid_qubit(left) && valid_qubit(right))
1011        }
1012    };
1013    if valid {
1014        Ok(())
1015    } else {
1016        Err(RuntimeError::InvalidProgram("invalid quantum operand"))
1017    }
1018}

1020fn quantum_op_qubits(operation: &QuantumOp) -> Vec<u32> {
1021    match operation {
1022        QuantumOp::Gate1 { qubit, .. }
1023        | QuantumOp::Pauli { qubit, .. }
1024        | QuantumOp::T { qubit, .. }
1025        | QuantumOp::Reset { qubit, .. }
1026        | QuantumOp::ConditionalPauli { qubit, .. } => vec![*qubit],
1027        QuantumOp::Gate2 {
1028            control, target, ..
1029        } => vec![*control, *target],
1030        QuantumOp::Measure { observable, .. } => {
1031            observable.terms.iter().map(|&(qubit, _)| qubit).collect()
1032        }
1033        QuantumOp::Depolarize1 { qubits, .. } | QuantumOp::PauliError { qubits, .. } => {
1034            qubits.to_vec()
1035        }
1036        QuantumOp::Depolarize2 { pairs, .. } => pairs
1037            .iter()
1038            .flat_map(|&(left, right)| [left, right])
1039            .collect(),
1040    }
1041}

1043fn validate_product(product: &PauliProduct, qubits: usize) -> Result<(), RuntimeError> {
1044    let mut seen = HashSet::new();
1045    if product
1046        .terms
1047        .iter()
1048        .any(|&(qubit, _)| qubit as usize >= qubits || !seen.insert(qubit))
1049    {
1050        Err(RuntimeError::InvalidProgram("invalid Pauli operand"))
1051    } else {
1052        Ok(())
1053    }
1054}

1056fn validate_parity(parity: &RecordParity, records: usize) -> Result<(), RuntimeError> {
1057    if parity
1058        .records
1059        .iter()
1060        .any(|&record| record as usize >= records)
1061    {
1062        Err(RuntimeError::InvalidProgram("invalid record operand"))
1063    } else {
1064        Ok(())
1065    }
1066}

1068fn validate_bool(expression: &BoolOp, bits: usize) -> Result<(), RuntimeError> {
1069    match expression {
1070        BoolOp::Call { body, inputs } => {
1071            if inputs.iter().any(|&bit| bit as usize >= bits) {
1072                return Err(RuntimeError::InvalidProgram("invalid function argument"));
1073            }
1074            validate_bool(body, inputs.len())
1075        }
1076        BoolOp::Const(_) => Ok(()),
1077        BoolOp::Copy(bit) => ((*bit as usize) < bits)
1078            .then_some(())
1079            .ok_or(RuntimeError::InvalidProgram("invalid bit operand")),
1080        BoolOp::Not(inner) => validate_bool(inner, bits),
1081        BoolOp::Parity { inputs, .. } => inputs
1082            .iter()
1083            .all(|&bit| (bit as usize) < bits)
1084            .then_some(())
1085            .ok_or(RuntimeError::InvalidProgram("invalid bit operand")),
1086        BoolOp::Xor(operands) | BoolOp::And(operands) | BoolOp::Or(operands) => {
1087            for operand in operands {
1088                validate_bool(operand, bits)?;
1089            }
1090            Ok(())
1091        }
1092        BoolOp::Select {
1093            condition,
1094            when_false,
1095            when_true,
1096        } => {
1097            validate_bool(condition, bits)?;
1098            validate_bool(when_false, bits)?;
1099            validate_bool(when_true, bits)
1100        }
1101    }
1102}

1105    stream.moments.iter().try_fold(0.0, |duration, moment| {
1106        if valid_duration(moment.duration) {
1107            Ok(duration + moment.duration)
1108        } else {
1109            Err(RuntimeError::InvalidProgram("invalid moment duration"))
1110        }
1111    })

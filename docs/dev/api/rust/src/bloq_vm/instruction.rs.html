1//! Backend-independent instructions executed by the Bloq VM.
2//!
3//! The types in this module deliberately contain no Bloq IR node, graph, or
4//! expression types. [`crate::lower()`] resolves those into dense identifiers,
5//! executable quantum streams, and structured control flow once up front.
6#![allow(
7    missing_docs,
8    reason = "the ISA containers and gate vocabulary are documented at their owning types"
9)]
10
11/// Dense task identifier in [`Program::tasks`].
12pub type TaskId = u32;
13/// Dense classical bit register.
14pub type BitId = u32;
15/// Dense simulator qubit.
16pub type QubitId = u32;
17/// Dense measurement record.
18pub type RecordId = u32;
19/// Runtime readiness signal, normally produced by an accepted RUS region.
20pub type ResourceId = u32;
21
22/// Default mock streaming-decoder latency, in physical memory rounds.
23pub const DEFAULT_DECODER_LATENCY_ROUNDS: u32 = 10;
24/// A lowered executable program.
25#[derive(Debug, Clone, PartialEq, serde::Serialize)]
26pub struct Program {
27    /// Number of simulator qubits.
28    pub qubit_count: u32,
29    /// Number of classical bit registers.
30    pub bit_count: u32,
31    /// Number of statically named measurement records.
32    pub record_count: u32,
33    /// All tasks, addressed by [`TaskId`]. Nested streams index this same arena.
34    pub tasks: Vec<Task>,
35    /// Backend-independent source geometry, indexed like `tasks`. Hand-built
36    /// programs may leave this empty.
37    pub task_origins: Vec<TaskOrigin>,
38    /// Top-level executable stream.
39    pub entry: Stream,
40    /// External logical patches injected when their source task is released.
41    pub inputs: Vec<LogicalInput>,
42    /// Terminal logical operators and their Pauli-frame registers.
43    pub outputs: Vec<LogicalOutput>,
44    /// Configured mock-decoder latency, in physical memory rounds.
45    pub decoder_latency_rounds: u32,
46}
47
48impl Default for Program {
49    fn default() -> Self {
50        Self {
51            qubit_count: 0,
52            bit_count: 0,
53            record_count: 0,
54            tasks: Vec::new(),
55            task_origins: Vec::new(),
56            entry: Stream::default(),
57            inputs: Vec::new(),
58            outputs: Vec::new(),
59            decoder_latency_rounds: DEFAULT_DECODER_LATENCY_ROUNDS,
60        }
61    }
62}

119impl Program {
120    /// Every instruction, including instructions owned by nested streams.
121    pub fn instructions(&self) -> impl Iterator<Item = &Instruction> {
122        self.tasks.iter().map(|task| &task.instruction)
123    }
124
125    /// Independently released sources in dense task order.
126    #[must_use]
127    pub fn sources(&self) -> Vec<SourceSchedule> {
128        self.tasks
129            .iter()
130            .enumerate()
131            .filter_map(|(task, source)| {
132                source.source.map(|role| SourceSchedule {
133                    task: task as TaskId,
134                    role,
135                    release: source.release,
136                    duration: source.duration,
137                })
138            })
139            .collect()
140    }
141
142    /// Render the complete lowered instruction stream as compact JSON.
143    ///
144    /// This is an inspection format, not a stable persistence protocol.
145    ///
146    /// # Errors
147    ///
148    /// Propagates JSON serialization errors.
149    pub fn to_json(&self) -> Result<String, serde_json::Error> {
150        serde_json::to_string(self)
151    }
152
153    /// Render the complete lowered instruction stream as readable JSON.
154    ///
155    /// This is an inspection format, not a stable persistence protocol.
156    ///
157    /// # Errors
158    ///
159    /// Propagates JSON serialization errors.
160    pub fn to_json_pretty(&self) -> Result<String, serde_json::Error> {
161        serde_json::to_string_pretty(self)
162    }
163}

165/// A set of tasks scheduled together. Dependencies, rather than vector order,
166/// define readiness; order only makes inspection deterministic.
167#[derive(Debug, Clone, Default, PartialEq, Eq, serde::Serialize)]
168pub struct Stream {
169    pub tasks: Box<[TaskId]>,
170    /// Declared bit result of a region body. `None` is constant false.
171    pub value: Option<BitId>,
172    /// Tasks whose boundary bindings the region exports.
173    pub bindings: Box<[TaskId]>,
174}

176/// One schedulable unit.
177#[derive(Debug, Clone, PartialEq, serde::Serialize)]
178pub struct Task {
179    /// Stable inspection label derived from the IR scope, node id, and role.
180    pub label: String,
181    /// Absolute earliest start time.
182    pub release: f64,
183    /// Source role, present only on independently released roots.
184    pub source: Option<SourceRole>,
185    /// All predecessors, including Bloq value, order, and quantum edges and
186    /// sequencing added for physical qubit reuse.
187    pub dependencies: Box<[TaskId]>,
188    /// Optional classical gate. A false gate suppresses physical work and
189    /// decoder requests, clears bindings, and writes `false` to `output`.
190    pub activation: Option<BitId>,
191    /// Classical result register, when this task produces one.
192    pub output: Option<BitId>,
193    /// Physical support retained for scheduling and wait events.
194    pub qubits: Box<[QubitId]>,
195    /// Nominal active duration. Dynamic instructions may extend it.
196    pub duration: f64,
197    pub instruction: Instruction,
198}

219/// Epoch from which decoder latency is measured.
220#[derive(Debug, Clone, Copy, PartialEq, Eq, serde::Serialize)]
221pub enum DecodeTiming {
222    Measurement,
223    FactoryCompletion,
224}

226/// One decoder query against an already-lowered `Observable`.
227///
228/// Corrected and flip outputs share one sampled decision for a request.
229/// The shared payload keeps their query fields aligned. The runtime caches
230/// decisions by observable task and attempt epoch, and rejects inconsistent
231/// inputs for that causal key.
232#[derive(Debug, Clone, Copy, PartialEq, serde::Serialize)]
233pub struct DecodeRequest {
234    /// Boolean result of this causal solve.
235    pub output: bloq_ir::ObservableOutput,
236    /// Task id of the `Observable` this estimates.
237    pub observable: TaskId,
238    /// The observable's program-level index.
239    pub index: u32,
240    /// Register holding the raw measured parity.
241    pub raw: BitId,
242    /// Duration of one physical memory round, the unit of decoder latency.
243    pub round_duration: f64,
244    /// Epoch decoder latency is measured from.
245    pub timing: DecodeTiming,
246}

248/// VM instruction set. Structured bodies refer to the shared task arena.
249#[derive(Debug, Clone, PartialEq, serde::Serialize)]
250pub enum Instruction {
251    Quantum(QuantumTask),
252    Eval(BoolOp),
253    Accumulate(RecordParity),
254    Bind(Box<[BoundaryBinding]>),
255    /// Shared bit parity and boundary bindings; only a complete Observable reads them.
256    ReadoutRecipe {
257        bits: Box<[BitId]>,
258        bindings: Box<[TaskId]>,
259    },
260    Observable {
261        index: u32,
262        bits: Box<[BitId]>,
263        bindings: Box<[TaskId]>,
264    },
265    /// Publish one observable's corrected parity or decoder flip.
266    Decode(DecodeRequest),
267    Discard(BoolOp),
268    /// Repeat a source-isolated body without rolling back global state or time.
269    Rus {
270        body: Stream,
271        restart: BoolOp,
272        owned_qubits: Box<[QubitId]>,
273        attempt_bits: Box<[BitId]>,
274        attempt_records: Box<[RecordId]>,
275        resource: ResourceId,
276        /// Optional explicit preparation before attempts after the first. Compiled
277        /// factories replay authored body resets and leave this empty.
278        retry_prepare: QuantumStream,
279        /// Exact terminal seam cycles used while attempt decoders are pending.
280        decoder_hold: Box<[MemoryCycle]>,
281        /// Physical cultivation exits after which GAP rounds begin.
282        cultivation_exits: Box<[TaskId]>,
283    },
284    /// Hold a live patch until every task in `until` is ready. Execute as many
285    /// whole syndrome rounds as fit, then idle only for a sub-round remainder.
286    WaitFor {
287        /// Sibling producer exits whose active work must finish.
288        until: Box<[TaskId]>,
289        memory: Box<[MemoryCycle]>,
290    },
291    SignalReady {
292        resource: ResourceId,
293    },
294    /// Statically requested memory rounds. The stream already contains the
295    /// exact full `rounds` circuit and executes once; `rounds` is inspection
296    /// metadata.
297    MemoryRounds {
298        rounds: u32,
299        stream: QuantumStream,
300        detectors: Box<[RecordParity]>,
301    },
302    /// A physical hold shorter than one available syndrome round.
303    Idle {
304        duration: f64,
305        noise_ratio: f64,
306        operations: Box<[QuantumOp]>,
307    },
308}

310/// Boolean bytecode with absolute register operands, or argument slots within a
311/// shared [`BoolOp::Call`] body.
312///
313/// Evaluators must load every
314/// listed operand, including unselected `Select` arms and duplicate parity
315/// inputs.
316#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize)]
317pub enum BoolOp {
318    Const(bool),
319    Copy(BitId),
320    Not(Box<BoolOp>),
321    Parity {
322        inputs: Box<[BitId]>,
323        constant: bool,
324    },
325    Xor(Box<[BoolOp]>),
326    And(Box<[BoolOp]>),
327    Or(Box<[BoolOp]>),
328    Select {
329        condition: Box<BoolOp>,
330        when_false: Box<BoolOp>,
331        when_true: Box<BoolOp>,
332    },
333    /// Evaluate a shared pure body. Copies in `body` name positions in `inputs`;
334    /// the invocation binds those positions to registers in its enclosing scope.
335    /// Every argument is required, including arguments unused by the body.
336    /// Values and timestamps are evaluated independently per call.
337    Call {
338        body: std::sync::Arc<BoolOp>,
339        inputs: Box<[BitId]>,
340    },
341}

525mod tests {
526    use serde_json::json;
527
528    use super::*;
529
530    #[test]
531    fn program_json_exposes_physical_stream_and_schedule_metadata() {
532        let quantum = Task {
533            label: "source".into(),
534            release: 2.0,
535            source: Some(SourceRole::Clifford),
536            dependencies: Box::new([]),
537            activation: None,
538            output: None,
539            qubits: Box::new([0]),
540            duration: 1.0,
541            instruction: Instruction::Quantum(QuantumTask {
542                alternatives: Box::new([QuantumAlternative {
543                    stream: QuantumStream {
544                        moments: Box::new([Moment {
545                            kind: Some(MomentKind::Rotation),
546                            duration: 1.0,
547                            operations: Box::new([QuantumOp::Gate1 {
548                                gate: Clifford1::H,
549                                qubit: 0,
550                            }]),
551                        }]),
552                    },
553                    ..QuantumAlternative::default()
554                }]),
555            }),
556        };
557        let dependent = Task {
558            label: "ready".into(),
559            release: 0.0,
560            source: None,
561            dependencies: Box::new([0]),
562            activation: None,
563            output: None,
564            qubits: Box::new([]),
565            duration: 0.0,
566            instruction: Instruction::SignalReady { resource: 0 },
567        };
568        let decode = Task {
569            label: "decode".into(),
570            release: 0.0,
571            source: None,
572            dependencies: Box::new([0]),
573            activation: None,
574            output: Some(0),
575            qubits: Box::new([]),
576            duration: 0.0,
577            instruction: Instruction::Decode(DecodeRequest {
578                output: bloq_ir::ObservableOutput::Corrected,
579                observable: 1,
580                index: 0,
581                raw: 1,
582                round_duration: 1.5,
583                timing: DecodeTiming::Measurement,
584            }),
585        };
586        let product = PauliProduct {
587            negative: false,
588            terms: Box::new([(0, Pauli::Z)]),
589        };
590        let program = Program {
591            qubit_count: 1,
592            bit_count: 2,
593            tasks: vec![quantum, dependent, decode],
594            task_origins: vec![
595                TaskOrigin {
596                    function: TaskFunction::Cube,
597                    sites: Box::new([[3, 4]]),
598                    members: Box::new([[3, 4, 5]]),
599                },
600                TaskOrigin::default(),
601                TaskOrigin::default(),
602            ],
603            entry: Stream {
604                tasks: Box::new([0, 1, 2]),
605                ..Stream::default()
606            },
607            inputs: vec![LogicalInput {
608                port: [3, 4, 5],
609                task: 0,
610                data_qubits: Box::new([0]),
611                stabilizers: Box::new([]),
612                x: product.clone(),
613                z: product.clone(),
614            }],
615            outputs: vec![LogicalOutput {
616                port: [3, 4, 6],
617                x: product.clone(),
618                z: product,
619                frame_x: 0,
620                frame_z: 1,
621            }],
622            ..Program::default()
623        };
624
625        let value: serde_json::Value = serde_json::from_str(&program.to_json().unwrap()).unwrap();
626        assert_eq!(value["entry"]["tasks"], json!([0, 1, 2]));
627        assert_eq!(value["tasks"][0]["source"], "clifford");
628        assert_eq!(value["tasks"][1]["dependencies"], json!([0]));
629        assert_eq!(
630            value["tasks"][0]["instruction"]["Quantum"]["alternatives"][0]["stream"]["moments"][0]
631                ["operations"][0]["Gate1"],
632            json!({ "gate": "H", "qubit": 0 })
633        );
634        // `DecodeRequest` is a newtype payload, so its fields stay flat under
635        // the variant tag exactly as the old struct variant serialized them.
636        assert_eq!(
637            value["tasks"][2]["instruction"]["Decode"],
638            json!({
639                "output": "Corrected",
640                "observable": 1,
641                "index": 0,
642                "raw": 1,
643                "round_duration": 1.5,
644                "timing": "Measurement",
645            })
646        );
647        assert_eq!(value["task_origins"][0]["function"], "cube");
648        assert_eq!(value["inputs"][0]["task"], 0);
649        assert_eq!(value["outputs"][0]["port"], json!([3, 4, 6]));
650    }

1//! Translate a [`bloq_ir::circuit::GateType`] into the engine's instruction set.
2//!
3//! Single-qubit Cliffords map by name onto [`Gate1Q`], whose own
4//! [`images`](Gate1Q::images) tableau is the engine's definition of the gate.
5//! Tests pin the signed images against Stim's gate definitions.
6//!
7//! Two-qubit gates need no table at all. `GateType::two_qubit_bases` already
8//! names the `<A>C<B>` axis pair the engine's own two-qubit instruction is
9//! parameterized by, so the translation is the identity on it.
10
11use crate::backend::{Gate1Q, Instruction, PauliBasis};
12use bloq_ir::circuit::GateType;
13
14/// The instruction `gate` contributes on one operand, or `None` for
15/// [`GateType::I`], which has nothing to apply.
16///
17/// Two-qubit gates consume operand *pairs* and are translated by the caller;
18/// passing one here panics.
19pub(crate) fn single_qubit_instruction(gate: GateType, qubit: usize) -> Option<Instruction> {
20    let instruction = match gate {
21        GateType::I => return None,
22        // Resets.
23        GateType::RX => Instruction::Reset {
24            basis: PauliBasis::X,
25            qubit,
26        },
27        GateType::RY => Instruction::Reset {
28            basis: PauliBasis::Y,
29            qubit,
30        },
31        GateType::RZ => Instruction::Reset {
32            basis: PauliBasis::Z,
33            qubit,
34        },
35        // T family: axis from bloq's clifford_proxy (T→Z, T_YZ→X, T_XZ→Y).
36        GateType::T => t(PauliBasis::Z, qubit, false),
37        GateType::T_DAG => t(PauliBasis::Z, qubit, true),
38        GateType::T_YZ => t(PauliBasis::X, qubit, false),
39        GateType::T_YZ_DAG => t(PauliBasis::X, qubit, true),
40        GateType::T_XZ => t(PauliBasis::Y, qubit, false),
41        GateType::T_XZ_DAG => t(PauliBasis::Y, qubit, true),
42        other => Instruction::Gate1 {
43            gate: single_qubit_clifford(other),
44            qubit,
45        },
46    };
47    Some(instruction)
48}
49
50fn t(basis: PauliBasis, qubit: usize, adjoint: bool) -> Instruction {
51    Instruction::T {
52        basis,
53        qubit,
54        adjoint,
55    }
56}
57
58/// The engine gate a single-qubit Clifford `GateType` names.
59///
60/// One-to-one and by name: the two enums carry the same stim gate set, spelled
61/// with underscores in the IR and in camel case in the engine.
62pub(crate) fn single_qubit_clifford(gate: GateType) -> Gate1Q {
63    match gate {
64        GateType::X => Gate1Q::X,
65        GateType::Y => Gate1Q::Y,
66        GateType::Z => Gate1Q::Z,
67        GateType::H => Gate1Q::H,
68        GateType::H_XY => Gate1Q::Hxy,
69        GateType::H_YZ => Gate1Q::Hyz,
70        GateType::H_NXY => Gate1Q::Hnxy,
71        GateType::H_NXZ => Gate1Q::Hnxz,
72        GateType::H_NYZ => Gate1Q::Hnyz,
73        GateType::SQRT_X => Gate1Q::SqrtX,
74        GateType::SQRT_X_DAG => Gate1Q::SqrtXDag,
75        GateType::SQRT_Y => Gate1Q::SqrtY,
76        GateType::SQRT_Y_DAG => Gate1Q::SqrtYDag,
77        GateType::S => Gate1Q::S,
78        GateType::S_DAG => Gate1Q::SDag,
79        GateType::C_XYZ => Gate1Q::Cxyz,
80        GateType::C_ZYX => Gate1Q::Czyx,
81        GateType::C_NXYZ => Gate1Q::Cnxyz,
82        GateType::C_XNYZ => Gate1Q::Cxnyz,
83        GateType::C_XYNZ => Gate1Q::Cxynz,
84        GateType::C_NZYX => Gate1Q::Cnzyx,
85        GateType::C_ZNYX => Gate1Q::Cznyx,
86        GateType::C_ZYNX => Gate1Q::Czynx,
87        other => unreachable!("non-single-qubit-Clifford gate {other:?} reached the gate table"),
88    }
89}

91#[cfg(test)]
92mod tests {
93    use super::{single_qubit_clifford, single_qubit_instruction};
94    use crate::backend::{Instruction, PauliBasis};
95    use bloq_ir::circuit::GateType;
96
97    /// Pin the `GateType` → [`Gate1Q`](crate::backend::Gate1Q) mapping by its
98    /// fingerprint: each gate's `(X→, Z→)` images, which equal stim's tableau
99    /// (`Tableau.from_named_gate`). The two enums are spelled almost
100    /// identically, so a mistyped arm is exactly the failure this catches; the
101    /// engine's own frame composition is pinned in `ticit`, not here.
102    #[test]
103    fn single_qubit_tableaux_match_stim() {
104        use PauliBasis::{X, Y, Z};
105        // (gate, X-image (axis, negated), Z-image (axis, negated)).
106        type Row = (GateType, (PauliBasis, bool), (PauliBasis, bool));
107        let cases: &[Row] = &[
108            (GateType::X, (X, false), (Z, true)),
109            (GateType::Y, (X, true), (Z, true)),
110            (GateType::Z, (X, true), (Z, false)),
111            (GateType::H, (Z, false), (X, false)),
112            (GateType::H_XY, (Y, false), (Z, true)),
113            (GateType::H_YZ, (X, true), (Y, false)),
114            (GateType::H_NXY, (Y, true), (Z, true)),
115            (GateType::H_NXZ, (Z, true), (X, true)),
116            (GateType::H_NYZ, (X, true), (Y, true)),
117            (GateType::SQRT_X, (X, false), (Y, true)),
118            (GateType::SQRT_X_DAG, (X, false), (Y, false)),
119            (GateType::SQRT_Y, (Z, true), (X, false)),
120            (GateType::SQRT_Y_DAG, (Z, false), (X, true)),
121            (GateType::S, (Y, false), (Z, false)),
122            (GateType::S_DAG, (Y, true), (Z, false)),
123            (GateType::C_XYZ, (Y, false), (X, false)),
124            (GateType::C_ZYX, (Z, false), (Y, false)),
125            (GateType::C_NXYZ, (Y, true), (X, true)),
126            (GateType::C_XNYZ, (Y, true), (X, false)),
127            (GateType::C_XYNZ, (Y, false), (X, true)),
128            (GateType::C_NZYX, (Z, true), (Y, true)),
129            (GateType::C_ZNYX, (Z, false), (Y, true)),
130            (GateType::C_ZYNX, (Z, true), (Y, false)),
131        ];
132        for &(gate, x_image, z_image) in cases {
133            assert_eq!(
134                single_qubit_clifford(gate).images(),
135                (x_image, z_image),
136                "{gate:?}: tableau mismatch vs stim"
137            );
138        }
139    }
140
141    /// Every gate the executor can meet on a one-qubit operand must translate.
142    /// A new `GateType` that reaches the `unreachable!` arm of the table fails
143    /// here rather than at the first shot that happens to use it.
144    #[test]
145    fn every_single_qubit_gate_translates() {
146        for gate in GateType::iter().filter(|gate| !gate.is_two_qubit_gate()) {
147            let instruction = single_qubit_instruction(gate, 0);
148            assert_eq!(
149                instruction.is_none(),
150                gate == GateType::I,
151                "{gate:?}: only the identity contributes no instruction"
152            );
153        }
154    }
155
156    /// The non-Clifford arms pick an axis that is *not* the gate's spelling
157    /// (`T_YZ` rotates about X, `T_XZ` about Y — bloq's `clifford_proxy`
158    /// convention), so unlike the Clifford arms they cannot be checked by name.
159    #[test]
160    fn reset_and_t_arms_pick_the_right_axis() {
161        use PauliBasis::{X, Y, Z};
162        let reset = |basis| Some(Instruction::Reset { basis, qubit: 3 });
163        let t = |basis, adjoint| {
164            Some(Instruction::T {
165                basis,
166                qubit: 3,
167                adjoint,
168            })
169        };
170        let cases = [
171            (GateType::RX, reset(X)),
172            (GateType::RY, reset(Y)),
173            (GateType::RZ, reset(Z)),
174            (GateType::T, t(Z, false)),
175            (GateType::T_DAG, t(Z, true)),
176            (GateType::T_YZ, t(X, false)),
177            (GateType::T_YZ_DAG, t(X, true)),
178            (GateType::T_XZ, t(Y, false)),
179            (GateType::T_XZ_DAG, t(Y, true)),
180        ];
181        for (gate, expected) in cases {
182            assert_eq!(single_qubit_instruction(gate, 3), expected, "{gate:?}");
183        }
184    }

10/// Cached Boolean queries over one immutable graph level.
11///
12/// Shared Compute ancestry preserves correlated guards. The graph must be
13/// acyclic and predicates must name existing bit producers. The borrow prevents
14/// graph edits from invalidating retained predicates and query results.
15#[derive(Debug)]
16pub struct PredicateAnalysis<'a> {
17    graph: &'a SubGraph,
18    diagram: BooleanDecisionDiagram,
19    memo: FxMap<ValueRef, DecisionId>,
20    variables: FxMap<ValueRef, usize>,
21    inputs: FxMap<BloqNodeId, FxMap<u32, ValueRef>>,
22    aliases: FxMap<ValueRef, (ValueRef, bool)>,
23    implications: FxMap<(Option<ValueRef>, ValueRef), bool>,
24    overlaps: FxMap<(Option<ValueRef>, Option<ValueRef>), bool>,
25}

34    /// Recognize a component whose member guards partition the input domain.
35    /// Side-table guards must be fixed by each choice; independent selections
36    /// still require pinning before timing or seam edits.
37    pub(crate) fn quantum_selections(
38        &mut self,
39        node: BloqNodeId,
40    ) -> Result<Option<Vec<QuantumSelection>>, Error> {
41        let quantum = self.graph[node].expect_quantum();
42        let mut choices = Vec::new();
43        let mut covered = DECISION_FALSE;
44        for guard in &quantum.guards {
45            if guard.instances.is_empty() {
46                continue;
47            }
48            let value = self.lower(Eval::Input(node, guard.input))?;
49            if value == DECISION_FALSE || choices.iter().any(|&(_, old)| old == value) {
50                continue;
51            }
52            if self.diagram.apply(BooleanOp::And, covered, value)? != DECISION_FALSE {
53                return Ok(None);
54            }
55            covered = self.diagram.apply(BooleanOp::Or, covered, value)?;
56            choices.push((guard.input, value));
57        }
58        if covered != DECISION_TRUE {
59            return Ok(None);
60        }
61        let mut selections = Vec::with_capacity(choices.len());
62        for (input, choice) in choices {
63            let mut values = FxMap::default();
64            for guard in &quantum.guards {
65                let value = self.lower(Eval::Input(node, guard.input))?;
66                let value = match self.diagram.constrain(value, choice)? {
67                    DECISION_FALSE => false,
68                    DECISION_TRUE => true,
69                    _ => return Ok(None),
70                };
71                values.insert(guard.input, value);
72            }
73            selections.push(QuantumSelection { input, values });
74        }
75        Ok(Some(selections))
76    }

106    pub(crate) fn using_work<T>(
107        &mut self,
108        work: &mut BooleanDecisionDiagram,
109        query: impl FnOnce(&mut Self) -> Result<T, Error>,
110    ) -> Result<T, Error> {
111        self.charge(work.steps().saturating_sub(self.steps()))?;

113        work.charge(self.steps().saturating_sub(work.steps()))?;
114        Ok(value)

131    /// Whether selected output guards overlap.
132    ///
133    /// # Errors
134    /// Returns an error for malformed predicates or exhausted Boolean resources.
135    pub fn overlap_values(
136        &mut self,
137        left: Option<ValueRef>,
138        right: Option<ValueRef>,
139    ) -> Result<bool, Error> {
140        self.charge(1)?;
141        let key = (left.min(right), left.max(right));
142        if let Some(&result) = self.overlaps.get(&key) {
143            return Ok(result);
144        }
145        let left = self.guard(left)?;
146        let right = self.guard(right)?;
147        let result = self.diagram.apply(BooleanOp::And, left, right)? != DECISION_FALSE;
148        self.overlaps.insert(key, result);
149        Ok(result)
150    }

152    /// Whether all required predicates can hold while every excluded
153    /// conjunction is false. Empty requirements are unconditional; an empty
154    /// excluded conjunction rules out every assignment.
155    ///
156    /// Shared Boolean ancestry is retained without enumerating branch choices.
157    ///
158    /// # Errors
159    /// Returns malformed predicate dataflow or exhausted Boolean resources.
160    pub fn coexist_without(
161        &mut self,
162        required: &[ValueRef],
163        excluded: &[Vec<ValueRef>],
164    ) -> Result<bool, Error> {
165        self.charge(
166            required
167                .len()
168                .saturating_add(excluded.len())
169                .saturating_add(1),
170        )?;
171        let mut domain = self.conjunction(required)?;
172        for conjunction in excluded {
173            self.charge(conjunction.len())?;
174            let active = self.conjunction(conjunction)?;
175            let inactive = self.diagram.negate(active)?;
176            domain = self.diagram.apply(BooleanOp::And, domain, inactive)?;
177            if domain == DECISION_FALSE {
178                return Ok(false);
179            }
180        }
181        Ok(domain != DECISION_FALSE)
182    }

184    fn conjunction(&mut self, predicates: &[ValueRef]) -> Result<DecisionId, Error> {
185        let mut result = DECISION_TRUE;
186        for &predicate in predicates {
187            let value = self.guard(Some(predicate))?;
188            result = self.diagram.apply(BooleanOp::And, result, value)?;
189        }
190        Ok(result)
191    }

208    pub(crate) fn implies_values(
209        &mut self,
210        left: Option<ValueRef>,
211        right: ValueRef,
212    ) -> Result<bool, Error> {
213        self.charge(1)?;
214        if let Some(&result) = self.implications.get(&(left, right)) {
215            return Ok(result);
216        }
217        if self.proves_conjunct(left.as_slice(), right)? {
218            self.implications.insert((left, right), true);
219            return Ok(true);
220        }
221        let source = self.guard(left)?;
222        let target = self.guard(Some(right))?;
223        let absent = self.diagram.negate(target)?;
224        let result = self.diagram.apply(BooleanOp::And, source, absent)? == DECISION_FALSE;
225        self.implications.insert((left, right), result);
226        Ok(result)
227    }

229    pub(crate) fn implies_all_values(
230        &mut self,
231        left: &[ValueRef],
232        right: ValueRef,
233    ) -> Result<bool, Error> {
234        if left.len() < 2 {
235            return self.implies_values(left.first().copied(), right);
236        }
237        self.charge(left.len())?;
238        if self.proves_conjunct(left, right)? {
239            return Ok(true);
240        }
241        let source = self.conjunction(left)?;
242        let target = self.guard(Some(right))?;
243        let absent = self.diagram.negate(target)?;
244        Ok(self.diagram.apply(BooleanOp::And, source, absent)? == DECISION_FALSE)
245    }

247    // Return the representative's truth when this value is true; compress shared paths.
248    fn normalize_alias(&mut self, mut value: ValueRef) -> Result<(ValueRef, bool), Error> {
249        let mut truth = true;
250        let mut path = FxMap::default();
251        let graph = self.graph;
252        'aliases: loop {
253            self.charge(1)?;
254            if let Some(&(canonical, positive)) = self.aliases.get(&value) {
255                truth = truth == positive;
256                value = canonical;
257                break;
258            }
259            if path.insert(value, truth).is_some() {
260                return Err(Error::InvalidMembership("cyclic predicate ancestry".into()));
261            }
262            let node = graph
263                .node(value.node)
264                .ok_or_else(|| Error::InvalidMembership(format!("unknown predicate {value:?}")))?;
265            if node.activation.is_some() || value.output != ObservableOutput::Corrected {
266                break;
267            }
268            let Some(ClassicalNode::Compute { expr }) = node.try_classical() else {
269                break;
270            };
271            let mut expr = expr;
272            let mut inverted = false;
273            let slot = loop {
274                self.charge(1)?;
275                match expr {
276                    ClassicalExpr::In(slot) => break *slot,
277                    ClassicalExpr::Not(inner) => {
278                        inverted = !inverted;
279                        expr = inner;
280                    }
281                    ClassicalExpr::Parity { inputs, constant } if inputs.len() == 1 => {
282                        inverted ^= *constant;
283                        break inputs[0];
284                    }
285                    _ => break 'aliases,
286                }
287            };
288            truth ^= inverted;
289            value = self.input(value.node, slot)?;
290        }
291        self.charge(path.len())?;
292        self.aliases.extend(
293            path.into_iter()
294                .map(|(alias, positive)| (alias, (value, truth == positive))),
295        );
296        Ok((value, truth))
297    }

299    // Each part records a truth forced by the source conjunction. This proves
300    // Boolean implication; full validation still checks every operand's dataflow.
301    fn proves_conjunct(&mut self, roots: &[ValueRef], target: ValueRef) -> Result<bool, Error> {
302        enum Part<'a> {
303            Value(ValueRef, bool),
304            Expr(BloqNodeId, &'a ClassicalExpr, bool),
305        }
306        if roots.is_empty() {
307            return Ok(false);
308        }
309        let (target, target_truth) = self.normalize_alias(target)?;
310        self.charge(roots.len())?;
311        let mut pending = roots
312            .iter()
313            .rev()
314            .copied()
315            .map(|value| Part::Value(value, true))
316            .collect::<Vec<_>>();
317        let mut seen = crate::FxSet::default();
318        let graph = self.graph;
319        while let Some(part) = pending.pop() {
320            self.charge(1)?;
321            match part {
322                Part::Value(value, truth) => {
323                    let (value, positive) = self.normalize_alias(value)?;
324                    let truth = truth == positive;
325                    let node = graph.node(value.node).ok_or_else(|| {
326                        Error::InvalidMembership(format!("unknown predicate {value:?}"))
327                    })?;
328                    if value == target && truth == target_truth {
329                        return Ok(true);
330                    }
331                    if !seen.insert((value, truth)) {
332                        continue;
333                    }
334                    // A false activated value need not have a false payload.
335                    if node.activation.is_some() && !truth {
336                        continue;
337                    }
338                    if value.output == ObservableOutput::Corrected
339                        && let Some(ClassicalNode::Compute { expr }) = node.try_classical()
340                    {
341                        pending.push(Part::Expr(value.node, expr, truth));
342                    }
343                    if let Some(slot) = node.activation {
344                        pending.push(Part::Value(self.input(value.node, slot)?, true));
345                    }
346                }
347                Part::Expr(node, ClassicalExpr::In(slot), truth) => {
348                    pending.push(Part::Value(self.input(node, *slot)?, truth));
349                }
350                Part::Expr(node, ClassicalExpr::Not(inner), truth) => {
351                    pending.push(Part::Expr(node, inner, !truth));
352                }
353                Part::Expr(node, ClassicalExpr::Parity { inputs, constant }, truth)
354                    if inputs.len() == 1 =>
355                {
356                    pending.push(Part::Value(self.input(node, inputs[0])?, truth ^ *constant));
357                }
358                Part::Expr(node, ClassicalExpr::And(operands), true) => {
359                    self.charge(operands.len())?;
360                    pending.extend(
361                        operands
362                            .iter()
363                            .rev()
364                            .map(|expr| Part::Expr(node, expr, true)),
365                    );
366                }
367                Part::Expr(node, ClassicalExpr::Or(operands), false) => {
368                    self.charge(operands.len())?;
369                    pending.extend(
370                        operands
371                            .iter()
372                            .rev()
373                            .map(|expr| Part::Expr(node, expr, false)),
374                    );
375                }
376                Part::Expr(node, ClassicalExpr::Select(operands), truth) => {
377                    self.charge(2)?;
378                    let [condition, when_false, when_true] = operands.as_ref();
379                    if matches!(when_true, ClassicalExpr::Const(bit) if *bit != truth) {
380                        pending.push(Part::Expr(node, when_false, truth));
381                        pending.push(Part::Expr(node, condition, false));
382                    } else if matches!(when_false, ClassicalExpr::Const(bit) if *bit != truth) {
383                        pending.push(Part::Expr(node, when_true, truth));
384                        pending.push(Part::Expr(node, condition, true));
385                    }
386                }
387                Part::Expr(_, ClassicalExpr::Const(bit), truth) if *bit != truth => {
388                    return Ok(true);
389                }
390                Part::Expr(_, _, _) => {}
391            }
392        }
393        Ok(false)
394    }

396    fn input(&mut self, node: BloqNodeId, slot: u32) -> Result<ValueRef, Error> {
397        if !self.inputs.contains_key(&node) {
398            let mut inputs = FxMap::default();
399            for input in self.graph.value_inputs(node) {
400                self.diagram.charge(1)?;
401                inputs.insert(input.slot, input.value_ref().expect("runtime input"));
402            }
403            self.inputs.insert(node, inputs);
404        }
405        self.inputs[&node]
406            .get(&slot)
407            .copied()
408            .ok_or(Error::MissingMembershipInput(slot))
409    }

411    // Both expression nesting and shared Compute ancestry use one explicit
412    // stack. A long predicate chain cannot consume the native call stack.
413    fn lower(&mut self, first: Eval<'_>) -> Result<DecisionId, Error> {
414        self.charge(1)?;
415        let mut pending = vec![first];
416        let mut values = Vec::new();
417        let mut visiting = crate::FxSet::default();
418        while let Some(next) = pending.pop() {
419            self.charge(1)?;
420            match next {
421                Eval::Node(node) => {
422                    if let Some(&value) = self.memo.get(&node) {
423                        values.push(value);
424                        continue;
425                    }
426                    if !visiting.insert(node) {
427                        return Err(Error::InvalidMembership("cyclic predicate ancestry".into()));
428                    }
429                    let weight = self.graph.node(node.node).ok_or_else(|| {
430                        Error::InvalidMembership(format!("unknown predicate {node:?}"))
431                    })?;
432                    pending.push(Eval::Store(node));
433                    if let Some(slot) = weight.activation {
434                        pending.push(Eval::Apply(BooleanOp::And, 2, DECISION_TRUE));
435                        pending.push(Eval::Input(node.node, slot));
436                    }
437                    if node.output == ObservableOutput::Corrected
438                        && let Some(ClassicalNode::Compute { expr }) = weight.try_classical()
439                    {
440                        pending.push(Eval::Expr(node.node, expr));
441                    } else {
442                        // Keep related predicate inputs together. Registry node ids
443                        // follow physical emission and can separate correlated bits,
444                        // making a compact predicate exponential in that ordering.
445                        let next = self.variables.len();
446                        let variable = *self.variables.entry(node).or_insert(next);
447                        values.push(self.diagram.make_node(
448                            variable,
449                            DECISION_FALSE,
450                            DECISION_TRUE,
451                        )?);
452                    }
453                }
454                Eval::Store(node) => {
455                    let value = *values.last().expect("evaluated node");
456                    self.memo.insert(node, value);
457                    visiting.remove(&node);
458                }
459                Eval::Input(node, slot) => {
460                    let producer = self.input(node, slot)?;
461                    pending.push(Eval::Node(producer));
462                }
463                Eval::Expr(node, expr) => match expr {
464                    ClassicalExpr::Const(value) => {
465                        values.push(if *value {
466                            DECISION_TRUE
467                        } else {
468                            DECISION_FALSE
469                        });
470                    }
471                    ClassicalExpr::In(slot) => pending.push(Eval::Input(node, *slot)),
472                    ClassicalExpr::Not(inner) => {
473                        pending.push(Eval::Negate);
474                        pending.push(Eval::Expr(node, inner));
475                    }
476                    ClassicalExpr::Parity { inputs, constant } => {
477                        self.charge(inputs.len())?;
478                        pending.push(Eval::Apply(
479                            BooleanOp::Xor,
480                            inputs.len(),
481                            if *constant {
482                                DECISION_TRUE
483                            } else {
484                                DECISION_FALSE
485                            },
486                        ));
487                        pending.extend(inputs.iter().rev().map(|&slot| Eval::Input(node, slot)));
488                    }
489                    ClassicalExpr::Xor(operands)
490                    | ClassicalExpr::And(operands)
491                    | ClassicalExpr::Or(operands) => {
492                        self.charge(operands.len())?;
493                        let op = match expr {
494                            ClassicalExpr::Xor(..) => BooleanOp::Xor,
495                            ClassicalExpr::And(..) => BooleanOp::And,
496                            _ => BooleanOp::Or,
497                        };
498                        pending.push(Eval::Apply(
499                            op,
500                            operands.len(),
501                            if op == BooleanOp::And {
502                                DECISION_TRUE
503                            } else {
504                                DECISION_FALSE
505                            },
506                        ));
507                        pending.extend(operands.iter().rev().map(|expr| Eval::Expr(node, expr)));
508                    }
509                    ClassicalExpr::Select(operands) => {
510                        pending.push(Eval::Select);
511                        pending.extend(operands.iter().rev().map(|expr| Eval::Expr(node, expr)));
512                    }
513                },
514                Eval::Apply(op, count, mut value) => {
515                    for _ in 0..count {
516                        value = self
517                            .diagram
518                            .apply(op, value, values.pop().expect("operand"))?;
519                    }
520                    values.push(value);
521                }
522                Eval::Negate => {
523                    let value = values.pop().expect("negation operand");
524                    values.push(self.diagram.negate(value)?);
525                }
526                Eval::Select => {
527                    let high = values.pop().expect("true operand");
528                    let low = values.pop().expect("false operand");
529                    let condition = values.pop().expect("selection predicate");
530                    let absent = self.diagram.negate(condition)?;
531                    let low = self.diagram.apply(BooleanOp::And, absent, low)?;
532                    let high = self.diagram.apply(BooleanOp::And, condition, high)?;
533                    values.push(self.diagram.apply(BooleanOp::Or, low, high)?);
534                }
535            }
536        }
537        Ok(values.pop().expect("evaluated predicate"))
538    }

542    /// Terminal choices share fixed input seams, so their waits can be composed
543    /// before selection. Changing seams or continuing membership still needs a
544    /// pinned path. Activated quantum regions also retain that requirement.
545    pub(crate) fn timing_requires_pins(&self) -> Result<bool, Error> {
546        fn guarded(level: &SubGraph, under_activation: bool) -> Result<bool, Error> {
547            let mut predicates = PredicateAnalysis::new(level);
548            for (id, node) in level.nodes() {
549                if let Some(quantum) = node.try_quantum() {
550                    if under_activation {
551                        return Ok(true);
552                    }
553                    if !quantum.guards.is_empty()
554                        && (level
555                            .outgoing(id)
556                            .any(|edge| matches!(edge.edge, crate::BloqEdge::Quantum(_)))
557                            || predicates.quantum_selections(id)?.is_none())
558                    {
559                        return Ok(true);
560                    }
561                }
562                if let Some(region) = node.try_region() {
563                    for (_, body) in region.bodies() {
564                        if guarded(body, under_activation || node.activation.is_some())? {
565                            return Ok(true);
566                        }
567                    }
568                }
569            }
570            Ok(level.edges().any(|edge| {
571                matches!(edge.edge, crate::BloqEdge::Quantum(quantum) if quantum.guard.is_some())
572            }))
573        }
574        guarded(self.top(), false)
575    }
576}

588/// SEM-MERGE legality is pairwise: aligned tick/repeat shapes, a unique partner
589/// for every reused body, and compatible operations on shared qubits. Checking
590/// every coexisting pair therefore certifies every selected set without listing
591/// its Boolean assignments. Record availability is checked separately by WF-7.
592pub(crate) fn validate_merge(
593    graph: &SubGraph,
594    id: BloqNodeId,
595    templates: &crate::BloqTemplatePool,
596    options: crate::lowering::InstantiationOptions<'_>,
597    predicates: &mut PredicateAnalysis<'_>,
598) -> Result<(), Error> {
599    let node = &graph[id];
600    let quantum = node.expect_quantum();
601    predicates.charge(
602        quantum
603            .instances
604            .len()
605            .saturating_add(quantum.guards.len())
606            .saturating_add(quantum.detector_bundles.len()),
607    )?;
608    for guard in &quantum.guards {
609        predicates.charge(
610            guard
611                .instances
612                .len()
613                .saturating_add(guard.detectors.len())
614                .saturating_add(guard.detector_bundles.len())
615                .saturating_add(guard.restarts.len())
616                .saturating_add(guard.detector_parities.len())
617                .saturating_add(guard.restart_parities.len()),
618        )?;
619    }
620    for parity in quantum.stored_parities() {
621        predicates.charge(parity.terms().len().saturating_add(1))?;
622    }
623    // Validate registration indices and uniqueness even for unreachable guards.
624    node.select_quantum_members(|_| Some(false))?;
625    let mut guards = FxMap::default();
626    for guard in &quantum.guards {
627        let producer = predicates.input(id, guard.input)?;
628        let value = predicates.guard(Some(producer))?;
629        for &instance in &guard.instances {
630            guards.insert(instance, value);
631        }
632    }
633    let mut pair = crate::BloqNode::from_members(Vec::new());
634    pair.provenance = node.provenance.clone();
635    let mut costs = FxMap::default();
636    let mut singleton = crate::FxSet::default();
637    let mut groups: FxMap<crate::TemplateId, (DecisionId, DecisionId)> = FxMap::default();
638    let mut shapes = crate::BloqTemplatePool::new();
639    let mut shape_ids = FxMap::default();
640    for instance in &quantum.instances {
641        let template = templates
642            .get(instance.template_id)
643            .ok_or(Error::UnknownTemplate(instance.template_id))?;
644        if let std::collections::hash_map::Entry::Vacant(entry) = costs.entry(instance.template_id)
645        {
646            let cost = crate::instantiation::circuit_work(&template.circuit);
647            predicates.charge(cost)?;
648            entry.insert(cost);
649            shape_ids.insert(
650                instance.template_id,
651                shapes.insert(crate::BloqTemplate::new(circuit_shape(&template.circuit))),
652            );
653        }
654        // Translation is the only instance-specific singleton check. Reuse the
655        // source footprint and materialize each template/provenance class once.
656        predicates.charge(template.qubits().len())?;
657        for &qubit in template.qubits() {
658            bloq_circuit::checked_translate_coordinate(qubit, instance.offset)?;
659        }
660        if singleton.insert((
661            instance.template_id,
662            instance.provenance.is_ideal_boundary(),
663        )) {
664            predicates.charge(costs[&instance.template_id])?;
665            pair.expect_quantum_mut().instances = vec![*instance];
666            pair.emission_plan_with_options(templates, options)?;
667        }
668        let value = guards.get(&instance.id).copied().unwrap_or(DECISION_TRUE);
669        // Keep the domains with at least one/two selected copies. The latter
670        // needs the pair merge's structural limits even at disjoint placements.
671        let (one, two) = groups
672            .entry(instance.template_id)
673            .or_insert((DECISION_FALSE, DECISION_FALSE));
674        let another = predicates.diagram.apply(BooleanOp::And, *one, value)?;
675        *two = predicates.diagram.apply(BooleanOp::Or, *two, another)?;
676        *one = predicates.diagram.apply(BooleanOp::Or, *one, value)?;
677    }
678    let groups: Vec<_> = groups
679        .into_iter()
680        .filter(|&(_, (guard, _))| guard != DECISION_FALSE)
681        .collect();
682    // Coexisting templates must align even when all their placements are
683    // disjoint. Strip coordinates, then use the same merge for shape legality.
684    for (index, &(left, (left_guard, multiple))) in groups.iter().enumerate() {
685        for &(right, (right_guard, _)) in groups.iter().take(index + 1) {
686            predicates.charge(1)?;
687            let coexisting = if left == right {
688                multiple
689            } else {
690                predicates
691                    .diagram
692                    .apply(BooleanOp::And, left_guard, right_guard)?
693            };
694            if coexisting == DECISION_FALSE {
695                continue;
696            }
697            predicates.charge(costs[&left].saturating_add(costs[&right]))?;
698            pair.expect_quantum_mut().instances = vec![
699                crate::TemplateInstance::new(
700                    crate::TemplateInstanceId(0),
701                    shape_ids[&left],
702                    glam::IVec2::ZERO,
703                ),
704                crate::TemplateInstance::new(
705                    crate::TemplateInstanceId(1),
706                    shape_ids[&right],
707                    glam::IVec2::ZERO,
708                ),
709            ];
710            pair.emission_plan(&shapes)?;
711        }
712    }
713
714    let mut occupants: FxMap<glam::IVec2, Vec<usize>> = FxMap::default();
715    let mut candidate_pairs = crate::FxSet::default();
716    let mut checked = crate::FxSet::default();
717    for (index, left) in quantum.instances.iter().enumerate() {
718        let left_guard = guards.get(&left.id).copied().unwrap_or(DECISION_TRUE);
719        if left_guard == DECISION_FALSE {
720            continue;
721        }
722        let footprint = templates[left.template_id].qubits();
723        predicates.charge(footprint.len())?;
724        for &qubit in footprint {
725            let qubit = bloq_circuit::checked_translate_coordinate(qubit, left.offset)?;
726            let others = occupants.entry(qubit).or_default();
727            for &other in others.iter() {
728                predicates.charge(1)?;
729                if !candidate_pairs.insert((index, other)) {
730                    continue;
731                }
732                let right = &quantum.instances[other];
733                let right_guard = guards.get(&right.id).copied().unwrap_or(DECISION_TRUE);
734                if predicates
735                    .diagram
736                    .apply(BooleanOp::And, left_guard, right_guard)?
737                    == DECISION_FALSE
738                {
739                    continue;
740                }
741                let offset = [
742                    i64::from(right.offset.x) - i64::from(left.offset.x),
743                    i64::from(right.offset.y) - i64::from(left.offset.y),
744                ];
745                if !checked.insert((
746                    left.template_id,
747                    right.template_id,
748                    offset,
749                    left.provenance.is_ideal_boundary(),
750                    right.provenance.is_ideal_boundary(),
751                )) {
752                    continue;
753                }
754                predicates
755                    .charge(costs[&left.template_id].saturating_add(costs[&right.template_id]))?;
756                pair.expect_quantum_mut().instances = vec![*left, *right];
757                pair.emission_plan_with_options(templates, options)?;
758            }
759            others.push(index);
760        }
761    }
762    Ok(())
763}

765/// Keep exactly the tick/repeat structure, including nonempty plain segments.
766/// The regular merge remains the sole authority for aligned-body legality.
767fn circuit_shape(source: &bloq_circuit::CoordCircuit) -> bloq_circuit::CoordCircuit {
768    use bloq_circuit::{BodyId, CircuitBody, CoordCircuit, GateType, Op};
769    let mut shape = CoordCircuit::new();
770    for index in 0..source.body_count() {
771        if index != 0 {
772            shape.add_body(CircuitBody::new());
773        }
774        let id = BodyId(index as u32);
775        let ops = shape.body_mut(id).expect("allocated shape body").ops_mut();
776        for op in source.body(id).expect("stored source body").ops() {
777            match op {
778                Op::Tick | Op::Repeat { .. } => ops.push(op.clone()),
779                _ if !matches!(ops.last(), Some(Op::Gate { .. })) => {
780                    ops.push(Op::Gate {
781                        gate: GateType::H,
782                        qubits: Vec::new(),
783                    });
784                }
785                _ => {}
786            }
787        }
788    }
789    shape
790        .set_entry_body(source.entry_body())
791        .expect("stored source entry");
792    shape
793}

795impl PredicateAnalysis<'_> {
796    /// Enumerate at most `limit` distinct, jointly reachable predicate tuples.
797    /// Shared Boolean ancestry preserves equal and complementary inputs without
798    /// enumerating all assignments to the underlying bit producers.
799    ///
800    /// # Errors
801    ///
802    /// Returns malformed predicate dataflow or exhausted Boolean resources.
803    pub fn values_up_to(
804        &mut self,
805        predicates: &[BloqNodeId],
806        limit: usize,
807    ) -> Result<Vec<Vec<bool>>, Error> {
808        self.charge(predicates.len())?;
809        let roots = predicates
810            .iter()
811            .map(|&node| self.guard(Some(node.into())))
812            .collect::<Result<Vec<_>, _>>()?;
813        Ok(self.diagram.values_up_to(&roots, limit)?)
814    }
815
816    /// Whether a joint assignment is consistent with the Boolean dataflow.
817    ///
818    /// # Errors
819    ///
820    /// Returns [`crate::lowering::NodeTemplateInstanceMergeError`] for malformed predicate
821    /// dataflow or exhausted Boolean resources.
822    pub fn assignment_reachable(
823        &mut self,
824        predicates: &[(BloqNodeId, bool)],
825    ) -> Result<bool, Error> {
826        Ok(self.assignment_root(predicates)? != DECISION_FALSE)
827    }
828
829    /// Additional predicate values fixed by one reachable assignment.
830    /// Unconditional Boolean constants remain for the caller's strict fold:
831    /// an unrelated `x ^ x` must still wait on `x` (SEM-EXPR).
832    pub(crate) fn values_fixed_by_assignment(
833        &mut self,
834        predicates: &[(ValueRef, bool)],
835        nodes: impl IntoIterator<Item = ValueRef>,
836    ) -> Result<Option<FxMap<ValueRef, bool>>, Error> {
837        let domain = self.assignment_value_root(predicates)?;
838        if domain == DECISION_FALSE {
839            return Ok(None);
840        }
841        let mut known = FxMap::default();
842        if domain == DECISION_TRUE {
843            return Ok(Some(known));
844        }
845        let mut fixed = FxMap::default();
846        for node in nodes {
847            let value = self.guard(Some(node))?;
848            if value == DECISION_FALSE || value == DECISION_TRUE {
849                continue;
850            }
851            let bit = if let Some(&bit) = fixed.get(&value) {
852                bit
853            } else {
854                let bit = match self.diagram.constrain(value, domain)? {
855                    DECISION_FALSE => Some(false),
856                    DECISION_TRUE => Some(true),
857                    _ => None,
858                };
859                fixed.insert(value, bit);
860                bit
861            };
862            if let Some(bit) = bit {
863                known.insert(node, bit);
864            }
865        }
866        Ok(Some(known))
867    }
868
869    fn assignment_root(&mut self, predicates: &[(BloqNodeId, bool)]) -> Result<DecisionId, Error> {
870        self.assignment_value_root(
871            &predicates
872                .iter()
873                .map(|&(node, value)| (node.into(), value))
874                .collect::<Vec<_>>(),
875        )
876    }
877
878    fn assignment_value_root(
879        &mut self,
880        predicates: &[(ValueRef, bool)],
881    ) -> Result<DecisionId, Error> {
882        self.charge(predicates.len().saturating_add(1))?;
883        let mut root = DECISION_TRUE;
884        for &(predicate, value) in predicates {
885            let mut condition = self.guard(Some(predicate))?;
886            if !value {
887                condition = self.diagram.negate(condition)?;
888            }
889            root = self.diagram.apply(BooleanOp::And, root, condition)?;
890            if root == DECISION_FALSE {
891                return Ok(root);
892            }
893        }
894        Ok(root)
895    }
896
897    /// Whether level-local bit predicates can all be true.
898    ///
899    /// Private because [`assignment_reachable`](Self::assignment_reachable) is
900    /// the public generalization (it takes a value per predicate); this is the
901    /// all-true special case, kept for its memoized one- and two-predicate
902    /// path through [`overlap`](Self::overlap).
903    fn can_coexist(&mut self, predicates: &[BloqNodeId]) -> Result<bool, Error> {
904        self.can_coexist_values(
905            &predicates
906                .iter()
907                .copied()
908                .map(Into::into)
909                .collect::<Vec<_>>(),
910        )
911    }
912
913    fn can_coexist_values(&mut self, predicates: &[ValueRef]) -> Result<bool, Error> {
914        if predicates.len() <= 2 {
915            return self.overlap_values(predicates.first().copied(), predicates.get(1).copied());
916        }
917        self.charge(predicates.len())?;
918        let mut root = DECISION_TRUE;
919        for &predicate in predicates {
920            let value = self.guard(Some(predicate))?;
921            root = self.diagram.apply(BooleanOp::And, root, value)?;
922            if root == DECISION_FALSE {
923                return Ok(false);
924            }
925        }
926        Ok(true)
927    }
928}

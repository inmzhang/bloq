1//! The human-readable Bloq IR text exchange format.
2//!
3//! Writing and re-parsing preserves templates (circuits with
4//! explicit measurement ids, side tables, `boundary_flows`), the node graph
5//! with exact node ids (holes from removals included, so the deterministic
6//! schedule survives), and all provenance channels.
7//! [`Bloq`]'s `Display` impl renders this format.
8//!
9//! # Layout
10//!
11//! ```text
12//! BLOQIR 1
13//! metadata bloq_compile.code_distance u64 3
14//! metadata bloq_compile.convention string fixed-bulk
15//! logical-input (0,0,0) instance i0 x X(8,0) z Z(8,0)
16//! logical-output (0,0,1) instance i0 x X(8,0) z Z(8,0)
17//!
18//! template t0 {
19//!   circuit {                           # entry body b0
20//!     H (0,0)
21//!     TICK
22//!     M(0.01) (0,0):m0                  # optional readout-flip probability
23//!     DEPOLARIZE1(0.001) (0,0)
24//!     MPP X(1,1)*X(1,3):m1
25//!     REPEAT 3 b1                       # bodies are flat, by id
26//!     CPAULI X[m0]@(0,0) Z[m2]@(1,0)
27//!   }
28//!   body b1 {
29//!     MX (1,0):m2
30//!   }
31//!   meas m9 @ (4,4)                     # registry records no op implies
32//!   detector -m0*m2 @ (0.5,1)           # optional sign; terms: mN | sN
33//!   detector body(b1) m2*s0
34//!   loop(b1,s0) init m0 next m2
35//!   flow X(0,0) -> Z(1,0)*X(2,0) meas m0*m2 sign -1 center (1,1) restart
36//!   restart m0*m2
37//! }
38//!
39//! bundle b0 owners t0 {
40//!   detector o0:m0 @ (0,1)             # owner slots are bundle-local
41//! }
42//!
43//! graph {
44//!   n0 quantum {
45//!     timeline rounds 3 6                 # cumulative rounds per occupied z layer
46//!     instance i0 t0 @ (8,0)
47//!     detector i0:m0*i1:m2 @ (1,2)      # node parity terms: iN:mK
48//!     use b0 i0 @ (0,0)                # bind owners to global instances
49//!     guard 0 b0                        # guard use index 0 as a whole
50//!     restart i0:m0
51//!     from blocks (0,0,0) (1,0,0)
52//!   }
53//!   n2 observable fragment measurements i0:m0*i0:m2 from generator 0
54//!   n3 compute in0 ^ !(in1 & in2) from frame x (0,0,1)
55//!   n6 observable fragment operators i0 input X(0,0)*Z(0,1)
56//!   n7 observable 0
57//!   result n7:flip                      # optional local bit result; default false
58//!   bindings n6                         # optional fragment/region exports; default empty
59//!   n8 discard in0
60//!   n9 rus in0 {
61//!     body { ... }                      # a nested graph level
62//!   }
63//!   n11 rus in0 source n1:flip {
64//!     body { ... }
65//!   }
66//!   n0 -> n2 order
67//!   n2 -> n7 compose 0
68//!   n6 -> n7 compose 1
69//!   n7 -> n3 value 0 flip                # omitted port means corrected
70//!   n3 -> n7 value 2 feedback 3
71//!   n0 -> n1 quantum (0,0,0)>(0,0,1) padding (0,0,0)>(0,0,1) offset (8,0) one t1 loop t2
72//! }
73//! ```
74//!
75//! `circuit bN { ... }` names a nonzero entry body; `circuit { ... }` means b0.
76//! All other bodies retain their explicit ids.
77//!
78//! Node ids (`nK`) are written explicitly and gaps in the sequence are
79//! reconstructed as vacant graph slots on parse. What is *not* preserved
80//! (by this codec or the binary one) is vacancy past the last live node and
81//! the free-slot reuse order — history that only affects the ids future
82//! `add_node` calls hand out, never the program's semantics.
83//!
84//! Metadata keys and string values use percent-escaped UTF-8 tokens; `~`
85//! represents an empty token. `#` starts a comment anywhere.
86//!
87//! The measurement registry is implied: `M`/`MX`/`MY`/`MPP` op targets define
88//! `(id, qubit)` pairs, and `meas` lines add the records no op implies.
89//!
90//! Parsing rejects structurally malformed input (duplicate node ids, dangling
91//! edge endpoints, one measurement id bound to two qubits, overflowing
92//! coordinate translations, exhausted measurement-output ids, an implausibly
93//! sparse node-id space, or sparse circuit body ids) but does not check IR
94//! semantics — run [`Bloq::validate`] on untrusted input. There is no
95//! cross-version compatibility: the reader rejects any header version other
96//! than its own.

114fn encode_metadata_token(value: &str) -> String {
115    if value.is_empty() {
116        return "~".to_owned();
117    }
118    let mut encoded = String::with_capacity(value.len());
119    for byte in value.bytes() {
120        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'_' | b'.') {
121            encoded.push(char::from(byte));
122        } else {
123            write!(encoded, "%{byte:02X}").expect("writing to a String cannot fail");
124        }
125    }
126    encoded
127}
128
129fn decode_metadata_token(token: &str) -> Result<String, &'static str> {
130    if token == "~" {
131        return Ok(String::new());
132    }
133    let source = token.as_bytes();
134    let mut decoded = Vec::with_capacity(source.len());
135    let mut index = 0;
136    while index < source.len() {
137        if source[index] != b'%' {
138            decoded.push(source[index]);
139            index += 1;
140            continue;
141        }
142        let hex = source
143            .get(index + 1..index + 3)
144            .ok_or("incomplete percent escape")?;
145        let digit = |byte: u8| match byte {
146            b'0'..=b'9' => Some(byte - b'0'),
147            b'a'..=b'f' => Some(byte - b'a' + 10),
148            b'A'..=b'F' => Some(byte - b'A' + 10),
149            _ => None,
150        };
151        let high = digit(hex[0]).ok_or("invalid percent escape")?;
152        let low = digit(hex[1]).ok_or("invalid percent escape")?;
153        decoded.push((high << 4) | low);
154        index += 3;
155    }
156    String::from_utf8(decoded).map_err(|_| "metadata token is not UTF-8")
157}

159impl Bloq {
160    /// Render this program in the text exchange format (also `Display`).
161    pub fn to_text(&self) -> String {
162        write::write_bloq_text(self)
163    }
164
165    /// Parse a program from the text exchange format.
166    ///
167    /// # Errors
168    ///
169    /// Returns [`TextParseError`] for malformed syntax, unsupported versions,
170    /// invalid references, or structurally invalid serialized data.
171    pub fn from_text(text: &str) -> Result<Self, TextParseError> {
172        let mut bloq = parse::parse_bloq_text(text)?;
173        bloq.top_mut().share_classical_data();
174        Ok(bloq)
175    }
176}
177
178impl fmt::Display for Bloq {
179    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
180        f.write_str(&self.to_text())
181    }
182}

407    /// Whether validation stopped at a resource allowance. Such a failure is
408    /// not evidence that the program violates the IR's well-formedness rules.
409    #[must_use]
410    pub fn is_resource_limited(&self) -> bool {
411        match self {
412            Self::BooleanResource(_) => true,
413            Self::InvalidInstanceMergeStructure { source, .. }
414            | Self::InvalidTemplateCircuit { source, .. } => source.is_resource_limited(),
415            _ => false,
416        }
417    }

420/// Quantum emission stages keyed by scoped node address.
421///
422/// Static stages retain an exact emission plan. Conditional stages are marked
423/// for selected materialization. Only [`Bloq::validate_with_plans`] certifies
424/// their full Boolean domain; [`Bloq::emission_plans`] does not.
425#[derive(Debug, Clone, Default)]
426pub struct ValidatedPlans {
427    plans: FxMap<LevelPath, FxMap<BloqNodeId, NodeEmissionPlan>>,
428    memberships: FxMap<LevelPath, FxSet<BloqNodeId>>,
429}
430
431impl ValidatedPlans {
432    fn insert(&mut self, path: LevelPath, node: BloqNodeId, plan: NodeEmissionPlan) {
433        self.plans.entry(path).or_default().insert(node, plan);
434    }
435
436    /// A static stage's emission plan. Conditional stages use [`Self::has_membership`].
437    #[must_use]
438    pub fn get(&self, path: &LevelPath, node: BloqNodeId) -> Option<&NodeEmissionPlan> {
439        self.plans.get(path)?.get(&node)
440    }
441
442    /// Whether this conditional stage requires selected materialization.
443    pub fn has_membership(&self, path: &LevelPath, node: BloqNodeId) -> bool {
444        self.memberships
445            .get(path)
446            .is_some_and(|nodes| nodes.contains(&node))
447    }
448
449    /// Every static `((path, node), plan)` entry, in unspecified order.
450    pub fn iter(&self) -> impl Iterator<Item = ((&LevelPath, BloqNodeId), &NodeEmissionPlan)> {
451        self.plans
452            .iter()
453            .flat_map(|(path, plans)| plans.iter().map(move |(&node, plan)| ((path, node), plan)))
454    }
455
456    /// The number of static and conditional quantum stages.
457    #[must_use]
458    pub fn len(&self) -> usize {
459        self.plans.values().map(FxMap::len).sum::<usize>()
460            + self.memberships.values().map(FxSet::len).sum::<usize>()
461    }
462
463    /// Whether the program has no quantum-node plans.
464    #[must_use]
465    pub fn is_empty(&self) -> bool {
466        self.plans.is_empty() && self.memberships.is_empty()
467    }
468}

470fn valid_value_source(graph: &SubGraph, value: crate::ValueRef) -> bool {
471    graph
472        .node(value.node)
473        .is_some_and(|node| match value.output {
474            crate::ObservableOutput::Flip => matches!(
475                node.try_classical(),
476                Some(ClassicalNode::Observable { index: Some(_), .. })
477            ),
478            crate::ObservableOutput::Corrected => {
479                node.try_region().is_some()
480                    || matches!(
481                        node.try_classical(),
482                        Some(ClassicalNode::Compute { .. } | ClassicalNode::Observable { .. })
483                    )
484            }
485        })
486}

492/// Checks unique value-input slots, linear quantum faces, and valid value-edge sources.
493// Spec rules WF-2/WF-3 (edge cardinality) and WF-16 (producer kind).
494fn validate_edge_cardinality(
495    graph: &SubGraph,
496    predicates: &mut crate::membership::PredicateAnalysis<'_>,
497) -> Result<(), BloqValidationError> {
498    for id in graph.node_ids() {
499        let mut seen_slots = BTreeSet::new();
500        for input in graph.incoming(id).filter_map(|edge| match edge.edge {
501            BloqEdge::Value { slot, role, output } => Some(crate::ValueInput {
502                slot: *slot,
503                producer: edge.source,
504                role,
505                output: Some(*output),
506            }),
507            BloqEdge::Compose { slot, role } => Some(crate::ValueInput {
508                slot: *slot,
509                producer: edge.source,
510                role,
511                output: None,
512            }),
513            _ => None,
514        }) {
515            if !seen_slots.insert(input.slot) {
516                return Err(BloqValidationError::DuplicateValueSlot {
517                    node: id,
518                    slot: input.slot,
519                });
520            }
521            let consumer = &graph[id];
522            let producer = &graph[input.producer].kind;
523            let valid_source = if let Some(output) = input.output {
524                valid_value_source(
525                    graph,
526                    crate::ValueRef {
527                        node: input.producer,
528                        output,
529                    },
530                )
531            } else {
532                consumer.activation != Some(input.slot)
533                    && consumes_observable_folds(&consumer.kind)
534                    && (matches!(
535                        producer.try_classical(),
536                        Some(ClassicalNode::Observable { .. })
537                    ) || matches!(producer, BloqNodeKind::Region(_)))
538            };
539            if !valid_source {
540                return Err(BloqValidationError::InvalidValueProducer {
541                    node: id,
542                    producer: input.producer,
543                });
544            }
545            if consumer.activation == Some(input.slot) && input.role != &ValueRole::Data {
546                return Err(BloqValidationError::InvalidActivation {
547                    node: id,
548                    rule: "WF-16",
549                    reason: "activation is not an observable fold",
550                });
551            }
552            // WF-16: fold roles carry runtime bits, never structural recipes.
553            if matches!(
554                input.role,
555                ValueRole::FeedbackFold { .. } | ValueRole::ReadoutFold
556            ) && (!consumes_observable_folds(&consumer.kind) || input.output.is_none())
557            {
558                return Err(BloqValidationError::InvalidObservableFold {
559                    node: id,
560                    producer: input.producer,
561                });
562            }
563        }
564        let mut seen_faces: FxMap<[i32; 3], Vec<Option<crate::ValueRef>>> = FxMap::default();
565        for edge in graph.outgoing(id) {
566            let BloqEdge::Quantum(quantum) = edge.edge else {
567                continue;
568            };
569            if let Some(guard) = quantum.guard
570                && (!valid_value_source(graph, guard) || !graph.has_path(guard.node, edge.target))
571            {
572                return Err(BloqValidationError::InvalidSeamGuard {
573                    node: edge.target,
574                    guard,
575                });
576            }
577            for seam in edge.edge.pipes() {
578                let guards = seen_faces.entry(seam.pipe.src.to_array()).or_default();
579                for &other in guards.iter() {
580                    if predicates
581                        .overlap_values(other, quantum.guard)
582                        .map_err(|error| BloqValidationError::InvalidInstanceMergeStructure {
583                            node: id,
584                            source: error,
585                        })?
586                    {
587                        return Err(BloqValidationError::QuantumEdgeFanout { node: id });
588                    }
589                }
590                guards.push(quantum.guard);
591            }
592        }
593    }
594    Ok(())
595}

597impl Bloq {
598    /// Validates IR well-formedness, including nested region bodies.
599    ///
600    /// Same-layer qubit overlap depends on source `BlockGraph` soft corridors
601    /// and is checked by `bloq_compile` lowering.
602    ///
603    /// WF-9 checks every reachable selected merge. [`Self::validate_with_plans`]
604    /// additionally returns static plans and certified conditional stages.
605    /// This entry point releases each merged circuit after checking its node,
606    /// bounding temporary circuit storage by the largest quantum stage. Within
607    /// one noiseless validation, successful unguarded merges are reused at the
608    /// same relative placements; absolute coordinates are still checked.
609    ///
610    /// # Errors
611    ///
612    /// Returns [`BloqValidationError`] for any violated IR well-formedness or
613    /// bounded-resource rule.
614    pub fn validate(&self) -> Result<(), BloqValidationError> {
615        self.validate_with_options(&InstantiationOptions::default())
616    }
617
618    /// Validate with explicit noise and resource limits, releasing emission
619    /// plans after each node instead of retaining them.
620    ///
621    /// # Errors
622    ///
623    /// Returns [`BloqValidationError`] for invalid IR, merge, noise, or
624    /// exhausted resources.
625    pub fn validate_with_options(
626        &self,
627        options: &InstantiationOptions<'_>,
628    ) -> Result<(), BloqValidationError> {
629        self.validate_inner(options, &mut None)
630    }
631
632    /// [`Self::validate`], additionally returning static [`NodeEmissionPlan`]s
633    /// and certified conditional stages, keyed by scoped address (see
634    /// [`ValidatedPlans`]). Plans are instantiated with `options`.
635    ///
636    /// Applies the same well-formedness rules as [`Self::validate`], charging
637    /// work for every materialized plan. SEM-MERGE applies noise once to the
638    /// *post*-merge circuit, and the returned plans carry the caller's `options`.
639    /// Conditional membership uses pairwise Boolean overlap checks.
640    /// A malformed [`NoiseModel`](crate::circuit::NoiseModel) also changes the outcome: its
641    /// probabilities are only checked while materializing noise, so a bad model
642    /// surfaces here (and only under noisy `options`) as
643    /// [`InvalidInstanceMergeStructure`](BloqValidationError::InvalidInstanceMergeStructure)
644    /// at the first quantum node.
645    ///
646    /// # Errors
647    ///
648    /// Returns [`BloqValidationError`] under the same conditions as
649    /// [`Self::validate_with_options`].
650    ///
651    /// # Panics
652    ///
653    /// Panics only if the internal plan collector is unexpectedly absent.
654    pub fn validate_with_plans(
655        &self,
656        options: &InstantiationOptions<'_>,
657    ) -> Result<ValidatedPlans, BloqValidationError> {
658        let mut plans = Some(ValidatedPlans::default());
659        self.validate_inner(options, &mut plans)?;
660        Ok(plans.expect("emission plan collection was requested"))
661    }

663    /// Materialize static quantum stages without a full well-formedness audit.
664    /// Guarded stages are marked for the consumer to select and materialize;
665    /// no Boolean proof of their entire choice domain is performed here.
666    ///
667    /// Required static merges and their noise application still fail here.
668    /// A consumer must report failures while materializing selected guarded
669    /// stages. Call [`Self::validate_with_plans`] explicitly for the complete
670    /// all-choice IR audit as well as plans.
671    ///
672    /// # Errors
673    ///
674    /// [`InvalidInstanceMergeStructure`](BloqValidationError::InvalidInstanceMergeStructure)
675    /// if a static quantum node's instances cannot merge (SEM-MERGE).
676    ///
677    /// # Caution
678    ///
679    /// This is not a substitute for [`Self::validate`]. Malformed IR outside
680    /// the materialized stages may not be reported by this method.
681    pub fn emission_plans(
682        &self,
683        options: &InstantiationOptions<'_>,
684    ) -> Result<ValidatedPlans, BloqValidationError> {
685        let mut plans = ValidatedPlans::default();
686        for (path, level) in self.levels() {
687            for (id, node) in level.nodes() {
688                let Some(quantum) = node.try_quantum() else {
689                    continue;
690                };
691                if !quantum.guards.is_empty() {
692                    plans
693                        .memberships
694                        .entry(path.clone())
695                        .or_default()
696                        .insert(id);
697                    continue;
698                }
699                let plan = node
700                    .emission_plan_with_options(self.templates(), *options)
701                    .map_err(
702                        |source| BloqValidationError::InvalidInstanceMergeStructure {
703                            node: id,
704                            source,
705                        },
706                    )?;
707                plans.insert(path.clone(), id, plan);
708            }
709        }
710        Ok(plans)
711    }

713    /// Shared checks; only consumers requesting emission retain WF-9 plans.
714    fn validate_inner(
715        &self,
716        options: &InstantiationOptions<'_>,
717        plans: &mut Option<ValidatedPlans>,
718    ) -> Result<(), BloqValidationError> {
719        let mut work = BooleanDecisionDiagram::with_limits(options.boolean_limits());
720        work.charge(
721            self.logical_inputs()
722                .len()
723                .saturating_add(self.logical_outputs().len()),
724        )?;
725        // Charge source containers before whole-program maps reserve their entries.
726        for (_, level) in self.levels() {
727            work.charge(level.node_count().saturating_add(level.edge_count()))?;
728            for (_, quantum) in level.quantum_nodes() {
729                work.charge(
730                    quantum
731                        .instances
732                        .len()
733                        .saturating_add(quantum.guards.len())
734                        .saturating_add(quantum.detectors.len())
735                        .saturating_add(quantum.detector_bundles.len())
736                        .saturating_add(quantum.restarts.len()),
737                )?;
738            }
739        }
740        // Templates are shared, so validate the pool once.
741        let template_measurements = self
742            .templates()
743            .iter()
744            .map(|(template_id, template)| validate_template(template_id, template, &mut work))
745            .collect::<Result<Vec<_>, _>>()?;
746        // A definition is shared by every use. Check its record coordinates
747        // once; uses below validate only their bindings and activation.
748        for (bundle_id, bundle) in self.detector_bundles().iter() {
749            work.charge(
750                bundle
751                    .detectors()
752                    .len()
753                    .saturating_add(bundle.owner_templates().len()),
754            )?;
755            for &template in bundle.owner_templates() {
756                if self.templates().get(template).is_none() {
757                    return Err(BloqValidationError::UnknownDetectorBundleTemplate {
758                        bundle: bundle_id,
759                        template,
760                    });
761                }
762            }
763            for row in bundle.detectors() {
764                work.charge(row.parity.terms().len().saturating_add(1))?;
765            }
766            bundle.used_owners()?;
767            for row in bundle.detectors() {
768                for term in row.parity.terms() {
769                    let term = match term {
770                        DetectorTerm::Measurement(term) => term,
771                        DetectorTerm::LoopState(state) => {
772                            return Err(BloqValidationError::DetectorBundleLoopStateUnsupported {
773                                bundle: bundle_id,
774                                state: *state,
775                            });
776                        }
777                    };
778                    let template = bundle.owner_templates()[term.owner as usize];
779                    if !template_measurements
780                        .get(template.0 as usize)
781                        .is_some_and(|records| records.contains(term.measurement))
782                    {
783                        return Err(BloqValidationError::UnknownTemplateMeasurement {
784                            template,
785                            measurement: term.measurement,
786                        });
787                    }
788                }
789            }
790        }
791        let all_instances = collect_all_instances(self)?;
792        for (_, level) in self.levels() {
793            for (_, quantum) in level.quantum_nodes() {
794                work.charge(quantum.detector_bundles.len())?;
795                for use_ in &quantum.detector_bundles {
796                    work.charge(use_.instances.len())?;
797                }
798                self.check_detector_bundle_bindings(quantum, |instance| {
799                    all_instances.get(&instance).copied()
800                })?;
801            }
802        }
803
804        // Memory-round insertion requires an unrolled template and one with a
805        // single top-level REPEAT.
806        for entry in self.pipe_padding() {
807            work.charge(1)?;
808            for template in [entry.one_round, entry.looped] {
809                if self.templates().get(template).is_none() {
810                    return Err(BloqValidationError::UnknownPipePaddingTemplate { template });
811                }
812            }
813            if !self.templates()[entry.one_round]
814                .circuit
815                .entry_top_level_repeats()
816                .is_empty()
817            {
818                return Err(BloqValidationError::PipePaddingOneRoundLooped {
819                    template: entry.one_round,
820                });
821            }
822            if self.templates()[entry.looped]
823                .circuit
824                .entry_top_level_repeats()
825                .len()
826                != 1
827            {
828                return Err(BloqValidationError::PipePaddingLoopedMalformed {
829                    template: entry.looped,
830                });
831            }
832        }
833
834        // Cross-region references use program-global instance ids (WF-5), while
835        // read-after-measure order remains graph-local.
836        if let Some(input) = self
837            .logical_inputs()
838            .iter()
839            .find(|input| !all_instances.contains_key(&input.instance))
840        {
841            return Err(BloqValidationError::LogicalInputUnknownInstance {
842                port: input.port,
843                instance: input.instance,
844            });
845        }
846
847        if let Some(output) = self
848            .logical_outputs()
849            .iter()
850            .find(|output| !all_instances.contains_key(&output.instance))
851        {
852            return Err(BloqValidationError::LogicalOutputUnknownInstance {
853                port: output.port,
854                instance: output.instance,
855            });
856        }
857
858        // U10 requires one X/Z stamp pair on top-level Compute nodes.
859        let mut frame_stamps: FxMap<IVec3, (bool, bool)> = FxMap::default();
860        for (path, level) in self.levels() {
861            for (id, weight) in level.nodes() {
862                if let Some(timeline) = weight
863                    .try_quantum()
864                    .and_then(|quantum| quantum.timeline.as_ref())
865                    && !valid_quantum_timeline(timeline, &weight.provenance)
866                {
867                    return Err(BloqValidationError::InvalidQuantumTimeline { node: id });
868                }
869                let crate::NodeProvenance::OutputFrame { port, basis } = &weight.provenance else {
870                    continue;
871                };
872                if !path.segments().is_empty()
873                    || !matches!(weight.try_classical(), Some(ClassicalNode::Compute { .. }))
874                {
875                    return Err(BloqValidationError::OutputFrameNotCompute {
876                        port: *port,
877                        node: id,
878                    });
879                }
880                let entry = frame_stamps.entry(*port).or_default();
881                let seen = match basis {
882                    crate::Basis::X => &mut entry.0,
883                    crate::Basis::Z => &mut entry.1,
884                };
885                if std::mem::replace(seen, true) {
886                    return Err(BloqValidationError::DuplicateOutputFrameStamp { port: *port });
887                }
888            }
889        }
890        let incomplete_stamp_pair = frame_stamps
891            .iter()
892            .filter(|(_, (x, z))| x != z)
893            .map(|(&port, _)| port);
894        let missing_logical_output_pair = self.logical_outputs().iter().filter_map(|logical| {
895            (frame_stamps.get(&logical.port).copied() != Some((true, true))).then_some(logical.port)
896        });
897        let incomplete = incomplete_stamp_pair
898            .chain(missing_logical_output_pair)
899            .min_by_key(|port| (port.x, port.y, port.z));
900        if let Some(port) = incomplete {
901            return Err(BloqValidationError::OutputFrameIncomplete { port });
902        }
903
904        // Observable indices share one global emission space (§4.2/U17a).
905        let mut observable_indices = FxSet::default();
906        validate_graph(
907            self.top(),
908            &LevelPath::default(),
909            self.templates(),
910            self.detector_bundles(),
911            &all_instances,
912            &template_measurements,
913            &mut observable_indices,
914            *options,
915            plans,
916            &mut MergeProofCache::default(),
917            &mut work,
918            ValidationCtx {
919                in_rus: false,
920                // Skip per-node scans when no template can restart.
921                pool_has_restarts: self
922                    .templates()
923                    .iter()
924                    .any(|(_, template)| !template.restarts.is_empty()),
925            },
926        )?;
927        self.validate_membership_references(*options, &mut work)?;
928        Ok(())
929    }

931    fn validate_membership_references(
932        &self,
933        options: InstantiationOptions<'_>,
934        work: &mut BooleanDecisionDiagram,
935    ) -> Result<(), BloqValidationError> {
936        let external_instances: FxSet<_> = self
937            .logical_inputs()
938            .iter()
939            .map(|input| input.instance)
940            .chain(self.logical_outputs().iter().map(|output| output.instance))
941            .collect();
942        let mut nested_cut_owners = FxMap::default();
943        let mut availability = FxMap::default();
944        for (path, level) in self.levels() {
945            for (id, quantum) in level.quantum_nodes() {
946                if !path.is_top_level() {
947                    for instance in &quantum.instances {
948                        if external_instances.contains(&instance.id) {
949                            nested_cut_owners.insert(instance.id, path.clone());
950                        }
951                    }
952                }
953                for guard in &quantum.guards {
954                    let producer = level
955                        .value_inputs(id)
956                        .find(|input| input.slot == guard.input)
957                        .expect("validated registration input")
958                        .value_ref()
959                        .expect("runtime input");
960                    for &instance in &guard.instances {
961                        availability.insert(instance, (path.clone(), producer));
962                    }
963                }
964            }
965        }
966        let mut predicates = FxMap::default();
967        // WF-7: external metadata is unconditional, so its cut owner must
968        // exist in every reachable selection too.
969        for instance in self
970            .logical_inputs()
971            .iter()
972            .map(|input| input.instance)
973            .chain(self.logical_outputs().iter().map(|output| output.instance))
974        {
975            if let Some(owner_path) = nested_cut_owners.get(&instance) {
976                let mut path = LevelPath::default();
977                for segment in owner_path.segments() {
978                    let level = self.level_at(&path).expect("cut owner ancestor");
979                    let node = &level[segment.region];
980                    let relation = predicates.entry(path.clone()).or_insert_with(|| {
981                        crate::membership::PredicateAnalysis::with_limits(
982                            level,
983                            options.boolean_limits(),
984                        )
985                    });
986                    let invalid = |source| BloqValidationError::InvalidInstanceMergeStructure {
987                        node: segment.region,
988                        source,
989                    };
990                    let active = if let Some(slot) = node.activation {
991                        let guard = level
992                            .value_inputs(segment.region)
993                            .find(|input| input.slot == slot)
994                            .expect("validated ancestor activation")
995                            .value_ref()
996                            .expect("runtime input");
997                        relation
998                            .using_work(work, |relation| relation.implies_values(None, guard))
999                            .map_err(invalid)?
1000                    } else {
1001                        true
1002                    };
1003                    // RUS executes its body at least once on every completed shot.
1004                    if !active {
1005                        return Err(invalid(NodeTemplateInstanceMergeError::InvalidMembership(
1006                            format!(
1007                                "external logical cut i{} is not always selected",
1008                                instance.0
1009                            ),
1010                        )));
1011                    }
1012                    path = path.child(segment.region, segment.body);
1013                }
1014            }
1015            if let Some((path, guard)) = availability.get(&instance) {
1016                let relation = predicates.entry(path.clone()).or_insert_with(|| {
1017                    crate::membership::PredicateAnalysis::with_limits(
1018                        self.level_at(path).expect("owner level"),
1019                        options.boolean_limits(),
1020                    )
1021                });
1022                let invalid = |source| BloqValidationError::InvalidInstanceMergeStructure {
1023                    node: guard.node,
1024                    source,
1025                };
1026                if !relation
1027                    .using_work(work, |relation| relation.implies_values(None, *guard))
1028                    .map_err(invalid)?
1029                {
1030                    return Err(invalid(NodeTemplateInstanceMergeError::InvalidMembership(
1031                        format!(
1032                            "external logical cut i{} is not always selected",
1033                            instance.0
1034                        ),
1035                    )));
1036                }
1037            }
1038        }
1039        if availability.is_empty() {
1040            return Ok(());
1041        }
1042        for (path, level) in self.levels() {
1043            let mut check =
1044                |node: BloqNodeId, guards: &[crate::ValueRef], instance: TemplateInstanceId| {
1045                    let Some((owner_path, required)) = availability.get(&instance) else {
1046                        return Ok(());
1047                    };
1048                    let mut gates = Vec::new();
1049                    let compatible = path.segments().starts_with(owner_path.segments());
1050                    if compatible {
1051                        if *owner_path == path {
1052                            gates.extend(guards.iter().copied());
1053                        } else {
1054                            let level = self.level_at(owner_path).expect("registered owner level");
1055                            let region = path.segments()[owner_path.segments().len()].region;
1056                            if let Some(slot) = level[region].activation {
1057                                gates.push(
1058                                    level
1059                                        .value_inputs(region)
1060                                        .find(|input| input.slot == slot)
1061                                        .expect("validated region activation")
1062                                        .value_ref()
1063                                        .expect("runtime input"),
1064                                );
1065                            }
1066                        }
1067                    }
1068                    let relation = predicates.entry(owner_path.clone()).or_insert_with(|| {
1069                        crate::membership::PredicateAnalysis::with_limits(
1070                            self.level_at(owner_path).expect("registered owner level"),
1071                            options.boolean_limits(),
1072                        )
1073                    });
1074                    let available = compatible
1075                        && relation
1076                            .using_work(work, |relation| {
1077                                relation.implies_all_values(&gates, *required)
1078                            })
1079                            .map_err(|source| {
1080                                BloqValidationError::InvalidInstanceMergeStructure { node, source }
1081                            })?;
1082                    if !available {
1083                        return Err(BloqValidationError::ConditionalReferenceUnavailable {
1084                            node,
1085                            instance,
1086                        });
1087                    }
1088                    Ok(())
1089                };
1090            for (id, node) in level.nodes() {
1091                let input = |slot| {
1092                    level
1093                        .value_inputs(id)
1094                        .find(|input| input.slot == slot)
1095                        .expect("validated guard input")
1096                        .value_ref()
1097                        .expect("runtime input")
1098                };
1099                let guard = node.activation.map(input);
1100                if let Some(classical) = node.try_classical() {
1101                    for term in classical.measurements() {
1102                        check(id, guard.as_slice(), term.instance)?;
1103                    }
1104                    for operator in classical.operators() {
1105                        check(id, guard.as_slice(), operator.instance)?;
1106                    }
1107                }
1108                if let Some(quantum) = node.try_quantum() {
1109                    for (index, detector) in quantum.detectors.iter().enumerate() {
1110                        let guard = quantum
1111                            .guards
1112                            .iter()
1113                            .find(|guard| guard.detectors.contains(&(index as u32)))
1114                            .map(|guard| input(guard.input));
1115                        for term in detector.parity.measurements() {
1116                            check(id, guard.as_slice(), term.instance)?;
1117                        }
1118                    }
1119                    for (index, use_) in quantum.detector_bundles.iter().enumerate() {
1120                        let bundle = self
1121                            .detector_bundles()
1122                            .get(use_.bundle)
1123                            .ok_or(crate::DetectorBundleError::UnknownBundle(use_.bundle))?;
1124                        let guard = quantum
1125                            .guards
1126                            .iter()
1127                            .find(|guard| guard.detector_bundles.contains(&(index as u32)))
1128                            .map(|guard| input(guard.input));
1129                        for &owner in bundle.used_owners()? {
1130                            check(id, guard.as_slice(), use_.instances[owner as usize])?;
1131                        }
1132                    }
1133                    for (index, restart) in quantum.restarts.iter().enumerate() {
1134                        let guard = quantum
1135                            .guards
1136                            .iter()
1137                            .find(|guard| guard.restarts.contains(&(index as u32)))
1138                            .map(|guard| input(guard.input));
1139                        for term in restart.parity.measurements() {
1140                            check(id, guard.as_slice(), term.instance)?;
1141                        }
1142                    }
1143                    for contribution in &quantum.guards {
1144                        for (rows, restart) in [
1145                            (&contribution.detector_parities, false),
1146                            (&contribution.restart_parities, true),
1147                        ] {
1148                            for (index, parity) in rows {
1149                                let mut guards = vec![input(contribution.input)];
1150                                if let Some(guard) = quantum.guards.iter().find(|guard| {
1151                                    if restart {
1152                                        guard.restarts.contains(index)
1153                                    } else {
1154                                        guard.detectors.contains(index)
1155                                    }
1156                                }) {
1157                                    guards.push(input(guard.input));
1158                                }
1159                                for term in parity.measurements() {
1160                                    check(id, &guards, term.instance)?;
1161                                }
1162                            }
1163                        }
1164                    }
1165                }
1166            }
1167        }
1168        Ok(())
1169    }
1170}

1172pub(crate) fn valid_quantum_timeline(
1173    timeline: &crate::QuantumTimeline,
1174    provenance: &crate::NodeProvenance,
1175) -> bool {
1176    let crate::NodeProvenance::BlockComponent { members } = provenance else {
1177        return false;
1178    };
1179    let Some(min_member_z) = members.iter().map(|member| member.pos.z).min() else {
1180        return false;
1181    };
1182    let ends = &timeline.layer_round_ends;
1183    let valid_rounds = ends.len() >= 2
1184        && ends.last().is_some_and(|&last| last > 0)
1185        && ends.windows(2).all(|pair| pair[0] <= pair[1]);
1186    let valid_span = ends
1187        .len()
1188        .checked_sub(1)
1189        .and_then(|span| i64::try_from(span).ok())
1190        .and_then(|span| i64::from(min_member_z).checked_add(span))
1191        .is_some_and(|z_end| z_end <= i64::from(i32::MAX));
1192    valid_rounds && valid_span
1193}

1219struct ValidationCtx {

1221    in_rus: bool,
1222    /// Whether any shared template carries restart syndromes.
1223    pool_has_restarts: bool,

1226/// A merge proof depends on ordered templates and relative placement, while
1227/// ownership, side tables, guards, and logical cuts remain per-node checks.
1228/// This cache belongs to one validation of one immutable template pool.
1229#[derive(Default)]
1230struct MergeProofCache {
1231    shapes: FxSet<Vec<(TemplateId, [i64; 2])>>,
1232    placements: FxSet<(TemplateId, IVec2)>,
1233}
1234
1235impl MergeProofCache {
1236    fn validate(
1237        &mut self,
1238        id: BloqNodeId,
1239        node: &BloqNode,
1240        templates: &BloqTemplatePool,
1241        options: InstantiationOptions<'_>,
1242        work: &mut BooleanDecisionDiagram,
1243    ) -> Result<(), BloqValidationError> {
1244        let instances = &node.expect_quantum().instances;
1245        work.charge(instances.len().saturating_add(1))?;
1246        let origin = instances
1247            .first()
1248            .map_or(IVec2::ZERO, |instance| instance.offset);
1249        // Subtraction must represent opposite i32 extrema without overflow.
1250        let shape = instances
1251            .iter()
1252            .map(|instance| {
1253                (
1254                    instance.template_id,
1255                    [
1256                        i64::from(instance.offset.x) - i64::from(origin.x),
1257                        i64::from(instance.offset.y) - i64::from(origin.y),
1258                    ],
1259                )
1260            })
1261            .collect::<Vec<_>>();
1262        let merge_error =
1263            |source| BloqValidationError::InvalidInstanceMergeStructure { node: id, source };
1264        if self.shapes.contains(&shape) {
1265            for instance in instances {
1266                let placement = (instance.template_id, instance.offset);
1267                work.charge(1)?;
1268                if self.placements.contains(&placement) {
1269                    continue;
1270                }
1271                let qubits = templates[instance.template_id].qubits();
1272                work.charge(qubits.len())?;
1273                // Includes unused registry entries and stored, unreachable bodies,
1274                // matching the full instance translator's coordinate preflight.
1275                for &coordinate in qubits {
1276                    bloq_circuit::checked_translate_coordinate(coordinate, instance.offset)
1277                        .map_err(NodeTemplateInstanceMergeError::from)
1278                        .map_err(merge_error)?;
1279                }
1280                self.placements.insert(placement);
1281            }
1282            return Ok(());
1283        }
1284
1285        for instance in instances {
1286            work.charge(crate::instantiation::circuit_work(
1287                &templates[instance.template_id].circuit,
1288            ))?;
1289        }
1290        // Template preflight alone does not prove measurement remapping inside
1291        // zero-count REPEAT bodies. Cache only successful full SEM-MERGE proofs.
1292        node.emission_plan_with_options(templates, options)
1293            .map_err(merge_error)?;
1294        self.shapes.insert(shape);
1295        self.placements.extend(
1296            instances
1297                .iter()
1298                .map(|instance| (instance.template_id, instance.offset)),
1299        );
1300        Ok(())
1301    }
1302}

1304/// Declared outputs remain valid even when the parent does not consume them.
1305fn validate_outputs(graph: &SubGraph) -> Result<(), BloqValidationError> {
1306    if let Some(value) = graph.value_output()
1307        && !valid_value_source(graph, value)
1308    {
1309        return Err(BloqValidationError::InvalidValueOutput { node: value.node });
1310    }
1311    let mut seen = FxSet::default();
1312    for &node in graph.boundary_outputs() {
1313        if !graph.node(node).is_some_and(|node| {
1314            node.try_region().is_some()
1315                || matches!(node.try_classical(), Some(ClassicalNode::Observable { .. }))
1316        }) {
1317            return Err(BloqValidationError::InvalidBoundaryOutput { node });
1318        }
1319        if !seen.insert(node) {
1320            return Err(BloqValidationError::DuplicateBoundaryOutput { node });
1321        }
1322    }
1323    Ok(())
1324}

1326/// Validates one graph level, then recurses into region bodies. `path` is the
1327/// level's [`LevelPath`], threaded down so each quantum node's plan is keyed by
1328/// its program-unique scoped address (matching `Bloq::levels`/`Bloq::walk`).
1329#[expect(
1330    clippy::too_many_arguments,
1331    reason = "whole-level validation shares scoped registries and bounded-work state"
1332)]
1333fn validate_graph(
1334    graph: &SubGraph,
1335    path: &LevelPath,
1336    templates: &BloqTemplatePool,
1337    bundles: &crate::DetectorBundlePool,
1338    all_instances: &FxMap<TemplateInstanceId, TemplateId>,
1339    template_measurements: &[&MeasurementSet],
1340    observable_indices: &mut FxSet<u32>,
1341    options: InstantiationOptions<'_>,
1342    plans: &mut Option<ValidatedPlans>,
1343    merge_proofs: &mut MergeProofCache,
1344    work: &mut BooleanDecisionDiagram,
1345    ctx: ValidationCtx,
1346) -> Result<(), BloqValidationError> {
1347    work.charge(graph.node_count().saturating_add(graph.edge_count()))?;
1348    // Hand-built Value edges may create cycles (IR spec §3).
1349    let order = graph
1350        .deterministic_emit_order()
1351        .map_err(|_| BloqValidationError::CyclicGraph)?;
1352    let mut predicates =
1353        crate::membership::PredicateAnalysis::with_limits(graph, options.boolean_limits());
1354    predicates.charge(work.steps()).map_err(|source| {
1355        BloqValidationError::InvalidInstanceMergeStructure {
1356            node: order.first().copied().unwrap_or(BloqNodeId(0)),
1357            source,
1358        }
1359    })?;
1360    validate_edge_cardinality(graph, &mut predicates)?;
1361    work.charge(predicates.steps().saturating_sub(work.steps()))?;
1362    validate_outputs(graph)?;
1363    let order_index = order
1364        .iter()
1365        .enumerate()
1366        .map(|(index, &id)| (id, index))
1367        .collect::<FxMap<_, _>>();
1368
1369    let mut instances = FxMap::default();
1370    for &id in &order {
1371        let node = &graph[id];
1372        let Some(quantum) = node.try_quantum() else {
1373            continue; // classical / region nodes carry no template instances
1374        };
1375        for instance in &quantum.instances {
1376            if templates.get(instance.template_id).is_none() {
1377                return Err(BloqValidationError::UnknownTemplate {
1378                    node: id,
1379                    template: instance.template_id,
1380                });
1381            }
1382            assert!(
1383                instances
1384                    .insert(
1385                        instance.id,
1386                        InstanceValidation {
1387                            template: instance.template_id,
1388                            owner: id,
1389                            order: order_index[&id],
1390                        },
1391                    )
1392                    .is_none(),
1393                "the whole-program instance preflight rejected duplicate ids"
1394            );
1395        }
1396    }
1397
1398    let mut path_scratch = graph.path_scratch();
1399    let mut verified_paths = FxSet::default();
1400    for (id, node) in graph.nodes() {
1401        if let Some(slot) = node.activation {
1402            if node.try_quantum().is_some() {
1403                return Err(BloqValidationError::InvalidActivation {
1404                    node: id,
1405                    rule: "WF-12",
1406                    reason: "quantum nodes use member registrations",
1407                });
1408            }
1409            if !graph.value_inputs(id).any(|input| input.slot == slot) {
1410                return Err(BloqValidationError::MissingClassicalInput { node: id, slot });
1411            }
1412        }
1413        let consumer_order = order_index[&id];
1414        if let Some(classical) = node.try_classical() {
1415            // Complete observables retain the same record and boundary-owner
1416            // checks as independently shared readout fragments.
1417            work.charge(classical.measurements().len())?;
1418            validate_instance_measurement_refs(
1419                id,
1420                classical.measurements().iter().copied(),
1421                &instances,
1422                all_instances,
1423                template_measurements,
1424                consumer_order,
1425            )?;
1426            work.charge(classical.operators().len())?;
1427            for operator in classical.operators() {
1428                if !all_instances.contains_key(&operator.instance) {
1429                    return Err(BloqValidationError::UnknownTemplateInstance {
1430                        node: id,
1431                        instance: operator.instance,
1432                    });
1433                }
1434            }
1435        }
1436        if let Some(ClassicalNode::Observable {
1437            index: Some(observable),
1438            ..
1439        }) = node.try_classical()
1440            && !observable_indices.insert(*observable)
1441        {
1442            return Err(BloqValidationError::DuplicateObservableIndex {
1443                node: id,
1444                index: *observable,
1445            });
1446        }
1447        let Some(quantum) = node.try_quantum() else {
1448            continue; // classical / region nodes carry no detectors/repeat states
1449        };
1450        for parity in quantum.stored_parities() {
1451            work.charge(parity.terms().len())?;
1452            validate_instance_parity_refs(
1453                graph,
1454                id,
1455                parity,
1456                &instances,
1457                all_instances,
1458                template_measurements,
1459                consumer_order,
1460                &mut path_scratch,
1461                &mut verified_paths,
1462            )?;
1463        }
1464        for use_ in &quantum.detector_bundles {
1465            work.charge(1)?;
1466            let bundle = bundles
1467                .get(use_.bundle)
1468                .ok_or(crate::DetectorBundleError::UnknownBundle(use_.bundle))?;
1469            for &owner in bundle.used_owners()? {
1470                work.charge(1)?;
1471                let instance = use_.instances[owner as usize];
1472                if let Some(source) = instances.get(&instance) {
1473                    if source.order > consumer_order
1474                        || (source.owner != id
1475                            && !verified_paths.contains(&(source.owner, id))
1476                            && !graph.has_path_backwards_with_scratch(
1477                                source.owner,
1478                                id,
1479                                &mut path_scratch,
1480                            ))
1481                    {
1482                        return Err(BloqValidationError::UnavailableDetectorBundleOwner {
1483                            node: id,
1484                            instance,
1485                        });
1486                    }
1487                    verified_paths.insert((source.owner, id));
1488                }
1489            }
1490        }
1491        if !ctx.in_rus
1492            && (!quantum.restarts.is_empty()
1493                || (ctx.pool_has_restarts
1494                    && quantum.instances.iter().any(|instance| {
1495                        templates
1496                            .get(instance.template_id)
1497                            .is_some_and(|template| !template.restarts.is_empty())
1498                    })))
1499        {
1500            return Err(BloqValidationError::RestartOutsideRepeatUntilSuccess { node: id });
1501        }
1502        // WF-9: the instance merge must succeed. Reuse successful relative-shape
1503        // proofs only when no noisy or retained emission plan is requested.
1504        // Retained plans let consumers emit without re-merging (IR-02).
1505        // Noise is applied post-merge, so a malformed NoiseModel surfaces here as
1506        // an InvalidInstanceMergeStructure (see `validate_with_plans`).
1507        let merge_error =
1508            |source| BloqValidationError::InvalidInstanceMergeStructure { node: id, source };
1509        if quantum.guards.is_empty() && plans.is_none() && options.is_noiseless() {
1510            merge_proofs.validate(id, node, templates, options, work)?;
1511        } else if quantum.guards.is_empty() {
1512            work.charge(quantum.instances.len())?;
1513            for instance in &quantum.instances {
1514                work.charge(crate::instantiation::circuit_work(
1515                    &templates[instance.template_id].circuit,
1516                ))?;
1517            }
1518            let plan = node
1519                .emission_plan_with_options(templates, options)
1520                .map_err(merge_error)?;
1521            if let Some(plans) = plans {
1522                plans.insert(path.clone(), id, plan);
1523            }
1524        } else {
1525            predicates
1526                .using_work(work, |predicates| {
1527                    crate::membership::validate_merge(graph, id, templates, options, predicates)
1528                })
1529                .map_err(merge_error)?;
1530            if let Some(plans) = plans {
1531                plans
1532                    .memberships
1533                    .entry(path.clone())
1534                    .or_default()
1535                    .insert(id);
1536            }
1537        }
1538
1539        if let Some(input) = graph
1540            .value_inputs(id)
1541            .find(|input| !quantum.guards.iter().any(|guard| guard.input == input.slot))
1542        {
1543            return Err(BloqValidationError::UnusedQuantumValueInput {
1544                node: id,
1545                slot: input.slot,
1546            });
1547        }
1548    }
1549
1550    // RUS failures restart rather than discard, including in nested bodies (§6).
1551    if ctx.in_rus
1552        && let Some((node, _)) = graph
1553            .nodes()
1554            .find(|(_, node)| matches!(node.try_classical(), Some(ClassicalNode::Discard { .. })))
1555    {
1556        return Err(BloqValidationError::DiscardInsideRepeatUntilSuccess { node });
1557    }
1558
1559    for (region_id, node) in graph.nodes() {
1560        let BloqNodeKind::Region(region) = &node.kind else {
1561            continue;
1562        };
1563        if ctx.in_rus {
1564            return Err(BloqValidationError::NestedRepeatUntilSuccess { node: region_id });
1565        }
1566        let RegionNode::RepeatUntilSuccess {
1567            body,
1568            restart_condition: predicate,
1569            restart_source,
1570        } = region;
1571        // Region predicates share Compute's linear-input contract (WF-11).
1572        let mut read = expression_inputs(region_id, predicate, work)?;
1573        if let Some(slot) = node.activation
1574            && !read.insert(slot)
1575        {
1576            return Err(BloqValidationError::InvalidActivation {
1577                node: region_id,
1578                rule: "WF-11",
1579                reason: "activation slot is also a predicate operand",
1580            });
1581        }
1582        let fed: BTreeSet<_> = graph
1583            .value_inputs(region_id)
1584            .map(|input| input.slot)
1585            .collect();
1586
1587        // A supplied RUS restart_source must produce a body-local value (WF-10).
1588        if let Some(source) = restart_source
1589            && !valid_value_source(body, *source)
1590        {
1591            return Err(BloqValidationError::InvalidRestartSource {
1592                node: region_id,
1593                restart_source: *source,
1594            });
1595        }
1596
1597        // Restart predicates may use incoming Value edges or a body-local
1598        // restart_source because edges cannot cross regions (§6/U17).
1599        if restart_source.is_none()
1600            && let Some(&slot) = read.difference(&fed).next()
1601        {
1602            return Err(BloqValidationError::MissingRegionSelectorInput {
1603                node: region_id,
1604                slot,
1605            });
1606        }
1607        if let Some(&slot) = fed.difference(&read).next() {
1608            return Err(BloqValidationError::UnusedClassicalInput {
1609                node: region_id,
1610                slot,
1611            });
1612        }
1613
1614        let body_ctx = ValidationCtx {
1615            in_rus: true,
1616            ..ctx
1617        };
1618        for (selector, body) in region.bodies() {
1619            validate_graph(
1620                body,
1621                &path.child(region_id, selector),
1622                templates,
1623                bundles,
1624                all_instances,
1625                template_measurements,
1626                observable_indices,
1627                options,
1628                plans,
1629                merge_proofs,
1630                work,
1631                body_ctx,
1632            )?;
1633        }
1634    }
1635
1636    // Compute/Discard expressions must read each fed Value slot exactly once.
1637    for (id, node) in graph.nodes() {
1638        let expr = match node.try_classical() {
1639            Some(ClassicalNode::Compute { expr }) => expr,
1640            Some(ClassicalNode::Discard { condition }) => condition,
1641            _ => continue,
1642        };
1643        let mut read = expression_inputs(id, expr, work)?;
1644        if let Some(slot) = node.activation
1645            && !read.insert(slot)
1646        {
1647            return Err(BloqValidationError::InvalidActivation {
1648                node: id,
1649                rule: "WF-11",
1650                reason: "activation slot is also a data operand",
1651            });
1652        }
1653        let fed: BTreeSet<_> = graph.value_inputs(id).map(|input| input.slot).collect();
1654        if let Some(&slot) = read.difference(&fed).next() {
1655            return Err(BloqValidationError::MissingClassicalInput { node: id, slot });
1656        }
1657        if let Some(&slot) = fed.difference(&read).next() {
1658            return Err(BloqValidationError::UnusedClassicalInput { node: id, slot });
1659        }
1660    }
1661
1662    Ok(())
1663}

1665/// WF-11: collect inputs without recursing through expression nesting. Only
1666/// affine expressions prohibit duplicates; every constant/operand costs work.
1667fn expression_inputs(
1668    node: BloqNodeId,
1669    expr: &ClassicalExpr,
1670    work: &mut BooleanDecisionDiagram,
1671) -> Result<BTreeSet<u32>, BloqValidationError> {
1672    work.charge(1)?;
1673    let mut pending = vec![expr];
1674    let mut read = BTreeSet::new();
1675    let mut linear = true;
1676    let mut duplicate = None;
1677    while let Some(expr) = pending.pop() {
1678        work.charge(1)?;
1679        let mut insert = |slot| {
1680            if !read.insert(slot) && duplicate.is_none() {
1681                duplicate = Some(slot);
1682            }
1683        };
1684        match expr {
1685            ClassicalExpr::In(slot) => insert(*slot),
1686            ClassicalExpr::Parity { inputs, .. } => {
1687                work.charge(inputs.len())?;
1688                for &slot in inputs {
1689                    insert(slot);
1690                }
1691            }
1692            ClassicalExpr::And(..) | ClassicalExpr::Or(..) | ClassicalExpr::Select(..) => {
1693                linear = false;
1694            }
1695            _ => {}
1696        }
1697        work.charge(expr.operands().len())?;
1698        pending.extend(expr.operands().iter().rev());
1699    }
1700    if linear && let Some(slot) = duplicate {
1701        return Err(BloqValidationError::DuplicateExprInput { node, slot });
1702    }
1703    Ok(read)
1704}

1713fn validate_template<'a>(
1714    template_id: TemplateId,
1715    template: &'a crate::BloqTemplate,
1716    work: &mut bloq_utils::boolean::BooleanDecisionDiagram,
1717) -> Result<&'a MeasurementSet, BloqValidationError> {
1718    // Charge stored input before the circuit-analysis cache may allocate or
1719    // walk it. Reused caches still obey the caller's declared input allowance.
1720    charge_template_work(template_id, work, template.circuit.body_count())?;
1721    for index in 0..template.circuit.body_count() {
1722        let ops = template
1723            .circuit
1724            .body(BodyId(index as u32))
1725            .expect("allocated body")
1726            .ops();
1727        charge_template_work(template_id, work, ops.len())?;
1728        for op in ops {
1729            let terms = match op {
1730                Op::Measure {
1731                    qubits,
1732                    measurements,
1733                    ..
1734                } => qubits.len().saturating_add(measurements.len()),
1735                Op::Gate { qubits, .. }
1736                | Op::Depolarize1 { qubits, .. }
1737                | Op::Depolarize2 { qubits, .. }
1738                | Op::PauliError { qubits, .. } => qubits.len(),
1739                Op::MPP {
1740                    products,
1741                    measurements,
1742                } => {
1743                    charge_template_work(template_id, work, products.len())?;
1744                    for product in products {
1745                        charge_template_work(template_id, work, product.len())?;
1746                    }
1747                    measurements.len()
1748                }
1749                Op::ConditionalPauli(corrections) => corrections.len(),
1750                Op::Tick | Op::Repeat { .. } => 0,
1751            };
1752            charge_template_work(template_id, work, terms)?;
1753        }
1754    }
1755    charge_template_work(
1756        template_id,
1757        work,
1758        template.circuit.meas_registry().records().len(),
1759    )?;
1760    let analysis = template.circuit_analysis().map_err(|source| {
1761        BloqValidationError::InvalidTemplateCircuit {
1762            template: template_id,
1763            source,
1764        }
1765    })?;
1766    let produced_measurements = &analysis.produced_measurements;
1767    if template.validation_cached() {
1768        return Ok(produced_measurements);
1769    }
1770
1771    let produced = |measurement| produced_measurements.contains(measurement);
1772
1773    charge_template_work(template_id, work, template.repeat_states.len())?;
1774    charge_template_work(template_id, work, template.detectors.len())?;
1775    charge_template_work(template_id, work, template.restarts.len())?;
1776    charge_template_work(template_id, work, template.boundary_flows.len())?;
1777    for parity in template
1778        .repeat_states
1779        .iter()
1780        .flat_map(|state| [&state.initial, &state.next])
1781        .chain(template.detectors.iter().map(|detector| &detector.parity))
1782        .chain(template.restarts.iter().map(|restart| &restart.parity))
1783    {
1784        charge_template_work(template_id, work, parity.terms().len())?;
1785    }
1786    for flow in &template.boundary_flows {
1787        charge_template_work(template_id, work, flow.measurements.len())?;
1788    }
1789
1790    let mut states_by_body: FxMap<BodyId, Vec<&crate::TemplateRepeatState>> = FxMap::default();
1791    let mut state_ids =
1792        FxSet::with_capacity_and_hasher(template.repeat_states.len(), Default::default());
1793    for state in &template.repeat_states {
1794        if !state_ids.insert(state.state) {
1795            return Err(BloqValidationError::DuplicateTemplateLoopState {
1796                template: template_id,
1797                state: state.state,
1798            });
1799        }
1800        validate_template_body_ref(template_id, state.body, &analysis.reachable_bodies)?;
1801        for parity in [&state.initial, &state.next] {
1802            validate_template_measurement_refs(template_id, parity, produced)?;
1803        }
1804        states_by_body.entry(state.body).or_default().push(state);
1805    }
1806
1807    let mut detectors_by_body: FxMap<BodyId, Vec<&crate::TemplateDetector>> = FxMap::default();
1808    for detector in &template.detectors {
1809        validate_template_measurement_refs(template_id, &detector.parity, produced)?;
1810        if let TemplateDetectorScope::RepeatBody { body } = detector.scope {
1811            validate_template_body_ref(template_id, body, &analysis.reachable_bodies)?;
1812            detectors_by_body.entry(body).or_default().push(detector);
1813        }
1814    }
1815
1816    // Side tables may name only executed measurement sites.
1817    for flow in &template.boundary_flows {
1818        for &measurement in &flow.measurements {
1819            if !produced_measurements.contains(measurement) {
1820                return Err(BloqValidationError::UnknownTemplateMeasurement {
1821                    template: template_id,
1822                    measurement,
1823                });
1824            }
1825        }
1826    }
1827    for restart in &template.restarts {
1828        validate_template_measurement_refs(template_id, &restart.parity, produced)?;
1829    }
1830
1831    for event in &analysis.repeat_events {
1832        for state in states_by_body.get(&event.body).into_iter().flatten() {
1833            validate_template_measurement_refs_before(
1834                template_id,
1835                &analysis.measurement_order,
1836                event.before,
1837                &state.initial,
1838            )?;
1839            validate_template_measurement_refs_before(
1840                template_id,
1841                &analysis.measurement_order,
1842                event.after,
1843                &state.next,
1844            )?;
1845        }
1846        for detector in detectors_by_body.get(&event.body).into_iter().flatten() {
1847            validate_template_measurement_refs_before(
1848                template_id,
1849                &analysis.measurement_order,
1850                event.after,
1851                &detector.parity,
1852            )?;
1853        }
1854    }
1855
1856    let available_states = validate_template_body_events(
1857        template_id,
1858        template,
1859        &analysis.body_order,
1860        &states_by_body,
1861        &detectors_by_body,
1862        work,
1863    )?;
1864    for detector in &template.detectors {
1865        if detector.scope == TemplateDetectorScope::TopLevel {
1866            validate_template_loop_state_refs(template_id, &detector.parity, &available_states)?;
1867        }
1868    }
1869    for restart in &template.restarts {
1870        validate_template_loop_state_refs(template_id, &restart.parity, &available_states)?;
1871    }
1872    template.cache_validation();
1873    Ok(produced_measurements)
1874}

1876fn validate_template_body_events<'a>(
1877    template_id: TemplateId,
1878    template: &'a crate::BloqTemplate,
1879    body_order: &[BodyId],
1880    states_by_body: &FxMap<BodyId, Vec<&'a crate::TemplateRepeatState>>,
1881    detectors_by_body: &FxMap<BodyId, Vec<&'a crate::TemplateDetector>>,
1882    work: &mut bloq_utils::boolean::BooleanDecisionDiagram,
1883) -> Result<FxSet<LoopStateId>, BloqValidationError> {
1884    #[derive(Default)]
1885    struct StateEffects {
1886        // Preserve first-reference order for the same diagnostic as the walk.
1887        required: Vec<LoopStateId>,
1888        provided: FxSet<LoopStateId>,
1889    }
1890    let mut effects = FxMap::<BodyId, StateEffects>::default();
1891    for &body in body_order {
1892        charge_template_work(template_id, work, 1)?;
1893        let mut effect = StateEffects::default();
1894        let mut required = FxSet::default();
1895        let mut require = |state| {
1896            if required.insert(state) {
1897                effect.required.push(state);
1898            }
1899        };
1900        for op in template
1901            .circuit
1902            .body(body)
1903            .expect("preflight checked bodies")
1904            .ops()
1905        {
1906            charge_template_work(template_id, work, 1)?;
1907            let Op::Repeat {
1908                body: child,
1909                repetitions,
1910            } = op
1911            else {
1912                continue;
1913            };
1914            if *repetitions == 0 {
1915                continue;
1916            }
1917            let states = states_by_body.get(child);
1918            let child_effect = &effects[child];
1919            charge_template_work(template_id, work, child_effect.required.len())?;
1920            charge_template_work(template_id, work, states.map_or(0, Vec::len))?;
1921            let declared: FxSet<_> = states
1922                .into_iter()
1923                .flatten()
1924                .map(|state| state.state)
1925                .collect();
1926            // Initial parities see the surrounding frame before this loop's
1927            // own declarations become available.
1928            for state in states.into_iter().flatten() {
1929                charge_template_work(template_id, work, state.initial.terms().len())?;
1930                for term in state.initial.terms() {
1931                    if let DetectorTerm::LoopState(state) = *term
1932                        && !effect.provided.contains(&state)
1933                    {
1934                        require(state);
1935                    }
1936                }
1937            }
1938            let in_iteration =
1939                |state| effect.provided.contains(&state) || declared.contains(&state);
1940            for &state in &child_effect.required {
1941                if !in_iteration(state) {
1942                    require(state);
1943                }
1944            }
1945            // Body detectors and next parities also see completed nested
1946            // declarations. Those nested states die when this frame exits.
1947            for parity in detectors_by_body
1948                .get(child)
1949                .into_iter()
1950                .flatten()
1951                .map(|detector| &detector.parity)
1952                .chain(states.into_iter().flatten().map(|state| &state.next))
1953            {
1954                charge_template_work(template_id, work, parity.terms().len())?;
1955                for term in parity.terms() {
1956                    if let DetectorTerm::LoopState(state) = *term
1957                        && !in_iteration(state)
1958                        && !child_effect.provided.contains(&state)
1959                    {
1960                        require(state);
1961                    }
1962                }
1963            }
1964            effect.provided.extend(declared);
1965        }
1966        effects.insert(body, effect);
1967    }
1968    let entry = effects
1969        .remove(&template.circuit.entry_body())
1970        .expect("entry was visited");
1971    if let Some(&state) = entry.required.first() {
1972        return Err(BloqValidationError::UnknownTemplateLoopState {
1973            template: template_id,
1974            state,
1975        });
1976    }
1977    Ok(entry.provided)
1978}

2051    clippy::too_many_arguments,
2052    reason = "parity validation needs both local and whole-program ownership tables"
2053)]
2054fn validate_instance_parity_refs(
2055    graph: &SubGraph,
2056    node: BloqNodeId,
2057    parity: &crate::NodeDetectorParity,
2058    instances: &FxMap<TemplateInstanceId, InstanceValidation>,
2059    all_instances: &FxMap<TemplateInstanceId, TemplateId>,
2060    template_measurements: &[&MeasurementSet],
2061    consumer_order: usize,
2062    path_scratch: &mut crate::PathScratch,
2063    verified_paths: &mut FxSet<(BloqNodeId, BloqNodeId)>,
2064) -> Result<(), BloqValidationError> {
2065    if let Some(state) = parity.terms().iter().find_map(|term| match *term {
2066        DetectorTerm::LoopState(state) => Some(state),
2067        DetectorTerm::Measurement(_) => None,
2068    }) {
2069        return Err(BloqValidationError::NodeLoopStateUnsupported { node, state });
2070    }
2071    validate_instance_measurement_refs(
2072        node,
2073        parity.measurements(),
2074        instances,
2075        all_instances,
2076        template_measurements,
2077        consumer_order,
2078    )?;
2079    for measurement in parity.measurements() {
2080        let Some(instance) = instances.get(&measurement.instance) else {
2081            continue;
2082        };
2083        let pair = (instance.owner, node);
2084        if instance.owner != node && !verified_paths.contains(&pair) {
2085            if !graph.has_path_backwards_with_scratch(instance.owner, node, path_scratch) {
2086                return Err(BloqValidationError::UnorderedNodeMeasurement {
2087                    node,
2088                    owner: instance.owner,
2089                    measurement: measurement.measurement,
2090                });
2091            }
2092            verified_paths.insert(pair);
2093        }
2094    }
2095    Ok(())
2096}

2098fn validate_instance_measurement_refs(
2099    node: BloqNodeId,
2100    measurements: impl IntoIterator<Item = InstanceMeasurement>,
2101    instances: &FxMap<TemplateInstanceId, InstanceValidation>,
2102    all_instances: &FxMap<TemplateInstanceId, TemplateId>,
2103    template_measurements: &[&MeasurementSet],
2104    consumer_order: usize,
2105) -> Result<(), BloqValidationError> {
2106    let mut cached = None;
2107    for measurement in measurements {
2108        // Canonical parities and compiler observable recipes group an instance's refs.
2109        if cached
2110            .as_ref()
2111            .is_none_or(|(instance, _, _)| *instance != measurement.instance)
2112        {
2113            // Cross-region refs use global instances; only local refs enforce
2114            // read order.
2115            let local = instances.get(&measurement.instance);
2116            let template = match local {
2117                Some(instance) => instance.template,
2118                None => all_instances.get(&measurement.instance).copied().ok_or(
2119                    BloqValidationError::UnknownTemplateInstance {
2120                        node,
2121                        instance: measurement.instance,
2122                    },
2123                )?,
2124            };
2125            let produced = template_measurements.get(template.0 as usize).copied();
2126            let future_owner = local
2127                .filter(|instance| instance.order > consumer_order)
2128                .map(|instance| instance.owner);
2129            cached = Some((measurement.instance, produced, future_owner));
2130        }
2131        let (_, produced, future_owner) = cached.expect("current instance was resolved");
2132        if !produced.is_some_and(|produced| produced.contains(measurement.measurement)) {
2133            return Err(BloqValidationError::UnknownInstanceMeasurement {
2134                node,
2135                instance: measurement.instance,
2136                measurement: measurement.measurement,
2137            });
2138        }
2139        if let Some(owner) = future_owner {
2140            return Err(BloqValidationError::FutureMeasurement {
2141                node,
2142                owner,
2143                measurement: measurement.measurement,
2144            });
2145        }
2146    }
2147    Ok(())
2148}

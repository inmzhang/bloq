12#[derive(Serialize)]
13enum KindRef<'a> {
14    Quantum(&'a Arc<QuantumNode>),
15    Classical(u32),
16    Region(&'a RegionNode),
17}
18
19#[derive(Serialize)]
20struct NodeRef<'a> {
21    kind: KindRef<'a>,
22    provenance: &'a NodeProvenance,
23    activation: Option<u32>,
24}

26#[derive(Deserialize)]
27enum Kind {
28    Quantum(#[serde(deserialize_with = "crate::ir::node::deserialize_quantum")] Arc<QuantumNode>),
29    Classical(u32),
30    Region(RegionNode),
31}
32
33#[derive(Deserialize)]
34struct Node {
35    kind: Kind,
36    provenance: NodeProvenance,
37    activation: Option<u32>,
38}

40struct Nodes<'a> {
41    graph: &'a SubGraph,
42    ids: crate::FxMap<*const ClassicalNode, u32>,
43}
44
45impl Serialize for Nodes<'_> {
46    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
47        let mut sequence = serializer.serialize_seq(Some(self.graph.node_count()))?;
48        for (id, node) in self.graph.nodes() {
49            let kind = match &node.kind {
50                BloqNodeKind::Quantum(data) => KindRef::Quantum(data),
51                BloqNodeKind::Classical(data) => KindRef::Classical(self.ids[&Arc::as_ptr(data)]),
52                BloqNodeKind::Region(data) => KindRef::Region(data),
53            };
54            sequence.serialize_element(&(
55                id.0,
56                NodeRef {
57                    kind,
58                    provenance: &node.provenance,
59                    activation: node.activation,
60                },
61            ))?;
62        }
63        sequence.end()
64    }
65}

67struct Edges<'a>(&'a SubGraph);
68
69impl Serialize for Edges<'_> {
70    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
71        super::serialize_edges(self.0, serializer)
72    }
73}

75pub(super) fn serialize<S: serde::Serializer>(
76    graph: &SubGraph,
77    serializer: S,
78) -> Result<S::Ok, S::Error> {
79    let mut definitions = Vec::new();
80    let mut ids = crate::FxMap::default();
81    let mut contents = crate::FxMap::default();
82    for (_, node) in graph.nodes() {
83        if let BloqNodeKind::Classical(data) = &node.kind
84            && let std::collections::hash_map::Entry::Vacant(entry) = ids.entry(Arc::as_ptr(data))
85        {
86            let id = if let Some(&id) = contents.get(data.as_ref()) {
87                id
88            } else {
89                let id = u32::try_from(definitions.len()).map_err(serde::ser::Error::custom)?;
90                contents.insert(data.as_ref(), id);
91                definitions.push(data.as_ref());
92                id
93            };
94            entry.insert(id);
95        }
96    }
97    let mut wire = serializer.serialize_struct("SubGraphWire", 5)?;
98    wire.serialize_field("classical", &definitions)?;
99    wire.serialize_field("nodes", &Nodes { graph, ids })?;
100    wire.serialize_field("edges", &Edges(graph))?;
101    wire.serialize_field("value_output", &graph.value_output())?;
102    wire.serialize_field("boundary_outputs", &graph.boundary_outputs())?;
103    wire.end()
104}

106pub(super) fn read_nodes<'de, A: serde::de::SeqAccess<'de>>(
107    sequence: &mut A,
108) -> Result<Vec<(u32, BloqNode)>, A::Error> {
109    let definitions: Vec<Arc<ClassicalNode>> = sequence
110        .next_element()?
111        .ok_or_else(|| A::Error::missing_field("classical"))?;
112    let nodes: Vec<(u32, Node)> = sequence
113        .next_element()?
114        .ok_or_else(|| A::Error::missing_field("nodes"))?;
115    nodes
116        .into_iter()
117        .map(|(id, node)| {
118            let kind = match node.kind {
119                Kind::Quantum(data) => BloqNodeKind::Quantum(data),
120                Kind::Region(data) => BloqNodeKind::Region(data),
121                Kind::Classical(index) => BloqNodeKind::Classical(Arc::clone(
122                    definitions.get(index as usize).ok_or_else(|| {
123                        A::Error::custom(format!("classical definition {index} does not exist"))
124                    })?,
125                )),
126            };
127            Ok((
128                id,
129                BloqNode {
130                    kind,
131                    provenance: node.provenance,
132                    activation: node.activation,
133                },
134            ))
135        })
136        .collect()
137}

139#[cfg(test)]
140mod tests {
141    use super::*;
142    use crate::{BloqEdge, BloqNodeId, ClassicalExpr, ValueRef};

144    #[test]
145    fn codecs_restore_shared_definitions_and_keep_invocations_independent() {
146        let mut graph = SubGraph::new();
147        let zero = graph.add_node(BloqNode::classical(ClassicalNode::Compute {
148            expr: ClassicalExpr::Const(false),
149        }));
150        let one = graph.add_node(BloqNode::classical(ClassicalNode::Compute {
151            expr: ClassicalExpr::Const(true),
152        }));
153        let definition = ClassicalNode::Compute {
154            expr: ClassicalExpr::parity([3, 3, 1], true),
155        };
156        let mut first = BloqNode::classical(definition.clone())
157            .with_provenance(NodeProvenance::Action { ordinal: 7 });
158        first.activation = Some(9);
159        let first = graph.add_node(first);
160        let mut second = BloqNode::classical(definition.clone())
161            .with_provenance(NodeProvenance::Generator { ordinal: 11 });
162        second.activation = Some(10);
163        let second = graph.add_node(second);
164        for (source, target, slot) in [
165            (zero, first, 3),
166            (one, first, 1),
167            (one, first, 9),
168            (one, second, 3),
169            (zero, second, 1),
170            (zero, second, 10),
171        ] {
172            graph.add_edge(source, target, BloqEdge::value(slot));
173        }
174        graph.set_value_output(Some(second.into()));
175        let shares_definition = |level: &SubGraph| match (&level[first].kind, &level[second].kind) {
176            (BloqNodeKind::Classical(left), BloqNodeKind::Classical(right)) => {
177                Arc::ptr_eq(left, right)
178            }
179            _ => unreachable!("both invocations are classical"),
180        };
181        assert!(!shares_definition(&graph));
182
183        let bytes = postcard::to_extend(&graph, Vec::new()).unwrap();
184        let json = serde_json::to_value(&graph).unwrap();
185        assert!(json.get("classical").is_none());
186        assert_eq!(
187            json["nodes"][first.0 as usize][1]["kind"]["Classical"],
188            serde_json::to_value(&definition).unwrap()
189        );
190        for mut restored in [
191            postcard::from_bytes::<SubGraph>(&bytes).unwrap(),
192            serde_json::from_value::<SubGraph>(json.clone()).unwrap(),
193        ] {
194            assert!(shares_definition(&restored));
195            assert_eq!(postcard::to_extend(&restored, Vec::new()).unwrap(), bytes);
196            assert_eq!(serde_json::to_value(&restored).unwrap(), json);
197            *restored
198                .node_mut(first)
199                .unwrap()
200                .try_classical_mut()
201                .unwrap() = ClassicalNode::Compute {
202                expr: ClassicalExpr::Const(false),
203            };
204            assert!(!shares_definition(&restored));
205            assert_eq!(restored[second].try_classical(), Some(&definition));
206            assert_ne!(restored[first].try_classical(), Some(&definition));
207            assert_eq!(graph[first].try_classical(), Some(&definition));
208        }
209    }

211    #[test]
212    fn binary_graph_rejects_out_of_range_classical_definition() {
213        let definitions = vec![ClassicalNode::Compute {
214            expr: ClassicalExpr::Const(false),
215        }];
216        for index in [0, 1, u32::MAX] {
217            let nodes = vec![(
218                0u32,
219                NodeRef {
220                    kind: KindRef::Classical(index),
221                    provenance: &NodeProvenance::None,
222                    activation: None,
223                },
224            )];
225            let wire = (
226                &definitions,
227                &nodes,
228                Vec::<()>::new(),
229                None::<ValueRef>,
230                Vec::<BloqNodeId>::new(),
231            );
232            let bytes = postcard::to_extend(&wire, Vec::new()).unwrap();
233            let restored = postcard::from_bytes::<SubGraph>(&bytes);
234            if index == 0 {
235                assert_eq!(
236                    restored.unwrap()[BloqNodeId(0)].try_classical(),
237                    Some(&definitions[0])
238                );
239            } else {
240                restored.unwrap_err();
241            }
242        }
243    }

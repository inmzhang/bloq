39/// The compiled bloq circuit IR: a graph of quantum, classical, and region
40/// nodes produced by `bloq_compile`, consumed by emitters and the verifier.
41///
42/// The top level is a [`SubGraph`] — the same graph-level type region bodies
43/// use — so every read accessor ([`SubGraph::node`], [`SubGraph::nodes`],
44/// [`SubGraph::incoming`], …) works uniformly at any nesting level. `Bloq`
45/// delegates the common calls and adds whole-program state (templates,
46/// frames, provenance) plus recursive traversal ([`Bloq::walk`],
47/// [`Bloq::levels`]).
48///
49/// Cloning shares graph levels, quantum payloads, and templates; edits detach
50/// the affected storage. Metadata and populated layout caches are still copied.
51// `Clone` is derived: cloning a populated `sorted_layout_coords` cache is
52// sound because the cache is a pure function of the cloned graph.
53#[derive(Debug, Clone, Default, serde::Serialize, serde::Deserialize)]
54pub struct Bloq {
55    templates: BloqTemplatePool,
56    detector_bundles: DetectorBundlePool,
57    top: SubGraph,
58    /// Namespaced producer provenance. It never affects IR semantics.
59    metadata: BTreeMap<String, MetadataValue>,
60    /// Physical input-patch metadata needed by preparation hooks.
61    /// Empty for hand-built programs and inputs without a complete pair.
62    #[serde(default)]
63    pub(super) logical_inputs: Vec<LogicalInput>,
64    /// Terminal logical operators needed by output-state execution hooks.
65    /// Empty for hand-built programs and outputs without a complete pair.
66    pub(super) logical_outputs: Vec<LogicalOutput>,
67    /// Memo for [`Bloq::insert_memory_rounds`]'s padding-template
68    /// specialization: [`PaddingVariant`] -> specialized template, so padding
69    /// many seams at one wait duration shares one pool entry. A
70    /// rebuild-on-demand cache, never serialized. Pool ids are append-only;
71    /// the template-retuning edit clears this memo when circuit contents
72    /// change.
73    ///
74    /// [`PaddingVariant`]: crate::edit::PaddingVariant
75    #[serde(skip)]
76    pub(crate) specialized_padding: crate::FxMap<crate::edit::PaddingVariant, TemplateId>,
77    /// Derived cache, rebuilt on demand — never serialized.
78    #[serde(skip)]
79    sorted_layout_coords: OnceLock<Result<Vec<IVec2>, CoordinateOverflowError>>,
80}

148    /// Replace the padding provenance on the unique top-level quantum edge
149    /// `from -> to`, one record per pipe in pipe order. Returns `false` when
150    /// that edge is missing, ambiguous, or carries a different pipe count.
151    ///
152    /// This is public only for the compiler crate; ordinary edits consume the
153    /// provenance through [`Bloq::insert_memory_rounds`].
154    #[doc(hidden)]
155    pub fn set_quantum_edge_padding(
156        &mut self,
157        from: BloqNodeId,
158        to: BloqNodeId,
159        padding: Vec<PipePadding>,
160    ) -> bool {
161        let edge_id = {
162            let mut edges = self
163                .graph()
164                .edges_connecting(
165                    NodeIndex::new(from.0 as usize),
166                    NodeIndex::new(to.0 as usize),
167                )
168                .filter(|edge| matches!(edge.weight(), BloqEdge::Quantum(_)));
169            let Some(edge) = edges.next() else {
170                return false;
171            };
172            if edges.next().is_some() {
173                return false;
174            }
175            edge.id()
176        };
177        let Some(BloqEdge::Quantum(edge)) = self.graph_mut_internal().edge_weight_mut(edge_id)
178        else {
179            unreachable!("selected a quantum edge")
180        };
181        if edge.pipes.len() != padding.len() {
182            return false;
183        }
184        for (seam, padding) in edge.pipes.iter_mut().zip(padding) {
185            seam.padding = Some(padding);
186        }
187        self.specialized_padding.clear();
188        true
189    }

296    /// The program's distinct layout qubit coordinates, sorted by `(x, y)`.
297    /// Computed once and cached; the cache is cleared on any mutation.
298    ///
299    /// # Errors
300    ///
301    /// Returns [`CoordinateOverflowError`] if an instance offset would move a
302    /// template qubit outside the `i32` coordinate lattice.
303    pub fn sorted_layout_coords(&self) -> Result<&[IVec2], CoordinateOverflowError> {
304        match self
305            .sorted_layout_coords
306            .get_or_init(|| self.collect_sorted_layout_coords())
307        {
308            Ok(coords) => Ok(coords),
309            Err(error) => Err(*error),
310        }
311    }
312
313    fn collect_sorted_layout_coords(&self) -> Result<Vec<IVec2>, CoordinateOverflowError> {
314        let mut qubits = FxSet::default();
315        for (_, node) in self.top.nodes() {
316            self.extend_node_qubits(node, &mut qubits)?;
317        }
318        let mut coords = qubits.into_iter().collect::<Vec<_>>();
319        coords.sort_by_key(|coord| (coord.x, coord.y));
320        Ok(coords)
321    }

323    fn extend_node_qubits(
324        &self,
325        node: &BloqNode,
326        qubits: &mut FxSet<IVec2>,
327    ) -> Result<(), CoordinateOverflowError> {
328        // A region's footprint is its bodies' footprint:
329        // arm/body instances resolve against the same shared template pool, so
330        // recurse rather than reporting a region as physically empty.
331        if let Some(region) = node.try_region() {
332            for (_, body) in region.bodies() {
333                for (_, body_node) in body.nodes() {
334                    self.extend_node_qubits(body_node, qubits)?;
335                }
336            }
337            return Ok(());
338        }
339        let Some(quantum) = node.try_quantum() else {
340            return Ok(()); // classical nodes carry no qubits
341        };
342        let mut seen: FxSet<(crate::TemplateId, IVec2)> = FxSet::default();
343        for instance in &quantum.instances {
344            // Instances repeat the same (template, offset) — one round per time
345            // layer, sometimes interleaved across layers — and contribute
346            // nothing new. Track all seen pairs so a repeat never re-walks the
347            // template's qubit list, not just consecutive duplicates.
348            if !seen.insert((instance.template_id, instance.offset)) {
349                continue;
350            }
351            if let Some(template) = self.templates.get(instance.template_id) {
352                let template_qubits = template.qubits();
353                qubits.reserve(template_qubits.len());
354                for &qubit in template_qubits {
355                    qubits.insert(checked_translate_coordinate(qubit, instance.offset)?);
356                }
357            }
358        }
359        Ok(())
360    }

377    /// RUS bodies contribute; this is not a runtime

420/// Mirror [`SubGraph`]'s read surface onto [`Bloq`], targeting the top level.
421///
422/// These are pure forwarding calls, so hand-writing them only invited drift —
423/// `open_value_producers`/`open_bit_producers` had no `Bloq` twin at all until
424/// this list made the mirror total. `Deref<Target = SubGraph>` would collapse
425/// them further, but a `Bloq` is not a smart pointer to its top level
426/// (C-DEREF), and `DerefMut` would hand out `&mut SubGraph` implicitly at every
427/// mutating call site, hiding the layout-cache contract the mutators above
428/// spell out.
429macro_rules! delegate_to_top {
430    ($(fn $name:ident($($arg:ident: $ty:ty),* $(,)?) -> $ret:ty;)*) => {
431        impl Bloq {
432            $(
433                #[doc = concat!(
434                    "The top level's [`SubGraph::", stringify!($name), "`]. Region body \
435                     levels are reached through [`Bloq::levels`] / [`Bloq::level_at`]."
436                )]
437                pub fn $name(&self $(, $arg: $ty)*) -> $ret {
438                    self.top.$name($($arg),*)
439                }
440            )*
441        }
442    };
443}

500/// Node-major CSR adjacency for one graph level.
501///
502/// petgraph threads each node's edges through a linked list inside the edge
503/// storage, so a Kahn walk pointer-chases ~E scattered words — at millions of
504/// edges that dominates the algorithm. Building CSR costs two sequential edge
505/// passes plus one E-sized `u32` array (and the offsets array), and the walk
506/// then reads each node's neighbors sequentially.
507struct CsrAdjacency {
508    /// `offsets[node]..offsets[node + 1]` is `node`'s slice in `targets`.
509    offsets: Vec<u32>,
510    targets: Vec<u32>,
511}
512
513impl CsrAdjacency {
514    fn of(graph: &StableDiGraph<BloqNode, BloqEdge>) -> Self {
515        let bound = graph.node_bound();
516        let mut offsets = vec![0u32; bound + 1];
517        for edge in graph.edge_references() {
518            offsets[edge.source().index() + 1] += 1;
519        }
520        for slot in 1..=bound {
521            offsets[slot] += offsets[slot - 1];
522        }
523        // Fill each node's slice from its end, decrementing the end offsets —
524        // no separate cursor array. Afterwards `offsets[i + 1]` is node `i`'s
525        // start, so rotating one slot left restores the invariant. The fill
526        // reverses the within-slice edge order, which no consumer of this
527        // structure observes.
528        let total = graph.edge_count() as u32;
529        let mut targets = vec![0u32; graph.edge_count()];
530        for edge in graph.edge_references() {
531            let from = edge.source().index();
532            offsets[from + 1] -= 1;
533            targets[offsets[from + 1] as usize] = edge.target().index() as u32;
534        }
535        offsets.remove(0);
536        offsets.push(total);
537        Self { offsets, targets }
538    }
539
540    fn slice(&self, node: usize) -> &[u32] {
541        &self.targets[self.offsets[node] as usize..self.offsets[node + 1] as usize]
542    }
543}

545/// The deterministic Kahn topological order of any [`BloqNode`]/[`BloqEdge`]
546/// graph (a [`Bloq`] or a region [`crate::SubGraph`] body). All edge kinds count
547/// equally toward indegree; determinism comes entirely from the min-heap keyed on
548/// node index — indices are unique, so ties never arise. Returns [`CycleDetected`]
549/// when a cycle leaves some node permanently un-emitted.
550///
551/// The walk reads a transient CSR copy of the adjacency (`CsrAdjacency`),
552/// roughly one `u32` per edge plus one per node slot, so neighbor scans are
553/// sequential instead of chasing petgraph's edge lists.
554pub(crate) fn deterministic_emit_order_of(
555    graph: &StableDiGraph<BloqNode, BloqEdge>,
556) -> Result<Vec<BloqNodeId>, CycleDetected> {
557    let adjacency = CsrAdjacency::of(graph);
558    // A node's indegree is bounded by the live edge count, which cannot
559    // approach u32::MAX within addressable memory.
560    let mut indegree = vec![0u32; graph.node_bound()];
561    for &target in &adjacency.targets {
562        indegree[target as usize] += 1;
563    }
564
565    let mut ready = BinaryHeap::new();
566    ready.extend(
567        graph
568            .node_identifiers()
569            .filter(|node| indegree[node.index()] == 0)
570            .map(|node| Reverse(node.index() as u32)),
571    );
572
573    let mut order = Vec::with_capacity(graph.node_count());
574    while let Some(Reverse(node)) = ready.pop() {
575        order.push(BloqNodeId(node));
576        // Determinism is entirely the min-heap's: node indices are unique (no
577        // ties) and each node is pushed exactly once, so push order cannot affect
578        // pop order.
579        for &target in adjacency.slice(node as usize) {
580            let target = target as usize;
581            indegree[target] -= 1;
582            if indegree[target] == 0 {
583                ready.push(Reverse(target as u32));
584            }
585        }
586    }
587
588    if order.len() != graph.node_count() {
589        return Err(CycleDetected);
590    }
591    Ok(order)
592}

594/// Whether the graph — plus `extra` directed edges (duplicates and parallel
595/// edges are allowed; extras naming vacant node slots are ignored) — is
596/// acyclic: the deterministic emit order's success condition, without
597/// materializing the order. Plain-stack Kahn over CSR adjacency, O(V+E) with
598/// no order or graph copy.
599pub(crate) fn is_acyclic_of(
600    graph: &StableDiGraph<BloqNode, BloqEdge>,
601    extra: &[(BloqNodeId, BloqNodeId)],
602) -> bool {
603    let adjacency = CsrAdjacency::of(graph);
604    let mut indegree = vec![0u32; graph.node_bound()];
605    for &target in &adjacency.targets {
606        indegree[target as usize] += 1;
607    }
608
609    // Extra edges grouped by source in a sorted CSR (extras are far fewer
610    // than graph edges). `source_marks` is a bitset over node slots so the
611    // per-pop probe stays O(1) for the overwhelmingly common non-source node.
612    let mut extras: Vec<(u32, u32)> = extra
613        .iter()
614        .filter(|&(from, to)| {
615            graph.contains_node(NodeIndex::new(from.0 as usize))
616                && graph.contains_node(NodeIndex::new(to.0 as usize))
617        })
618        .map(|&(from, to)| (from.0, to.0))
619        .collect();
620    extras.sort_unstable();
621    for &(_, to) in &extras {
622        indegree[to as usize] += 1;
623    }
624    let mut sources = Vec::new();
625    let mut starts = Vec::new();
626    let mut source_marks = vec![0u64; graph.node_bound().div_ceil(u64::BITS as usize)];
627    for (index, &(from, _)) in extras.iter().enumerate() {
628        if sources.last() != Some(&from) {
629            sources.push(from);
630            starts.push(index);
631            source_marks[from as usize >> 6] |= 1 << (from & 63);
632        }
633    }
634
635    let mut stack: Vec<u32> = graph
636        .node_identifiers()
637        .filter(|node| indegree[node.index()] == 0)
638        .map(|node| node.index() as u32)
639        .collect();
640    let mut visited = 0usize;
641    while let Some(node) = stack.pop() {
642        visited += 1;
643        for &target in adjacency.slice(node as usize) {
644            let target = target as usize;
645            indegree[target] -= 1;
646            if indegree[target] == 0 {
647                stack.push(target as u32);
648            }
649        }
650        if source_marks[node as usize >> 6] >> (node & 63) & 1 == 0 {
651            continue;
652        }
653        let group = sources
654            .binary_search(&node)
655            .expect("marked nodes are exactly the extra-edge sources");
656        let start = starts[group];
657        let end = starts.get(group + 1).copied().unwrap_or(extras.len());
658        for &(_, to) in &extras[start..end] {
659            let to = to as usize;
660            indegree[to] -= 1;
661            if indegree[to] == 0 {
662                stack.push(to as u32);
663            }
664        }
665    }
666
667    visited == graph.node_count()
668}

980    /// The pre-optimization min-heap Kahn, kept as the correctness oracle.
981    fn heap_order(
982        graph: &StableDiGraph<BloqNode, BloqEdge>,
983    ) -> Result<Vec<BloqNodeId>, CycleDetected> {
984        let mut indegree = vec![0usize; graph.node_bound()];
985        for edge in graph.edge_references() {
986            indegree[edge.target().index()] += 1;
987        }
988        let mut ready = BinaryHeap::new();
989        ready.extend(
990            graph
991                .node_identifiers()
992                .filter(|node| indegree[node.index()] == 0)
993                .map(|node| Reverse(node.index())),
994        );
995        let mut order = Vec::with_capacity(graph.node_count());
996        while let Some(Reverse(node)) = ready.pop() {
997            order.push(BloqNodeId(node as u32));
998            for target in
999                graph.neighbors_directed(NodeIndex::new(node), petgraph::Direction::Outgoing)
1000            {
1001                let target = target.index();
1002                indegree[target] -= 1;
1003                if indegree[target] == 0 {
1004                    ready.push(Reverse(target));
1005                }
1006            }
1007        }
1008        if order.len() != graph.node_count() {
1009            return Err(CycleDetected);
1010        }
1011        Ok(order)
1012    }

1015    fn lcg(seed: u64) -> impl FnMut() -> u64 {
1016        let mut state = seed;

1019                .wrapping_mul(6364136223846793005)
1020                .wrapping_add(1442695040888963407);
1021            state >> 33
1022        }
1023    }

1032    /// Random graph with parallel edges, holes (every third node removed), and
1033    /// — when `cyclic` — a chain over the live nodes closed by a backward edge,
1034    /// so the cycle survives the removals.
1035    fn random_graph(
1036        rng: &mut impl FnMut() -> u64,
1037        nodes: usize,
1038        edges: usize,
1039        cyclic: bool,
1040    ) -> StableDiGraph<BloqNode, BloqEdge> {
1041        let mut graph = StableDiGraph::new();
1042        for _ in 0..nodes {
1043            graph.add_node(placeholder_node());
1044        }
1045        for _ in 0..edges {
1046            let from = NodeIndex::new((rng() % nodes as u64) as usize);
1047            let to = NodeIndex::new((rng() % nodes as u64) as usize);
1048            if from != to {
1049                graph.add_edge(from, to, BloqEdge::Order);
1050                if rng().is_multiple_of(4) {
1051                    graph.add_edge(from, to, BloqEdge::Order);
1052                }
1053            }
1054        }
1055        for slot in (0..nodes).step_by(3) {
1056            graph.remove_node(NodeIndex::new(slot));
1057        }
1058        let live: Vec<NodeIndex> = graph.node_identifiers().collect();
1059        if cyclic && live.len() > 1 {
1060            for pair in live.windows(2) {
1061                graph.add_edge(pair[0], pair[1], BloqEdge::Order);
1062            }
1063            let (first, last) = (live[0], live[live.len() - 1]);
1064            graph.add_edge(last, first, BloqEdge::Order);
1065        }
1066        graph
1067    }

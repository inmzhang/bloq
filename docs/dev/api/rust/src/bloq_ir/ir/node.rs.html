12/// What source-level construct an IR node realizes.
13#[derive(Debug, Clone, PartialEq, serde::Serialize, serde::Deserialize)]
14pub enum NodeProvenance {
15    /// A quantum node realizing one connected component of source blocks (its
16    /// `members`), fused because they share physical qubits on a layer.
17    BlockComponent {
18        /// Source blocks in the component.
19        members: Vec<SourceBlockRef>,
20    },
21    /// A quantum node realizing one temporal pipe: the seam layer between two
22    /// block layers. Normally a temporal pipe will not compile to real quantum
23    /// circuits, the currently only use case if for temporal hadamard pipe
24    /// which includes a realignment layer.
25    TemporalPipe {
26        /// Realized temporal pipe.
27        pipe: TemporalPipeRef,
28    },
29    /// Compiler-only replacement of a source spatial Port. The virtual
30    /// temporal-Port node has no lattice cell of its own; `role` places it just
31    /// before or after `source` on the doubled-z schedule.
32    SpatialPortSubstitution {
33        /// Source port position.
34        source: IVec3,
35        /// Source port role.
36        role: bloq_utils::PortRole,
37    },
38    /// Memory-round padding inserted at a `Quantum` seam or region terminal
39    /// post-compile ([`crate::Bloq::insert_memory_rounds`]). Like
40    /// [`Self::TemporalPipe`] it lives on the pipe's layer, but it carries the
41    /// wait duration so consumers can find padding nodes and read
42    /// their round count without inspecting the template's repeat structure.
43    MemoryPadding {
44        /// Padded pipe.
45        pipe: TemporalPipeRef,
46        /// Inserted memory rounds.
47        rounds: u32,
48    },
49    /// A classical node realizing stabilizer generator `ordinal`'s readout
50    /// recipe: its complete `Observable`, any shared fragments, and the
51    /// corrected-value `Compute`s derived from them.
52    Generator {
53        /// Source generator ordinal.
54        ordinal: u32,
55    },
56    /// A classical node lowered from the source action at `ordinal` (program
57    /// action order): a named readout's complete `Observable`, a
58    /// condition/selector `Compute`, or a `Discard` sink.
59    Action {
60        /// Source action ordinal.
61        ordinal: u32,
62    },
63    /// A source branch's resolved selector, shared by every assembly stage.
64    BranchSelector {
65        /// Source selector name.
66        name: String,
67    },
68    /// A frame-sign `Compute`: the `basis`-frame correction bit for source
69    /// output port `port`. The stamp **is** the frame
70    /// table: [`crate::Bloq::output_frames`] derives its [`crate::FramePair`]s from
71    /// these stamps, so there is no side table to drift. Validation enforces
72    /// that a stamp sits on a top-level `Compute` and that each output carries
73    /// exactly one X and one Z stamp.
74    // Spec rule WF-15.
75    OutputFrame {
76        /// Source output port.
77        port: IVec3,
78        /// Corrected Pauli basis.
79        basis: Basis,
80    },
81    /// No provenance: hand-built or synthetic (e.g. a constant producer).
82    None,
83}
84
85/// Human-readable one-line description, e.g. `blocks (0,0,0) (1,0,0)` or
86/// `pipe (0,0,0)>(0,0,1)`. Coordinate/pipe spelling matches the `.bloq` text
87/// format so the two read alike; consumers are the program view and the Stim
88/// backend's per-node provenance comments.
89impl fmt::Display for NodeProvenance {
90    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
91        let coord3 = |f: &mut fmt::Formatter<'_>, p: IVec3| write!(f, "({},{},{})", p.x, p.y, p.z);
92        match self {
93            NodeProvenance::None => write!(f, "synthetic"),
94            NodeProvenance::BlockComponent { members } => {
95                write!(f, "blocks")?;
96                for member in members {
97                    write!(f, " ")?;
98                    coord3(f, member.pos)?;
99                }
100                Ok(())
101            }
102            NodeProvenance::TemporalPipe { pipe } => {
103                write!(f, "pipe ")?;
104                coord3(f, pipe.src)?;
105                write!(f, ">")?;
106                coord3(f, pipe.dst)?;
107                if pipe.hadamard {
108                    write!(f, " H")?;
109                }
110                Ok(())
111            }
112            NodeProvenance::SpatialPortSubstitution { source, role } => {
113                write!(f, "spatial-port {} ", role.as_str())?;
114                coord3(f, *source)
115            }
116            NodeProvenance::MemoryPadding { pipe, rounds } => {
117                write!(f, "padding ")?;
118                coord3(f, pipe.src)?;
119                write!(f, ">")?;
120                coord3(f, pipe.dst)?;
121                write!(f, " rounds {rounds}")
122            }
123            NodeProvenance::Generator { ordinal } => write!(f, "generator {ordinal}"),
124            NodeProvenance::Action { ordinal } => write!(f, "action {ordinal}"),
125            NodeProvenance::BranchSelector { name } => write!(f, "selector {name}"),
126            NodeProvenance::OutputFrame { port, basis } => {
127                let basis = match basis {
128                    Basis::X => "x",
129                    Basis::Z => "z",
130                };
131                write!(f, "frame {basis} ")?;
132                coord3(f, *port)
133            }
134        }
135    }
136}

138/// A graph node, tagged by what kind of work it represents.
139///
140/// `kind` is what the node computes; `provenance` is why it exists — its tie
141/// back to the source block graph, a stabilizer generator, a source action, or
142/// an output frame ([`NodeProvenance`]).
143#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
144pub struct BloqNode {
145    /// Node payload.
146    pub kind: BloqNodeKind,
147    /// Source provenance.
148    pub provenance: NodeProvenance,
149    /// Optional activation `Value` slot for a classical or region node. When false,
150    /// bit results are zero, boundary bindings are empty, and no read, decoder
151    /// request, region-body execution, or discard occurs. It is not a data operand.
152    pub activation: Option<u32>,
153}

155/// What a [`BloqNode`] computes.
156#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
157pub enum BloqNodeKind {
158    /// Shared until quantum data is edited. Large physical payloads do not
159    /// inflate classical node slots or copy during unrelated graph edits.
160    Quantum(#[serde(deserialize_with = "deserialize_quantum")] Arc<QuantumNode>),
161    /// Shared immutable classical definition. Activation, inputs and provenance
162    /// belong to this node's invocation; editing detaches only its definition.
163    Classical(Arc<ClassicalNode>),
164    /// Structured control-flow region.
165    Region(RegionNode),
166}

194/// Classical logic and observable recipes. Composition edges include a child's
195/// measurements and boundary bindings; value edges select a Boolean output.
196#[derive(Debug, Clone, PartialEq, Eq, Hash, serde::Serialize, serde::Deserialize)]
197pub enum ClassicalNode {
198    /// Combinational logic over incoming value slots.
199    Compute {
200        /// Boolean expression.
201        expr: ClassicalExpr,
202    },
203    /// Observable recipe. Indexed recipes expose corrected values and
204    /// predicted decoder flips. Unindexed fragments expose their assembled parity and
205    /// compose without an independent decoder request.
206    Observable {
207        /// Graph-local decoder query identity; absent for recipe fragments.
208        index: Option<u32>,
209        /// Local records, preserving order and occurrences.
210        #[serde(with = "crate::binary::instance_measurements")]
211        measurements: Vec<InstanceMeasurement>,
212        /// Boundary bindings with physical ownership.
213        operators: Vec<InstanceBoundaryOperator>,
214    },
215    /// Reject the shot when the condition holds.
216    Discard {
217        /// Rejection condition.
218        condition: ClassicalExpr,
219    },
220}

222impl ClassicalNode {
223    /// A complete observable with an empty recipe.
224    pub fn observable(index: u32) -> Self {
225        Self::Observable {
226            index: Some(index),
227            measurements: Vec::new(),
228            operators: Vec::new(),
229        }
230    }
231
232    /// A composable recipe without its own decoder query.
233    pub fn observable_fragment(
234        measurements: Vec<InstanceMeasurement>,
235        operators: Vec<InstanceBoundaryOperator>,
236    ) -> Self {
237        Self::Observable {
238            index: None,
239            measurements,
240            operators,
241        }
242    }
243
244    /// An empty composable recipe.
245    pub fn fragment() -> Self {
246        Self::observable_fragment(Vec::new(), Vec::new())
247    }
248
249    /// Record terms owned directly by this producer.
250    pub fn measurements(&self) -> &[InstanceMeasurement] {
251        match self {
252            Self::Observable { measurements, .. } => measurements,
253            _ => &[],
254        }
255    }
256
257    /// Boundary bindings owned directly by this producer.
258    pub fn operators(&self) -> &[InstanceBoundaryOperator] {
259        match self {
260            Self::Observable { operators, .. } => operators,
261            _ => &[],
262        }
263    }

266/// Combinational logic over a classical node's incoming `Value` edges.
267///
268/// [`ClassicalExpr::In`] reads the edge whose `slot` equals its index
269/// (SSA-by-slot): the slot is carried on the edge, so the binding survives edge
270/// reordering under edits rather than depending on iteration order.
271#[derive(Debug, Clone, PartialEq, Eq, Hash, serde::Serialize, serde::Deserialize)]
272pub enum ClassicalExpr {
273    /// Read one incoming edge slot.
274    In(u32),
275    /// A compile-time-constant bit — an input-less producer. Carries a
276    /// deterministic feedback sign (constant-sign folds to no op).
277    Const(bool),
278    /// Boolean negation.
279    Not(Box<ClassicalExpr>),
280    /// Ordered parity operands; the empty parity is false.
281    Xor(Box<[ClassicalExpr]>),
282    /// Ordered conjunction operands; the empty conjunction is true.
283    And(Box<[ClassicalExpr]>),
284    /// Ordered disjunction operands; the empty disjunction is false.
285    Or(Box<[ClassicalExpr]>),
286    /// `[condition, when_false, when_true]`. All three operands must be known,
287    /// including the unselected value; this is combinational dataflow (SEM-EXPR).
288    Select(Box<[ClassicalExpr; 3]>),
289    /// Affine parity of ordered input slots and a constant XOR term.
290    /// Repeated slots stay present: every input must be known even when its
291    /// Boolean contributions cancel (SEM-EXPR). No expression nodes per slot.
292    Parity {
293        /// Ordered input slots.
294        inputs: Box<[u32]>,
295        /// Constant XOR term.
296        constant: bool,
297    },
298}
299
300impl ClassicalExpr {
301    /// Build an affine parity, retaining input order and multiplicity.
302    #[must_use]
303    pub fn parity(inputs: impl IntoIterator<Item = u32>, constant: bool) -> Self {
304        let inputs: Box<[_]> = inputs.into_iter().collect();
305        match inputs.as_ref() {
306            [] => Self::Const(constant),
307            [slot] if !constant => Self::In(*slot),
308            _ => Self::Parity { inputs, constant },
309        }
310    }
311
312    /// Build a parity list, collapsing empty and singleton lists.
313    /// Preserves operand order, duplicates, and nested expressions.
314    ///
315    /// # Panics
316    ///
317    /// Panics only if the collected operand count changes before extraction.
318    #[must_use]
319    pub fn xor(operands: impl IntoIterator<Item = Self>) -> Self {
320        let mut operands: Vec<_> = operands.into_iter().collect();
321        match operands.len() {
322            0 => Self::Const(false),
323            1 => operands
324                .pop()
325                .expect("the single operand was counted above"),
326            _ => Self::Xor(operands.into_boxed_slice()),
327        }
328    }
329
330    /// Select a value with explicit false/true operand order (SEM-EXPR).
331    /// Literal branches use smaller operators without dropping unknown inputs.
332    #[must_use]
333    pub fn select(condition: Self, when_false: Self, when_true: Self) -> Self {
334        match (when_false, when_true) {
335            (Self::Const(false), Self::Const(true)) => condition,
336            (Self::Const(true), Self::Const(false)) => Self::Not(Box::new(condition)),
337            (Self::Const(false), when_true) => Self::And(Box::new([condition, when_true])),
338            (when_false, Self::Const(true)) => Self::Or(Box::new([condition, when_false])),
339            // A negated gate needs two boxes; its selection needs only one.
340            (when_false, when_true) => Self::Select(Box::new([condition, when_false, when_true])),
341        }
342    }
343
344    /// Direct expression children in evaluation order. Compact [`Self::Parity`]
345    /// stores slots directly; use [`Self::for_each_input`] to visit all inputs.
346    pub fn operands(&self) -> &[Self] {
347        match self {
348            Self::In(_) | Self::Const(_) | Self::Parity { .. } => &[],
349            Self::Not(inner) => std::slice::from_ref(inner),
350            Self::Xor(operands) | Self::And(operands) | Self::Or(operands) => operands,
351            Self::Select(operands) => operands.as_ref(),
352        }
353    }
354
355    /// Fold the expression over three-valued inputs: `None` (an input that is
356    /// unevaluable in the caller's context) propagates. The reference
357    /// evaluator every consumer shares.
358    pub fn eval(&self, input: &mut impl FnMut(u32) -> Option<bool>) -> Option<bool> {
359        match self {
360            Self::In(slot) => input(*slot),
361            Self::Const(bit) => Some(*bit),
362            Self::Parity { inputs, constant } => inputs
363                .iter()
364                .try_fold(*constant, |value, &slot| Some(value ^ input(slot)?)),
365            Self::Not(inner) => inner.eval(input).map(|bit| !bit),
366            Self::Xor(operands) => operands
367                .iter()
368                .try_fold(false, |value, expr| Some(value ^ expr.eval(input)?)),
369            Self::And(operands) => operands
370                .iter()
371                .try_fold(true, |value, expr| Some(value & expr.eval(input)?)),
372            Self::Or(operands) => operands
373                .iter()
374                .try_fold(false, |value, expr| Some(value | expr.eval(input)?)),
375            Self::Select(operands) => {
376                let [condition, when_false, when_true] = operands.as_ref();
377                let condition = condition.eval(input)?;
378                let when_false = when_false.eval(input)?;
379                let when_true = when_true.eval(input)?;
380                Some(if condition { when_true } else { when_false })
381            }
382        }
383    }
384
385    /// Combine evaluated operands under Kleene logic: unlike [`Self::eval`], a
386    /// known `false` decides `And` and a known `true` decides `Or` beside
387    /// unknown operands, and `Select` with an unknown condition is known when
388    /// both arms agree. `operands` holds the input values for `In` and
389    /// `Parity`, and the operand values otherwise.
390    #[must_use]
391    pub fn combine_kleene(&self, operands: &[Option<bool>]) -> Option<bool> {
392        match self {
393            Self::In(_) => operands[0],
394            Self::Const(bit) => Some(*bit),
395            Self::Not(_) => operands[0].map(|bit| !bit),
396            Self::And(_) => {
397                if operands.contains(&Some(false)) {
398                    Some(false)
399                } else {
400                    (!operands.contains(&None)).then_some(true)
401                }
402            }
403            Self::Or(_) => {
404                if operands.contains(&Some(true)) {
405                    Some(true)
406                } else {
407                    (!operands.contains(&None)).then_some(false)
408                }
409            }
410            Self::Xor(_) => operands
411                .iter()
412                .try_fold(false, |value, &bit| Some(value ^ bit?)),
413            Self::Parity { constant, .. } => operands
414                .iter()
415                .try_fold(*constant, |value, &bit| Some(value ^ bit?)),
416            Self::Select(_) => match operands[0] {
417                Some(condition) => operands[1 + usize::from(condition)],
418                None if operands[1] == operands[2] => operands[1],
419                None => None,
420            },
421        }
422    }
423
424    /// Visit every `In(slot)` leaf, left to right.
425    pub fn for_each_input(&self, f: &mut impl FnMut(u32)) {
426        match self {
427            Self::In(slot) => f(*slot),
428            Self::Parity { inputs, .. } => inputs.iter().for_each(|&slot| f(slot)),
429            _ => {}
430        }
431        for operand in self.operands() {
432            operand.for_each_input(f);
433        }
434    }
435
436    /// Replace original leaves in order, moving replacement subtrees into
437    /// place: slot remapping (`f` returns another `In`) and producer inlining
438    /// (`f` returns a subtree) in one operation.
439    ///
440    /// Newly inserted leaves are not visited again.
441    pub(crate) fn replace_inputs(&mut self, f: &mut impl FnMut(u32) -> ClassicalExpr) {
442        match self {
443            Self::In(slot) => *self = f(*slot),
444            Self::Const(_) => {}
445            Self::Parity { inputs, constant } => {
446                let mut operands = Vec::with_capacity(inputs.len());
447                for &slot in inputs.iter() {
448                    operands.push(f(slot));
449                }
450                if operands
451                    .iter()
452                    .all(|operand| matches!(operand, Self::In(_)))
453                {
454                    for (slot, operand) in inputs.iter_mut().zip(operands) {
455                        let Self::In(mapped) = operand else {
456                            unreachable!("all replacement operands were checked as inputs")
457                        };
458                        *slot = mapped;
459                    }
460                } else {
461                    let expr = Self::xor(operands);
462                    *self = if *constant {
463                        Self::Not(Box::new(expr))
464                    } else {
465                        expr
466                    };
467                }
468            }
469            Self::Not(inner) => inner.replace_inputs(f),
470            Self::Xor(operands) | Self::And(operands) | Self::Or(operands) => {
471                for operand in operands {
472                    operand.replace_inputs(f);
473                }
474            }
475            Self::Select(operands) => {
476                for operand in operands.as_mut() {
477                    operand.replace_inputs(f);
478                }
479            }
480        }
481    }
482
483    /// Whether the expression is XOR-expressible (`In`/`Const`/`Not`/`Xor`/`Parity`
484    /// only). A static backend folds a linear condition into
485    /// `OBSERVABLE_INCLUDE`; a nonlinear (`And`/`Or`/`Select`) one needs a
486    /// dynamic backend.
487    pub fn is_linear(&self) -> bool {
488        match self {
489            Self::In(_) | Self::Const(_) | Self::Parity { .. } => true,
490            Self::Not(inner) => inner.is_linear(),
491            Self::Xor(operands) => operands.iter().all(Self::is_linear),
492            Self::And(..) | Self::Or(..) | Self::Select(..) => false,
493        }
494    }
495
496    /// Compact affine subexpressions after graph rewrites. Never cancel input
497    /// slots: Boolean equality alone does not preserve unavailable inputs.
498    pub(crate) fn needs_affine_compaction(&self) -> bool {
499        if matches!(self, Self::Not(_) | Self::Xor(_)) && self.is_linear() {
500            return true;
501        }
502        self.operands().iter().any(Self::needs_affine_compaction)
503    }
504
505    /// Compact affine subexpressions while preserving ordered input occurrences.
506    pub(crate) fn compact_affine(&mut self) {
507        if matches!(self, Self::In(_) | Self::Const(_) | Self::Parity { .. }) {
508            return;
509        }
510        if self.is_linear() {
511            let constant = self
512                .eval(&mut |_| Some(false))
513                .expect("constant assignment");
514            let mut inputs = Vec::new();
515            self.for_each_input(&mut |slot| inputs.push(slot));
516            *self = Self::parity(inputs, constant);
517            return;
518        }
519        match self {
520            Self::Not(inner) => inner.compact_affine(),
521            Self::Xor(operands) | Self::And(operands) | Self::Or(operands) => {
522                for operand in operands {
523                    operand.compact_affine();
524                }
525            }
526            Self::Select(operands) => {
527                for operand in operands.as_mut() {
528                    operand.compact_affine();
529                }
530            }
531            _ => unreachable!("leaves returned above"),
532        }
533    }

536/// Which temporal face of an instance a boundary operator binds.
537///
538/// The face fixes emission position: an `Input` (−Z) operator reads the qubit
539/// state entering the instance, so it emits *before* the instance circuit; an
540/// `Output` (+Z) operator reads on exit, so it emits *after*.
541#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, serde::Serialize, serde::Deserialize)]
542pub enum BoundaryFace {
543    /// State entering the instance.
544    Input,
545    /// State leaving the instance.
546    Output,
547}

549/// A symbolic logical operator at one template-instance boundary.
550///
551/// This is one binding payload element of [`ClassicalNode::Observable`]. It names the
552/// operator by instance reference (never per-shot bits), in instance-global
553/// qubit coordinates — a translated [`PauliMap`] at one temporal `face`.
554#[derive(Debug, Clone, PartialEq, Eq, Hash, serde::Serialize, serde::Deserialize)]
555pub struct InstanceBoundaryOperator {
556    /// Owning template instance.
557    pub instance: TemplateInstanceId,
558    /// Bound temporal face.
559    pub face: BoundaryFace,
560    /// Logical Pauli operator.
561    pub operator: PauliMap,
562}

564/// A structured control-flow region.
565///
566/// Its interface is the body's declared bit result and boundary bindings. Its predicate arrives on a
567/// `Value` edge; its body is a [`SubGraph`] over the same global template pool.
568#[derive(Debug, Clone)]
569pub enum RegionNode {
570    /// Region carrying a postselection/restart predicate. Execution retries a
571    /// sandboxed body until its restart predicate is false, subject to a
572    /// configured execution cap. Decoder results affect retries only through
573    /// the explicit predicate; physical restart parities can abort an attempt early.
574    ///
575    /// `restart_condition` reads incoming `Value` slots; unfed slots read the
576    /// optional body-local `restart_source`. Constant predicates need no source.
577    /// The body's declared result remains independent of the restart predicate.
578    RepeatUntilSuccess {
579        /// Retried body.
580        body: SubGraph,
581        /// Restart predicate.
582        restart_condition: ClassicalExpr,
583        /// Optional body-local value feeding the predicate's unfed slots.
584        /// Absent when every predicate input is supplied by incoming edges;
585        /// constant predicates need no source.
586        // Spec rule WF-10.
587        restart_source: Option<super::ValueRef>,
588    },
589}
590
591// Keep the existing RUS binary discriminant (1). The removed variant (0)
592// must be rejected rather than interpreted as a retry region.
593#[derive(serde::Serialize, serde::Deserialize)]
594enum RegionWire<B, E> {
595    Removed,
596    RepeatUntilSuccess {
597        body: B,
598        restart_condition: E,
599        restart_source: Option<super::ValueRef>,
600    },
601}
602
603impl serde::Serialize for RegionNode {
604    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
605        let Self::RepeatUntilSuccess {
606            body,
607            restart_condition,
608            restart_source,
609        } = self;
610        RegionWire::RepeatUntilSuccess {
611            body,
612            restart_condition,
613            restart_source: *restart_source,
614        }
615        .serialize(serializer)
616    }
617}

619impl<'de> serde::Deserialize<'de> for RegionNode {
620    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
621        match RegionWire::deserialize(deserializer)? {
622            RegionWire::Removed => Err(serde::de::Error::custom("removed region variant")),
623            RegionWire::RepeatUntilSuccess {
624                body,
625                restart_condition,
626                restart_source,
627            } => Ok(Self::RepeatUntilSuccess {
628                body,
629                restart_condition,

664/// Cumulative round cuts projecting a multi-layer block onto source z layers.
665#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
666pub struct QuantumTimeline {
667    /// Cumulative logical-round ends, one entry per occupied source z layer.
668    /// The first entry belongs to the node's lowest source layer, derived from
669    /// [`BloqNode::layer`]. Equal adjacent entries represent a layer containing
670    /// no circuit rounds.
671    pub layer_round_ends: Vec<u32>,
672}

674/// A template-backed quantum node: its ops come from instantiating the
675/// referenced templates from the pool, not from any per-node circuit.
676///
677/// Instances must have merge-compatible template entry bodies: aligned
678/// tick/repeat segments, matching repeat counts, and no same-segment qubit
679/// overlap except duplicate measurements or duplicate same-basis resets.
680/// [`Bloq::validate`](crate::Bloq::validate) enforces this for debug
681/// compilation, so instantiation and backends may assume it.
682#[derive(Debug, Clone, Default, serde::Serialize, serde::Deserialize)]
683pub struct QuantumNode {
684    /// Placed circuit templates.
685    pub instances: Vec<TemplateInstance>,
686    /// Cross-template detectors.
687    pub detectors: Vec<NodeDetector>,
688    /// Shared cross-template detector rows bound to this node's instances.
689    pub detector_bundles: Vec<DetectorBundleUse>,
690    /// Cross-instance restart syndromes; see [`NodeRestart`].
691    pub restarts: Vec<NodeRestart>,
692    /// Source-layer projection for a multi-z block component. Single-layer and
693    /// synthetic nodes need no timeline.
694    pub timeline: Option<QuantumTimeline>,
695    /// Conditional registrations in this component. Unlisted members and side
696    /// tables are common to every selection. All inputs must be ready before
697    /// the component starts; registration itself performs no quantum work.
698    pub guards: Vec<QuantumGuard>,
699}
700
701/// Members and side tables enabled by one quantum node's `Value` input.
702///
703/// Each instance/detector/restart may occur in at most one registration; a
704/// compound condition is an ordinary `Compute` producer feeding `input`.
705#[derive(Debug, Clone, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
706pub struct QuantumGuard {
707    /// Incoming selector slot.
708    pub input: u32,
709    /// Conditionally enabled instances.
710    pub instances: Vec<TemplateInstanceId>,
711    /// Conditionally enabled detector indices.
712    pub detectors: Vec<u32>,
713    /// Conditionally enabled detector bundle use indices.
714    pub detector_bundles: Vec<u32>,
715    /// Conditionally enabled restart indices.
716    pub restarts: Vec<u32>,
717    /// Parity contributions XORed into selected detector rows by this input.
718    pub detector_parities: Vec<(u32, crate::NodeDetectorParity)>,
719    /// Parity contributions XORed into selected restart rows by this input.
720    pub restart_parities: Vec<(u32, crate::NodeDetectorParity)>,
721}
722
723impl QuantumNode {
724    /// Every parity stored on this node: the inline detector rows, then the
725    /// restart rows, then each guard's side-table contributions.
726    ///
727    /// Pooled [`DetectorBundleUse`] rows are deliberately *not* included — they
728    /// live in the program's bundle pool, so a caller that needs whole-node
729    /// detector coverage walks [`Self::detector_bundles`] as well.
730    pub(crate) fn stored_parities(&self) -> impl Iterator<Item = &crate::NodeDetectorParity> {
731        self.detectors
732            .iter()
733            .map(|detector| &detector.parity)
734            .chain(self.restarts.iter().map(|restart| &restart.parity))
735            .chain(self.guards.iter().flat_map(|guard| {
736                guard
737                    .detector_parities
738                    .iter()
739                    .chain(&guard.restart_parities)
740                    .map(|(_, parity)| parity)
741            }))
742    }
743}

747        match &self.kind {
748            BloqNodeKind::Quantum(_) => "Quantum",
749            BloqNodeKind::Region(region) => region.kind_name(),

775        Self::quantum(QuantumNode::default()).with_provenance(provenance)
776    }

778    /// A template-backed quantum node with no provenance. Its payload shares

782            kind: BloqNodeKind::Quantum(quantum.into()),

791            kind: BloqNodeKind::Region(region),
792            provenance: NodeProvenance::None,
793            activation: None,
794        }
795    }

797    /// A classical-dataflow node ([`ClassicalNode`]), with no provenance.
798    /// Lowering stamps the real provenance ([`NodeProvenance::Generator`] /
799    /// [`NodeProvenance::Action`] / [`NodeProvenance::OutputFrame`]) via
800    /// [`Self::with_provenance`] where it knows the origin.
801    pub fn classical(node: impl Into<Arc<ClassicalNode>>) -> Self {
802        Self {
803            kind: BloqNodeKind::Classical(node.into()),
804            provenance: NodeProvenance::None,
805            activation: None,
806        }
807    }
808
809    /// The same node with its provenance replaced — builder-style, for lowering
810    /// sites that know what a classical node realizes.
811    pub fn with_provenance(mut self, provenance: NodeProvenance) -> Self {
812        self.provenance = provenance;
813        self
814    }
815
816    /// The node's quantum payload, **panicking** on a non-quantum node (the
817    /// rustc `expect_*` convention for panicking accessors). Most passes only
818    /// ever see quantum nodes (lowering produces no others on the static
819    /// path), so they call this directly; a pass that must tolerate other
820    /// kinds uses [`Self::try_quantum`] instead.
821    ///
822    /// # Panics
823    ///
824    /// Panics when this node is classical or a region.
825    pub fn expect_quantum(&self) -> &QuantumNode {
826        self.try_quantum()
827            .expect("expect_quantum() called on a non-quantum BloqNode")
828    }
829
830    /// Mutable twin of [`Self::expect_quantum`]; detaches shared quantum data.
831    /// Panics on a non-quantum node.
832    ///
833    /// # Panics
834    ///
835    /// Panics when this node is classical or a region.
836    pub fn expect_quantum_mut(&mut self) -> &mut QuantumNode {
837        match &mut self.kind {
838            BloqNodeKind::Quantum(quantum) => Arc::make_mut(quantum),
839            BloqNodeKind::Classical(_) | BloqNodeKind::Region(_) => {
840                panic!("expect_quantum_mut() called on a non-quantum BloqNode")
841            }
842        }
843    }
844
845    /// The node's quantum payload, or `None` for a non-quantum (classical or
846    /// region) node.
847    pub fn try_quantum(&self) -> Option<&QuantumNode> {
848        match &self.kind {
849            BloqNodeKind::Quantum(quantum) => Some(quantum),
850            BloqNodeKind::Classical(_) | BloqNodeKind::Region(_) => None,
851        }
852    }
853
854    /// The node's classical payload, or `None` for a non-classical node.
855    pub fn try_classical(&self) -> Option<&ClassicalNode> {
856        self.kind.try_classical()
857    }
858
859    /// Mutable classical payload, detaching a shared definition before editing.
860    pub fn try_classical_mut(&mut self) -> Option<&mut ClassicalNode> {
861        self.kind.try_classical_mut()
862    }
863
864    /// The node's region payload, or `None` for a non-region node.
865    pub fn try_region(&self) -> Option<&RegionNode> {
866        match &self.kind {
867            BloqNodeKind::Region(region) => Some(region),
868            BloqNodeKind::Quantum(_) | BloqNodeKind::Classical(_) => None,
869        }
870    }
871
872    /// Source block members, or an empty slice for another provenance kind.
873    pub fn block_members(&self) -> &[SourceBlockRef] {
874        match &self.provenance {
875            NodeProvenance::BlockComponent { members } => members,
876            _ => &[],
877        }
878    }
879
880    /// The spliced node's wait duration in syndrome-extraction rounds, or
881    /// `None` for anything that is not a wait on a seam.
882    ///
883    pub fn memory_rounds(&self) -> Option<u32> {
884        match &self.provenance {
885            NodeProvenance::MemoryPadding { rounds, .. } => Some(*rounds),
886            _ => None,
887        }
888    }
889
890    /// The node's spatial layer on the doubled-z axis: block layers land on
891    /// even values (`2z`) and the pipe seams between them on odd values
892    /// (`2z + 1`), so temporally adjacent blocks and their connecting seam
893    /// stay strictly ordered. Classical, frame, and action nodes have no
894    /// spatial layer and report `0`.
895    pub fn layer(&self) -> i64 {
896        match &self.provenance {
897            NodeProvenance::BlockComponent { members } => {
898                2 * i64::from(members.iter().map(|member| member.pos.z).min().unwrap_or(0))
899            }
900            NodeProvenance::TemporalPipe { pipe } | NodeProvenance::MemoryPadding { pipe, .. } => {
901                2 * i64::from(pipe.src.z.min(pipe.dst.z)) + 1
902            }
903            // Spec rule SEM-SPATIAL-PORT: positionless boundary layer.
904            NodeProvenance::SpatialPortSubstitution { source, role } => {
905                2 * i64::from(source.z) + if role.has_input_boundary() { -1 } else { 1 }
906            }
907            // Classical/frame/action nodes have no spatial layer of their own.
908            NodeProvenance::Generator { .. }
909            | NodeProvenance::Action { .. }
910            | NodeProvenance::BranchSelector { .. }
911            | NodeProvenance::OutputFrame { .. }
912            | NodeProvenance::None => 0,
913        }
914    }

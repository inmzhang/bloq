17/// Read a saved Bloq IR program, picking the codec by extension and falling
18/// back to "try text, then binary" for a renamed artifact.
19///
20/// Neither decoder validates what it decodes (see `Bloq::from_binary`), so
21/// callers acting on the result decide whether to run `Bloq::validate` first.
22pub(crate) fn load_bloq_ir(input: &Path) -> eyre::Result<Bloq> {
23    let bytes = std::fs::read(input).wrap_err_with(|| format!("read {}", input.display()))?;
24    match input.extension().and_then(|ext| ext.to_str()) {
25        Some(ext) if ext == bloq_ir::BLOQ_BINARY_EXTENSION => Bloq::from_binary(&bytes)
26            .wrap_err_with(|| format!("decode {} as binary Bloq IR", input.display())),
27        Some(ext) if ext == bloq_ir::BLOQ_TEXT_EXTENSION => {
28            let text = String::from_utf8(bytes)
29                .wrap_err_with(|| format!("read {} as UTF-8 text", input.display()))?;
30            Bloq::from_text(&text)
31                .wrap_err_with(|| format!("parse {} as Bloq IR text", input.display()))
32        }
33        _ => match std::str::from_utf8(&bytes) {
34            Ok(text) => Bloq::from_text(text).or_else(|_| decode_binary_fallback(input, &bytes)),
35            Err(_) => decode_binary_fallback(input, &bytes),
36        },
37    }
38}
39
40fn decode_binary_fallback(input: &Path, bytes: &[u8]) -> eyre::Result<Bloq> {
41    Bloq::from_binary(bytes).wrap_err_with(|| {
42        format!(
43            "decode {} as Bloq IR (tried text and binary; name it .{} or .{} to pick a codec)",
44            input.display(),
45            bloq_ir::BLOQ_TEXT_EXTENSION,
46            bloq_ir::BLOQ_BINARY_EXTENSION
47        )
48    })
49}

51/// Reject an existing output path that names the input file through another
52/// spelling, symbolic link, or hard link.
53pub(crate) fn reject_input_output_alias(input: &Path, output: &Path) -> eyre::Result<()> {
54    if output
55        .try_exists()
56        .wrap_err_with(|| format!("inspect output file {}", output.display()))?
57        && same_file::is_same_file(input, output).wrap_err_with(|| {
58            format!(
59                "compare input file {} with output file {}",
60                input.display(),
61                output.display()
62            )
63        })?
64    {
65        eyre::bail!(
66            "output file {} aliases the input file; choose a different output path",
67            output.display()
68        );
69    }
70    Ok(())
71}

73/// Text or binary output shared by compile and emit commands.
74#[derive(Debug)]
75enum EmittedArtifact {
76    Text(String),
77    Binary(Vec<u8>),
78}
79
80impl EmittedArtifact {
81    fn bytes(&self) -> &[u8] {
82        match self {
83            Self::Text(text) => text.as_bytes(),
84            Self::Binary(bytes) => bytes,
85        }
86    }
87
88    fn write_stdout(&self) -> eyre::Result<()> {
89        terminal::write_stdout(self.bytes()).wrap_err("write artifact to stdout")?;
90        if let Self::Text(text) = self
91            && !text.ends_with('\n')
92        {
93            terminal::write_stdout(b"\n").wrap_err("terminate text artifact on stdout")?;
94        }
95        Ok(())
96    }
97}

99impl BuiltInBackend {
100    fn emit(
101        self,
102        program: &Bloq,
103        options: &bloq_stim::BloqStimOptions,
104    ) -> eyre::Result<EmittedArtifact> {
105        Ok(match self {
106            Self::Stim => EmittedArtifact::Text(
107                bloq_stim::emit_bloq_stim_with(program, options).wrap_err("emit Stim text")?,
108            ),
109            Self::IrText => EmittedArtifact::Text(program.to_text()),
110            Self::IrBinary => EmittedArtifact::Binary(program.to_binary()),
111        })
112    }
113}

115/// Replace `path` only after a complete same-directory temporary file exists.
116fn atomic_write(path: &Path, bytes: &[u8]) -> std::io::Result<()> {
117    atomic_write_with(path, |temporary| std::fs::write(temporary, bytes))
118}
119
120fn atomic_write_with(
121    path: &Path,
122    write: impl FnOnce(&Path) -> std::io::Result<()>,
123) -> std::io::Result<()> {
124    let parent = path
125        .parent()
126        .filter(|parent| !parent.as_os_str().is_empty())
127        .unwrap_or_else(|| Path::new("."));
128    path.file_name().ok_or_else(|| {
129        std::io::Error::new(
130            std::io::ErrorKind::InvalidInput,
131            "output path has no file name",
132        )
133    })?;
134    let permissions = match std::fs::symlink_metadata(path) {
135        Ok(metadata) if metadata.file_type().is_symlink() => {
136            return Err(std::io::Error::new(
137                std::io::ErrorKind::InvalidInput,
138                format!(
139                    "refusing to replace symbolic-link output {}",
140                    path.display()
141                ),
142            ));
143        }
144        Ok(metadata) => {
145            // Replacing via rename only checks the parent directory. Preserve
146            // the previous write contract by checking the target itself too.
147            drop(std::fs::OpenOptions::new().write(true).open(path)?);
148            Some(metadata.permissions())
149        }
150        Err(error) if error.kind() == std::io::ErrorKind::NotFound => None,
151        Err(error) => return Err(error),
152    };
153
154    let mut builder = tempfile::Builder::new();
155    #[cfg(unix)]
156    {
157        use std::os::unix::fs::PermissionsExt as _;
158
159        // Match ordinary file creation: 0666 filtered through the process umask.
160        builder.permissions(std::fs::Permissions::from_mode(0o666));
161    }
162    let mut temporary = builder.tempfile_in(parent)?;
163    write(temporary.path())?;
164    temporary.flush()?;
165    if let Some(permissions) = permissions {
166        temporary.as_file().set_permissions(permissions)?;
167    }
168    temporary
169        .persist(path)
170        .map(|_| ())
171        .map_err(|error| error.error)
172}

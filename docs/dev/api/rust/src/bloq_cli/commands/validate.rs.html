7/// Check a saved Bloq IR program against the full T2 well-formedness gate.
8///
9/// Ordinary compilation and emission do not run this full audit in either
10/// build mode, and neither decoder — `from_text` / `from_binary` — runs it.
11/// This command is how a program that was written to disk, hand-edited, or
12/// received from elsewhere gets checked on demand.
13///
14/// This deliberate off-hot-path check includes per-node SEM-MERGE.
15pub(crate) fn run(input: &Path, quiet: bool) -> eyre::Result<()> {
16    let program = super::load_bloq_ir(input)?;
17    program
18        .validate()
19        .wrap_err_with(|| format!("validate {}", input.display()))?;
20    if !quiet {
21        terminal::status(
22            "Validated",
23            format!(
24                "{} ({} quantum node(s), {} template(s))",
25                input.display(),
26                program.quantum_node_count(),
27                program.templates().len()
28            ),
29        );
30    }
31    Ok(())
32}

8/// Emit a backend target from a saved Bloq IR program.
9///
10/// The counterpart to `compile`, which goes from a `.blog` block graph to the
11/// same targets in one step. Splitting the two lets a compiled program be
12/// stored once and re-emitted — and makes IR-to-IR codec conversion fall out
13/// for free.
14pub(crate) fn run(options: EmitCmd, quiet: bool) -> eyre::Result<()> {
15    if options.align_moments && options.backend != BuiltInBackend::Stim {
16        eyre::bail!("--align-moments requires --backend stim");
17    }
18    let input = &options.input;
19    let program = super::load_bloq_ir(input)?;
20    let trust = if options.validate {
21        bloq_stim::InputTrust::Checked
22    } else {
23        bloq_stim::InputTrust::Trusted
24    };
25    // The Stim backend checks only when requested through `trust`. IR exchange
26    // codecs do not check, so request the same full gate here explicitly.
27    if options.validate && options.backend != BuiltInBackend::Stim {
28        program
29            .validate()
30            .wrap_err_with(|| format!("validate {}", input.display()))?;
31    }
32
33    let emitted = options.backend.emit(
34        &program,
35        &bloq_stim::BloqStimOptions::new()
36            .with_trust(trust)
37            .with_align_moments(options.align_moments),
38    )?;
39
40    if options.print {
41        return emitted.write_stdout();
42    }
43
44    let path = options
45        .output
46        .unwrap_or_else(|| input.with_extension(options.backend.file_extension()));
47    super::reject_input_output_alias(input, &path)?;
48    atomic_write(&path, emitted.bytes())
49        .wrap_err_with(|| format!("write output file {}", path.display()))?;
50    if !quiet {
51        terminal::status(
52            "Emitted",
53            format!("{} -> {}", input.display(), path.display()),
54        );
55    }
56    Ok(())
57}

64    #[test]
65    fn align_moments_rejects_ir_backend_before_loading() {
66        let options = EmitCmd {
67            input: "missing.bloq".into(),
68            backend: BuiltInBackend::IrText,
69            output: None,
70            print: false,
71            validate: true,
72            align_moments: true,
73        };
74
75        let error = run(options, true).expect_err("moment alignment only changes Stim emission");
76
77        assert!(error.to_string().contains("requires --backend stim"));
78    }

80    #[test]
81    fn ir_exchange_runs_full_validation_only_when_requested() {
82        let dir = tempdir().expect("create tempdir");
83        let input = dir.path().join("invalid.bloqir");
84        std::fs::write(&input, "BLOQIR 1\ngraph {\n  n0 compute in0\n}\n")
85            .expect("write structurally invalid IR");
86        let options = |validate| EmitCmd {
87            input: input.clone(),
88            backend: BuiltInBackend::IrText,
89            output: Some(dir.path().join("roundtrip.bloqir")),
90            print: false,
91            validate,
92            align_moments: false,
93        };
94
95        run(options(false), true).expect("IR exchange does not run a full audit");
96        assert!(dir.path().join("roundtrip.bloqir").exists());
97        let error = run(options(true), true).expect_err("explicit audit rejects invalid IR");
98        assert!(format!("{error:?}").contains("WF-11"));
99    }

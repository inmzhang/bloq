3/// Append a bracketed Stim instruction tag, escaping delimiters and line breaks.
4/// Shared by circuit emission and tools that rewrite existing Stim text.
5pub fn write_stim_tag(output: &mut String, tag: &str) {
6    output.push('[');
7    for character in tag.chars() {
8        match character {
9            ']' => output.push_str("\\C"),
10            '\\' => output.push_str("\\B"),
11            '\n' => output.push_str("\\n"),
12            '\r' => output.push_str("\\r"),
13            other => output.push(other),
14        }
15    }
16    output.push(']');
17}

19/// Append an integer's decimal digits. Uses `itoa` instead of `write!` because
20/// emission spends a measurable share of its time formatting qubit indices,
21/// measurement lookbacks, and coordinates, and the `core::fmt` machinery costs
22/// several times more than `itoa`'s direct digit loop.
23pub(crate) fn push_int<T: itoa::Integer>(output: &mut String, value: T) {
24    output.push_str(itoa::Buffer::new().format(value));
25}
26
27/// Append a coordinate/argument, preferring the integer spelling (`1`, not
28/// `1.0`) whenever the value is integral. Stim writes detector and qubit
29/// coordinates as plain integers, so an integral `f64` must render without a
30/// fractional part; genuinely fractional values fall back to `f64` `Display`.
31pub(crate) fn push_number_prefer_int(output: &mut String, value: f64) {
32    // i64::MAX rounds up to 2^63 as f64; exclude that endpoint before casting.
33    if value >= i64::MIN as f64 && value < -(i64::MIN as f64) {
34        let as_int = value as i64;
35        if as_int as f64 == value {
36            return push_int(output, as_int);
37        }
38    }
39    write!(output, "{value}").expect("writing to a String is infallible");
40}

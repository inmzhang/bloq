1//! Compile-time producer of the IR's memory-padding provenance
2//! ([`bloq_ir::PipePadding`]).
3//!
4//! [`record_edge_padding`] stamps every padding-capable quantum edge, including
5//! region boundaries, with
6//! precompiled one-round and looped padding templates plus the offsets placing
7//! them over the edge's cross-section, so
8//! [`bloq_ir::Bloq::insert_memory_rounds`] can splice waits into a deserialized
9//! program without the source graph or a recompile. A temporal Hadamard is a
10//! real IR node, so its parent and child edges are stamped independently from
11//! their adjacent cube faces.
12

22/// The round count the recorded looped padding template is compiled at. Any
23/// value whose repeat body survives chunk normalization (>= 3, i.e. two or
24/// more repetitions) works; insertion specializes this padding template's
25/// repetition count to the requested duration.
26const LOOPED_PADDING_ROUNDS: u32 = 3;
27
28/// The patch inputs a pipe's padding template is compiled from, plus the
29/// per-block offset placing it.
30///
31/// The patch is a plain `d x d` surface code face named by the basis on its
32/// `y`-normal edges. That one bit is the whole space: a cube admits a temporal
33/// pipe only when its odd-basis axis is spatial, and neither the temporal
34/// basis, spatial connectivity, nor the layer's CX-slot count can reach a
35/// padding round. `fixed_bulk::tests::seam_padding_matches_the_cube_derived_patch`
36/// pins this against cube-derived padding.
37struct PipePatch {
38    y_face_basis: Basis,
39    offset: IVec2,
40}
41
42/// Immutable padding templates shared by every compilation that reaches this
43/// cache — one [`crate::CompileContext`], or all the contexts sharing a
44/// [`crate::SharedCompileCache`] shard. Program-local template ids are still
45/// assigned by each output [`Bloq`].
46///
47/// The lock is internal so the padding pass can hold `&self` for its whole walk
48/// without serializing concurrent compilations behind it: only the map probe
49/// and insert are guarded, never [`compile_padding_templates`].
50#[derive(Debug, Default)]
51pub(crate) struct PaddingTemplateCache {
52    templates: Mutex<crate::FxMap<Basis, CachedPaddingTemplates>>,
53}
54
55impl PaddingTemplateCache {
56    fn get_or_compile(
57        &self,
58        y_face_basis: Basis,
59        distance: u32,
60    ) -> Result<CachedPaddingTemplates, CompileError> {
61        if let Some(templates) = crate::cache::lock(&self.templates).get(&y_face_basis) {
62            return Ok(templates.clone());
63        }
64        let templates = compile_padding_templates(y_face_basis, distance)?;
65        // A concurrent miss on the same patch may have won the race; its entry
66        // is just as good, and keeping it avoids invalidating what it returned.
67        Ok(crate::cache::lock(&self.templates)
68            .entry(y_face_basis)
69            .or_insert(templates)
70            .clone())
71    }
72
73    pub(crate) fn prepare(
74        &self,
75        needed: bool,
76        distance: u32,
77    ) -> Result<PreparedPaddingTemplates, CompileError> {
78        if !needed {
79            return Ok(PreparedPaddingTemplates::default());
80        }
81        let mut templates = crate::FxMap::default();
82        for basis in [Basis::X, Basis::Z] {
83            templates.insert(basis, self.get_or_compile(basis, distance)?);
84        }
85        Ok(PreparedPaddingTemplates { templates })
86    }
87}

96/// The block whose side of the seam the wait sits on.
97///
98/// A wait idles the patch its *lower* node handed over, so the lower node's
99/// block is the anchor. Around a materialized temporal Hadamard only one of the
100/// two IR edges has a lower node that owns a block — the other's lower node is
101/// the Hadamard itself — and there the upper node's block is the one whose patch
102/// idles. If the other endpoint is a region, the temporal-pipe node itself
103/// identifies which side of the Hadamard the seam occupies.
104fn seam_anchor(
105    bloq: &Bloq,
106    graph: &BlockGraph,
107    endpoints: (BloqNodeId, BloqNodeId),
108    pipe: &TemporalPipeRef,
109) -> Option<IVec3> {
110    let (from, to) = endpoints;
111    let (lower, upper) = pipe.endpoints_by_z();
112    if owns_endpoint(bloq, graph, from, lower) {
113        return Some(lower);
114    }
115    if owns_endpoint(bloq, graph, to, upper) {
116        return Some(upper);
117    }
118    // A materialized temporal Hadamard owns the pipe rather than either
119    // endpoint. On its upper edge the waiting patch is already Hadamard-
120    // transformed, including when the upper endpoint lowers to a region.
121    let is_pipe_node = |id| {
122        matches!(
123            bloq.node(id).map(|node| &node.provenance),
124            Some(NodeProvenance::TemporalPipe { pipe: node_pipe }) if node_pipe == pipe
125        )
126    };
127    if is_pipe_node(from) {
128        return Some(upper);
129    }
130    if is_pipe_node(to) {
131        return Some(lower);
132    }
133    // A region on an ordinary pipe (a RUS escape or selective fix) leaves the
134    // seam anchored on its source patch.
135    let is_region = |id| matches!(bloq.node(id), Some(node) if node.try_region().is_some());
136    (is_region(from) || is_region(to)).then_some(lower)
137}

139/// Whether IR node `id` was lowered from the block covering `endpoint`.
140/// Resolved through [`BlockGraph::get_endpoint_block`] so a block addressed by
141/// its far end (a patch rotation) still matches its own source ref.
142fn owns_endpoint(bloq: &Bloq, graph: &BlockGraph, id: BloqNodeId, endpoint: IVec3) -> bool {
143    let Some(block) = graph.get_endpoint_block(endpoint) else {
144        return false;
145    };
146    matches!(
147        bloq.node(id).map(|node| &node.provenance),
148        Some(NodeProvenance::BlockComponent { members })
149            if members.iter().any(|member| member.pos == block.pos())
150    )
151}

153/// The patch one IR edge's temporal pipe waits on.
154///
155/// The basis is read off the *pipe*, not off a cube: `infer_pipe_basis_from_endpoint`
156/// reports the pipe's face bases as seen from one end, deriving them from
157/// whichever end is a cube and flipping across a Hadamard. That is what lets a
158/// seam terminating in a port, a `Y`, or a `T` injection resolve at all — those
159/// blocks have no bases of their own, and the pipe's are exactly what idles.
160fn edge_pipe_patch(
161    bloq: &Bloq,
162    graph: &BlockGraph,
163    endpoints: (BloqNodeId, BloqNodeId),
164    pipe: &TemporalPipeRef,
165    distance: u32,
166) -> Result<Option<PipePatch>, CompileError> {
167    let (from, to) = endpoints;
168    let (Some(from_node), Some(to_node)) = (bloq.node(from), bloq.node(to)) else {
169        return Ok(None);
170    };
171    let addressable =
172        |node: &bloq_ir::BloqNode| node.try_quantum().is_some() || node.try_region().is_some();
173    if !addressable(from_node) || !addressable(to_node) {
174        return Ok(None);
175    }
176    // The positionless temporal half shares its source address with the
177    // expanded cube. Its virtual seam idles that cube's actual patch.
178    if pipe.src == pipe.dst
179        && (matches!(to_node.provenance, NodeProvenance::SpatialPortSubstitution { source, role: PortRole::Output } if source == pipe.src)
180            || matches!(from_node.provenance, NodeProvenance::SpatialPortSubstitution { source, role } if source == pipe.src && role.has_input_boundary()))
181    {
182        let Some(BlockKind::Cube(kind)) = graph.get_block(pipe.src).map(bloq_graph::Block::kind)
183        else {
184            return Ok(None);
185        };
186        return Ok(Some(PipePatch {
187            y_face_basis: kind.y(),
188            offset: crate::compile::block_xy_offset(pipe.src.truncate(), distance)?,
189        }));
190    }
191    let Some(anchor) = seam_anchor(bloq, graph, endpoints, pipe) else {
192        return Ok(None);
193    };
194    let Some(source) = graph.get_pipe(pipe.src, pipe.dst) else {
195        return Ok(None);
196    };
197    let [_, y_face_basis, _] = graph.infer_pipe_basis_from_endpoint(source, anchor);
198    let Some(y_face_basis) = y_face_basis else {
199        return Ok(None);
200    };
201    Ok(Some(PipePatch {
202        y_face_basis,
203        // A temporal pipe's two endpoints share a column, so the seam's
204        // cross-section sits at the anchor's own layout position.
205        offset: crate::compile::block_xy_offset(anchor.truncate(), distance)?,
206    }))
207}

209/// Stamp each padding-capable `Quantum` edge in `bloq` with one
210/// [`PipePadding`] per pipe. One-round and looped
211/// ([`LOOPED_PADDING_ROUNDS`]) templates are deduped per patch shape.
212///
213/// Edges padding cannot fully cover — empty seams, unsupported endpoints, or a
214/// pipe with no resolvable patch — retain empty provenance rather than
215/// failing compilation. Region-boundary records also drive
216/// [`bloq_ir::MemoryRoundTarget::After`].
217/// Template compilation errors on a supported cube are real compile bugs and
218/// propagate.
219pub(crate) fn record_edge_padding(
220    prepared: &PreparedPaddingTemplates,
221    bloq: &mut Bloq,
222    graph: &BlockGraph,
223    distance: u32,
224) -> Result<(), CompileError> {
225    // Only top-level seams carry source pipes today (region-body Quantum edges
226    // are synthetic), so this walk covers both ordinary and region-boundary
227    // padding. Collected first: recording mutates the template pool.
228    let edges: Vec<(BloqNodeId, BloqNodeId, Vec<TemporalPipeRef>)> = bloq
229        .edges()
230        .filter_map(|edge| match edge.edge {
231            BloqEdge::Quantum(quantum) => Some((
232                edge.source,
233                edge.target,
234                quantum.pipes.iter().map(|seam| seam.pipe).collect(),
235            )),
236            BloqEdge::Value { .. } | BloqEdge::Compose { .. } | BloqEdge::Order => None,
237        })
238        .collect();
239
240    let mut by_patch: crate::FxMap<Basis, PaddingTemplates> = crate::FxMap::default();
241    for (from, to, pipes) in edges {
242        if pipes.is_empty() {
243            continue;
244        }
245        let mut patches = Vec::with_capacity(pipes.len());
246        for pipe in &pipes {
247            let Some(patch) = edge_pipe_patch(bloq, graph, (from, to), pipe, distance)? else {
248                patches.clear();
249                break;
250            };
251            patches.push(patch);
252        }
253        if patches.len() != pipes.len() {
254            continue;
255        }
256
257        let mut refs = Vec::with_capacity(pipes.len());
258        for patch in &patches {
259            let key = patch.y_face_basis;
260            let templates = match by_patch.get(&key) {
261                Some(&templates) => templates,
262                None => {
263                    let cached = prepared
264                        .templates
265                        .get(&key)
266                        .expect("module-object preparation covers both patch bases")
267                        .clone();
268                    let templates = PaddingTemplates {
269                        one_round: bloq.add_shared_template(cached.one_round),
270                        looped: bloq.add_shared_template(cached.looped),
271                    };
272                    by_patch.insert(key, templates);
273                    templates
274                }
275            };
276            refs.push(PipePadding {
277                offset: patch.offset,
278                one_round: templates.one_round,
279                looped: templates.looped,
280            });
281        }
282        assert!(
283            bloq.set_quantum_edge_padding(from, to, refs),
284            "lowering emits one quantum edge per node pair"
285        );
286    }
287    Ok(())
288}

290/// The two padding templates registered for one cube patch. A named pair so
291/// the one-round and looped ids cannot be transposed at construction (both are
292/// `TemplateId`, so a positional tuple would swap silently).
293#[derive(Debug, Clone, Copy)]
294struct PaddingTemplates {
295    one_round: bloq_ir::TemplateId,
296    looped: bloq_ir::TemplateId,
297}
298
299#[derive(Debug, Clone)]
300struct CachedPaddingTemplates {
301    one_round: Arc<bloq_ir::lowering::BloqTemplate>,
302    looped: Arc<bloq_ir::lowering::BloqTemplate>,
303}
304
305/// Compile the one-round and looped padding templates for one patch shape.
306fn compile_padding_templates(
307    y_face_basis: Basis,
308    distance: u32,
309) -> Result<CachedPaddingTemplates, CompileError> {
310    // Both round counts pad the same patch shape, so the entry point builds its
311    // Theta(d^2) tiles once and compiles each count against it.
312    let [one_round, looped] =
313        compile_seam_padding_rounds(y_face_basis, distance, [1, LOOPED_PADDING_ROUNDS])?;
314    Ok(CachedPaddingTemplates { one_round, looped })
315}

318mod tests {
319    use bloq_ir::{LevelPath, MemoryRoundTarget};

323    #[test]
324    fn virtual_spatial_inputs_support_physical_memory_padding() {
325        for role in ["input", "multiplex"] {
326            let graph = BlockGraph::from_blog_text(&format!(
327                "BLOG 1.0\n0: Port [-1,0,0] role={role}\n1: ZXZ [0,0,0]\n2: Port [0,0,1] role=output\n0 -> +X\n1 -> +Z\n"
328            )).unwrap();
329            let mut program = crate::compile(&graph, 3).unwrap();
330            let input = program.nodes().find(|(_, node)| matches!(node.provenance,
331                NodeProvenance::SpatialPortSubstitution { role, .. } if role.has_input_boundary()
332            )).unwrap().0;
333            let target = program
334                .outgoing(input)
335                .find(|edge| matches!(edge.edge, BloqEdge::Quantum(_)))
336                .unwrap()
337                .target;
338            program
339                .insert_memory_rounds(
340                    MemoryRoundTarget::Edge {
341                        path: LevelPath::default(),
342                        from: input,
343                        to: target,
344                    },
345                    2,
346                )
347                .unwrap();
348            program.validate().unwrap();
349            let report = bloq_vm::run_bloq_with_io(
350                &program,
351                4,
352                7,
353                |sim, ctx| {
354                    for input in ctx.inputs {
355                        sim.cx(input.qubit, sim.num_qubits())?;
356                    }
357                    Ok(())
358                },
359                |_, _| Ok(()),
360            )
361            .unwrap();
362            assert_eq!(report.discarded, 0);
363            assert!(
364                report
365                    .detectors
366                    .iter()
367                    .all(|detector| detector.constant && detector.value == Some(false))
368            );
369        }
370    }

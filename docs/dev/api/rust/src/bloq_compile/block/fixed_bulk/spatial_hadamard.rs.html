27/// What one endpoint cube contributes to the wall.
28///
29/// The wall owns the circuit of the region it stands in, so it has to reproduce
30/// the collapse decisions its neighbours make for the data column they share
31/// with it — the cubes emit the same resets and readouts, and the two copies
32/// dedup when the node's instances merge.
33#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
34pub(crate) struct WallSide {
35    /// The cube's temporal basis `z()`: the basis it collapses data in.
36    pub(crate) temporal_basis: Basis,
37    /// The endpoint's complete pipe connectivity. The wall derives data
38    /// preparation, readout, perpendicular arms, and shared corners from this
39    /// single source of truth.
40    pub(crate) connectivity: Connectivity,
41}
42
43/// Everything a positive-axis spatial Hadamard wall is compiled from, and its
44/// template cache key. The code distance is supplied separately, exactly as it
45/// is for the realignment: it is a property of the whole compilation, not of
46/// one pipe.
47#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
48pub(crate) struct SpatialHadamardKey {
49    pub(crate) axis: UDirection,
50    /// Syndrome rounds of the endpoint cubes, which the wall matches round for
51    /// round.
52    pub(crate) rounds: u32,
53    /// The negative-axis cube's perpendicular face basis. Both halves of an
54    /// end row agree on whether it survives, because the positive-axis cube's
55    /// corresponding basis is its flip.
56    pub(crate) boundary_basis: Basis,
57    pub(crate) minus: WallSide,
58    pub(crate) plus: WallSide,
59}
60
61impl SpatialHadamardKey {
62    fn resets(self, side: WallSide) -> bool {
63        !side.connectivity.has_pipe(Direction::ZMINUS)
64    }
65
66    fn measures(self, side: WallSide) -> bool {
67        !side.connectivity.has_pipe(Direction::ZPLUS)
68    }
69
70    fn arm(self, side: WallSide, positive: bool) -> bool {
71        let direction = match (self.axis, positive) {
72            (UDirection::X, true) => Direction::YPLUS,
73            (UDirection::X, false) => Direction::YMINUS,
74            (UDirection::Y, true) => Direction::XPLUS,
75            (UDirection::Y, false) => Direction::XMINUS,
76            (UDirection::Z, _) => unreachable!("a spatial wall has an X or Y axis"),
77        };
78        side.connectivity.has_open_edge(direction)
79    }
80}

86/// CX slots per wall round.
87const WALL_CX_SLOTS: usize = 6;
88
89/// The negative-axis half's X-wall data corners and the slots they occupy.
90/// Slot `k` is moment `k + 1`; the GHZ grows at slots 0/1 and is undone at
91/// slots 4/5, so its data interactions sit strictly between.
92const X_MINUS_SLOTS: [(Corner, usize); 2] = [(Corner::TL, 1), (Corner::BL, 3)];
93/// The positive-axis X-wall half's, one slot later on both counts.
94const X_PLUS_SLOTS: [(Corner, usize); 2] = [(Corner::TR, 2), (Corner::BR, 4)];
95/// The Y wall is the diagonal transpose of the X wall.
96const Y_MINUS_SLOTS: [(Corner, usize); 2] = [(Corner::BR, 1), (Corner::BL, 3)];
97const Y_PLUS_SLOTS: [(Corner, usize); 2] = [(Corner::TR, 2), (Corner::TL, 4)];
98
99const GROW_MINUS_SLOT: usize = 0;
100const GROW_PLUS_SLOT: usize = 1;
101const UNDO_MINUS_SLOT: usize = 4;
102const UNDO_PLUS_SLOT: usize = 5;
103
104/// One extended stabilizer: two ancillas straddling the consumed column, the
105/// link qubit chaining them into a GHZ state, and up to four data qubits split
106/// between the two cubes.
107struct WallTile {
108    /// The negative-axis ancilla. Its checkerboard colour is [`basis`].
109    ///
110    /// [`basis`]: Self::basis
111    minus: IVec2,
112    /// The positive-axis ancilla, coloured [`basis`]`.flip()`. It is the
113    /// chain's `MX` readout in forward rounds; the negative-axis ancilla is in
114    /// reversed rounds.
115    ///
116    /// [`basis`]: Self::basis
117    plus: IVec2,
118    basis: Basis,
119    /// Active negative-half data, paired with the slot that drives it.
120    minus_data: SmallVec<[(IVec2, usize); 2]>,
121    /// Active positive-half data, in `basis.flip()`.
122    plus_data: SmallVec<[(IVec2, usize); 2]>,
123    axis: UDirection,
124}
125
126impl WallTile {
127    /// Detector coordinate: the middle of the consumed column, on the tile's row.
128    fn center(&self) -> IVec2 {
129        (self.minus + self.plus) / 2
130    }
131
132    fn link_for(&self, reversed: bool) -> IVec2 {
133        let side = if reversed { -1 } else { 1 };
134        self.minus + wall_coord(self.axis, 1, side)
135    }
136}
137
138fn wall_coord(axis: UDirection, across: i32, along: i32) -> IVec2 {
139    match axis {
140        UDirection::X => IVec2::new(across, along),
141        UDirection::Y => IVec2::new(along, across),
142        UDirection::Z => unreachable!("a spatial wall has an X or Y axis"),
143    }
144}
145
146fn along_coord(axis: UDirection, qubit: IVec2) -> i32 {
147    match axis {
148        UDirection::X => qubit.y,
149        UDirection::Y => qubit.x,
150        UDirection::Z => unreachable!("a spatial wall has an X or Y axis"),
151    }
152}
153
154/// Coordinates along `side`'s seam-adjacent data column.
155///
156/// The `d` interior rows are always there; a perpendicular arm opens the cube's
157/// perpendicular edge, which activates that boundary cell's outward corners
158/// and extends the column one row past the wall's end.
159fn has_seam_data(distance: u32, key: SpatialHadamardKey, side: WallSide, along: i32) -> bool {
160    let d = distance as i32;
161    match along {
162        -1 => key.arm(side, false),
163        along if along == 2 * d + 1 => key.arm(side, true),
164        along => (1..2 * d).contains(&along),
165    }
166}
167
168/// Whether `along` is one of the two rows a perpendicular arm adds. Those
169/// qubits sit on an open spatial edge, so their cube prepares and reads them
170/// out unconditionally — the same override `grid.rs` applies.
171fn is_arm_row(distance: u32, along: i32) -> bool {
172    along == -1 || along == 2 * distance as i32 + 1
173}
174
175/// The wall's extended stabilizers, bottom row first.
176///
177/// One per measure row of the cubes' cell grid, minus the end row whose colour
178/// disagrees with the cubes' wall-perpendicular boundary basis: there the cubes
179/// keep their own weight-2 boundary plaquettes, which share exactly one data
180/// qubit with the would-be end tile and would anticommute with it. The two end
181/// rows always disagree with each other (`d` is odd), so exactly one survives.
182fn wall_tiles(distance: u32, key: SpatialHadamardKey) -> Vec<WallTile> {
183    let d = distance as i32;
184    let mut tiles = Vec::with_capacity(distance as usize + 1);
185    let (minus_slots, plus_slots) = match key.axis {
186        UDirection::X => (X_MINUS_SLOTS, X_PLUS_SLOTS),
187        UDirection::Y => (Y_MINUS_SLOTS, Y_PLUS_SLOTS),
188        UDirection::Z => unreachable!("a spatial wall has an X or Y axis"),
189    };
190    for along in (0..=2 * d).step_by(2) {
191        let minus = wall_coord(key.axis, 2 * d, along);
192        let plus = wall_coord(key.axis, 2 * d + 2, along);
193        let basis = checkerboard_basis(minus);
194        if (along == 0 || along == 2 * d) && basis != key.boundary_basis {
195            continue;
196        }
197        let half = |anchor: IVec2, side: WallSide, slots: [(Corner, usize); 2]| {
198            slots
199                .into_iter()
200                .filter_map(|(corner, slot)| {
201                    let qubit = anchor + corner.to_ivec2();
202                    has_seam_data(distance, key, side, along_coord(key.axis, qubit))
203                        .then_some((qubit, slot))
204                })
205                .collect::<SmallVec<[(IVec2, usize); 2]>>()
206        };
207        tiles.push(WallTile {
208            minus,
209            plus,
210            basis,
211            minus_data: half(minus, key.minus, minus_slots),
212            plus_data: half(plus, key.plus, plus_slots),
213            axis: key.axis,
214        });
215    }
216    tiles
217}

219/// Which temporal face a collapse map describes.
220#[derive(Debug, Clone, Copy, PartialEq, Eq)]
221enum Face {
222    Init,
223    Meas,
224}
225
226/// The data qubits the wall collapses at `face`, in their owning cube's temporal
227/// basis. Mirrors the cube rule: collapse unless a pipe carries the state across
228/// that face, and always on an open spatial edge.
229fn collapse_map(
230    distance: u32,
231    key: SpatialHadamardKey,
232    tiles: &[WallTile],
233    face: Face,
234) -> crate::FxMap<IVec2, Basis> {
235    let mut map = crate::FxMap::default();
236    let mut collect = |data: &SmallVec<[(IVec2, usize); 2]>, side: WallSide| {
237        let collapses = match face {
238            Face::Init => key.resets(side),
239            Face::Meas => key.measures(side),
240        };
241        for &(qubit, _) in data {
242            if collapses || is_arm_row(distance, along_coord(key.axis, qubit)) {
243                map.insert(qubit, side.temporal_basis);
244            }
245        }
246    };
247    for tile in tiles {
248        collect(&tile.minus_data, key.minus);
249        collect(&tile.plus_data, key.plus);
250    }
251    map
252}

258fn push_interaction(
259    cx: &mut Vec<IVec2>,
260    cz: &mut Vec<IVec2>,
261    ancilla: IVec2,
262    data: IVec2,
263    basis: Basis,
264) {
265    match basis {
266        Basis::X => cx.extend([ancilla, data]),
267        Basis::Z => cz.extend([ancilla, data]),
268    }
269}

271fn data_slot(slot: usize, reversed: bool) -> usize {
272    if reversed {
273        WALL_CX_SLOTS - 1 - slot
274    } else {
275        slot
276    }
277}

279/// Emit one wall round.
280fn emit_wall_round(
281    tiles: &[WallTile],
282    init_data: &DataBatches,
283    meas_data: &DataBatches,
284    reversed: bool,
285) -> Result<(CoordCircuit, MeasurementIndex), CompileError> {
286    let mut circuit = CoordCircuit::new();
287    circuit.do_gate(
288        GateType::RX,
289        tiles
290            .iter()
291            .map(|tile| if reversed { tile.plus } else { tile.minus }),
292    )?;
293    circuit.do_gate(
294        GateType::RZ,
295        tiles.iter().map(|tile| tile.link_for(reversed)),
296    )?;
297    circuit.do_gate(
298        GateType::RZ,
299        tiles
300            .iter()
301            .map(|tile| if reversed { tile.minus } else { tile.plus }),
302    )?;
303    for (basis, qubits) in init_data {
304        circuit.do_gate(reset_gate(*basis), qubits.iter().copied())?;
305    }
306    circuit.tick();
307
308    let mut measurements = MeasurementIndex::default();
309    for slot in 0..WALL_CX_SLOTS {
310        let mut cx = Vec::new();
311        let mut cz = Vec::new();
312        for tile in tiles {
313            let link = tile.link_for(reversed);
314            let (start, end) = if reversed {
315                (tile.plus, tile.minus)
316            } else {
317                (tile.minus, tile.plus)
318            };
319            let bracket = match slot {
320                GROW_MINUS_SLOT => Some([start, link]),
321                GROW_PLUS_SLOT => Some([link, end]),
322                UNDO_MINUS_SLOT => Some([link, start]),
323                UNDO_PLUS_SLOT => Some([end, link]),
324                _ => None,
325            };
326            cx.extend(bracket.into_iter().flatten());
327            for (anchor, basis, data) in [
328                (tile.minus, tile.basis, &tile.minus_data),
329                (tile.plus, tile.basis.flip(), &tile.plus_data),
330            ] {
331                for &(qubit, at) in data {
332                    if data_slot(at, reversed) == slot {
333                        push_interaction(&mut cx, &mut cz, anchor, qubit, basis);
334                    }
335                }
336            }
337        }
338        circuit.do_gate(GateType::CX, cx)?;
339        circuit.do_gate(GateType::CZ, cz)?;
340        circuit.tick();
341    }
342
343    let measured = tiles
344        .iter()
345        .map(|tile| if reversed { tile.minus } else { tile.plus })
346        .collect::<Vec<_>>();
347    for (&qubit, id) in measured
348        .iter()
349        .zip(circuit.measure(PauliBasis::X, measured.iter().copied()))
350    {
351        measurements.record(qubit, id);
352    }
353    // The bracket uncomputes the other endpoint ancilla and link into Z.
354    // Discard them instead of creating detector-only helper readouts.
355    for (basis, qubits) in meas_data {
356        for (&qubit, id) in qubits
357            .iter()
358            .zip(circuit.measure(PauliBasis::from(*basis), qubits.iter().copied()))
359        {
360            measurements.record(qubit, id);
361        }
362    }
363    Ok((circuit, measurements))
364}

370/// The collapse rule applied to a mixed-basis stabilizer: once per half, against
371/// that half's own Pauli, then recombined. A half that meets an opposite-basis
372/// collapse kills the whole flow, exactly as a uniform tile's would.
373fn collapse_extended_stabilizer(
374    tile: &WallTile,
375    collapsed_data: Option<&crate::FxMap<IVec2, Basis>>,
376) -> Option<(PauliMap, Vec<IVec2>)> {
377    let pauli = Pauli::from(tile.basis);
378    let halves = [
379        (&tile.minus_data, tile.basis, pauli),
380        (&tile.plus_data, tile.basis.flip(), pauli.flip()),
381    ];
382    let mut support = PauliMap::empty();
383    let mut collapsed = Vec::new();
384    for (data, basis, pauli) in halves {
385        let qubits: SmallVec<[IVec2; 2]> = data.iter().map(|&(qubit, _)| qubit).collect();
386        let (half_support, half_collapsed) =
387            collapse_matching_data_boundary(&qubits, basis, pauli, collapsed_data)?;
388        for (qubit, pauli) in half_support.iter() {
389            support.insert(*qubit, *pauli);
390        }
391        collapsed.extend(half_collapsed);
392    }
393    Some((support, collapsed))
394}

396/// Creator/consumer pair per extended stabilizer, collapsed against the round's
397/// init/meas maps — the same shape [`standard_round_flows`] gives an ordinary
398/// tile, with the chain's single `MX` standing in for the ancilla record.
399///
400/// [`standard_round_flows`]: super::utils
401fn wall_round_flows(
402    tiles: &[WallTile],
403    measurements: &MeasurementIndex,
404    init_data: Option<&crate::FxMap<IVec2, Basis>>,
405    meas_data: Option<&crate::FxMap<IVec2, Basis>>,
406    reversed: bool,
407) -> Vec<Flow> {
408    let mut flows = Vec::with_capacity(tiles.len() * 4);
409    for tile in tiles {
410        let ancilla =
411            measurements.expect_measurement(if reversed { tile.minus } else { tile.plus });
412
413        if let Some((start, _)) = collapse_extended_stabilizer(tile, init_data) {
414            flows.push(
415                Flow::new(start, PauliMap::empty())
416                    .with_measurements([ancilla])
417                    .with_center(tile.center()),
418            );
419        }
420
421        if let Some((end, mut collapsed)) = collapse_extended_stabilizer(tile, meas_data) {
422            collapsed.sort_unstable_by_key(|qubit| (qubit.y, qubit.x));
423            flows.push(
424                Flow::new(PauliMap::empty(), end)
425                    .with_measurements(
426                        collapsed
427                            .into_iter()
428                            .map(|qubit| measurements.expect_measurement(qubit))
429                            .chain(std::iter::once(ancilla)),
430                    )
431                    .with_center(tile.center()),
432            );
433        }
434    }
435    flows
436}

442/// Pipe-frame crossing key: the plus-side Pauli is the minus-side Pauli flipped.
443fn crossing_key(axis: UDirection, minus_pauli: Pauli) -> LocalStabilizer {
444    let (minus, plus) = match axis {
445        UDirection::X => (Direction::XMINUS, Direction::XPLUS),
446        UDirection::Y => (Direction::YMINUS, Direction::YPLUS),
447        UDirection::Z => unreachable!("a spatial wall has an X or Y axis"),
448    };
449    LocalStabilizer::isolated(Pauli::I)
450        .with_arm(minus, minus_pauli, false)
451        .with_arm(plus, minus_pauli.flip(), false)
452}

454/// Crossing gateway: free init-check records restore support killed by resets;
455/// final readout and temporal-face operators remain cube-owned.
456fn build_gateway(
457    key: SpatialHadamardKey,
458    tiles: &[WallTile],
459    init_data: &crate::FxMap<IVec2, Basis>,
460    init_chunk: &Chunk,
461) -> ObservableGateway {
462    let introduces_joint_checks = key.minus.connectivity.has_pipe(Direction::ZMINUS)
463        && key.plus.connectivity.has_pipe(Direction::ZMINUS)
464        && init_data.is_empty();
465    let measurements = MeasurementIndex::from_circuit(&init_chunk.circuit);
466    let mut gateway = ObservableGateway::new();
467    for basis in [Basis::X, Basis::Z] {
468        let records: Vec<u32> = tiles
469            .iter()
470            .filter(|tile| {
471                tile.basis == basis
472                    && ((introduces_joint_checks && basis == key.boundary_basis)
473                        || collapse_extended_stabilizer(tile, Some(init_data)).is_none())
474            })
475            .map(|tile| measurements.expect_measurement(tile.plus))
476            .collect();
477        let mut entry = GatewayEntry::default();
478        if !records.is_empty() {
479            entry.measurements.push(ChunkMeasurements {
480                chunk_index: 0,
481                measurements: records,
482            });
483        }
484        gateway.insert(crossing_key(key.axis, Pauli::from(basis)), entry);
485    }
486    gateway
487}

489fn wall_chunk(
490    tiles: &[WallTile],
491    init_data: Option<&crate::FxMap<IVec2, Basis>>,
492    meas_data: Option<&crate::FxMap<IVec2, Basis>>,
493    reversed: bool,
494) -> Result<Chunk, CompileError> {
495    let (circuit, measurements) = emit_wall_round(
496        tiles,
497        &sorted_batches(init_data, [Basis::X, Basis::Z]),
498        &sorted_batches(meas_data, [Basis::X, Basis::Z]),
499        reversed,
500    )?;
501    let flows = wall_round_flows(tiles, &measurements, init_data, meas_data, reversed);
502    Ok(Chunk { circuit, flows })
503}

505/// Compile a positive-axis spatial Hadamard wall in the negative endpoint's
506/// local coordinates.
507///
508/// The stage structure is the cube's, round for round, so template composition,
509/// loop flattening and the instance merge need nothing new — the wall's chunks
510/// line up moment for moment with every cube in its z-layer component.
511pub(crate) fn compile_spatial_hadamard(
512    distance: u32,
513    key: SpatialHadamardKey,
514) -> Result<CompiledTemplate, CompileError> {
515    let tiles = wall_tiles(distance, key);
516    let init_data = collapse_map(distance, key, &tiles, Face::Init);
517    let meas_data = collapse_map(distance, key, &tiles, Face::Meas);
518
519    let make_chunk =
520        |init_data, meas_data, reversed| wall_chunk(&tiles, init_data, meas_data, reversed);
521    let init_chunk = make_chunk(Some(&init_data), None, false)?;
522    let gateway = build_gateway(key, &tiles, &init_data, &init_chunk);
523    let repetitions = cube_bulk_repetitions(key.rounds);
524    let forward_bulk = make_chunk(None, None, false)?;
525    let reverse_bulk = make_chunk(None, None, true)?;
526    let meas = make_chunk(None, Some(&meas_data), repetitions.is_multiple_of(2))?;
527
528    let mut chunks = vec![ChunkOrLoop::Single(Box::new(init_chunk))];
529    chunks.push(ChunkOrLoop::Loop {
530        body: vec![reverse_bulk.clone(), forward_bulk],
531        repetitions: repetitions / 2,
532    });
533    if repetitions % 2 == 1 {
534        chunks.push(ChunkOrLoop::Single(Box::new(reverse_bulk)));
535    }
536    chunks.push(ChunkOrLoop::Single(Box::new(meas)));
537
538    Ok(Arc::new(LoweringTemplate::from_chunks(chunks, gateway)?))
539}

765    /// Splice chunks as instance lowering does, deduplicating shared resets and
766    /// measurements so merged wall/cube flows test the real moment schedule.
767    fn splice(chunks: &[Chunk]) -> Chunk {
768        let moments: Vec<Vec<Vec<Op>>> = chunks
769            .iter()
770            .map(|chunk| {
771                split_ops_into_moments(
772                    chunk
773                        .circuit
774                        .body(chunk.circuit.entry_body())
775                        .expect("entry body")
776                        .ops(),
777                )
778            })
779            .collect();
780        let depth = moments[0].len();
781        assert!(
782            moments.iter().all(|chunk| chunk.len() == depth),
783            "spliced chunks must share a moment structure"
784        );
785
786        let mut circuit = CoordCircuit::new();
787        let mut remaps: Vec<crate::FxMap<u32, u32>> = vec![Default::default(); chunks.len()];
788        for moment in 0..depth {
789            let mut measured: crate::FxMap<IVec2, u32> = Default::default();
790            let mut reset: crate::FxSet<IVec2> = Default::default();
791            for (index, chunk_moments) in moments.iter().enumerate() {
792                for op in &chunk_moments[moment] {
793                    match op {
794                        Op::Gate { gate, qubits } if gate.is_reset() => {
795                            let fresh: Vec<IVec2> = qubits
796                                .iter()
797                                .copied()
798                                .filter(|qubit| reset.insert(*qubit))
799                                .collect();
800                            circuit.do_gate(*gate, fresh).expect("reset");
801                        }
802                        Op::Gate { gate, qubits } => {
803                            circuit
804                                .do_gate(*gate, qubits.iter().copied())
805                                .expect("gate");
806                        }
807                        Op::Measure {
808                            basis,
809                            qubits,
810                            measurements,
811                            ..
812                        } => {
813                            let fresh: Vec<IVec2> = qubits
814                                .iter()
815                                .copied()
816                                .filter(|qubit| !measured.contains_key(qubit))
817                                .collect();
818                            let ids = circuit.measure(*basis, fresh.iter().copied());
819                            for (&qubit, id) in fresh.iter().zip(ids) {
820                                measured.insert(qubit, id);
821                            }
822                            for (qubit, &local) in qubits.iter().zip(measurements) {
823                                remaps[index].insert(local, measured[qubit]);
824                            }
825                        }
826                        _ => {}
827                    }
828                }
829            }
830            if moment + 1 < depth {
831                circuit.tick();
832            }
833        }
834
835        let flows = chunks
836            .iter()
837            .zip(&remaps)
838            .flat_map(|(chunk, remap)| {
839                chunk.flows.iter().map(move |flow| {
840                    let mut spliced = flow.clone();
841                    spliced.measurements =
842                        flow.measurements.iter().map(|local| remap[local]).collect();
843                    spliced
844                })
845            })
846            .collect();
847
848        Chunk { circuit, flows }
849    }

851    /// The CZ gallery's endpoint pair: `XZX` threaded through both temporal
852    /// faces, `XXZ` terminated with both perpendicular arms.
853    fn cz_merged_rounds(distance: u32) -> Vec<Vec<Chunk>> {
854        let d = distance as i32;
855        let stride = IVec2::new(2 * d + 2, 0);
856
857        let left_connectivity = Connectivity::ISOLATED
858            .with_hadamard(Direction::XPLUS)
859            .with_pipe(Direction::ZMINUS)
860            .with_pipe(Direction::ZPLUS);
861        let right_connectivity = Connectivity::ISOLATED
862            .with_hadamard(Direction::XMINUS)
863            .with_pipe(Direction::YPLUS)
864            .with_pipe(Direction::YMINUS);
865
866        let (left_patch, left_init, left_meas) = build_regular_patch(
867            CubeKind::XZX,
868            distance,
869            left_connectivity,
870            LayerSchedule::Extended,
871        );
872        let (right_patch, right_init, right_meas) = build_spatial_patch(
873            CubeKind::XXZ,
874            distance,
875            right_connectivity,
876            LayerSchedule::Extended,
877        );
878        let right_patch = translated_patch(&right_patch, stride);
879        let right_init = translated_map(&right_init, stride);
880        let right_meas = translated_map(&right_meas, stride);
881
882        let wall = key(
883            CubeKind::XZX.y(),
884            threaded(CubeKind::XZX.z()),
885            WallSide {
886                connectivity: Connectivity::ISOLATED
887                    .with_pipe(Direction::YPLUS)
888                    .with_pipe(Direction::YMINUS),
889                ..terminated(CubeKind::XXZ.z())
890            },
891        );
892        let tiles = wall_tiles(distance, wall);
893        let wall_init = collapse_map(distance, wall, &tiles, Face::Init);
894        let wall_meas = collapse_map(distance, wall, &tiles, Face::Meas);
895
896        let round = |left_data: Option<&crate::FxMap<IVec2, Basis>>,
897                     right_data: Option<&crate::FxMap<IVec2, Basis>>,
898                     wall_data: Option<&crate::FxMap<IVec2, Basis>>,
899                     face: Face| {
900            let (left_init, left_meas) = match face {
901                Face::Init => (left_data, None),
902                Face::Meas => (None, left_data),
903            };
904            let (right_init, right_meas) = match face {
905                Face::Init => (right_data, None),
906                Face::Meas => (None, right_data),
907            };
908            let (wall_init, wall_meas) = match face {
909                Face::Init => (wall_data, None),
910                Face::Meas => (None, wall_data),
911            };
912            vec![
913                make_surface_code_chunk(&left_patch, left_init, left_meas).expect("left round"),
914                make_surface_code_chunk(&right_patch, right_init, right_meas).expect("right round"),
915                wall_chunk(&tiles, wall_init, wall_meas, false).expect("wall round"),
916            ]
917        };
918
919        vec![
920            round(
921                Some(&left_init),
922                Some(&right_init),
923                Some(&wall_init),
924                Face::Init,
925            ),
926            round(None, None, None, Face::Init),
927            round(
928                Some(&left_meas),
929                Some(&right_meas),
930                Some(&wall_meas),
931                Face::Meas,
932            ),
933        ]
934    }

952    /// Number of logical degrees of freedom a stabilizer set leaves: data qubits
953    /// minus the GF(2) rank of the generators, over the `(X | Z)` bit layout.
954    fn logical_freedom(stabilizers: &[PauliMap]) -> usize {
955        let qubits: Vec<IVec2> = {
956            let mut qubits: Vec<IVec2> = stabilizers
957                .iter()
958                .flat_map(|stabilizer| stabilizer.iter().map(|(&qubit, _)| qubit))
959                .collect::<crate::FxSet<_>>()
960                .into_iter()
961                .collect();
962            qubits.sort_unstable_by_key(|qubit| (qubit.y, qubit.x));
963            qubits
964        };
965        let column = |qubit: IVec2| {
966            qubits
967                .iter()
968                .position(|candidate| *candidate == qubit)
969                .expect("stabilizer support is in the qubit list")
970        };
971        let width = 2 * qubits.len();
972        let mut rows: Vec<Vec<bool>> = stabilizers
973            .iter()
974            .map(|stabilizer| {
975                let mut row = vec![false; width];
976                for (&qubit, &pauli) in stabilizer.iter() {
977                    let offset = if pauli == bloq_circuit::Pauli::X {
978                        0
979                    } else {
980                        1
981                    };
982                    row[2 * column(qubit) + offset] = true;
983                }
984                row
985            })
986            .collect();
987
988        let mut rank = 0;
989        for bit in 0..width {
990            let Some(pivot) = (rank..rows.len()).find(|&row| rows[row][bit]) else {
991                continue;
992            };
993            rows.swap(rank, pivot);
994            let pivot_row = rows[rank].clone();
995            for (index, row) in rows.iter_mut().enumerate() {
996                if index == rank || !row[bit] {
997                    continue;
998                }
999                for (target, source) in row.iter_mut().zip(&pivot_row).skip(bit) {
1000                    *target ^= *source;
1001                }
1002            }
1003            rank += 1;
1004        }
1005        qubits.len() - rank
1006    }

1026    /// Rank completeness: the wall and a plain merge of the same geometry keep
1027    /// the same logical count.
1028    #[rstest]
1029    fn wall_preserves_the_merged_logical_count(
1030        #[values(3, 5)] distance: u32,
1031        #[values(0, 1, 2, 3)] arm_mask: u8,
1032    ) {
1033        let d = distance as i32;
1034        let stride = IVec2::new(2 * d + 2, 0);
1035        let arm_pipes = |connectivity: Connectivity| {
1036            [Direction::YPLUS, Direction::YMINUS]
1037                .into_iter()
1038                .enumerate()
1039                .filter(|(index, _)| arm_mask & (1 << index) != 0)
1040                .fold(connectivity, |connectivity, (_, direction)| {
1041                    connectivity.with_pipe(direction)
1042                })
1043        };
1044
1045        let hadamard = {
1046            let left = cube_stabilizers(
1047                CubeKind::XZX,
1048                distance,
1049                Connectivity::ISOLATED.with_hadamard(Direction::XPLUS),
1050                IVec2::ZERO,
1051            );
1052            let right = cube_stabilizers(
1053                CubeKind::XXZ,
1054                distance,
1055                arm_pipes(Connectivity::ISOLATED.with_hadamard(Direction::XMINUS)),
1056                stride,
1057            );
1058            let wall = key(
1059                CubeKind::XZX.y(),
1060                terminated(CubeKind::XZX.z()),
1061                WallSide {
1062                    connectivity: arm_pipes(Connectivity::ISOLATED),
1063                    ..terminated(CubeKind::XXZ.z())
1064                },
1065            );
1066            let tiles = wall_tiles(distance, wall);
1067            assert_eq!(tiles.len() as u32, distance);
1068            left.into_iter()
1069                .chain(right)
1070                .chain(tiles.iter().map(|tile| {
1071                    collapse_extended_stabilizer(tile, None)
1072                        .expect("an uncollapsed stabilizer always survives")
1073                        .0
1074                }))
1075                .collect::<Vec<_>>()
1076        };
1077
1078        let plain = {
1079            let left = cube_stabilizers(
1080                CubeKind::XZX,
1081                distance,
1082                Connectivity::ISOLATED.with_pipe(Direction::XPLUS),
1083                IVec2::ZERO,
1084            );
1085            let right = cube_stabilizers(
1086                CubeKind::ZZX,
1087                distance,
1088                arm_pipes(Connectivity::ISOLATED.with_pipe(Direction::XMINUS)),
1089                stride,
1090            );
1091            left.into_iter().chain(right).collect::<Vec<_>>()
1092        };
1093
1094        assert_eq!(
1095            logical_freedom(&hadamard),
1096            logical_freedom(&plain),
1097            "d={distance} arm_mask={arm_mask}"
1098        );
1099    }

1119    /// crossing runs *along* — colour `P` — and only when the seam columns'
1120    /// resets would otherwise destroy that support, which is `P != z_L`.
1121    ///

1123    /// temporal faces (`z_L = X`) and the second one terminates with both
1124    /// perpendicular arms (`z_R = Z`). The `Z` crossing is perpendicular to the second cube's
1125    /// resets, so it takes the `Z`-coloured tiles' records; the `X` crossing
1126    /// runs parallel to both collapses and takes none.
1127    #[rstest]
1128    fn crossing_records_are_the_tiles_the_string_runs_along(#[values(3, 5)] distance: u32) {

1130            CubeKind::XZX.y(),
1131            threaded(CubeKind::XZX.z()),
1132            WallSide {
1133                connectivity: Connectivity::ISOLATED
1134                    .with_pipe(Direction::YPLUS)
1135                    .with_pipe(Direction::YMINUS),
1136                ..terminated(CubeKind::XXZ.z())
1137            },
1138        );

1140        let gateway = &template.observable_gateway;
1141
1142        let tiles = wall_tiles(distance, wall);
1143        let init_data = collapse_map(distance, wall, &tiles, Face::Init);
1144        let init_chunk = wall_chunk(&tiles, Some(&init_data), None, false).expect("init round");
1145        let measurements = MeasurementIndex::from_circuit(&init_chunk.circuit);
1146
1147        // `z_L = X`, so the `X` crossing is parallel to both halves' collapse
1148        // bases and needs nothing.
1149        let parallel = gateway
1150            .lookup(crossing_key(UDirection::X, Pauli::from(CubeKind::XZX.z())))
1151            .expect("the parallel crossing has an entry");

1174        assert_eq!(perpendicular.measurements.len(), 1);
1175        assert_eq!(perpendicular.measurements[0].chunk_index, 0);
1176        assert_eq!(perpendicular.measurements[0].measurements, expected);
1177
1178        // The wall spans no temporal face, so it never carries an operator.
1179        for entry in [&parallel, &perpendicular] {
1180            assert!(entry.operator_in.is_empty() && entry.operator_out.is_empty());
1181        }

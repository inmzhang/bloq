1//! Y-basis initialization and measurement block compilation.
2//!
3//! Reference: arXiv:2302.07395 "Inplace Access to the Surface Code Y Basis"
4//!
5//! Y blocks perform in-place Y-basis initialization or measurement. They connect
6//! to cubes via temporal (Z-axis) pipes only, with at most 1 neighbor.
7//!
8//! The compiled circuit has 3 chunks:
9//! - **Transition**: diagonal twist round, with the final ancilla basis changes
10//!   folded into the measurement layer
11//! - **Degenerate bulk** (loop, `d//2` repetitions): standard syndrome extraction on rotated patch
12//! - **Final**: syndrome extraction + mixed-basis data measurement
13//!
14//! Initialization = time_reversed(Measurement).
15//!
16//! # Coordinate system
17//!
18//! All coordinates are native bloq coordinates (`IVec2`, Y-up). On the
19//! `d × d` rotated surface code patch:
20//!
21//! - **Data qubits** sit at odd `(x, y)` positions: `(2i+1, 2j+1)`.
22//! - **Measure qubits** sit at even `(x, y)` positions: `(2i, 2j)`.
23//!   Boundary ancillas have one coordinate equal to `0` or `2d`.
24//!
25//! The **twist diagonal** `(q.x - q.y) / 2 + d - 1` partitions the patch:
26//! - `< d-1`: upper-left (UL) region
27//! - `= d-1`: twist line (`q.x == q.y` for measure qubits)
28//! - `> d-1`: down-right (DR) region
29//!
30//! Measure-qubit neighbors in the tile grid correspond to fixed offsets:
31//! - One step up: `IVec2(0, 2)`
32//! - One step left: `IVec2(-2, 0)`
33//! - Data-qubit corners: `Corner::TR`, `Corner::BL`, etc. (`±1, ±1`)
34//!
35//! Patch center for observables: `IVec2(d, d)` (a data qubit when `d` is odd).
36

57/// Compile a Y block into init/meas chunks.
58pub(super) fn compile_y(
59    boundary_basis: Basis,
60    connectivity: Connectivity,
61    distance: u32,
62) -> Result<CompiledTemplate, CompileError> {
63    let is_init = connectivity.has_pipe(Direction::ZPLUS);
64
65    // padding_rounds >= 1 because distance >= 3 (odd) so distance / 2 >= 1.
66    let padding_rounds = distance / 2;
67    let (chunks, gateway) = build_y_chunks(boundary_basis, distance, padding_rounds, is_init)?;
68
69    Ok(Arc::new(LoweringTemplate::from_chunks(chunks, gateway)?))
70}

72/// Build the three Y-block chunks (transition, bulk loop, final) and observable gateway.
73///
74/// For measurement: chunks are in forward order.
75/// For initialization: chunks are time-reversed and reordered.
76///
77fn build_y_chunks(
78    boundary_basis: Basis,
79    d: u32,
80    padding_rounds: u32,
81    is_init: bool,
82) -> Result<(Vec<ChunkOrLoop>, ObservableGateway), CompileError> {
83    let degenerate_patch = make_degenerate_patch(d, boundary_basis, false);
84    let padding_patch = if is_init {
85        make_degenerate_patch(d, boundary_basis, true)
86    } else {
87        degenerate_patch.clone()
88    };
89    let padding = make_surface_code_chunk(&padding_patch, None, None)?;
90    let data_basis =
91        get_final_measure_data_basis(&degenerate_patch.data_set(), boundary_basis, d as i32);
92
93    let (transition, transition_ids) = if is_init {
94        make_y_transition_round_reversed(d, boundary_basis)?
95    } else {
96        make_y_transition_round(d, boundary_basis)?
97    };
98
99    let chunks = if is_init {
100        vec![
101            ChunkOrLoop::Single(Box::new(make_surface_code_chunk(
102                &degenerate_patch,
103                Some(&data_basis),
104                None,
105            )?)),
106            ChunkOrLoop::Loop {
107                body: vec![padding],
108                repetitions: padding_rounds,
109            },
110            ChunkOrLoop::Single(Box::new(transition)),
111        ]
112    } else {
113        vec![
114            ChunkOrLoop::Single(Box::new(transition)),
115            ChunkOrLoop::Loop {
116                body: vec![padding],
117                repetitions: padding_rounds,
118            },
119            ChunkOrLoop::Single(Box::new(make_surface_code_chunk(
120                &degenerate_patch,
121                None,
122                Some(&data_basis),
123            )?)),
124        ]
125    };
126    let gateway = build_y_observable_gateway(boundary_basis, d, is_init, &transition_ids);
127
128    Ok((chunks, gateway))
129}

131fn build_y_observable_gateway(
132    boundary_basis: Basis,
133    d: u32,
134    is_init: bool,
135    transition_ids: &MeasurementIndex,
136) -> ObservableGateway {
137    let pipe_dir = if is_init {
138        Direction::ZPLUS
139    } else {
140        Direction::ZMINUS
141    };
142    let key = LocalStabilizer::new(Pauli::Y, Connectivity::ISOLATED.with_pipe(pipe_dir));
143
144    let transition_chunk_index = if is_init { 2 } else { 0 };
145
146    let sets = TransitionSets::new(d, boundary_basis);
147    let obs_meas_qubits = if !is_init {
148        collect_observable_meas_qubits(&sets, d as i32, boundary_basis)
149    } else {
150        collect_observable_meas_qubits_reversed(&sets, d as i32, boundary_basis)
151    };
152
153    let measurements = vec![ChunkMeasurements {
154        chunk_index: transition_chunk_index,
155        measurements: obs_meas_qubits
156            .into_iter()
157            .map(|q| transition_ids.expect_measurement(q))
158            .collect(),
159    }];
160
161    // The Y-basis boundary operator is the product of the patch's X and Z
162    // middle lines, which meet at the center data qubit: X·Z = Y
163    // there, with X or Z along each arm — exactly the `operator` the neighbouring
164    // cube emits for its Y observable. The Y block touches a cube on one temporal
165    // face only, so the cross rides `operator_out` when it initialises (a `+Z`
166    // pipe carries the state upward) and `operator_in` when it measures (`−Z`).
167    let x_line = logical_line_operator(d, Basis::X, boundary_basis, Pauli::X);
168    let z_line = logical_line_operator(d, Basis::Z, boundary_basis, Pauli::Z);
169    let cross = &x_line ^ &z_line;
170    let (operator_in, operator_out) = if is_init {
171        (PauliMap::empty(), cross)
172    } else {
173        (cross, PauliMap::empty())
174    };
175
176    let mut gateway = ObservableGateway::new();
177    gateway.insert(
178        key,
179        GatewayEntry {
180            measurements,
181            operator_in,
182            operator_out,
183        },
184    );
185    gateway
186}

188/// Compute the twist diagonal value for a qubit as an integer.
189///
190/// Unified formula for both data qubits (odd coords) and measure qubits (even
191/// coords): `(q.x - q.y) / 2 + d - 1`. Works because `q.x - q.y` is always
192/// even on the surface code lattice.
193fn twist_diagonal(q: IVec2, d: i32) -> i32 {
194    (q.x - q.y) / 2 + d - 1
195}

197/// Build the degenerate patch after Y-basis transition.
198///
199/// `top_basis` is the top/bottom boundary basis of the *standard* patch
200/// before the transition. It determines both the degenerate boundary layout
201/// and the CX hook orientation (passed through as `horizontal_hook_basis`).
202fn make_degenerate_patch(d: u32, top_basis: Basis, reverse_schedule: bool) -> Patch {
203    // The transition rotates the boundary assignment a quarter turn: top and
204    // right take `top_basis.flip()` and `top_basis`, bottom and left the
205    // reverse.
206    let flipped = top_basis.flip();
207    let boundaries = EdgeBases {
208        top: flipped,
209        bottom: top_basis,
210        left: top_basis,
211        right: flipped,
212    };
213    make_rectangular_surface_code_patch(d, d, &boundaries, top_basis, reverse_schedule)
214}

216/// Pre-computed measure qubit classification for the transition round.
217///
218/// Shared between `make_y_transition_round` and `build_y_observable_gateway`
219/// to avoid duplicating the classification logic.
220struct TransitionSets {
221    used: crate::FxSet<IVec2>,
222    xs: crate::FxSet<IVec2>,
223    zs: crate::FxSet<IVec2>,
224    top_row: crate::FxSet<IVec2>,
225    left_col: crate::FxSet<IVec2>,
226}
227
228impl TransitionSets {
229    fn new(d: u32, top_basis: Basis) -> Self {
230        let start_patch = make_normal_surface_code_patch(d, top_basis);
231        let end_patch = make_degenerate_patch(d, top_basis, false);
232        Self::from_patches(d, &start_patch, &end_patch)
233    }
234
235    fn from_patches(d: u32, start_patch: &Patch, end_patch: &Patch) -> Self {
236        let used: crate::FxSet<IVec2> = start_patch
237            .used_set()
238            .union(&end_patch.used_set())
239            .copied()
240            .collect();
241
242        let xs: crate::FxSet<IVec2> = used
243            .iter()
244            .copied()
245            .filter(|&q| is_measure_qubit(q) && checkerboard_basis(q) == Basis::X)
246            .collect();
247        let zs: crate::FxSet<IVec2> = used
248            .iter()
249            .copied()
250            .filter(|&q| is_measure_qubit(q) && checkerboard_basis(q) == Basis::Z)
251            .collect();
252
253        let top_row: crate::FxSet<IVec2> = used
254            .iter()
255            .copied()
256            .filter(|&q| q.y == 2 * d as i32 && is_measure_qubit(q))
257            .collect();
258        let left_col: crate::FxSet<IVec2> = used
259            .iter()
260            .copied()
261            .filter(|&q| q.x == 0 && is_measure_qubit(q))
262            .collect();
263
264        Self {
265            used,
266            xs,
267            zs,
268            top_row,
269            left_col,
270        }
271    }
272}

274/// Qubits split by the coarse twist band `q.x - q.y`: upper-left below `-2`,
275/// down-right at or above `+2`, middle in between.
276#[derive(Default)]
277struct DiagSplit {
278    ul: crate::FxSet<IVec2>,
279    md: crate::FxSet<IVec2>,
280    dr: crate::FxSet<IVec2>,
281}
282
283impl DiagSplit {
284    fn new(qs: &crate::FxSet<IVec2>) -> Self {
285        let mut split = Self::default();
286        for &q in qs {
287            let diff = q.x - q.y;
288            let band = if diff < -2 {
289                &mut split.ul
290            } else if diff >= 2 {
291                &mut split.dr
292            } else {
293                &mut split.md
294            };
295            band.insert(q);
296        }
297        split
298    }
299
300    fn ul_md(&self) -> crate::FxSet<IVec2> {
301        self.ul.union(&self.md).copied().collect()
302    }
303}

305struct TransitionBoundaries {
306    old_x: crate::FxSet<IVec2>,
307    new_x: crate::FxSet<IVec2>,
308    old_z: crate::FxSet<IVec2>,
309    new_z: crate::FxSet<IVec2>,
310}
311
312impl TransitionBoundaries {
313    fn new(sets: &TransitionSets, top_basis: Basis) -> Self {
314        if top_basis == Basis::X {
315            Self {
316                old_x: sets.top_row.clone(),
317                new_x: sets.left_col.clone(),
318                old_z: sets.left_col.clone(),
319                new_z: sets.top_row.clone(),
320            }
321        } else {
322            Self {
323                old_x: sets.left_col.clone(),
324                new_x: sets.top_row.clone(),
325                old_z: sets.top_row.clone(),
326                new_z: sets.left_col.clone(),
327            }
328        }
329    }
330}

338struct TransitionRoundContext {
339    d: i32,
340    standard_patch: Patch,
341    degenerate_patch: Patch,
342    sets: TransitionSets,
343    xs: DiagSplit,
344    zs: DiagSplit,
345    boundaries: TransitionBoundaries,
346    x_order: [Corner; 4],
347    z_order: [Corner; 4],
348}
349
350impl TransitionRoundContext {
351    fn new(d: u32, top_basis: Basis) -> Self {
352        let standard_patch = make_normal_surface_code_patch(d, top_basis);
353        let degenerate_patch = make_degenerate_patch(d, top_basis, false);
354        let sets = TransitionSets::from_patches(d, &standard_patch, &degenerate_patch);
355        let xs = DiagSplit::new(&sets.xs);
356        let zs = DiagSplit::new(&sets.zs);
357        let boundaries = TransitionBoundaries::new(&sets, top_basis);
358
359        Self {
360            d: d as i32,
361            standard_patch,
362            degenerate_patch,
363            sets,
364            xs,
365            zs,
366            boundaries,
367            x_order: tile_cx_schedule(Basis::X, top_basis).compact(),
368            z_order: tile_cx_schedule(Basis::Z, top_basis).compact(),
369        }
370    }
371
372    fn boundary_targets(&self, side: BoundarySide) -> (Vec<IVec2>, Vec<IVec2>) {
373        match side {
374            BoundarySide::Old => (
375                swapped_boundary_targets(
376                    &self.sets.xs,
377                    &self.boundaries.new_x,
378                    &self.boundaries.old_x,
379                ),
380                swapped_boundary_targets(
381                    &self.sets.zs,
382                    &self.boundaries.new_z,
383                    &self.boundaries.old_z,
384                ),
385            ),
386            BoundarySide::New => (
387                swapped_boundary_targets(
388                    &self.sets.xs,
389                    &self.boundaries.old_x,
390                    &self.boundaries.new_x,
391                ),
392                swapped_boundary_targets(
393                    &self.sets.zs,
394                    &self.boundaries.old_z,
395                    &self.boundaries.new_z,
396                ),
397            ),
398        }
399    }
400}

402fn is_measure_qubit(q: IVec2) -> bool {
403    q.x % 2 == 0 && q.y % 2 == 0
404}

406fn swapped_boundary_targets(
407    all: &crate::FxSet<IVec2>,
408    excluded: &crate::FxSet<IVec2>,
409    included: &crate::FxSet<IVec2>,
410) -> Vec<IVec2> {
411    sorted_unique_coords(
412        all.difference(excluded)
413            .copied()
414            .chain(included.iter().copied()),
415    )
416}

418fn active_targets(
419    all: &crate::FxSet<IVec2>,
420    excluded: &crate::FxSet<IVec2>,
421) -> crate::FxSet<IVec2> {
422    all.difference(excluded).copied().collect()
423}

425fn transition_rotation_targets(used: &crate::FxSet<IVec2>, d: i32) -> (Vec<IVec2>, Vec<IVec2>) {
426    let h_targets = sorted_unique_coords(
427        used.iter()
428            .copied()
429            .filter(|&q| twist_diagonal(q, d) < d - 1),
430    );
431    let twist_targets = sorted_unique_coords(
432        used.iter()
433            .copied()
434            .filter(|&q| is_measure_qubit(q) && twist_diagonal(q, d) == d - 1),
435    );
436    (h_targets, twist_targets)
437}

439fn emit_transition_measurements(
440    circuit: &mut CoordCircuit,
441    mx_targets: Vec<IVec2>,
442    my_target: Option<IVec2>,
443    mz_targets: Vec<IVec2>,
444) {
445    let mut measurement_bases = crate::FxMap::default();
446    for target in mx_targets {
447        measurement_bases.insert(target, PauliBasis::X);
448    }
449    for target in mz_targets {
450        measurement_bases.insert(target, PauliBasis::Z);
451    }
452    if let Some(target) = my_target {
453        measurement_bases.insert(target, PauliBasis::Y);
454    }
455
456    for basis in [PauliBasis::X, PauliBasis::Y, PauliBasis::Z] {
457        let targets = sorted_unique_coords(
458            measurement_bases
459                .iter()
460                .filter_map(|(&target, &target_basis)| (target_basis == basis).then_some(target)),
461        );
462        if !targets.is_empty() {
463            circuit.measure(basis, targets);
464        }
465    }
466}

468fn patch_flows(
469    patch: &Patch,
470    measurement_ids: &MeasurementIndex,
471    direction: TileFlow,
472    mids_for_tile: impl Fn(IVec2, Basis) -> Vec<IVec2>,
473) -> Vec<Flow> {
474    patch
475        .tiles()
476        .iter()
477        .map(|tile| {
478            let measurements = mids_for_tile(tile.measure_qubit(), tile.basis())
479                .into_iter()
480                .map(|q| measurement_ids.expect_measurement(q));
481            tile_flow(tile, direction, measurements)
482        })
483        .collect()
484}

486/// Generate CX pairs: for each qubit in `qs` (visited in `(y, x)` order), if
487/// `q + delta` is in `used`, produce the pair ordered by `sign`
488/// (+1 = measure-qubit-first, -1 = target-first).
489fn toward(
490    qs: &crate::FxSet<IVec2>,
491    delta: Corner,
492    sign: i32,
493    used: &crate::FxSet<IVec2>,
494) -> Vec<IVec2> {
495    let offset = delta.to_ivec2();
496    let mut pairs: Vec<[IVec2; 2]> = Vec::new();
497    for q in sorted_unique_coords(qs.iter().copied()) {
498        let target = q + offset;
499        if used.contains(&target) {
500            let pair = if sign > 0 { [q, target] } else { [target, q] };
501            pairs.push(pair);
502        }
503    }
504    pairs.into_iter().flatten().collect()
505}

513/// Build the Y-basis transition round chunk, with the measurement index its
514/// flows were resolved against — the observable gateway needs the same one.
515fn make_y_transition_round(
516    d: u32,
517    top_basis: Basis,
518) -> Result<(Chunk, MeasurementIndex), CompileError> {
519    make_transition_round(d, top_basis, TransitionDirection::Forward)
520}
521
522fn make_y_transition_round_reversed(
523    d: u32,
524    top_basis: Basis,
525) -> Result<(Chunk, MeasurementIndex), CompileError> {
526    make_transition_round(d, top_basis, TransitionDirection::Reversed)
527}
528
529fn make_transition_round(
530    d: u32,
531    top_basis: Basis,
532    direction: TransitionDirection,
533) -> Result<(Chunk, MeasurementIndex), CompileError> {
534    let ctx = TransitionRoundContext::new(d, top_basis);
535    let mut circuit = CoordCircuit::new();
536
537    match direction {
538        TransitionDirection::Forward => emit_forward_transition(&mut circuit, &ctx, top_basis)?,
539        TransitionDirection::Reversed => emit_reversed_transition(&mut circuit, &ctx, top_basis)?,
540    }
541
542    // The transition emits its measurements through `CoordCircuit::measure`
543    // batches rather than a recording helper, so the index is recovered once
544    // here and shared by every consumer.
545    let measurement_ids = MeasurementIndex::from_circuit(&circuit);
546    Ok((
547        Chunk {
548            flows: transition_flows(&ctx, &measurement_ids, top_basis, direction),
549            circuit,
550        },
551        measurement_ids,
552    ))
553}
554
555fn emit_forward_transition(
556    circuit: &mut CoordCircuit,
557    ctx: &TransitionRoundContext,
558    top_basis: Basis,
559) -> Result<(), CompileError> {
560    emit_boundary_resets(circuit, ctx, BoundarySide::Old, None)?;
561    emit_outer_interactions(circuit, ctx, [3, 2])?;
562    emit_diagonal_interactions(circuit, ctx)?;
563    emit_corner_interactions(circuit, ctx)?;
564    emit_middle_z_cy(circuit, ctx)?;
565    emit_rotation_layer(circuit, ctx, GateType::S)?;
566    emit_measurement_layer(
567        circuit,
568        ctx,
569        BoundarySide::New,
570        Some(y_corner_qubit(ctx.d, top_basis)),
571    );
572    Ok(())
573}
574
575fn emit_reversed_transition(
576    circuit: &mut CoordCircuit,
577    ctx: &TransitionRoundContext,
578    top_basis: Basis,
579) -> Result<(), CompileError> {
580    emit_boundary_resets(
581        circuit,
582        ctx,
583        BoundarySide::New,
584        Some(y_corner_qubit(ctx.d, top_basis)),
585    )?;
586    emit_rotation_layer(circuit, ctx, GateType::S_DAG)?;
587    emit_middle_z_cy(circuit, ctx)?;
588    emit_corner_interactions(circuit, ctx)?;
589    emit_diagonal_interactions(circuit, ctx)?;
590    emit_outer_interactions(circuit, ctx, [2, 3])?;
591    emit_measurement_layer(circuit, ctx, BoundarySide::Old, None);
592    Ok(())
593}
594
595fn emit_boundary_resets(
596    circuit: &mut CoordCircuit,
597    ctx: &TransitionRoundContext,
598    side: BoundarySide,
599    ry_target: Option<IVec2>,
600) -> Result<(), CompileError> {
601    let (rx_targets, rz_targets) = ctx.boundary_targets(side);
602    circuit.do_gate(GateType::RX, rx_targets)?;
603    if let Some(target) = ry_target {
604        circuit.do_gate(GateType::RY, std::iter::once(target))?;
605    }
606    circuit.do_gate(GateType::RZ, rz_targets)?;
607    circuit.tick();
608    Ok(())
609}

611fn emit_outer_interactions(
612    circuit: &mut CoordCircuit,
613    ctx: &TransitionRoundContext,
614    order: [usize; 2],
615) -> Result<(), CompileError> {
616    let xs_active = active_targets(&ctx.sets.xs, &ctx.boundaries.new_x);
617    let zs_active = active_targets(&ctx.sets.zs, &ctx.boundaries.new_z);
618
619    for idx in order {
620        emit_cx_pair(circuit, &xs_active, ctx.x_order[idx], 1, &ctx.sets.used)?;
621        emit_cx_pair(circuit, &zs_active, ctx.z_order[idx], -1, &ctx.sets.used)?;
622        circuit.tick();
623    }
624    Ok(())
625}
626
627fn emit_diagonal_interactions(
628    circuit: &mut CoordCircuit,
629    ctx: &TransitionRoundContext,
630) -> Result<(), CompileError> {
631    emit_cx_pair(circuit, &ctx.xs.ul, ctx.x_order[1], -1, &ctx.sets.used)?;
632    emit_cx_pair(circuit, &ctx.zs.ul_md(), ctx.z_order[1], 1, &ctx.sets.used)?;
633    emit_gate_pair(
634        circuit,
635        GateType::CY,
636        &ctx.xs.md,
637        ctx.x_order[1],
638        1,
639        &ctx.sets.used,
640    )?;
641    emit_cx_pair(circuit, &ctx.xs.dr, ctx.x_order[1], 1, &ctx.sets.used)?;
642    emit_cx_pair(circuit, &ctx.zs.dr, ctx.z_order[1], -1, &ctx.sets.used)?;
643    circuit.tick();
644    Ok(())
645}

647fn emit_corner_interactions(
648    circuit: &mut CoordCircuit,
649    ctx: &TransitionRoundContext,
650) -> Result<(), CompileError> {
651    emit_cx_pair(circuit, &ctx.xs.ul, ctx.x_order[3], -1, &ctx.sets.used)?;
652    emit_cx_pair(circuit, &ctx.zs.ul, ctx.x_order[3], 1, &ctx.sets.used)?;

698fn transition_flows(
699    ctx: &TransitionRoundContext,
700    measurement_ids: &MeasurementIndex,
701    top_basis: Basis,
702    direction: TransitionDirection,
703) -> Vec<Flow> {
704    let reversed = matches!(direction, TransitionDirection::Reversed);
705    // The transition carries the standard patch to the degenerate one; running
706    // it backwards swaps which patch is the input.
707    let (input_patch, output_patch) = if reversed {
708        (&ctx.degenerate_patch, &ctx.standard_patch)
709    } else {
710        (&ctx.standard_patch, &ctx.degenerate_patch)
711    };
712
713    let mut input_flows = patch_flows(
714        input_patch,
715        measurement_ids,
716        TileFlow::Consume,
717        |m, tile_basis| transition_in_flow_mids(m, tile_basis, ctx.d, top_basis, reversed),
718    );
719    let mut output_flows = patch_flows(
720        output_patch,
721        measurement_ids,
722        TileFlow::Create,
723        |m, tile_basis| {
724            if reversed {
725                standard_out_flow_mids(m, tile_basis, ctx.d, top_basis)
726            } else {
727                degenerate_out_flow_mids(m, ctx.d, top_basis)
728            }
729        },
730    );
731
732    // The twist turns the degenerate-patch stabilizer adjacent to the Y corner
733    // negative. Stim's signed flow check pins this convention in both time
734    // directions.
735    let y_corner = y_corner_qubit(ctx.d, top_basis);
736    let offset = if top_basis == Basis::X {
737        IVec2::ONE
738    } else {
739        IVec2::NEG_ONE
740    };
741    let signed_flows = if reversed {
742        &mut input_flows
743    } else {
744        &mut output_flows
745    };
746    signed_flows
747        .iter_mut()
748        .find(|flow| flow.center == Some(y_corner + offset))
749        .expect("the degenerate patch contains the Y-corner stabilizer")
750        .sign = true;
751
752    input_flows.extend(output_flows);
753    input_flows
754}

758    circuit: &mut CoordCircuit,
759    qs: &crate::FxSet<IVec2>,

761    sign: i32,

767/// Emit an arbitrary 2-qubit gate with `toward()` pairs, skipping if empty.
768fn emit_gate_pair(
769    circuit: &mut CoordCircuit,
770    gate: GateType,
771    qs: &crate::FxSet<IVec2>,
772    delta: Corner,
773    sign: i32,
774    used: &crate::FxSet<IVec2>,
775) -> Result<(), CompileError> {

783/// The single MY corner qubit for the transition round.
784fn y_corner_qubit(d: i32, top_basis: Basis) -> IVec2 {
785    if top_basis == Basis::X {
786        // data_coord(0, d-1, d) = (1, 1)
787        IVec2::new(1, 1)
788    } else {
789        // data_coord(d-1, 0, d) = (2d-1, 2d-1)
790        IVec2::new(2 * d - 1, 2 * d - 1)
791    }
792}

794/// Collect the measurement qubits for the observable flow in Y measurement block.
795///
796/// Includes the MY corner qubit plus measure qubits in the upper-right and
797/// down-left quadrants (relative to the center at `(d, d)`).
798fn collect_observable_meas_qubits(sets: &TransitionSets, d: i32, top_basis: Basis) -> Vec<IVec2> {
799    let center = IVec2::new(d, d);
800
801    let my_target = y_corner_qubit(d, top_basis);
802
803    let (mset_ur, mset_dl) = if top_basis == Basis::Z {
804        (&sets.xs, &sets.zs)
805    } else {
806        (&sets.zs, &sets.xs)
807    };
808
809    let mut ms = vec![my_target];
810
811    // Upper-right quadrant (bloq Y-up: large x, large y)
812    for &q in mset_ur.iter().chain(sets.top_row.iter()) {
813        if q.x > center.x && q.y > center.y {
814            ms.push(q);
815        }
816    }
817    // Down-left quadrant (bloq Y-up: small x, small y)
818    for &q in mset_dl.iter().chain(sets.left_col.iter()) {
819        if q.x < center.x && q.y < center.y {
820            ms.push(q);
821        }
822    }
823    sorted_unique_coords(ms)
824}

826/// Collect the measurement qubits for the observable flow in Y init block.
827fn collect_observable_meas_qubits_reversed(
828    sets: &TransitionSets,
829    d: i32,
830    top_basis: Basis,
831) -> Vec<IVec2> {
832    let center = IVec2::new(d, d);
833
834    let mut ms = Vec::new();
835    let (mset0, mset1) = if top_basis == Basis::Z {
836        (&sets.zs, &sets.xs)
837    } else {
838        (&sets.xs, &sets.zs)
839    };
840    for &q in mset0.iter().chain(sets.top_row.iter()) {
841        if (q.x > center.x && q.y > q.x) || (q.x < center.x && q.y <= q.x) {
842            ms.push(q);
843        }
844    }
845    for &q in mset1.iter().chain(sets.left_col.iter()) {
846        if (q.x > center.x && q.y <= q.x && q.y > center.y)
847            || (q.x < center.x && q.y > q.x && q.y < center.y)
848        {
849            ms.push(q);
850        }
851    }
852    sorted_unique_coords(ms)
853}

855/// Measure-qubit offset: one tile step upward in bloq coords.
856const UP: IVec2 = IVec2::new(0, 2);
857/// Measure-qubit offset: one tile step leftward in bloq coords.
858const LEFT: IVec2 = IVec2::new(-2, 0);
859
860/// One tile step along the twist, as a measure-qubit offset.
861///
862/// The forward round borrows the partner above for an `X`-top patch and to the
863/// left for a `Z`-top one; running the round backwards borrows the other way.
864fn twist_step(top_basis: Basis, reversed: bool) -> IVec2 {
865    if (top_basis == Basis::X) ^ reversed {
866        UP
867    } else {
868        LEFT
869    }
870}
871
872/// Measurement references for a transition round's **input** flow, in both
873/// time directions.
874///
875/// `m` is the tile's measure qubit. The patch splits into three: the twist
876/// line `m.x == m.y`, the upper-left bulk `m.x < m.y`, and everything else —
877/// the down-right region plus the two boundary strips, which all read their
878/// own ancilla.
879///
880/// The forward and reverse rounds differ only in which way the twist leans:
881/// upper-left tiles borrow their partner one step along it, with `Z` tiles
882/// leaning the opposite way from `X` tiles. (The reverse round used to test
883/// all four patch edges where the forward round tested two; the extra two lie
884/// in the down-right region, which answers `[m]` either way, so the wider test
885/// was vacuous.)
886fn transition_in_flow_mids(
887    m: IVec2,
888    tile_basis: Basis,
889    d: i32,
890    top_basis: Basis,
891    reversed: bool,
892) -> Vec<IVec2> {
893    if m.x == m.y {
894        let partner = m + twist_step(top_basis, reversed);
895        return if reversed {
896            vec![partner, m]
897        } else {
898            vec![m, partner]
899        };
900    }
901    if m.x < m.y && m.x != 0 && m.y != 2 * d {
902        let lean = reversed ^ (tile_basis == Basis::Z);
903        return vec![m + twist_step(top_basis, lean)];
904    }
905    vec![m]
906}
907
908/// Measurement references for the **degenerate** patch's creator flows — the
909/// forward round's output interface.
910fn degenerate_out_flow_mids(m: IVec2, d: i32, top_basis: Basis) -> Vec<IVec2> {
911    let is_xtop = top_basis == Basis::X;
912
913    // Special corner cases at the twist-line endpoints.
914    // (2d-2, 2d-2): top-right twist endpoint (Z-top only).
915    if m.x == 2 * d - 2 && m.y == 2 * d - 2 && !is_xtop {
916        return vec![m, m + LEFT, m + UP, m + Corner::TR.to_ivec2()];
917    }
918    // (2, 2): bottom-left twist endpoint (X-top only).
919    if m.x == 2 && m.y == 2 && is_xtop {
920        return vec![m, m + LEFT, m + UP, m + Corner::BL.to_ivec2()];
921    }
922    // (2, 2d): top boundary singleton (Z-top only).
923    if m.x == 2 && m.y == 2 * d && !is_xtop {
924        return vec![m];
925    }
926    // (0, 2(d-1)): left boundary singleton (X-top only).
927    if m.x == 0 && m.y == 2 * (d - 1) && is_xtop {
928        return vec![m];
929    }
930
931    // Top row: pair with left neighbor.
932    if m.y == 2 * d {
933        return vec![m, m + LEFT];
934    }
935    // Left column: pair with upper neighbor.
936    if m.x == 0 {
937        return vec![m, m + UP];
938    }
939
940    // Twist line: pair with both neighbors.
941    if m.x == m.y {
942        return vec![m, m + LEFT, m + UP];
943    }
944
945    vec![m]
946}

948/// Measurement references for the **standard** patch's creator flows — the
949/// reverse round's output interface.
950///
951/// Kept separate from [`degenerate_out_flow_mids`]: the two describe different
952/// patches, not two views of one rule.
953fn standard_out_flow_mids(m: IVec2, tile_basis: Basis, d: i32, top_basis: Basis) -> Vec<IVec2> {
954    if tile_basis == top_basis.flip() && m.x == 2 {
955        return vec![m, m + LEFT];
956    }
957    if tile_basis == top_basis && m.y == 2 * d - 2 {
958        return vec![m + UP, m];
959    }
960    vec![m]
961}

963fn get_final_measure_data_basis(
964    data_qubits: &crate::FxSet<IVec2>,
965    top_basis: Basis,
966    d: i32,
967) -> crate::FxMap<IVec2, Basis> {
968    data_qubits
969        .iter()
970        .map(|&q| {
971            // In tile coords: tx < ty (Z-top) or tx <= ty (X-top) determines basis.
972            // In bloq coords: q.x + q.y < 2*d or q.x + q.y <= 2*d.
973            let basis = if top_basis == Basis::Z {
974                if q.x + q.y < 2 * d {
975                    Basis::Z
976                } else {
977                    Basis::X
978                }
979            } else if q.x + q.y <= 2 * d {
980                Basis::X
981            } else {
982                Basis::Z
983            };
984            (q, basis)
985        })
986        .collect()
987}

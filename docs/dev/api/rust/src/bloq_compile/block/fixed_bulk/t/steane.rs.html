1//! Steane-code stage library: the cultivation half of a T block.
2//!
3//! Ports the three msc-ls cultivation stages as data-driven chunks (Port-block
4//! precedent — hand-written circuit + flows, no round machinery):
5//!
6//! 1. [`injection_chunk`] — 13-tick unitary (non-FT) encoding of the magic
7//!    state into the d=3 Steane code (`injection_generator`, STE:495). The
8//!    tick-8 seed is the **true non-Clifford `T`**, reversing msc-ls's
9//!    `S_DAG` stand-in orientation to match bloq's positive logical-Z
10//!    convention. Clifford-only consumers substitute it via
11//!    [`GateType::clifford_proxy`] (`T -> S`); the editor program view and
12//!    simulator see the real gate.
13//! 2. [`syndrome_round_chunk`] — one superdense ZX syndrome round
14//!    (`zx_syndrome_extraction_after_injection_generator`, STE:708; the
15//!    variant the msc-ls surgery pipeline composes, LSCG:373). All 8
16//!    measurements are post-selected → restart parities.
17//! 3. [`check_chunk`] — the transversal Hadamard-test check certifying the T
18//!    state (`check_generator`, STE:828): the true non-Clifford `T … T_DAG`
19//!    conjugation sandwich, whose `S … S_DAG` proxy is the complex conjugate
20//!    of msc-ls's `S_DAG … S`; flag MX
21//!    on Steane qubit 0, reverse cascade, 6 post-selected ancilla MXs. The
22//!    single msc-ls `perform_check` implicitly performs the papers' *pair* of
23//!    "Check T" boxes.
24//!
25//! Coordinates are authored in msc-ls space and transported per variant
26//! through [`SurgeryLayout::map_msc`] — the rotation only. The merge-basis
27//! dual deliberately leaves cultivation untouched: the stage
28//! prepares the same magic state regardless of how the escape measures it;
29//! only the surgery and readout dualize.
30//!
31//! ## Flow decomposition
32//!
33//! Every msc-ls cultivation detector is a post-selected single measurement,
34//! deterministic only given the codespace prepared upstream. In chunk-flow
35//! terms that determinism is a *chain closure*, so the stages decompose like
36//! a standard syndrome round:
37//!
38//! - injection emits one creator flow `∅ → S` per Steane stabilizer;
39//!   every flow in these chunks is restart-flagged, because every closure of
40//!   a cultivation stabilizer chain — here or in the escape template — is
41//!   post-selected;
42//! - the syndrome round consumes each stabilizer (closing a restart chain
43//!   whose parity is exactly the msc-ls post-selected single) and re-creates
44//!   it toward the check;
45//! - the check has no stabilizer-closing measurement (its ancilla web only
46//!   reconstructs syndromes jointly with the flag), so its stabilizer flows
47//!   are pass-throughs; the three X-face pass-throughs splice in the flag.
48//!
49//! One msc-ls post-selection is not a stabilizer chain at all: the check's
50//! flag reads the *logical* operator the cultivated state is an eigenstate of.
51//! It travels its own chain — injection creates, the syndrome round transports,
52//! the check reads back — described in [`transversal_y`].
53//!
54//! The measurement sets on each flow were derived from the msc-ls reference
55//! circuits with stim (`has_flow`, minimal subsets) and are pinned semantically
56//! by the `verify_flows` tests below. This module holds the canonical
57//! restart-parity accounting: 14 template-local entries against msc-ls's 15,
58//! the one deviation being the qubit-2 shuttle readout described in
59//! [`syndrome_round_chunk`].
60

72/// The magic-state injection stage (STE:495): 13 ticks, no measurements.
73/// Flows: one restart-flagged creator per Steane stabilizer at the
74/// post-injection positions.
75pub(crate) fn injection_chunk(layout: &SurgeryLayout) -> Result<Chunk, CompileError> {
76    let mut b = StageBuilder::new(layout);
77    // msc-ls resets each qubit mid-sweep, right before its first CX — mixing
78    // reset and two-qubit ticks. Every such qubit idles from circuit start to
79    // its reset, so hoisting all resets into the opening tick is
80    // circuit-equivalent and keeps each tick to a single op kind.
81    b.gate(
82        GateType::RX,
83        &[
84            (5, 13),
85            (3, 13),
86            (3, 15),
87            (1, 13),
88            (6, 12),
89            (3, 11),
90            (5, 15),
91            (2, 12),
92            (2, 10),
93            (0, 14),
94        ],
95    )?;
96    b.gate(GateType::RZ, &[(4, 12), (4, 14), (2, 14)])?;
97    b.tick();
98    b.cx(&[((5, 13), (4, 12)), ((3, 13), (4, 14)), ((3, 15), (2, 14))])?;
99    b.tick();
100    b.cx(&[((5, 13), (4, 14)), ((3, 13), (2, 14))])?;
101    b.tick();
102    b.cx(&[
103        ((3, 13), (4, 12)),
104        ((3, 15), (4, 14)),
105        ((1, 13), (2, 14)),
106        ((6, 12), (5, 13)),
107    ])?;
108    b.tick();
109    b.cx(&[
110        ((3, 11), (4, 12)),
111        ((5, 15), (4, 14)),
112        ((2, 12), (1, 13)),
113        ((5, 13), (6, 12)),
114    ])?;
115    b.tick();
116    b.cx(&[((2, 12), (3, 11)), ((4, 14), (5, 15)), ((2, 14), (1, 13))])?;
117    b.tick();
118    b.cx(&[((4, 12), (3, 11)), ((1, 13), (2, 12))])?;
119    b.tick();
120    b.cx(&[((3, 11), (2, 12))])?;
121    b.tick();
122    // The seed of the magic state: the true non-Clifford `T` (bloq keeps the
123    // real gate and lets the Clifford proxy substitute it for Clifford-only
124    // consumers — see module docs). DELIBERATE DEVIATION from msc-ls (whose
125    // stand-in is `S_DAG`, i.e. a `T_DAG` seed): msc-ls's implicit logical
126    // frame is `Z_L = −Z₁Z₃Z₅` / `Y_L = Y₁Y₄Y₆`, in which its `T_DAG` seed IS
127    // logical T; bloq's frame is the all-positive `Z̄ = +Z₁Z₃Z₅` /
128    // `Ȳ = iX̄Z̄ = −Y₁Y₄Y₆` (the escape gateway's strings), where the seed
129    // transports as `C₂ Z_v C₂† = +Z₁Z₄Z₆ ≡ +Z̄` (CX-only, sign-free), so a
130    // `T` seed is the positive-frame logical T̄ and the delivered state is
131    // exactly `|T̄⟩ = T̄|+̄⟩` — eliminating the output-frame conversion
132    // constant. The check sandwich below flips with it, keeping seed and
133    // certification consistent.
134    b.gate(GateType::T, &[(2, 12)])?;
135    b.tick();
136    b.cx(&[((3, 11), (2, 12))])?;
137    b.tick();
138    b.cx(&[((1, 13), (2, 12))])?;
139    b.tick();
140    b.cx(&[((2, 10), (3, 11)), ((0, 14), (1, 13))])?;
141    b.tick();
142    b.cx(&[((3, 11), (2, 10)), ((1, 13), (0, 14))])?;
143
144    let mut flows: Vec<Flow> = stabilizer_supports(layout, &STEANE_INJECTED)
145        .map(|support| restart_flow(PauliMap::empty(), support, &[], &[]))
146        .collect();
147    // The logical anchor (see [`transversal_y`]): injection seeds the magic
148    // state, so the anchor is a creator here.
149    flows.push(
150        restart_flow(
151            PauliMap::empty(),
152            transversal_y(layout, &STEANE_INJECTED),
153            &[],
154            &[],
155        )
156        .with_sign(true),
157    );
158    Ok(Chunk {
159        circuit: b.circuit,
160        flows,
161    })
162}

164/// One superdense ZX syndrome round after injection (STE:708). Moves qubit 1
165/// to its final home and qubit 2 out and back.
166///
167/// Deviates from msc-ls: qubit 2's vacated home is *not* re-initialized
168/// mid-round. The move-back `CX((6,12),(5,11)); CX((5,11),(6,12))` is then not
169/// a clean move — the pair ends stabilized by `Z(6,12)Z(5,11)`, so `MX(5,11)`
170/// comes out random and its byproduct rides every chain carrying X on qubit 2.
171/// Each such chain absorbs it (see the rec tables below), so nothing is lost
172/// except msc-ls's standalone `MX(5,11)` post-selection: 14 template-local
173/// parities where the reference declares 15.
174pub(crate) fn syndrome_round_chunk(layout: &SurgeryLayout) -> Result<Chunk, CompileError> {
175    // Measurements, in emission order:
176    // `[MX(1,13), MZ(2,14), MZ(0,14), MX(4,14), MZ(5,13), MX(3,11), MZ(4,12), MX(5,11)]`.
177    // Consumers close the injection creators into restart parities. Measurement
178    // 2 is the qubit-1 move byproduct, deterministic on its own; measurement 7
179    // is the qubit-2 one and is not (module docs above), so it folds into the
180    // 0235 X face it shuttled through.
181    const CONSUMER_RECS: [&[u32]; 6] = [&[0], &[1], &[3, 7], &[4], &[5], &[6]];
182    // Correlation of each re-created stabilizer with the round's outcomes (the
183    // mid-round qubit moves entangle faces, hence the cross-face Z sets). The
184    // un-re-initialized (5,11) rides the 0246 X face on re-creation.
185    const CREATOR_RECS: [&[u32]; 6] = [&[0], &[6], &[3], &[], &[5, 7], &[1, 4, 6]];
186    const LOCAL_RESTARTS: [&[u32]; 1] = [&[2]];
187
188    let mut b = StageBuilder::new(layout);
189    b.gate(GateType::RX, &[(1, 13)])?;
190    b.gate(GateType::RZ, &[(2, 14), (1, 15)])?;
191    b.gate(GateType::RX, &[(4, 14)])?;
192    b.gate(GateType::RZ, &[(5, 13)])?;
193    b.gate(GateType::RX, &[(3, 11)])?;
194    b.gate(GateType::RZ, &[(4, 12), (5, 11)])?;
195    b.tick();
196    b.cx(&[
197        ((0, 14), (1, 15)),
198        ((1, 13), (2, 14)),
199        ((4, 14), (5, 13)),
200        ((3, 11), (4, 12)),
201        ((6, 12), (5, 11)),
202    ])?;
203    b.tick();
204    b.cx(&[
205        ((3, 15), (2, 14)),
206        ((0, 14), (1, 13)),
207        ((5, 15), (4, 14)),
208        ((6, 12), (5, 13)),
209        ((2, 10), (3, 11)),
210        ((3, 13), (4, 12)),
211    ])?;
212    b.tick();
213    b.cx(&[
214        ((1, 15), (0, 14)),
215        ((3, 13), (4, 14)),
216        ((2, 12), (3, 11)),
217        ((5, 11), (6, 12)),
218    ])?;
219    b.tick();
220    b.cx(&[
221        ((2, 12), (1, 13)),
222        ((3, 13), (2, 14)),
223        ((3, 15), (4, 14)),
224        ((5, 11), (4, 12)),
225    ])?;
226    b.tick();
227    b.cx(&[
228        ((1, 13), (2, 12)),
229        ((2, 14), (3, 13)),
230        ((4, 14), (3, 15)),
231        ((4, 12), (5, 11)),
232    ])?;
233    b.tick();
234    b.cx(&[
235        ((2, 14), (3, 15)),
236        ((4, 14), (3, 13)),
237        ((3, 11), (2, 10)),
238        ((6, 12), (5, 11)),
239    ])?;
240    b.tick();
241    b.cx(&[
242        ((2, 14), (1, 15)),
243        ((4, 14), (5, 15)),
244        ((5, 13), (6, 12)),
245        ((3, 11), (2, 12)),
246        ((4, 12), (3, 13)),
247    ])?;
248    b.tick();
249    b.cx(&[
250        ((1, 13), (2, 14)),
251        ((4, 14), (5, 13)),
252        ((3, 11), (4, 12)),
253        ((5, 11), (6, 12)),
254    ])?;
255    b.tick();
256    let measurements = [
257        b.measure(PauliBasis::X, (1, 13)),
258        b.measure(PauliBasis::Z, (2, 14)),
259        b.measure(PauliBasis::Z, (0, 14)),
260        b.measure(PauliBasis::X, (4, 14)),
261        b.measure(PauliBasis::Z, (5, 13)),
262        b.measure(PauliBasis::X, (3, 11)),
263        b.measure(PauliBasis::Z, (4, 12)),
264        b.measure(PauliBasis::X, (5, 11)),
265    ];
266
267    let mut flows = Vec::with_capacity(14);
268    let supports_in = stabilizer_supports(layout, &STEANE_INJECTED);
269    let supports_out = stabilizer_supports(layout, &STEANE_FINAL);
270    for (index, (support_in, support_out)) in supports_in.zip(supports_out).enumerate() {
271        flows.push(restart_flow(
272            support_in,
273            PauliMap::empty(),
274            CONSUMER_RECS[index],
275            &measurements,
276        ));
277        flows.push(restart_flow(
278            PauliMap::empty(),
279            support_out,
280            CREATOR_RECS[index],
281            &measurements,
282        ));
283    }
284    flows.extend(
285        LOCAL_RESTARTS.iter().map(|parity| {
286            restart_flow(PauliMap::empty(), PauliMap::empty(), parity, &measurements)
287        }),
288    );
289    // The logical anchor rides along untouched — it commutes with everything
290    // the round measures — but it does ride the two Z-face ancillas it shares
291    // support with (`[MZ(2,14), MZ(5,13)]`) and, because `Y(6,12)` carries X on
292    // qubit 2, the dirty move-back's byproduct `MX(5,11)` as well. Folding that
293    // third outcome in is what keeps the anchor a deterministic chain without
294    // the re-init: the shuttle readout is individually random, but its
295    // randomness *is* the anchor's byproduct.
296    flows.push(restart_flow(
297        transversal_y(layout, &STEANE_INJECTED),
298        transversal_y(layout, &STEANE_FINAL),
299        &[1, 4, 7],
300        &measurements,
301    ));
302    Ok(Chunk {
303        circuit: b.circuit,
304        flows,
305    })
306}

308/// The transversal Hadamard-test check (STE:828). Qubit positions are
309/// unchanged; the flag rides the open X-face chains toward the escape
310/// template (module docs).
311pub(crate) fn check_chunk(layout: &SurgeryLayout) -> Result<Chunk, CompileError> {
312    // Measurements, in emission order: flag `MX(3,13)` then the 6 ancilla MXs
313    // `[(3,11), (4,12), (5,13), (4,14), (1,13), (2,14)]`. X-face pass-throughs
314    // splice in the flag (index 0); Z faces pass through untouched.
315    const PASSTHROUGH_RECS: [&[u32]; 6] = [&[0], &[], &[0], &[], &[0], &[]];
316    // Template-local restart parities: the 5 standalone-deterministic ancilla
317    // MXs plus the flag⊗`MX(4,12)` pair. Neither half of that pair is a Pauli
318    // flow on its own — both measure the *logical* `(X̄+Ȳ)/√2` — so the flag's
319    // own msc-ls post-selection rides the logical anchor chain instead
320    // ([`transversal_y`]), and `MX(4,12)` follows from the two together.
321    const LOCAL_RESTARTS: [&[u32]; 6] = [&[1], &[3], &[4], &[5], &[6], &[0, 2]];
322    // The ancilla-web CX cascade, one slice per tick; the check replays it in
323    // reverse after the flag (the circuit is a palindrome around the flag).
324    const CASCADE_TICKS: [&[(MscQubit, MscQubit)]; 4] = [
325        &[
326            ((3, 11), (2, 10)),
327            ((4, 12), (3, 13)),
328            ((5, 13), (6, 12)),
329            ((4, 14), (5, 15)),
330            ((1, 13), (2, 12)),
331            ((2, 14), (1, 15)),
332        ],
333        &[((2, 12), (3, 11)), ((4, 14), (5, 13)), ((3, 13), (2, 14))],
334        &[((3, 13), (2, 12)), ((4, 14), (3, 15))],
335        &[((3, 13), (4, 14))],
336    ];
337
338    let mut b = StageBuilder::new(layout);
339    // msc-ls resets the ancilla web and applies the transversal T† in one
340    // tick; two ticks keep the reset and rotation kinds unmixed.
341    b.gate(
342        GateType::RX,
343        &[(3, 11), (4, 12), (5, 13), (4, 14), (1, 13), (2, 14)],
344    )?;
345    b.tick();
346    // The transversal non-Clifford conjugation `T … flag-MX … T_DAG` certifying
347    // the magic state. DELIBERATE DEVIATION from msc-ls (`S_DAG … S` stand-in,
348    // i.e. `T_DAG … T`), mirroring the seed flip above: the measured operator
349    // is `⊗(T† X T) = ⊗(X−Y)/√2`, transversally the positive-frame logical
350    // `(X̄+Ȳ)/√2` (X-faces → (−Y)⁴-faces = X·Z-face products, Z-faces fixed —
351    // the stabilizer group is unchanged), whose +1 eigenstate is the
352    // positive-frame `|T̄⟩` the flipped seed injects.
353    b.gate(GateType::T, &STEANE_FINAL)?;
354    for tick in CASCADE_TICKS {
355        b.tick();
356        b.cx(tick)?;
357    }
358    b.tick();
359    let flag = b.measure(PauliBasis::X, (3, 13));
360    b.tick();
361    b.gate(GateType::RX, &[(3, 13)])?;
362    for tick in CASCADE_TICKS.iter().rev() {
363        b.tick();
364        b.cx(tick)?;
365    }
366    b.tick();
367    b.gate(GateType::T_DAG, &STEANE_FINAL)?;
368    b.tick();
369    let measurements = [
370        flag,
371        b.measure(PauliBasis::X, (3, 11)),
372        b.measure(PauliBasis::X, (4, 12)),
373        b.measure(PauliBasis::X, (5, 13)),
374        b.measure(PauliBasis::X, (4, 14)),
375        b.measure(PauliBasis::X, (1, 13)),
376        b.measure(PauliBasis::X, (2, 14)),
377    ];
378
379    let mut flows: Vec<Flow> = stabilizer_supports(layout, &STEANE_FINAL)
380        .enumerate()
381        .map(|(index, support)| {
382            restart_flow(
383                support.clone(),
384                support,
385                PASSTHROUGH_RECS[index],
386                &measurements,
387            )
388        })
389        .collect();
390    flows.extend(
391        LOCAL_RESTARTS.iter().map(|parity| {
392            restart_flow(PauliMap::empty(), PauliMap::empty(), parity, &measurements)
393        }),
394    );
395    // The check reads the logical anchor back and does not re-create it: the
396    // flag is exactly that readout, which is what makes msc-ls's standalone
397    // `MX(3,13)` post-selection expressible here (see [`transversal_y`]).
398    flows.push(
399        restart_flow(
400            transversal_y(layout, &STEANE_FINAL),
401            PauliMap::empty(),
402            &[0],
403            &measurements,
404        )
405        .with_sign(true),
406    );
407    Ok(Chunk {
408        circuit: b.circuit,
409        flows,
410    })
411}

413/// The transversal `Y⊗7` at the given positions: the *logical* anchor of the
414/// cultivated magic state, and the one post-selection that is not a stabilizer
415/// chain.
416///
417/// Conjugating the check's flag `X(3,13)` back through the ancilla cascade
418/// (whose `RX`s pin every ancilla factor to `+1`) and then through the
419/// transversal seed gate turns it into `⊗(T†XT) = ⊗(X−Y)/√2`, i.e. the logical
420/// `(X̄+Ȳ)/√2` — deterministic precisely because cultivation delivers exactly
421/// `|T̄⟩` (`tests::cultivation_delivers_exact_t_state`). The final `MX(4,12)`
422/// measures the same operator, which is why the two are a Pauli flow together
423/// while neither is alone.
424///
425/// Under the Clifford proxy that carries flow verification (`T -> S`) the same
426/// operator is `⊗(S†XS) = ⊗(−Y) = −Y⊗7`, an honest Pauli, so the anchor is a
427/// stabilizer of the injected state and the chain
428/// *injection creates → syndrome round transports → check reads back* verifies
429/// with stim and composes to the single-measurement restart `{flag}` msc-ls
430/// declares. The proxy is what makes the chain expressible; the parity it
431/// yields is deterministic under the true non-Clifford `T` as well.
432///
433/// `Y` is dual-invariant, so — like the rest of cultivation — this needs no
434/// dual-aware basis.
435fn transversal_y(layout: &SurgeryLayout, positions: &[MscQubit; 7]) -> PauliMap {
436    positions
437        .iter()
438        .map(|&q| (layout.map_msc(q), Pauli::Y))
439        .collect()
440}

442/// The 6 Steane stabilizer supports in table order (`2 * face + {X: 0, Z: 1}`),
443/// at the given qubit positions, transported per variant.
444fn stabilizer_supports<'a>(
445    layout: &'a SurgeryLayout,
446    positions: &'a [MscQubit; 7],
447) -> impl Iterator<Item = PauliMap> + 'a {
448    FACES.iter().flat_map(move |face| {
449        [Pauli::X, Pauli::Z].map(move |pauli| {
450            face.iter()
451                .map(|&qubit| (layout.map_msc(positions[qubit]), pauli))
452                .collect()
453        })
454    })
455}

457/// A restart-flagged flow between the given interfaces, carrying the
458/// `table`-indexed measurements. With both interfaces empty this is a
459/// template-local restart parity (a deterministic, post-selected product).
460fn restart_flow(start: PauliMap, end: PauliMap, table: &[u32], measurements: &[u32]) -> Flow {
461    Flow::new(start, end)
462        .with_measurements(recs(table, measurements))
463        .with_marker(FlowMarker::Restart)
464}
465
466fn recs<'a>(table: &'a [u32], measurements: &'a [u32]) -> impl Iterator<Item = u32> + 'a {
467    table.iter().map(move |&index| measurements[index as usize])
468}

470/// The cultivation template (stages 1–2): injection + one
471/// superdense syndrome round + the Hadamard-test check, composed standalone.
472/// No observable gateway — the escape template owns the in/out story; the six
473/// boundary flows are the restart-flagged open Steane chains the escape
474/// template's first merge round closes.
475pub(crate) fn cultivation_template(
476    layout: &SurgeryLayout,
477) -> Result<LoweringTemplate, CompileError> {
478    let chunks = vec![
479        ChunkOrLoop::Single(Box::new(injection_chunk(layout)?)),
480        ChunkOrLoop::Single(Box::new(syndrome_round_chunk(layout)?)),
481        ChunkOrLoop::Single(Box::new(check_chunk(layout)?)),
482    ];
483    LoweringTemplate::from_chunks(chunks, ObservableGateway::new())
484}

486/// Transliterates one msc-ls tick sequence, mapping every msc-ls coordinate
487/// through the variant layout.
488struct StageBuilder<'a> {
489    layout: &'a SurgeryLayout,
490    circuit: CoordCircuit,
491}
492
493impl<'a> StageBuilder<'a> {
494    fn new(layout: &'a SurgeryLayout) -> Self {
495        Self {
496            layout,
497            circuit: CoordCircuit::new(),
498        }
499    }
500
501    fn gate(&mut self, gate: GateType, qubits: &[MscQubit]) -> Result<(), CompileError> {
502        let targets: Vec<IVec2> = qubits.iter().map(|&q| self.layout.map_msc(q)).collect();
503        Ok(self.circuit.do_gate(gate, targets)?)
504    }
505
506    fn cx(&mut self, pairs: &[(MscQubit, MscQubit)]) -> Result<(), CompileError> {
507        let targets: Vec<IVec2> = pairs
508            .iter()
509            .flat_map(|&(control, target)| {
510                [self.layout.map_msc(control), self.layout.map_msc(target)]
511            })
512            .collect();
513        Ok(self.circuit.do_gate(GateType::CX, targets)?)
514    }
515
516    fn measure(&mut self, basis: PauliBasis, qubit: MscQubit) -> u32 {
517        self.circuit.measure(basis, [self.layout.map_msc(qubit)])[0]
518    }
519
520    fn tick(&mut self) {

539    /// Simulate the full cultivation stage alone (injection, syndrome round,
540    /// check) with the TRUE `T` seed (`bloq_vm`), and pin the delivered
541    /// logical Bloch vector against the exact canonical edge strings the
542    /// escape gateway anchors on (`Z̄ = Z{1,3,5}` seam line, `X̄ = X{1,4,6}`
543    /// west edge, `Ȳ := iX̄Z̄ = Y₁X₄X₆Z₃Z₅`).
544    ///
545    /// Cultivation delivers **exactly** `|T̄⟩ = T̄|+̄⟩` in that positive
546    /// frame — ⟨X̄⟩ = +1/√2, ⟨Ȳ⟩ = +1/√2, ⟨Z̄⟩ = 0, all stabilizers +1 —
547    /// with no Pauli dressing and no sign constant. (The dressed variants
548    /// are distinguishable by the (⟨X̄⟩, ⟨Ȳ⟩) sign pattern: ideal (+,+),
549    /// X̄-dressed (+,−), Z̄-dressed (−,−), Ȳ-dressed (−,+).) With
550    /// `escape::tests::gateway_record_folds_equal_physical_byproduct` this
551    /// pins the frame-constant-free output frames
552    /// lowered by `registry/readouts.rs`.
553    #[test]
554    fn cultivation_delivers_exact_t_state() {
555        use bloq_vm::{CircuitExecutor, EnginePauli, EnginePauliString as PauliString, Simulator};
556
557        let layout = layout(Direction::YMINUS, 3);
558        let template = cultivation_template(&layout).unwrap();
559        let circuit = &template.program_template.circuit;
560        let executor = CircuitExecutor::new(circuit).expect("valid cultivation circuit");
561        let n = executor.qubit_count();
562        let coord_to_index = circuit.build_coord_to_index();
563
564        let at = |q: usize| coord_to_index[&layout.map_msc(STEANE_FINAL[q])] as usize;
565        // Every string below names distinct qubits, so this is a set of sites
566        // rather than a Pauli product — no phase to carry.
567        let engine_pauli = |pauli| match pauli {
568            Pauli::X => EnginePauli::X,
569            Pauli::Y => EnginePauli::Y,
570            Pauli::Z => EnginePauli::Z,
571            Pauli::I => unreachable!("Pauli strings omit identity terms"),
572        };
573        let string = |terms: &[(usize, Pauli)]| -> PauliString {
574            PauliString::from_terms(n, terms.iter().map(|&(q, p)| (at(q), engine_pauli(p))))
575        };
576
577        let x_bar = string(&[(1, Pauli::X), (4, Pauli::X), (6, Pauli::X)]);
578        let z_bar = string(&[(1, Pauli::Z), (3, Pauli::Z), (5, Pauli::Z)]);
579        let y_bar = string(&[
580            (1, Pauli::Y),
581            (4, Pauli::X),
582            (6, Pauli::X),
583            (3, Pauli::Z),
584            (5, Pauli::Z),
585        ]);
586        let stabilizers: Vec<PauliString> = FACES
587            .iter()
588            .flat_map(|face| {
589                [Pauli::X, Pauli::Z]
590                    .map(|p| string(&face.iter().map(|&q| (q, p)).collect::<Vec<_>>()))
591            })
592            .collect();
593
594        const TOL: f64 = 1e-9;
595        let m = circuit.num_measurements();
596        let mut accepted_shots = 0;
597        for seed in 0..8u64 {
598            let mut sim = Simulator::with_seed(n, seed);
599            let rec = executor.run_shot(&mut sim).unwrap();
600            let accepted = (0..m).all(|id| rec.get(id) == Some(false));
601            if !accepted {
602                continue;
603            }
604            accepted_shots += 1;
605            for (i, stabilizer) in stabilizers.iter().enumerate() {
606                let value = sim.peek_observable_expectation(stabilizer).unwrap();
607                assert!(
608                    (value - 1.0).abs() < TOL,
609                    "seed {seed}: stabilizer {i} expectation {value}, want +1"
610                );
611            }
612            let ex = sim.peek_observable_expectation(&x_bar).unwrap();
613            let ey = sim.peek_observable_expectation(&y_bar).unwrap();
614            let ez = sim.peek_observable_expectation(&z_bar).unwrap();
615            let inv_sqrt2 = std::f64::consts::FRAC_1_SQRT_2;
616            assert!(
617                (ex - inv_sqrt2).abs() < TOL && (ey - inv_sqrt2).abs() < TOL && ez.abs() < TOL,
618                "seed {seed}: delivered Bloch ({ex:+.4}, {ey:+.4}, {ez:+.4}), \
619                 want exactly |T̄⟩ = (+1/√2, +1/√2, 0)"
620            );
621        }
622        assert!(
623            accepted_shots > 0,
624            "no accepted cultivation shot across the seed batch"
625        );
626    }

714    /// Composing the three stages into a template routes every post-selected
715    /// parity into the `restarts` side table (no ordinary detectors) and
716    /// leaves the six Steane stabilizers open — restart-flagged — for the
717    /// escape template.
718    #[test]
719    fn cultivation_template_restarts_match_msc_ls_post_selection() {
720        let layout = layout(Direction::YMINUS, 3);
721        let template = cultivation_template(&layout).expect("cultivation stages compose");
722        let program = &template.program_template;
723
724        assert!(
725            program.detectors.is_empty(),
726            "cultivation emits no decoder-fed detectors"
727        );
728
729        // Template measurement ids: syndrome round 0..=7, check 8..=14.
730        // Syndrome parities are msc-ls's 8 singles minus the standalone
731        // `MX(5,11)` (id 7), which folds into the 0235 face closure (id 3)
732        // now that the mid-round re-init is gone. Check parities are the 5
733        // standalone ancilla singles, the flag⊗MX(4,12) pair, and the logical
734        // anchor's closure — which carries ids 1, 4 (the Z-face ancillas its
735        // transport rides) and 7 (the shuttle byproduct). Ids 1 and 4 are also
736        // declared alone, so the closure still spans msc-ls's standalone flag
737        // `MX(3,13)` up to id 7, the one deviation
738        // (`escape::tests::restart_set_spans_msc_ls_post_selection`).
739        let expected: crate::FxSet<DetectorParity> = [
740            &[0][..],
741            &[1],
742            &[2],
743            &[3, 7],
744            &[4],
745            &[5],
746            &[6],
747            &[9],
748            &[11],
749            &[12],
750            &[13],
751            &[14],
752            &[8, 10],
753            &[1, 4, 7, 8],
754        ]
755        .into_iter()
756        .map(|parity| DetectorParity::from_measurements(parity.iter().copied()))
757        .collect();
758        let actual: crate::FxSet<DetectorParity> = program
759            .restarts
760            .iter()
761            .map(|restart| restart.parity.clone())
762            .collect();
763        assert_eq!(actual, expected);
764
765        // Boundary: exactly the six stabilizer chains, all restart-flagged,
766        // each entering the escape template with its accumulated syndrome
767        // history (the flag, template id 8, rides every X face).
768        assert_eq!(program.boundary_flows.len(), 6);
769        let ends: crate::FxSet<PauliMap> = program
770            .boundary_flows
771            .iter()
772            .map(|flow| flow.end.clone())
773            .collect();
774        let expected_ends: crate::FxSet<PauliMap> =
775            stabilizer_supports(&layout, &STEANE_FINAL).collect();
776        assert_eq!(ends, expected_ends);
777        for flow in &program.boundary_flows {
778            assert!(flow.start.is_empty());
779            assert_eq!(flow.marker, FlowMarker::Restart);
780        }
781        let x0145: PauliMap = FACES[0]
782            .iter()
783            .map(|&qubit| (layout.map_msc(STEANE_FINAL[qubit]), Pauli::X))
784            .collect();
785        let flag_chain = program
786            .boundary_flows
787            .iter()
788            .find(|flow| flow.end == x0145)
789            .expect("X0145 stays open");
790        let mut measurements = flag_chain.measurements.to_vec();
791        measurements.sort_unstable();
792        assert_eq!(measurements, [0, 8], "syndrome MX(1,13) plus the flag");
793    }

795    /// The rotated variants place the Steane spill outside the patch on the
796    /// chosen side (the cell the placement pass must reserve).
797    #[rstest]
798    #[case(Direction::YMINUS, IVec2::NEG_Y)]
799    #[case(Direction::XPLUS, IVec2::X)]
800    #[case(Direction::YPLUS, IVec2::Y)]
801    #[case(Direction::XMINUS, IVec2::NEG_X)]
802    fn steane_spill_lands_on_chosen_side(
803        #[case] side: Direction,
804        #[case] outward: IVec2,
805        #[values(false, true)] dual: bool,
806    ) {
807        let distance = 3;
808        let layout = family_layout(side, dual, distance);
809        let patch_max = 2 * distance as i32 - 1;
810        for position in STEANE_FINAL {
811            let mapped = layout.map_msc(position);
812            let out_of_patch =
813                mapped.x < 1 || mapped.x > patch_max || mapped.y < 1 || mapped.y > patch_max;
814            assert!(
815                out_of_patch,
816                "Steane qubit {position:?} maps into the patch"
817            );
818            let overshoot = mapped - IVec2::splat(distance as i32);
819            assert!(
820                overshoot.dot(outward) > 0,
821                "Steane qubit {position:?} spills toward the wrong side"
822            );
823        }
824    }

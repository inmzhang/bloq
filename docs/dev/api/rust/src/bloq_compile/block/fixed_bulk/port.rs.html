1//! Temporal open-port block compilation.
2//!
3//! A `Port` is an ideal (noiseless) open temporal boundary that terminates a
4//! logical operator at a temporal face without closing it into an init/meas cube.
5//! It does two things:
6//!
7//! 1. **Anchors the boundary** — a single [`Op::MPP`] jointly measures every patch
8//!    stabilizer (one product per tile); composed with the neighbouring cube's
9//!    syndrome round, each record pairs into a boundary detector like a cube seam.
10//! 2. **Passes the logical through** — its observable gateway carries the X and Z
11//!    lines on *both* faces (`operator_in == operator_out`); the seam toward the
12//!    cube cancels and the open face survives as the boundary `OBSERVABLE_INCLUDE`.
13//!
14//! An ordinary `+Z` pipe is an **input** port (past boundary), and `−Z` is an
15//! **output** (future). Multiplex uses the input orientation plus a corrected
16//! single-qubit output. The patch is rebuilt from the neighbouring cube's
17//! boundary basis.

36/// Compile a temporal port into a single stabilizer-MPP chunk plus a passthrough
37/// observable gateway.
38pub(crate) fn compile_port(
39    boundary_basis: Basis,
40    connectivity: Connectivity,
41    distance: u32,
42) -> Result<CompiledTemplate, CompileError> {
43    // `+Z` pipe ⇒ input port (past boundary); `−Z` pipe ⇒ output port (future).
44    let is_input = connectivity.has_pipe(Direction::ZPLUS);
45    let pipe_dir = if is_input {
46        Direction::ZPLUS
47    } else {
48        Direction::ZMINUS
49    };
50
51    let patch = make_normal_surface_code_patch(distance, boundary_basis);
52    let chunk = make_port_mpp_chunk(&patch, is_input)?;
53    let gateway = build_port_observable_gateway(boundary_basis, distance, pipe_dir);
54
55    let chunks = vec![ChunkOrLoop::Single(Box::new(chunk))];
56    Ok(Arc::new(LoweringTemplate::from_chunks(chunks, gateway)?))
57}

59/// Compile the temporal-input half of a Multiplex spatial Port.
60pub(crate) fn compile_multiplex_port(
61    boundary_basis: Basis,
62    distance: u32,
63    output_qubit: IVec2,
64) -> Result<CompiledTemplate, CompileError> {
65    let patch = make_normal_surface_code_patch(distance, boundary_basis);
66    let mut circuit = CoordCircuit::new();
67    circuit.do_gate(GateType::RX, [output_qubit])?;
68    circuit.tick();
69
70    let mut split = logical_line_operator(distance, Basis::Z, boundary_basis, Pauli::Z);
71    split.insert(output_qubit, Pauli::Z);
72    let control = circuit.measure_pauli_products([split])?[0];
73    circuit.tick();
74    circuit
75        .body_mut(circuit.entry_body())
76        .expect("entry body exists")
77        .ops_mut()
78        .push(Op::ConditionalPauli(vec![ConditionalCorrection {
79            pauli: PauliBasis::X,
80            control,
81            target: output_qubit,
82        }]));
83    circuit.tick();
84
85    let products = patch
86        .tiles()
87        .iter()
88        .map(super::super::patch::Tile::pauli_map);
89    let measurements = circuit.measure_pauli_products(products)?;
90    let flows = patch
91        .tiles()
92        .iter()
93        .zip(measurements)
94        .map(|(tile, measurement)| tile_flow(tile, TileFlow::Create, [measurement]))
95        .collect::<Vec<_>>();
96    let connectivity = Connectivity::ISOLATED.with_pipe(Direction::ZPLUS);
97    let mut gateway = build_port_observable_gateway(boundary_basis, distance, Direction::ZPLUS);
98    let key = LocalStabilizer::new(Pauli::X, connectivity);
99    let mut x_entry = gateway.lookup(key).expect("input gateway spans X");
100    x_entry.operator_out.insert(output_qubit, Pauli::X);
101    gateway.insert(key, x_entry);
102
103    let anticommutes = |map: &PauliMap| matches!(map.get(&output_qubit), Some(Pauli::Z | Pauli::Y));
104    if flows
105        .iter()
106        .any(|flow| anticommutes(&flow.start) || anticommutes(&flow.end))
107        || [Pauli::X, Pauli::Y, Pauli::Z].into_iter().any(|pauli| {
108            let entry = gateway
109                .lookup(LocalStabilizer::new(pauli, connectivity))
110                .expect("Multiplex gateway spans every Pauli");
111            anticommutes(&entry.operator_in) || anticommutes(&entry.operator_out)
112        })
113    {
114        return Err(CompileError::ConditionalPauliFeedbackUnsupported);
115    }
116    Ok(Arc::new(
117        LoweringTemplate::from_chunks_with_commuting_feedback(
118            vec![ChunkOrLoop::Single(Box::new(Chunk { circuit, flows }))],
119            gateway,
120        )?,
121    ))
122}

124/// Build the single MPP chunk measuring every patch stabilizer. Each tile's
125/// record anchors a flow matching the cube's stabilizer flow across the seam: an
126/// input port flows *out* toward the future cube (`start` empty), an output port
127/// flows *in* from the past cube (`end` empty).
128fn make_port_mpp_chunk(patch: &Patch, is_input: bool) -> Result<Chunk, CompileError> {
129    let mut circuit = CoordCircuit::new();
130
131    let measurements = circuit.measure_pauli_products(
132        patch
133            .tiles()
134            .iter()
135            .map(super::super::patch::Tile::pauli_map),
136    )?;
137
138    let direction = if is_input {
139        TileFlow::Create
140    } else {
141        TileFlow::Consume
142    };
143    let flows = patch
144        .tiles()
145        .iter()
146        .zip(&measurements)
147        .map(|(tile, &measurement)| tile_flow(tile, direction, [measurement]))
148        .collect();
149
150    Ok(Chunk { circuit, flows })
151}

153/// Build the passthrough observable gateway. The logical X and Z lines run
154/// perpendicular through the patch centre ([`logical_line_operator`]). Each key
155/// carries the same line on `operator_in` and `operator_out`, so the cube-facing
156/// face cancels at the seam and the open face survives as the boundary
157/// observable.
158fn build_port_observable_gateway(
159    boundary_basis: Basis,
160    distance: u32,
161    pipe_dir: Direction,
162) -> ObservableGateway {
163    let connectivity = Connectivity::ISOLATED.with_pipe(pipe_dir);
164
165    let mut gateway = ObservableGateway::new();
166    for (basis, pauli) in [(Basis::X, Pauli::X), (Basis::Z, Pauli::Z)] {
167        let line = logical_line_operator(distance, basis, boundary_basis, pauli);
168        gateway.insert(
169            LocalStabilizer::new(pauli, connectivity),
170            GatewayEntry {
171                measurements: Vec::new(),
172                operator_in: line.clone(),
173                operator_out: line,
174            },
175        );
176    }
177    gateway
178}

187    fn pipe_dir(is_input: bool) -> Direction {
188        if is_input {
189            Direction::ZPLUS

195    #[rstest]
196    fn test_compile_port_emits_single_mpp_chunk(
197        #[values(Basis::X, Basis::Z)] boundary_basis: Basis,
198        #[values(3, 5, 7)] distance: u32,
199        #[values(true, false)] is_input: bool,
200    ) {
201        let connectivity = Connectivity::ISOLATED.with_pipe(pipe_dir(is_input));
202        let template = compile_port(boundary_basis, connectivity, distance).expect("port compiles");
203
204        // Exactly one stabilizer measurement per patch tile, all from a single MPP.
205        let patch = make_normal_surface_code_patch(distance, boundary_basis);
206        let circuit = &template.program_template.circuit;
207        assert_eq!(
208            circuit.num_measurements() as usize,
209            patch.tiles().len(),
210            "one MPP record per patch stabilizer"
211        );
212        let mpp_count = circuit
213            .body(circuit.entry_body())
214            .expect("entry body exists")
215            .ops()
216            .iter()
217            .filter(|op| matches!(op, Op::MPP { .. }))
218            .count();
219        assert_eq!(mpp_count, 1, "all stabilizers ride one MPP instruction");
220    }

222    #[rstest]
223    fn test_port_gateway_is_passthrough(
224        #[values(Basis::X, Basis::Z)] boundary_basis: Basis,
225        #[values(true, false)] is_input: bool,
226    ) {
227        let connectivity = Connectivity::ISOLATED.with_pipe(pipe_dir(is_input));
228        let template = compile_port(boundary_basis, connectivity, 3).expect("port compiles");
229        let gateway = &template.observable_gateway;
230        assert_eq!(gateway.len(), 2, "one passthrough key per logical basis");
231
232        for pauli in [Pauli::X, Pauli::Z] {
233            let key = LocalStabilizer::new(pauli, connectivity);
234            assert!(
235                gateway.contains_key(&key),
236                "missing {pauli} passthrough key"
237            );
238            let entry = &gateway[&key];
239            assert!(
240                entry.measurements.is_empty(),
241                "gateway carries operators only"
242            );
243            assert_eq!(
244                entry.operator_in, entry.operator_out,
245                "{pauli} face operators must be a passthrough"
246            );
247            assert!(
248                !entry.operator_in.is_empty(),
249                "{pauli} operator is non-empty"
250            );
251        }
252    }

254    #[rstest]
255    fn test_standalone_port_chunk_has_open_boundary_flow(
256        #[values(Basis::X, Basis::Z)] boundary_basis: Basis,
257        #[values(true, false)] is_input: bool,
258    ) {
259        // The single MPP chunk is open on one temporal face (it composes with
260        // the neighbouring cube), so every flow survives as boundary residual:
261        // an input port flows out toward the future cube (`start` empty), an
262        // output port flows in from the past cube (`end` empty).
263        let patch = make_normal_surface_code_patch(3, boundary_basis);
264        let chunk = make_port_mpp_chunk(&patch, is_input).unwrap();
265        let chunks = vec![ChunkOrLoop::Single(Box::new(chunk))];
266        let template = LoweringTemplate::from_chunks(chunks, ObservableGateway::new()).unwrap();
267        let boundary = &template.program_template.boundary_flows;
268        assert_eq!(boundary.len(), patch.tiles().len());
269        for flow in boundary {
270            if is_input {
271                assert!(flow.start.is_empty() && !flow.end.is_empty(), "{flow}");
272            } else {
273                assert!(!flow.start.is_empty() && flow.end.is_empty(), "{flow}");
274            }
275        }
276    }

278    #[test]
279    fn multiplex_port_splits_x_support_onto_its_corrected_output() {
280        let output = ivec2(-8, 0);
281        let template = compile_multiplex_port(Basis::X, 3, output).unwrap();
282        let ops = template
283            .program_template
284            .circuit
285            .body(template.program_template.circuit.entry_body())
286            .unwrap()
287            .ops();
288        let [
289            Op::Gate {
290                gate: GateType::RX, ..
291            },
292            Op::Tick,
293            Op::MPP {
294                products,
295                measurements,
296            },
297            Op::Tick,
298            Op::ConditionalPauli(corrections),
299            Op::Tick,
300            Op::MPP { .. },
301        ] = ops
302        else {
303            panic!("unexpected Multiplex circuit")
304        };
305        assert_eq!(products[0].get(&output), Some(&Pauli::Z));
306        assert!(matches!(
307            corrections.as_slice(),
308            [correction]
309                if correction.pauli == PauliBasis::X
310                    && correction.control == measurements[0]
311                    && correction.target == output
312        ));
313
314        let connectivity = Connectivity::ISOLATED.with_pipe(Direction::ZPLUS);
315        for pauli in [Pauli::X, Pauli::Y] {
316            let entry = template
317                .observable_gateway
318                .lookup(LocalStabilizer::new(pauli, connectivity))
319                .unwrap();
320            assert_eq!(entry.operator_in.get(&output), None);
321            assert_eq!(entry.operator_out.get(&output), Some(&Pauli::X));
322        }
323        assert_eq!(
324            template
325                .observable_gateway
326                .lookup(LocalStabilizer::new(Pauli::Z, connectivity))
327                .unwrap()
328                .operator_out
329                .get(&output),
330            None
331        );
332    }

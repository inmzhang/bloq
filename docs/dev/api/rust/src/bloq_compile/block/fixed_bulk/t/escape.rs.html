1//! Escape template constructor: the lattice-surgery half of a T block.
2//!
3//! Ports the msc-ls ZXZ escape stage (`lattice_surgery_generator_zxz`,
4//! STE:1325–1692, composed by LSCG:349–476): three merge rounds interleaving
5//! the Steane superdense cycles with the surgery-seam stabilizers and the
6//! intermediate patch's surface rounds; destructive color-code readout; the
7//! top-row stabilizer recovery with the `x_ab` history splice; downward code
8//! expansion to `d`; and `d_color = 3` stabilization rounds.
9//!
10//! ## Chunk decomposition
11//!
12//! [`escape_chunks`] exposes five `Single` chunks for flow/distance tests:
13//! `[merge, E1, S1, S2, S3]`. The persisted template compacts the identical
14//! `S1..S3` tail into `REPEAT 3`. Decoder latency is represented by a separate
15//! memory-padding node after the escape; it never rewrites this template.
16//!
17//! The whole 18-tick merge stage is **one chunk**: every Steane superdense
18//! cycle crosses the msc-ls round boundaries (a cycle spans ~10 ticks), so
19//! per-round chunks would need artificial ancilla-entangled interface
20//! stabilizers at each cut. One chunk keeps every post-selected msc-ls parity
21//! a plain chunk-local flow, with only the six cultivation faces entering and
22//! the intermediate-patch stabilizers (plus `X_AB`) leaving.
23//!
24//! `E1` is the recovery/expansion round and `S1..S3` the stabilization
25//! rounds, all plain `make_surface_code_chunk` products over the final patch
26//! (flat `Single`s, not a `Loop`: only 3 rounds, and the distance harness
27//! needs top-level detectors).
28//!
29//! ## Restart parities
30//!
31//! msc-ls emits each post-selected parity as one single-measurement detector,
32//! deterministic given everything upstream is pinned. Chunk-local flows must
33//! be deterministic *within the chunk*, so consecutive same-cycle
34//! measurements pair up (`[m_r1 ⊕ m_r2]` instead of msc's `[m_r2]`); the
35//! restart condition ("any parity odd → restart") depends only on the span of
36//! the parity set, which matches msc's — the equivalence test must
37//! compare spans, not listings.
38//!
39//! ## Fixed-bulk convention
40//!
41//! Surface code stabilizer bases are the base-frame checkerboard colors
42//! carried through the variant transform (equal to
43//! `checkerboard_basis(mapped position)` by the D₄ parity;
44//! pinned by test). The msc-ls construction *logic* (which plaquettes are
45//! weight-2, slot order, seam interleaving) is what gets ported; gate bases
46//! thread the dual helpers, so the X-type-seam family is the H-conjugate of
47//! the base circuit.

95/// Build the escape template's chunks and observable gateway: `[merge, E1,
96/// S1, S2, S3]`, all `Single`.
97fn escape_chunks(
98    layout: &SurgeryLayout,
99) -> Result<(Vec<ChunkOrLoop>, ObservableGateway), CompileError> {
100    let (merge, handles) = merge_chunk(layout)?;
101    let final_patch = make_normal_surface_code_patch(layout.distance, layout.top_basis());
102    let init = expansion_init(layout);
103    let mut e1 = make_surface_code_chunk(&final_patch, (!init.is_empty()).then_some(&init), None)?;
104    discard_orphan_resplit_consumers(layout, &mut e1);
105    let gateway = build_gateway(layout, &handles, &e1, &final_patch);
106
107    // `S1..S3` are the same plain round over the same patch, so it is built
108    // once and cloned (`escape_template` folds the tail into a `REPEAT 3`).
109    let stabilization = make_surface_code_chunk(&final_patch, None, None)?;
110    let mut chunks = vec![
111        ChunkOrLoop::Single(Box::new(merge)),
112        ChunkOrLoop::Single(Box::new(e1)),
113    ];
114    chunks.extend(
115        std::iter::repeat_n(stabilization, 3).map(|round| ChunkOrLoop::Single(Box::new(round))),
116    );
117    Ok((chunks, gateway))
118}

120/// The escape template: standalone composition of [`escape_chunks`]. Its
121/// boundary flows are the six start-open (restart-flagged) Steane chains
122/// toward the cultivation template and the end-open standard-face creators.
123pub(crate) fn escape_template(layout: &SurgeryLayout) -> Result<LoweringTemplate, CompileError> {
124    let (mut chunks, gateway) = escape_chunks(layout)?;
125    let mut stabilization = chunks.split_off(2).into_iter();
126    let ChunkOrLoop::Single(round) = stabilization
127        .next()
128        .expect("escape has a stabilization tail")
129    else {
130        unreachable!("escape stabilization rounds are flat chunks")
131    };
132    debug_assert_eq!(stabilization.count(), 2);
133    chunks.push(ChunkOrLoop::Loop {
134        body: vec![*round],
135        repetitions: 3,
136    });
137    LoweringTemplate::from_chunks(chunks, gateway)
138}

144/// Whether the msc-frame surface cell `(i, j)` (row, column) belongs to the
145/// intermediate patch (msc anchors it at its top-left corner, LSCG:314).
146fn in_intermediate(i: i32, j: i32, d_int: i32) -> bool {
147    i < d_int && j < d_int
148}
149
150/// Downward-expansion data init (LSCG:159–183): new data reset X on and below
151/// the `i >= j` diagonal, Z above it; empty when `d_int == d`. Keys are final
152/// block-frame coordinates, values dual-aware bases.
153fn expansion_init(layout: &SurgeryLayout) -> crate::FxMap<IVec2, Basis> {
154    let d = layout.distance as i32;
155    let d_int = layout.intermediate_distance() as i32;
156    let mut init = crate::FxMap::default();
157    if d_int == d {
158        return init;
159    }
160    for i in 0..d {
161        for j in 0..d {
162            if in_intermediate(i, j, d_int) {
163                continue;
164            }
165            let base = if i >= j { Basis::X } else { Basis::Z };
166            let msc = (SURFACE_OFFSET.0 + 2 * j, SURFACE_OFFSET.1 + 2 * i);
167            init.insert(layout.map_msc(msc), layout.basis(base));
168        }
169    }
170    init
171}

173/// Seam-side X two-weights deleted during the merge (LSCG:402–407), as final
174/// block-frame ancilla positions. The first entry is the `x_ab`-splice tile
175/// (msc `(2, 16)` = `A_AB`); the rest are re-split with no history.
176fn deleted_seam_two_weights(layout: &SurgeryLayout) -> Vec<IVec2> {
177    let d_int = layout.intermediate_distance() as i32;
178    (0..d_int)
179        .filter(|j| j % 2 == 0 && *j < d_int - 1)
180        .map(|j| layout.map_msc((SURFACE_OFFSET.0 + 2 * j + 1, SURFACE_OFFSET.1 - 1)))
181        .collect()
182}
183
184/// Drop E1's consumer flows for the recovered two-weights that re-split with
185/// no measurement history (all deleted seam-side tiles except the
186/// `x_ab`-splice one): their first post-merge outcome is nondeterministic
187/// (msc recovers them with `already_satisfied = False`), and an unmatched
188/// consumer would otherwise pollute the template's boundary residual.
189fn discard_orphan_resplit_consumers(layout: &SurgeryLayout, e1: &mut Chunk) {
190    let orphans = &deleted_seam_two_weights(layout)[1..];
191    e1.flows.retain(|flow| {
192        !(flow.end.is_empty() && flow.center.is_some_and(|center| orphans.contains(&center)))
193    });
194}

200/// One repeatedly-measured surface stabilizer during the merge: the msc-ls
201/// `Surface{X,Z}SyndromeMeasurement` action tables (`surface_code.py:23–249`)
202/// normalized to reset at slot 0 and measure at slot 5 (CX slots keep their
203/// msc positions, so the neighbor-conflict structure is untouched; the
204/// ancilla idles across the shifts).
205struct DriverTile {
206    ancilla: MscQubit,
207    /// Base-frame stabilizer basis (== the checkerboard color after mapping).
208    basis: Basis,
209    /// Data coupling per CX slot 1..=4, msc frame.
210    slots: [Option<MscQubit>; 4],
211    /// Surgery-seam stabilizer: round comparisons post-selected, no creator.
212    seam: bool,
213    /// Measurement key per round.
214    rounds: Vec<MeasKey>,
215}
216
217impl DriverTile {
218    fn new(ancilla: MscQubit, basis: Basis, slots: [Option<MscQubit>; 4]) -> Self {
219        Self {
220            ancilla,
221            basis,
222            slots,
223            seam: false,
224            rounds: Vec::new(),
225        }
226    }
227
228    fn four_weight(ancilla: MscQubit, basis: Basis) -> Self {
229        let (x, y) = ancilla;
230        let (lt, lb, rt, rb) = (
231            (x - 1, y - 1),
232            (x - 1, y + 1),
233            (x + 1, y - 1),
234            (x + 1, y + 1),
235        );
236        let slots = match basis {
237            Basis::Z => [Some(lt), Some(lb), Some(rt), Some(rb)],
238            Basis::X => [Some(lt), Some(rt), Some(lb), Some(rb)],
239        };
240        Self::new(ancilla, basis, slots)
241    }
242
243    /// The two-weight patterns the merge actually runs (the seam-side X
244    /// `TWO_WEIGHT_DOWN`s are deleted; E1's `make_surface_code_chunk`
245    /// re-creates them natively).
246    fn two_weight(ancilla: MscQubit, basis: Basis, pattern: TwoWeight) -> Self {
247        let (x, y) = ancilla;
248        let (lt, lb, rt, rb) = (
249            (x - 1, y - 1),
250            (x - 1, y + 1),
251            (x + 1, y - 1),
252            (x + 1, y + 1),
253        );
254        let slots = match (basis, pattern) {
255            (Basis::Z, TwoWeight::Down) => [None, Some(lb), None, Some(rb)],
256            (Basis::X, TwoWeight::Up) => [Some(lt), Some(rt), None, None],
257            (Basis::Z, TwoWeight::Left) => [Some(lt), Some(lb), None, None],
258            (Basis::Z, TwoWeight::Right) => [None, None, Some(rt), Some(rb)],
259            _ => unreachable!("two-weight pattern unused by the merge stage"),
260        };
261        Self::new(ancilla, basis, slots)
262    }
263
264    /// Emit this tile's action for the given round slot.
265    fn run(&mut self, slot: usize, b: &mut EscapeBuilder) -> Result<(), CompileError> {
266        match slot {
267            0 => b.reset(self.basis, self.ancilla),
268            5 => {
269                let key = b.measure(self.basis, self.ancilla)?;
270                self.rounds.push(key);
271            }
272            _ => {
273                if let Some(data) = self.slots[slot - 1] {
274                    match self.basis {
275                        Basis::Z => b.cx(data, self.ancilla)?,
276                        Basis::X => b.cx(self.ancilla, data)?,
277                    }
278                }
279            }
280        }
281        Ok(())
282    }
283
284    /// The stabilizer's support in final block-frame coordinates.
285    fn support(&self, layout: &SurgeryLayout) -> PauliMap {
286        let pauli = layout.pauli(Pauli::from(self.basis));
287        self.slots
288            .iter()
289            .flatten()
290            .map(|&q| (layout.map_msc(q), pauli))
291            .collect()
292    }
293
294    fn center(&self, layout: &SurgeryLayout) -> IVec2 {
295        layout.map_msc(self.ancilla)
296    }
297}

307/// The stabilizers measured every merge round: the surgery seam (STE:1348–58)
308/// plus the intermediate patch minus its deleted seam-side two-weights
309/// (LSCG:98–127, 402–407).
310fn merge_driver_tiles(d_int: i32) -> Vec<DriverTile> {
311    let (ox, oy) = SURFACE_OFFSET;
312    let mut tiles = Vec::new();
313
314    // Seam: one four-weight (bridging STEANE_5/STEANE_3 to the patch) plus
315    // (d_int − 1)/2 − 1 pure-surface two-weights.
316    let mut seam = vec![DriverTile::four_weight((4, 16), Basis::Z)];
317    for i in 0..((d_int - 1) / 2 - 1) {
318        seam.push(DriverTile::two_weight(
319            (8 + 4 * i, 16),
320            Basis::Z,
321            TwoWeight::Down,
322        ));
323    }
324    for mut tile in seam {
325        tile.seam = true;
326        tiles.push(tile);
327    }
328
329    for i in 0..d_int {
330        for j in 0..d_int {
331            let (x, y) = (ox + 2 * j, oy + 2 * i);
332            // The i == 0 seam-side X two-weights are deleted during the merge.
333            if i == d_int - 1 && j % 2 == 1 {
334                tiles.push(DriverTile::two_weight(
335                    (x + 1, y + 1),
336                    Basis::X,
337                    TwoWeight::Up,
338                ));
339            }
340            if j == 0 && i % 2 == 1 {
341                tiles.push(DriverTile::two_weight(
342                    (x - 1, y + 1),
343                    Basis::Z,
344                    TwoWeight::Right,
345                ));
346            }
347            if j == d_int - 1 && i % 2 == 0 && i < d_int - 1 {
348                tiles.push(DriverTile::two_weight(
349                    (x + 1, y + 1),
350                    Basis::Z,
351                    TwoWeight::Left,
352                ));
353            }
354            if i < d_int - 1 && j < d_int - 1 {
355                let basis = if (i + j) % 2 == 0 { Basis::Z } else { Basis::X };
356                tiles.push(DriverTile::four_weight((x + 1, y + 1), basis));
357            }
358        }
359    }
360    tiles
361}

367/// Handle for a pooled measurement's record id, resolved via the table
368/// [`EscapeBuilder::finish`] returns (ids are only allocated when the
369/// measurement's moment is emitted).
370#[derive(Clone, Copy, Default)]
371struct MeasKey(usize);
372
373/// Transliterates the msc-ls tick stream, coalescing resets and measurements
374/// into shared moments by construction: CX batches emit per msc tick (the
375/// moment skeleton), while resets and measurements pool across ticks and only
376/// flush — each pool as one kind-homogeneous moment — when a later op touches
377/// a pooled qubit. Flushing the whole pool at the first forced point is the
378/// greedy interval-stabbing optimum, so the merge stage keeps a handful of
379/// reset/measurement moments instead of one per msc reset/readout tick.
380///
381/// All ops within one msc tick act on disjoint qubits, so any serialization
382/// is circuit-equivalent; bloq tick indices and measurement record order are
383/// therefore not 1:1 with msc-ls's (flows resolve ids through [`MeasKey`]).
384struct EscapeBuilder<'a> {
385    layout: &'a SurgeryLayout,
386    circuit: CoordCircuit,
387    /// Current msc tick's CX operands.
388    cx: Vec<IVec2>,
389    /// Resets awaiting their forced moment, in call order.
390    resets: Vec<(GateType, IVec2)>,
391    /// Measurements awaiting their forced moment, in call order.
392    measures: Vec<(PauliBasis, IVec2, MeasKey)>,
393    /// Record id per [`MeasKey`], filled at flush time.
394    ids: Vec<u32>,
395    /// Whether any moment was emitted yet (tick separators are lazy).
396    dirty: bool,
397}
398
399impl<'a> EscapeBuilder<'a> {
400    fn new(layout: &'a SurgeryLayout) -> Self {
401        Self {
402            layout,
403            circuit: CoordCircuit::new(),
404            cx: Vec::new(),
405            resets: Vec::new(),
406            measures: Vec::new(),
407            ids: Vec::new(),
408            dirty: false,
409        }
410    }
411
412    fn has_pending_reset(&self, qubit: IVec2) -> bool {
413        self.resets.iter().any(|&(_, q)| q == qubit)
414    }
415
416    fn has_pending_measure(&self, qubit: IVec2) -> bool {
417        self.measures.iter().any(|&(_, q, _)| q == qubit)
418    }
419
420    /// Pool a dual-aware reset of the given base-frame basis. A pending
421    /// measurement of the qubit flushes first (measure-before-reset order);
422    /// these flush rules keep the two pools qubit-disjoint, so flushing a
423    /// whole pool never reorders same-qubit ops.
424    fn reset(&mut self, base: Basis, qubit: MscQubit) {
425        let qubit = self.layout.map_msc(qubit);
426        if self.has_pending_measure(qubit) {
427            self.flush_measures();
428        }
429        self.resets.push((self.layout.reset_gate(base), qubit));
430    }
431
432    /// Queue a dual-aware CX (base-frame control/target order), first
433    /// flushing any pool holding one of its operands — those moments must
434    /// precede this tick's CX moment.
435    fn cx(&mut self, control: MscQubit, target: MscQubit) -> Result<(), CompileError> {
436        let pair = self
437            .layout
438            .cx_operands(self.layout.map_msc(control), self.layout.map_msc(target));
439        for qubit in pair {
440            if self.has_pending_measure(qubit) {
441                self.flush_measures();
442            }
443            if self.has_pending_reset(qubit) {
444                self.flush_resets()?;
445            }
446        }
447        self.cx.extend(pair);
448        Ok(())
449    }
450
451    /// Pool a dual-aware measurement of the given base-frame basis; the
452    /// returned key resolves after [`Self::finish`]. A pending reset of the
453    /// qubit flushes first (reset-before-measure order).
454    fn measure(&mut self, base: Basis, qubit: MscQubit) -> Result<MeasKey, CompileError> {
455        let qubit = self.layout.map_msc(qubit);
456        if self.has_pending_reset(qubit) {
457            self.flush_resets()?;
458        }
459        if self.has_pending_measure(qubit) {
460            self.flush_measures();
461        }
462        let key = MeasKey(self.ids.len());
463        self.ids.push(u32::MAX);
464        self.measures
465            .push((self.layout.basis(base).into(), qubit, key));
466        Ok(key)
467    }
468
469    /// Start a fresh moment.
470    fn next_moment(&mut self) {
471        if self.dirty {
472            self.circuit.tick();
473        }
474        self.dirty = true;
475    }
476
477    /// Emit every pooled reset as one moment (grouped per gate type — all
478    /// reset flavors are one moment kind).
479    fn flush_resets(&mut self) -> Result<(), CompileError> {
480        if self.resets.is_empty() {
481            return Ok(());
482        }
483        self.next_moment();
484        for (gate, group) in group_by_key(std::mem::take(&mut self.resets)) {
485            self.circuit.do_gate(gate, group)?;
486        }
487        Ok(())
488    }
489
490    /// Emit every pooled measurement as one moment, resolving their keys.
491    fn flush_measures(&mut self) {
492        if self.measures.is_empty() {
493            return;
494        }
495        self.next_moment();
496        let pooled = std::mem::take(&mut self.measures)
497            .into_iter()
498            .map(|(basis, qubit, key)| (basis, (qubit, key)))
499            .collect();
500        for (basis, batch) in group_by_key(pooled) {
501            let (qubits, keys): (Vec<_>, Vec<_>) = batch.into_iter().unzip();
502            for (key, id) in keys.into_iter().zip(self.circuit.measure(basis, qubits)) {
503                self.ids[key.0] = id;
504            }
505        }
506    }
507
508    /// End the current msc tick: emit its CX moment. Pools stay pooled — they
509    /// flush when a later op forces them, coalescing across msc ticks.
510    fn tick(&mut self) -> Result<(), CompileError> {
511        if self.cx.is_empty() {
512            return Ok(());
513        }
514        self.next_moment();
515        Ok(self
516            .circuit
517            .do_gate(GateType::CX, std::mem::take(&mut self.cx))?)
518    }
519
520    /// Flush the remaining pools and return the circuit plus the
521    /// key → record id table.
522    fn finish(mut self) -> Result<(CoordCircuit, Vec<u32>), CompileError> {
523        self.flush_measures();
524        self.flush_resets()?;
525        Ok((self.circuit, self.ids))
526    }
527}

529/// Bucket a pooled batch by key, preserving first-seen key order.
530///
531/// A pool flushes as one moment, but the emitters take one gate type (or one
532/// basis) per call, so the moment is issued as a handful of same-key batches.
533fn group_by_key<K: PartialEq, V>(items: Vec<(K, V)>) -> Vec<(K, Vec<V>)> {
534    let mut groups: Vec<(K, Vec<V>)> = Vec::new();
535    for (key, value) in items {
536        match groups.iter_mut().find(|(existing, _)| *existing == key) {
537            Some((_, values)) => values.push(value),
538            None => groups.push((key, vec![value])),
539        }
540    }
541    groups
542}

557/// Build the single merge-stage chunk: 18 msc ticks interleaving the ZXZ
558/// Steane segments (STE:1370–1692), the seam drivers, and the intermediate
559/// patch's surface rounds (LSCG:395–421, `SURFACE_SYNDROME_EXTRACTION_OFFSET
560/// = 0`).
561fn merge_chunk(layout: &SurgeryLayout) -> Result<(Chunk, MergeHandles), CompileError> {
562    let d_int = layout.intermediate_distance() as i32;
563    let mut tiles = merge_driver_tiles(d_int);
564    let mut b = EscapeBuilder::new(layout);
565
566    // Steane-side measurement keys, named after the msc-ls sources.
567    let mut a1a = [MeasKey::default(); 3];
568    let mut ax0145 = [MeasKey::default(); 2];
569    let mut az0145 = [MeasKey::default(); 2];
570    let mut ax0235 = [MeasKey::default(); 2];
571    let mut az0235 = [MeasKey::default(); 2];
572    let mut ax0246 = [MeasKey::default(); 2];
573    let mut az0246 = [MeasKey::default(); 2];
574
575    let run_tiles = |tiles: &mut Vec<DriverTile>,
576                     slot: usize,
577                     b: &mut EscapeBuilder|
578     -> Result<(), CompileError> {
579        for tile in tiles.iter_mut() {
580            tile.run(slot, b)?;
581        }
582        Ok(())
583    };

585    // The merge runs as three uniform 7-moment rounds — [R][CX×5][M] — the
586    // hand-optimized repacking of the msc-ls 18-tick stream (each call keeps
587    // its "msc t = n" provenance). Driver tiles occupy CX ticks 1–4 of each
588    // round; the 5th CX tick is pure Steane. Resets and measurements pool in
589    // the builder and flush as the round's single reset/measurement moment.
590
591    // ---- Round 1: superdense cycle 1 (0145 completes), qubit 2 out, first
592    // seam readout. -------------------------------------------------------
593    b.reset(Basis::X, A_0145_4);
594    b.reset(Basis::Z, A_0145_015);
595    b.reset(Basis::X, A_0235_035);
596    b.reset(Basis::Z, A_0235_2);
597    b.reset(Basis::X, A_0246_46);
598    b.reset(Basis::Z, A_0246_02);
599    b.reset(Basis::Z, STEANE_2_);
600    b.reset(Basis::Z, A_1A_R); // msc t = 1
601    for i in 0..d_int {
602        for j in 0..d_int {
603            b.reset(
604                Basis::X,
605                (SURFACE_OFFSET.0 + 2 * j, SURFACE_OFFSET.1 + 2 * i),
606            );
607        }
608    }
609    run_tiles(&mut tiles, 0, &mut b)?;
610    b.tick()?;
611
612    // CX 1 — entangle ancilla pairs; move qubit 2 out (msc t = 1).
613    b.cx(A_0145_4, A_0145_015)?;
614    b.cx(A_0235_035, A_0235_2)?;
615    b.cx(A_0246_46, A_0246_02)?;
616    b.cx(STEANE[2], STEANE_2_)?;
617    run_tiles(&mut tiles, 1, &mut b)?;
618    b.tick()?;
619
620    // CX 2 — CX(1) (msc t = 2).
621    b.cx(STEANE[5], A_0145_015)?;
622    b.cx(STEANE[0], A_0235_035)?;
623    b.cx(STEANE[2], A_0235_2)?;
624    b.cx(STEANE[6], A_0246_46)?;
625    b.cx(STEANE[1], A_1A_R)?;
626    run_tiles(&mut tiles, 2, &mut b)?;
627    b.tick()?;
628
629    // CX 3 — CX(2); qubit 2 lands at STEANE_2_ (msc t = 3).
630    b.cx(STEANE[1], A_0145_015)?;
631    b.cx(STEANE[5], A_0235_035)?;
632    b.cx(STEANE[4], A_0246_46)?;
633    b.cx(STEANE[0], A_0246_02)?;
634    b.cx(STEANE_2_, STEANE[2])?;
635    b.cx(SURFACE_A, A_1A_R)?;
636    run_tiles(&mut tiles, 3, &mut b)?;
637    b.tick()?;
638
639    // CX 4 — CX(3) (msc t = 4). Qubit 2's vacated home is deliberately *not*
640    // re-initialized, the same shuttle deviation the cultivation syndrome round
641    // takes (`steane::syndrome_round_chunk`): the round-2 move-back is then not
642    // a clean move and its `MX(5,11)` byproduct is no longer an independent
643    // post-selection — it folds into the chains that carry X on qubit 2.
644    b.cx(STEANE[4], A_0145_4)?;
645    b.cx(STEANE[0], A_0145_015)?;
646    b.cx(STEANE[3], A_0235_035)?;
647    b.cx(STEANE_2_, A_0246_02)?;
648    run_tiles(&mut tiles, 4, &mut b)?;
649    b.tick()?;
650
651    // CX 5 — disentangle 0145; CX(4) (msc t = 5).
652    b.cx(A_0145_4, A_0145_015)?;
653    b.cx(A_0235_035, STEANE[5])?;
654    b.cx(A_0246_02, STEANE_2_)?;
655    b.tick()?;
656
657    // Round-1 measurements — first seam readout (msc t = 4), Z0145 cycle-1
658    // readout (msc t = 6), surface round 1.
659    a1a[0] = b.measure(Basis::Z, A_1A_R)?;
660    ax0145[0] = b.measure(Basis::X, A_0145_4)?;
661    az0145[0] = b.measure(Basis::Z, A_0145_015)?;
662    run_tiles(&mut tiles, 5, &mut b)?;
663
664    // ---- Round 2: whole 0145 cycle 2, CX(5)/CX(6) of 0235/0246, qubit 2
665    // back home, remaining seam readouts, destructive m1. ------------------
666    b.reset(Basis::X, A_0145_4); // msc t = 7
667    b.reset(Basis::Z, A_0145_015); // msc t = 7
668    b.reset(Basis::Z, A_1A_L); // msc t = 7
669    b.reset(Basis::Z, A_1A_R); // msc t = 9
670    run_tiles(&mut tiles, 0, &mut b)?;
671    b.tick()?;
672
673    // CX 1 — CX(5) heads (msc t = 6); qubit 2 move-back starts (msc t = 6);
674    // re-entangle 0145 (msc t = 8); left seam entangles (msc t = 8).
675    b.cx(A_0235_035, STEANE[0])?;
676    b.cx(A_0246_46, STEANE[6])?;
677    b.cx(STEANE[2], STEANE_2_)?;
678    b.cx(A_0145_4, A_0145_015)?;
679    b.cx(STEANE[1], A_1A_L)?;
680    run_tiles(&mut tiles, 1, &mut b)?;
681    b.tick()?;
682
683    // CX 2 — CX(6) (msc t = 7); left seam couples the patch (msc t = 9);
684    // 0145 cycle-2 CX(1) head (msc t = 9).
685    b.cx(A_0235_035, STEANE[3])?;
686    b.cx(A_0235_2, STEANE[2])?;
687    b.cx(A_0246_46, STEANE[4])?;
688    b.cx(A_0246_02, STEANE[0])?;
689    b.cx(SURFACE_A, A_1A_L)?;
690    b.cx(STEANE[1], A_0145_015)?;
691    run_tiles(&mut tiles, 2, &mut b)?;
692    b.tick()?;
693
694    // CX 3 — disentangle 0235/0246, qubit 2 home (msc t = 8); 0145 cycle-2
695    // CX(1)/CX(2) tails (msc t = 9/10); right seam re-entangles (msc t = 10).
696    b.cx(A_0235_035, A_0235_2)?;
697    b.cx(A_0246_46, A_0246_02)?;
698    b.cx(STEANE_2_, STEANE[2])?;
699    b.cx(STEANE[4], A_0145_4)?;
700    b.cx(STEANE[0], A_0145_015)?;
701    b.cx(STEANE[1], A_1A_R)?;
702    run_tiles(&mut tiles, 3, &mut b)?;
703    b.tick()?;
704
705    // CX 4 — 0145 cycle-2 CX(3) (msc t = 11); right seam couples the patch
706    // (msc t = 11).
707    b.cx(STEANE[5], A_0145_015)?;
708    b.cx(SURFACE_A, A_1A_R)?;
709    run_tiles(&mut tiles, 4, &mut b)?;
710    b.tick()?;
711
712    // CX 5 — disentangle 0145 cycle 2 (msc t = 12).
713    b.cx(A_0145_4, A_0145_015)?;
714    b.tick()?;
715
716    // Round-2 measurements — 0235/0246 cycle-1 readouts + the move byproduct
717    // (msc t = 9), both remaining seam readouts (msc t = 10/12), 0145
718    // cycle-2 readouts (msc t = 13), destructive m1 (msc t = 11), surface
719    // round 2.
720    ax0235[0] = b.measure(Basis::X, A_0235_035)?;
721    az0235[0] = b.measure(Basis::Z, A_0235_2)?;
722    ax0246[0] = b.measure(Basis::X, A_0246_46)?;
723    az0246[0] = b.measure(Basis::Z, A_0246_02)?;
724    let steane2_move = b.measure(Basis::X, STEANE_2_)?;
725    a1a[1] = b.measure(Basis::Z, A_1A_L)?;
726    a1a[2] = b.measure(Basis::Z, A_1A_R)?;
727    ax0145[1] = b.measure(Basis::X, A_0145_4)?;
728    az0145[1] = b.measure(Basis::Z, A_0145_015)?;
729    let m1 = b.measure(Basis::X, STEANE[1])?;
730    run_tiles(&mut tiles, 5, &mut b)?;
731
732    // ---- Round 3: 0235/0246 cycle 2 (qubit 2 shuttles out again), the
733    // destructive readout, the x_ab splice, surface round 3. ---------------
734    b.reset(Basis::X, A_0235_035); // msc t = 10
735    b.reset(Basis::Z, A_0235_2); // msc t = 10
736    b.reset(Basis::X, A_0246_46); // msc t = 10
737    b.reset(Basis::Z, A_0246_02); // msc t = 10
738    b.reset(Basis::Z, STEANE_2_); // msc t = 10
739    b.reset(Basis::X, A_AB); // msc t = 14
740    run_tiles(&mut tiles, 0, &mut b)?;
741    b.tick()?;
742
743    // CX 1 — entangle 0235/0246 (msc t = 11); qubit 2 out (msc t = 11).
744    b.cx(A_0235_035, A_0235_2)?;
745    b.cx(A_0246_46, A_0246_02)?;
746    b.cx(STEANE[2], STEANE_2_)?;
747    run_tiles(&mut tiles, 1, &mut b)?;
748    b.tick()?;
749
750    // CX 2 — cycle-2 CX(1) (msc t = 12).
751    b.cx(STEANE[6], A_0246_46)?;
752    b.cx(STEANE[0], A_0246_02)?;
753    b.cx(STEANE[5], A_0235_035)?;
754    run_tiles(&mut tiles, 2, &mut b)?;
755    b.tick()?;
756
757    // CX 3 — cycle-2 CX(2) (msc t = 13); A_AB couples SURFACE_A (msc t = 15).
758    b.cx(A_AB, SURFACE_A)?;
759    b.cx(STEANE[4], A_0246_46)?;
760    b.cx(STEANE_2_, A_0246_02)?;
761    b.cx(STEANE[0], A_0235_035)?;
762    b.cx(STEANE[2], A_0235_2)?;
763    run_tiles(&mut tiles, 3, &mut b)?;
764    b.tick()?;
765
766    // CX 4 — cycle-2 CX(3) (msc t = 14); disentangle 0246 (msc t = 15); A_AB
767    // couples SURFACE_B (msc t = 16).
768    b.cx(A_AB, SURFACE_B)?;
769    b.cx(STEANE[3], A_0235_035)?;
770    b.cx(A_0246_46, A_0246_02)?;
771    run_tiles(&mut tiles, 4, &mut b)?;
772    b.tick()?;
773
774    // CX 5 — disentangle 0235 (msc t = 15).
775    b.cx(A_0235_035, A_0235_2)?;
776    b.tick()?;
777
778    // Round-3 measurements — cycle-2 readouts (msc t = 16), the destructive
779    // readout (msc t = 13–16), x_ab (msc t = 17), surface round 3.
780    az0246[1] = b.measure(Basis::Z, A_0246_02)?;
781    az0235[1] = b.measure(Basis::Z, A_0235_2)?;
782    let m0 = b.measure(Basis::X, STEANE[0])?;
783    let m2 = b.measure(Basis::X, STEANE[2])?;
784    let m2_ = b.measure(Basis::X, STEANE_2_)?;
785    let m4 = b.measure(Basis::X, STEANE[4])?;
786    let m5 = b.measure(Basis::X, STEANE[5])?;
787    let m6 = b.measure(Basis::X, STEANE[6])?;
788    let m3 = b.measure(Basis::X, STEANE[3])?;
789    ax0246[1] = b.measure(Basis::X, A_0246_46)?;
790    ax0235[1] = b.measure(Basis::X, A_0235_035)?;
791    let m_ab = b.measure(Basis::X, A_AB)?;
792    run_tiles(&mut tiles, 5, &mut b)?;
793
794    // Pooled measurements only receive record ids when their moment flushes:
795    // resolve every key now that the circuit is complete.
796    let (circuit, ids) = b.finish()?;
797    let rec = |key: MeasKey| ids[key.0];
798    let a1a = a1a.map(rec);
799    let ax0145 = ax0145.map(rec);
800    let az0145 = az0145.map(rec);
801    let ax0235 = ax0235.map(rec);
802    let az0235 = az0235.map(rec);
803    let ax0246 = ax0246.map(rec);
804    let az0246 = az0246.map(rec);
805    let [steane2_move, m0, m1, m2, m2_, m3, m4, m5, m6, m_ab] =
806        [steane2_move, m0, m1, m2, m2_, m3, m4, m5, m6, m_ab].map(rec);
807
808    // ---- Flows ----------------------------------------------------------
809    let face = |face: usize, pauli: Pauli| -> PauliMap {
810        FACES[face]
811            .iter()
812            .map(|&q| (layout.map_msc(STEANE[q]), layout.pauli(pauli)))
813            .collect()
814    };
815    let restart = |start: PauliMap, recs: &[u32]| {
816        Flow::new(start, PauliMap::empty())
817            .with_measurements(recs.iter().copied())
818            .with_marker(FlowMarker::Restart)
819    };
820    let local_restart = |recs: &[u32]| restart(PauliMap::empty(), recs);
821
822    // Steane-side chains and parities. Every closure of a cultivation chain
823    // and every superdense-cycle comparison is post-selected (design §3.2).
824    // Rec sets were derived against the ported circuit with stim
825    // (`flow_generators`, unsigned) and are pinned by `verify_flows`. The
826    // non-obvious sets:
827    // - the round-2 Z0145 comparison picks up the Z the 0246 ancilla dumps
828    //   onto STEANE_0 at t=7 (`CX(A_0246_02, STEANE_0)`), absorbed by its
829    //   readout;
830    // - cycle-2 of 0235/0246 re-extracts Z only; its Z readout is
831    //   deterministic through the pair entanglement (0235 alone; 0246 with
832    //   the Z the 0235 ancilla dumps onto qubit 2 at t=7, absorbed by
833    //   `az0235[0]`);
834    // - the terminal reconstructions (STE:1682–87) splice against the
835    //   cycle-1 X readouts so each parity closes within the chunk;
836    // - the seam-anticommuting X0145 face survives as X0145·X_A·X_B (the
837    //   seam dumps cancel pairwise) until the destructive readout closes it
838    //   as `[m0, m1, m4, m5, m_ab]` (STE:1686) — the §4 cross-template
839    //   restart carrying the cultivation flag;
840    // - both hand-rolled `Z_1A` round comparisons are post-selected
841    //   (STE:1557, 1601);
842    // - with qubit 2's old home never re-initialized (the deliberate deviation
843    //   above), the move byproduct `steane2_move` is not deterministic alone:
844    //   it folds into the X0235 consumer and the 0246 destructive
845    //   reconstruction — msc-ls's standalone `MX (5,11)` post-selection is one
846    //   fewer independent constraint here.
847    let mut flows = vec![
848        restart(face(0, Pauli::Z), &[az0145[0]]),
849        local_restart(&[ax0145[0]]),
850        local_restart(&[az0145[0], az0145[1], az0246[0]]),
851        local_restart(&[ax0145[1]]),
852        restart(face(1, Pauli::X), &[ax0235[0], steane2_move]),
853        restart(face(1, Pauli::Z), &[az0235[0]]),
854        restart(face(2, Pauli::X), &[ax0246[0]]),
855        restart(face(2, Pauli::Z), &[az0246[0]]),
856        local_restart(&[ax0235[1]]),
857        local_restart(&[az0235[1]]),
858        local_restart(&[ax0246[1]]),
859        local_restart(&[az0246[0], az0246[1], az0235[0]]),
860        local_restart(&[ax0235[0], m0, m2, m2_, m3, m5]),
861        local_restart(&[ax0246[0], steane2_move, m0, m2, m2_, m4, m6]),
862        restart(face(0, Pauli::X), &[m0, m1, m4, m5, m_ab]),
863        local_restart(&[a1a[0], a1a[1]]),
864        local_restart(&[a1a[1], a1a[2]]),
865    ];
866
867    // Surface stabilizers: first-round determinism for X tiles (|+⟩ init),
868    // round comparisons, and creators toward E1 are decoder detectors. Seam
869    // tiles compare post-selected and never re-create (the seam splits at
870    // readout).
871    for tile in &tiles {
872        let rounds: Vec<u32> = tile.rounds.iter().map(|&key| rec(key)).collect();
873        debug_assert_eq!(rounds.len(), 3);
874        let marker = |post_selected: bool| {
875            if post_selected {
876                FlowMarker::Restart
877            } else {
878                FlowMarker::Detector
879            }
880        };
881        let center = tile.center(layout);
882        let comparison = |a: u32, b: u32, post_selected: bool| {
883            Flow::new(PauliMap::empty(), PauliMap::empty())
884                .with_measurements([a, b])
885                .with_center(center)
886                .with_marker(marker(post_selected))
887        };
888        if tile.seam {
889            flows.push(comparison(rounds[0], rounds[1], true));
890            flows.push(comparison(rounds[1], rounds[2], true));
891            continue;
892        }
893        if tile.basis == Basis::X {
894            flows.push(
895                Flow::new(PauliMap::empty(), PauliMap::empty())
896                    .with_measurements([rounds[0]])
897                    .with_center(center)
898                    .with_marker(FlowMarker::Detector),
899            );
900        }
901        flows.push(comparison(rounds[0], rounds[1], false));
902        flows.push(comparison(rounds[1], rounds[2], false));
903        flows.push(
904            Flow::new(PauliMap::empty(), tile.support(layout))
905                .with_measurements([rounds[2]])
906                .with_center(center),
907        );
908    }
909
910    // The recovered `x_ab`-splice stabilizer: `MX(A_AB)` measures X_A·X_B
911    // (A_AB leaves reset |+⟩ and coupled both), so the tile E1 re-creates at
912    // the same ancilla starts with history (LSCG:438–439).
913    let x_ab_support: PauliMap = [SURFACE_A, SURFACE_B]
914        .into_iter()
915        .map(|q| (layout.map_msc(q), layout.pauli(Pauli::X)))
916        .collect();
917    flows.push(
918        Flow::new(PauliMap::empty(), x_ab_support)
919            .with_measurements([m_ab])
920            .with_center(layout.map_msc(A_AB)),
921    );
922
923    let seam_first_round = std::iter::once(a1a[0])
924        .chain(tiles.iter().filter(|t| t.seam).map(|t| rec(t.rounds[0])))
925        .collect();
926
927    Ok((
928        Chunk { circuit, flows },
929        MergeHandles {
930            seam_first_round,
931            logical_readout: [m1, m4, m6],
932        },
933    ))
934}

940/// Build the observable gateway. A T block is non-Clifford: it
941/// *terminates* the stabilizer flow, so its logical correlation surface cannot
942/// pass through — it must be **anchored** by an endpoint operator to be defined
943/// deterministically. Each entry encodes one such surface:
944///
945/// - `operator_in` — where the surface **starts**: the Steane logical edge
946///   string the cultivation produces (`Z_L` on the seam-line qubits
947///   `{1, 3, 5}`, `X_L` on the west edge `{1, 4, 6}` ⊥ the seam). Not a −Z
948///   *boundary* operator (nothing enters a T block from below); it is the
949///   magic-state source endpoint, closed internally by the cultivation seam
950///   when the templates compose.
951/// - `measurements` — how the surface is **transported**: the first-round seam
952///   records (`Z_L`) / destructive readout `m1, m4, m6` (`X_L`), then the
953///   E1-round stabilizers of the strip that shifts it from the seam-adjacent
954///   edge to the block midline.
955/// - `operator_out` — the surface's **final form** at the +Z face midline: the
956///   contract a downstream block connects to.
957///
958/// Movers come uniformly from E1 (deviating from the design's "merge round 1
959/// when d_int == d" note): the mover strip includes the seam-side two-weights,
960/// which are deleted during the whole merge, so round 1 cannot supply them —
961/// E1, which recovers every final-patch tile, always can.
962fn build_gateway(
963    layout: &SurgeryLayout,
964    handles: &MergeHandles,
965    e1: &Chunk,
966    final_patch: &Patch,
967) -> ObservableGateway {
968    let d = layout.distance as i32;
969    // E1 tile measurement ids by ancilla (every tile's creator carries its
970    // own single measurement).
971    let e1_ids: crate::FxMap<IVec2, u32> = e1
972        .flows
973        .iter()
974        .filter(|flow| flow.start.is_empty() && !flow.end.is_empty())
975        .filter_map(|flow| Some((flow.center?, flow.measurements[0])))
976        .collect();
977
978    // Base-frame final patch (top basis X) picks the mover strips; the map
979    // transports them per variant. It *is* the escape's final patch unless the
980    // variant flipped the top basis.
981    let x_patch = (layout.top_basis() != Basis::X)
982        .then(|| make_normal_surface_code_patch(layout.distance, Basis::X));
983    let base_patch = x_patch.as_ref().unwrap_or(final_patch);
984    let movers = |strip: &dyn Fn(IVec2) -> bool, basis: Basis| -> Vec<u32> {
985        base_patch
986            .tiles()
987            .iter()
988            .filter(|tile| tile.basis() == basis && strip(tile.measure_qubit()))
989            .map(|tile| {
990                *e1_ids
991                    .get(&layout.map(tile.measure_qubit()))
992                    .expect("every base-patch tile exists in the E1 round")
993            })
994            .collect()
995    };
996    // `Z_L` is a data row, `X_L` a data column; both are logical for any odd
997    // index in `1..=2d − 1`. Landing them on row/column `line` costs the
998    // plaquettes between `line` and where the merge left the surface: the
999    // seam-adjacent row `y = 1` for `Z_L`, the column `x = 2d − 1` opposite it
1000    // for `X_L` (the mirrored embedding puts msc's west column at +x).
1001    let z_rep = |line: i32| {
1002        (
1003            (0..d)
1004                .map(|i| {
1005                    (
1006                        layout.map(IVec2::new(2 * i + 1, line)),
1007                        layout.pauli(Pauli::Z),
1008                    )
1009                })
1010                .collect::<PauliMap>(),
1011            movers(&|m: IVec2| (2..line).contains(&m.y), Basis::Z),
1012        )
1013    };
1014    let x_rep = |line: i32| {
1015        (
1016            (0..d)
1017                .map(|i| {
1018                    (
1019                        layout.map(IVec2::new(line, 2 * i + 1)),
1020                        layout.pauli(Pauli::X),
1021                    )
1022                })
1023                .collect::<PauliMap>(),
1024            movers(&|m: IVec2| (line + 1..2 * d - 1).contains(&m.x), Basis::X),
1025        )
1026    };
1027    let (midline_row, z_movers) = z_rep(d);
1028    let (midline_col, x_movers) = x_rep(d);
1029
1030    // The correlation-surface start endpoints: the Steane logical edge strings
1031    // (`Z_L` on the seam line {1,3,5}, `X_L` on the west edge {1,4,6}).
1032    let steane_edge = |qubits: &[usize], pauli: Pauli| -> PauliMap {
1033        qubits
1034            .iter()
1035            .map(|&q| (layout.map_msc(STEANE[q]), layout.pauli(pauli)))
1036            .collect()
1037    };
1038
1039    let connectivity = Connectivity::ISOLATED.with_pipe(Direction::ZPLUS);
1040    let entry = |merge_recs: &[u32],
1041                 mover_recs: Vec<u32>,

1512    /// Row-reduced GF(2) basis of a set of parities over (qubit, ordinal)
1513    /// columns — two parity sets define the same restart predicate iff their
1514    /// bases agree.
1515    fn gf2_row_basis(parities: &[Vec<(IVec2, u32)>]) -> Vec<Vec<(IVec2, u32)>> {
1516        let mut columns: Vec<(IVec2, u32)> = parities.iter().flatten().copied().collect();
1517        columns.sort_unstable_by_key(|&(qubit, ordinal)| (qubit.x, qubit.y, ordinal));
1518        columns.dedup();
1519        let index_of: crate::FxMap<(IVec2, u32), usize> = columns
1520            .iter()
1521            .enumerate()
1522            .map(|(index, &key)| (key, index))
1523            .collect();
1524
1525        let mut rows: Vec<Vec<u64>> = parities
1526            .iter()
1527            .map(|parity| {
1528                let mut row = vec![0u64; columns.len().div_ceil(64)];
1529                for key in parity {
1530                    // XOR, not OR: a parity listing a measurement twice
1531                    // cancels it.
1532                    row[index_of[key] / 64] ^= 1 << (index_of[key] % 64);
1533                }
1534                row
1535            })
1536            .collect();
1537
1538        // Gaussian elimination to reduced row echelon form.
1539        let mut pivot_rows: Vec<Vec<u64>> = Vec::new();
1540        for col in 0..columns.len() {
1541            let (word, bit) = (col / 64, 1u64 << (col % 64));
1542            let Some(pivot) = rows.iter().position(|row| row[word] & bit != 0) else {
1543                continue;
1544            };
1545            let pivot_row = rows.swap_remove(pivot);
1546            for row in &mut rows {
1547                if row[word] & bit != 0 {
1548                    for (lhs, rhs) in row.iter_mut().zip(&pivot_row) {
1549                        *lhs ^= rhs;
1550                    }
1551                }
1552            }
1553            for row in &mut pivot_rows {
1554                if row[word] & bit != 0 {
1555                    for (lhs, rhs) in row.iter_mut().zip(&pivot_row) {
1556                        *lhs ^= rhs;
1557                    }
1558                }
1559            }
1560            pivot_rows.push(pivot_row);
1561        }
1562
1563        pivot_rows
1564            .iter()
1565            .map(|row| {
1566                columns
1567                    .iter()
1568                    .enumerate()
1569                    .filter(|(col, _)| row[col / 64] & (1 << (col % 64)) != 0)
1570                    .map(|(_, &key)| key)
1571                    .collect()
1572            })
1573            .collect()
1574    }

1811    /// Perfect Steane initialization (msc's `perform_perfect_steane_*`): one
1812    /// MPP moment measuring the six face stabilizers plus the gateway's
1813    /// `operator_in` — the surface's start endpoint, materialized as an MPP
1814    /// boundary so the observable is deterministic. Stabilizer records anchor
1815    /// creators that fuse into the merge chunk's restart closures; the
1816    /// `operator_in` record feeds the observable directly.
1817    fn steane_port_chunk(layout: &SurgeryLayout, operator_in: &PauliMap) -> (Chunk, u32) {
1818        let steane_map = |qubits: &[usize], p: Pauli| -> PauliMap {
1819            qubits
1820                .iter()
1821                .map(|&q| (layout.map_msc(STEANE[q]), layout.pauli(p)))
1822                .collect()
1823        };
1824        let mut products: Vec<PauliMap> = FACES
1825            .iter()
1826            .flat_map(|face| [Pauli::X, Pauli::Z].map(|p| steane_map(face, p)))
1827            .collect();
1828        products.push(operator_in.clone());
1829
1830        let mut circuit = CoordCircuit::new();
1831        let recs = circuit
1832            .measure_pauli_products(products.iter().cloned())
1833            .unwrap();
1834        let flows = products[..6]
1835            .iter()
1836            .zip(&recs)
1837            .map(|(stabilizer, &m)| {
1838                Flow::new(PauliMap::empty(), stabilizer.clone()).with_measurements([m])
1839            })
1840            .collect();
1841        (Chunk { circuit, flows }, recs[6])
1842    }

1845    /// the last stabilization round's creators) plus the logical line.
1846    fn cap_chunk(patch: &Patch, logical: &PauliMap) -> (Chunk, u32) {

1848            .tiles()
1849            .iter()
1850            .map(crate::block::patch::Tile::pauli_map)

1871    /// with every detector *and restart* parity as a plain `DETECTOR` line
1872    /// (msc-faithful: its `.stim` carries the post-selected parities as
1873    /// ordinary detectors and keeps the post-selection index list on the
1874    /// Python side) and observable 0 = perfect initialization ⊕ gateway
1875    /// records ⊕ perfect readout.
1876    fn merged_distance_stim(layout: &SurgeryLayout, pauli: Pauli) -> String {
1877        let (escape, mut gateway) = escape_chunks(layout).unwrap();
1878        let key = LocalStabilizer::new(
1879            layout.pauli(pauli),
1880            Connectivity::ISOLATED.with_pipe(Direction::ZPLUS),
1881        );
1882        let gateway_entry = gateway.lookup(key).unwrap();
1883
1884        // Materialize the surface's endpoints: operator_in as the perfect
1885        // Steane port, operator_out as the perfect capped readout.
1886        let (port, port_logical) = steane_port_chunk(layout, &gateway_entry.operator_in);
1887        let final_patch = make_normal_surface_code_patch(layout.distance, layout.top_basis());
1888        let (cap, _) = cap_chunk(&final_patch, &gateway_entry.operator_out);
1889
1890        let mut chunks = vec![ChunkOrLoop::Single(Box::new(port))];
1891        chunks.extend(escape);
1892        chunks.push(ChunkOrLoop::Single(Box::new(cap)));

1894        let template = LoweringTemplate::from_chunks(chunks, gateway).unwrap();

1896        assert!(
1897            program.boundary_flows.is_empty(),
1898            "port and cap close every chain"
1899        );

1902            .detectors
1903            .iter()
1904            .map(|detector| detector.parity.measurements().collect())
1905            .chain(
1906                program
1907                    .restarts
1908                    .iter()
1909                    .map(|restart| restart.parity.measurements().collect()),
1910            )
1911            .collect();
1912        // The port is chunk 0, so its chunk-local logical id is already a
1913        // template id; the cap logical MPP is the template's last measurement.
1914        let mut observable = vec![port_logical, program.circuit.num_measurements() - 1];
1915        let resolved = template.observable_gateway.lookup(key).unwrap();
1916        observable.extend(
1917            resolved
1918                .measurements
1919                .iter()
1920                .flat_map(|chunk| chunk.measurements.iter().copied()),
1921        );
1922        bloq_stim::emit_annotated_stim(&program.circuit, &detectors, &[(0, observable)]).unwrap()
1923    }

1979        let d_int = layout.intermediate_distance() as i32;
1980        let final_patch = make_normal_surface_code_patch(layout.distance, layout.top_basis());

1982        // NOTE: the real compiler runs `discard_orphan_resplit_consumers`
1983        // here (the recovered seam-side two-weights genuinely have no merge
1984        // history). This isolated harness keeps them so the perfect port can
1985        // constrain the seam boundary — otherwise that edge is
1986        // under-initialized and X_L collapses to a spurious weight-3 (a
1987        // harness artifact, not a code property; the full-merge distance is
1988        // pinned separately by `escape_distance_through_merge`).
1989        let e1 = make_surface_code_chunk(&final_patch, Some(&init), None).unwrap();

1992        // recovered seam-side two-weights), plus the intermediate-patch
1993        // logical representative whose extension onto the full patch is
1994        // init-deterministic (Z: the seam-side row y = 1; X: the corner
1995        // column x = 2d − 1 — both extend across qubits the expansion

2002            .map(|flow| flow.start.clone())

2007                .collect()

2012                line(0..d, &|i| IVec2::new(2 * i + 1, 1)),
2013            ),
2014            _ => (
2015                line(0..d_int, &|i| IVec2::new(2 * d - 1, 2 * i + 1)),

2018        };
2019        products.push(port_line);
2020        let mut circuit = CoordCircuit::new();
2021        let recs = circuit
2022            .measure_pauli_products(products.iter().cloned())
2023            .unwrap();
2024        let port_logical = *recs.last().unwrap();
2025        let flows = products[..recs.len() - 1]
2026            .iter()
2027            .zip(&recs)
2028            .map(|(stabilizer, &m)| {
2029                Flow::new(PauliMap::empty(), stabilizer.clone()).with_measurements([m])
2030            })
2031            .collect();
2032        let port = Chunk { circuit, flows };
2033

2035        let mut chunks = vec![
2036            ChunkOrLoop::Single(Box::new(port)),
2037            ChunkOrLoop::Single(Box::new(e1)),

2054            .detectors
2055            .iter()
2056            .map(|detector| detector.parity.measurements().collect())
2057            .collect();

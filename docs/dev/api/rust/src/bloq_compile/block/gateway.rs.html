11const CENTER_BITS: u32 = 2;
12const ARM_BITS: u32 = 3;
13const ARM_MASK: u32 = 0b111;
14const ARM_BASIS_MASK: u32 = 0b11;
15const ARM_HADAMARD_BIT: u32 = 0b100;
16
17/// A packed description of an observable's local flow through one block.
18///
19/// The low 2 bits store the junction Pauli. Each of the 6 directions then
20/// uses 3 bits: 2 for the arm Pauli (`I/X/Z/Y`) and 1 for the Hadamard flag.
21///
22/// This is the physical flow shape consumed by an observable gateway, not the
23/// coordinatewise restriction of a graph stabilizer row. Every Pauli component
24/// carried by an arm also acts at the junction, so [`Self::with_arm`]
25/// reconstructs that component in the center. Center-only components remain
26/// possible for observables anchored or terminated inside the block.
27///
28/// This keeps the key compact while still allowing mixed local operators such
29/// as an `X` arm on one face and a `Z` arm on another.
30#[derive(Clone, Copy, PartialEq, Eq, Hash)]
31pub(crate) struct LocalStabilizer(u32);
32
33impl LocalStabilizer {
34    /// Create a uniform-basis key with the given pipe subset.
35    pub(crate) fn new(center_basis: Pauli, connectivity: Connectivity) -> Self {
36        let mut local = Self::isolated(center_basis);
37        for dir in Direction::iter() {
38            if connectivity.has_pipe(dir) {
39                local = local.with_arm(dir, center_basis, connectivity.has_hadamard(dir));
40            }
41        }
42        local
43    }
44
45    /// Create a key with only a junction Pauli and no arm support.
46    pub(crate) const fn isolated(center_basis: Pauli) -> Self {
47        Self(pauli_bits(center_basis))
48    }
49
50    /// Physical junction Pauli at this block.
51    pub(crate) fn center_basis(self) -> Pauli {
52        decode_pauli_bits(self.0 & ARM_BASIS_MASK)
53    }
54
55    /// Pauli carried by the given arm, or `I` if the arm is inactive.
56    pub(crate) fn arm_basis(self, dir: Direction) -> Pauli {
57        decode_pauli_bits((self.0 >> arm_offset(dir)) & ARM_BASIS_MASK)
58    }
59
60    /// Whether the given direction carries a local operator arm.
61    pub(crate) fn has_arm(self, dir: Direction) -> bool {
62        self.arm_basis(dir) != Pauli::I
63    }
64
65    /// Whether the given arm corresponds to a Hadamard pipe.
66    pub(crate) fn has_hadamard(self, dir: Direction) -> bool {
67        self.has_arm(dir) && (self.0 >> arm_offset(dir)) & ARM_HADAMARD_BIT != 0
68    }
69
70    /// Add a previously inactive arm with the given basis and Hadamard flag.
71    ///
72    /// An arm is a logical flow into the block junction, so its Pauli
73    /// components are also reconstructed in the center.
74    pub(crate) fn with_arm(mut self, dir: Direction, basis: Pauli, hadamard: bool) -> Self {
75        debug_assert!(!self.has_arm(dir), "a local flow arm is added only once");
76        self.0 |= pauli_bits(basis);
77        let shift = arm_offset(dir);
78        self.0 &= !(ARM_MASK << shift);
79        self.0 |= pauli_bits(basis) << shift;
80        if basis != Pauli::I && hadamard {
81            self.0 |= ARM_HADAMARD_BIT << shift;
82        }
83        self
84    }
85
86    /// Build the requested Pauli component of a local operator.
87    ///
88    /// The component exists if either the center node or any incident arm has
89    /// support in the requested basis. `Pauli::I` has no support and returns
90    /// `None` immediately.
91    pub(crate) fn component(self, basis: Pauli) -> Option<Self> {
92        if basis == Pauli::I {
93            return None;
94        }
95
96        let has_center_support = self.center_basis() & basis;
97        let mut has_support = has_center_support;
98        let mut component = Self::isolated(if has_center_support { basis } else { Pauli::I });
99
100        for dir in Direction::iter() {
101            let arm_basis = self.arm_basis(dir);
102            if arm_basis & basis {
103                has_support = true;
104                component = component.with_arm(dir, basis, self.has_hadamard(dir));
105            }
106        }
107
108        has_support.then_some(component)
109    }
110
111    /// Iterate over active arm directions.
112    pub(crate) fn arm_dirs(self) -> impl Iterator<Item = Direction> {
113        Direction::iter().filter(move |dir| self.has_arm(*dir))
114    }
115
116    /// Connectivity induced by the active operator arms.
117    pub(crate) fn connectivity(self) -> Connectivity {
118        let mut connectivity = Connectivity::ISOLATED;
119        for dir in Direction::iter() {
120            if self.has_arm(dir) {
121                connectivity = if self.has_hadamard(dir) {
122                    connectivity.with_hadamard(dir)
123                } else {
124                    connectivity.with_pipe(dir)
125                };
126            }
127        }
128        connectivity
129    }
130
131    /// Shared arm Pauli when every active arm uses the same basis.
132    pub(crate) fn uniform_arm_basis(self) -> Option<Pauli> {
133        let mut basis = None;
134        for dir in self.arm_dirs() {
135            let arm_basis = self.arm_basis(dir);
136            match basis {
137                Some(existing) if existing != arm_basis => return None,
138                Some(_) => {}
139                None => basis = Some(arm_basis),
140            }
141        }
142        basis
143    }
144
145    /// Number of active pipe directions.
146    pub(crate) fn weight(self) -> u32 {
147        self.arm_dirs().count() as u32
148    }
149
150    /// Whether no pipes carry the operator (weight == 0).
151    pub(crate) fn is_isolated(self) -> bool {
152        self.weight() == 0
153    }
154}

156impl fmt::Debug for LocalStabilizer {
157    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
158        if self.is_isolated() {
159            return write!(f, "LocalStabilizer({}, ISOLATED)", self.center_basis());
160        }
161
162        if self.uniform_arm_basis() == Some(self.center_basis()) {
163            write!(
164                f,
165                "LocalStabilizer({}, {:?})",
166                self.center_basis(),
167                self.connectivity(),
168            )?;
169            return Ok(());
170        }
171
172        write!(f, "LocalStabilizer({}, Arms(", self.center_basis())?;
173        let mut first = true;
174        for dir in Direction::iter() {
175            let arm_basis = self.arm_basis(dir);
176            if arm_basis == Pauli::I {
177                continue;
178            }
179            if !first {
180                write!(f, " ")?;
181            }
182            first = false;
183            if self.has_hadamard(dir) {
184                write!(f, "{dir}:{arm_basis}H")?;
185            } else {
186                write!(f, "{dir}:{arm_basis}")?;
187            }
188        }
189        write!(f, "))")
190    }
191}

195        fmt::Debug::fmt(self, f)

205/// Gateway-key connectivity for a chosen set of a block's pipe directions.
206///
207/// Keys never carry Hadamard flags: the pipe owning the X↔Z flip (a realignment
208/// node, or a spatial Hadamard wall) answers it in its own gateway, arm Paulis
209/// are already frame-local (`incident_edge_pauli` flips the stored Pauli when
210/// read across a Hadamard edge), and observable lowering builds every query key
211/// flag-free — so a flagged key would be unreachable by construction.
212pub(crate) fn gateway_key_connectivity(dirs: impl IntoIterator<Item = Direction>) -> Connectivity {
213    dirs.into_iter()
214        .fold(Connectivity::ISOLATED, Connectivity::with_pipe)
215}

224/// Record parity and support-only boundary operators for one local flow.
225///
226/// Operators use block-local coordinates at the −Z and +Z temporal faces;
227/// an empty map means the flow does not cross that face. Lowering translates
228/// operator support and binds template records to instance measurements.
229#[derive(Debug, Clone, PartialEq, Eq, Default)]
230pub(crate) struct GatewayEntry {
231    pub(crate) measurements: Vec<ChunkMeasurements>,
232    pub(crate) operator_in: PauliMap,
233    pub(crate) operator_out: PauliMap,
234}
235
236impl GatewayEntry {
237    /// XOR-combine multiple exact gateway entries into one mixed observable entry.
238    ///
239    /// Measurements combine by symmetric difference (a record appearing an even
240    /// number of times cancels). The boundary operators combine by the pointwise
241    /// Pauli product (`PauliMap` `Mul`), so decomposing a `Y` request into its
242    /// `X` and `Z` components and XOR-ing them reconstructs the `Y = X·Z`
243    /// boundary operator on each face.
244    pub(crate) fn xor<'a>(entries: impl IntoIterator<Item = &'a GatewayEntry>) -> Self {
245        let mut by_chunk: BTreeMap<usize, Vec<u32>> = BTreeMap::new();
246        let mut operator_in = PauliMap::empty();
247        let mut operator_out = PauliMap::empty();
248        for entry in entries {
249            for chunk_measurements in &entry.measurements {
250                by_chunk
251                    .entry(chunk_measurements.chunk_index)
252                    .or_default()
253                    .extend(&chunk_measurements.measurements);
254            }
255            operator_in = &operator_in ^ &entry.operator_in;
256            operator_out = &operator_out ^ &entry.operator_out;
257        }
258
259        let measurements = by_chunk
260            .into_iter()
261            .filter_map(|(chunk_index, bucket)| {
262                let measurements = crate::xor_toggled_sorted(bucket);
263                (!measurements.is_empty()).then_some(ChunkMeasurements {
264                    chunk_index,
265                    measurements,
266                })
267            })
268            .collect();
269
270        Self {
271            measurements,
272            operator_in,
273            operator_out,
274        }
275    }
276}

278/// Maps operator routing -> measurements. Missing key = forbidden configuration.
279#[derive(Debug, Clone, PartialEq, Eq, Default)]
280pub(crate) struct ObservableGateway(crate::FxMap<LocalStabilizer, GatewayEntry>);
281
282impl ObservableGateway {
283    pub(crate) fn new() -> Self {
284        Self::default()
285    }
286
287    pub(crate) fn insert(
288        &mut self,
289        key: LocalStabilizer,
290        entry: GatewayEntry,
291    ) -> Option<GatewayEntry> {
292        self.0.insert(key, entry)
293    }
294
295    /// Rewrite every measurement id from chunk-local space into template space.
296    ///
297    /// Block compilers build the gateway while measurements are still numbered
298    /// per chunk; [`LoweringTemplate::from_chunks`] calls this once the template
299    /// numbering is assigned, so observable lowering maps gateway measurements
300    /// straight to instance measurements without re-deriving the chunk-local
301    /// translation. Each `ChunkMeasurements::chunk_index` is retained as the
302    /// originating chunk, but its `measurements` are sorted, XOR-canonical
303    /// template ids afterward so repeated lowering can reuse the range directly.
304    pub(crate) fn resolve_to_template_ids(&mut self, mut resolve: impl FnMut(usize, u32) -> u32) {
305        for entry in self.0.values_mut() {
306            for chunk in &mut entry.measurements {
307                for measurement in &mut chunk.measurements {
308                    *measurement = resolve(chunk.chunk_index, *measurement);
309                }
310                chunk.measurements =
311                    crate::xor_toggled_sorted(std::mem::take(&mut chunk.measurements));
312            }
313        }
314    }
315
316    /// Shift every entry's originating chunk index, for harnesses that
317    /// prepend chunks (e.g. a perfect-initialization port) before handing the
318    /// gateway to `LoweringTemplate::from_chunks`.
319    #[cfg(test)]
320    pub(crate) fn shift_chunk_indices(&mut self, offset: usize) {
321        for entry in self.0.values_mut() {
322            for chunk in &mut entry.measurements {
323                chunk.chunk_index += offset;
324            }
325        }
326    }
327
328    #[cfg(test)]
329    pub(crate) fn contains_key(&self, key: &LocalStabilizer) -> bool {
330        self.0.contains_key(key)
331    }
332
333    #[cfg(test)]
334    pub(crate) fn keys(&self) -> impl Iterator<Item = &LocalStabilizer> {
335        self.0.keys()
336    }
337
338    #[cfg(test)]
339    pub(crate) fn len(&self) -> usize {
340        self.0.len()
341    }
342
343    /// Resolve one local observable request.
344    ///
345    /// Current behavior is intentionally simple:
346    /// 1. exact-key lookup
347    /// 2. fallback decomposition into global X/Z components and XOR-combine
348    ///
349    /// This is not a general GF(2) basis solver over arbitrary gateway keys.
350    pub(crate) fn lookup(&self, local_stabilizer: LocalStabilizer) -> Option<GatewayEntry> {
351        self.lookup_ref(local_stabilizer).map(Cow::into_owned)
352    }
353
354    pub(crate) fn lookup_ref(
355        &self,
356        local_stabilizer: LocalStabilizer,
357    ) -> Option<Cow<'_, GatewayEntry>> {
358        if let Some(entry) = self.0.get(&local_stabilizer) {
359            return Some(Cow::Borrowed(entry));
360        }
361
362        let entries = [Pauli::X, Pauli::Z]
363            .into_iter()
364            .filter_map(|basis| local_stabilizer.component(basis))
365            .map(|component| self.0.get(&component))
366            .collect::<Option<Vec<_>>>()?;
367        (!entries.is_empty()).then(|| Cow::Owned(GatewayEntry::xor(entries)))
368    }
369}

379const fn pauli_bits(pauli: Pauli) -> u32 {
380    pauli as u32
381}
382
383fn decode_pauli_bits(bits: u32) -> Pauli {
384    match bits {
385        0 => Pauli::I,
386        1 => Pauli::X,
387        2 => Pauli::Z,
388        3 => Pauli::Y,
389        _ => unreachable!("Pauli is encoded in two bits"),
390    }
391}

393fn arm_offset(dir: Direction) -> u32 {
394    CENTER_BITS + dir.index() as u32 * ARM_BITS
395}

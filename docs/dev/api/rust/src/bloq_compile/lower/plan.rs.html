60/// A spatial Hadamard pipe, oriented from its smaller endpoint position so that
61/// `src` is the wall's negative-axis cube — the one whose local coordinates the
62/// wall template is built in, and at whose offset its instance is placed.
63#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
64pub(crate) struct SpatialPipeRef {
65    pub(crate) src: IVec3,
66    pub(crate) dst: IVec3,
67}
68
69impl SpatialPipeRef {
70    pub(super) fn new(src: IVec3, dst: IVec3) -> Self {
71        if src.to_array() <= dst.to_array() {
72            Self { src, dst }
73        } else {
74            Self { src: dst, dst: src }
75        }
76    }
77
78    fn sort_key(self) -> ([i32; 3], [i32; 3]) {
79        (self.src.to_array(), self.dst.to_array())
80    }
81
82    pub(crate) fn axis(self) -> UDirection {
83        match (self.dst.x - self.src.x, self.dst.y - self.src.y) {
84            (delta, 0) if delta > 0 => UDirection::X,
85            (0, delta) if delta > 0 => UDirection::Y,
86            _ => unreachable!("spatial Hadamard walls run along +X or +Y"),
87        }
88    }
89
90    pub(crate) fn arm_directions(self) -> (bloq_graph::Direction, bloq_graph::Direction) {
91        match self.axis() {
92            UDirection::X => (bloq_graph::Direction::XMINUS, bloq_graph::Direction::XPLUS),
93            UDirection::Y => (bloq_graph::Direction::YMINUS, bloq_graph::Direction::YPLUS),
94            UDirection::Z => unreachable!("spatial Hadamard walls are not temporal"),
95        }
96    }
97}

119impl LinkedPlanInput {
120    pub(crate) fn from_module_graph(
121        source: &BlockGraph,
122        spatial_ports: &SpatialPortExpansionMap,
123        linked_positions: impl IntoIterator<Item = IVec3>,
124    ) -> Self {
125        let linked_positions = linked_positions.into_iter().collect::<crate::FxSet<_>>();
126        debug_assert!(
127            linked_positions
128                .iter()
129                .all(|&position| source.get_block(position).is_some())
130        );
131
132        let mut seen = crate::FxSet::default();
133        let mut pipes = linked_positions
134            .iter()
135            .flat_map(|&position| source.pipes_at(position))
136            .filter(|pipe| seen.insert(*pipe))
137            .filter_map(|pipe| {
138                let (src, dst) = pipe.endpoints();
139                let owners = (
140                    source
141                        .get_endpoint_block(src)
142                        .expect("pipe source endpoint is owned by a block")
143                        .pos(),
144                    source
145                        .get_endpoint_block(dst)
146                        .expect("pipe target endpoint is owned by a block")
147                        .pos(),
148                );
149                if !linked_positions.contains(&owners.0) || !linked_positions.contains(&owners.1) {
150                    return None;
151                }
152                let pipe_ref = temporal_pipe_ref(pipe);
153                let template =
154                    (!pipe.dir().is_spatial() && pipe.is_hadamard()).then(|| TemplatePlan {
155                        top_basis: temporal_hadamard_top_basis(source, &pipe_ref),
156                        origin: temporal_hadamard_origin(&pipe_ref),
157                    });
158                Some(LinkedPipePlan {
159                    pipe: pipe.clone(),
160                    owners,
161                    template,
162                })
163            })
164            .collect::<Vec<_>>();
165        pipes.sort_unstable_by_key(|record| {
166            let (left, right) = record.pipe.endpoints();
167            let (left, right) = if left.to_array() <= right.to_array() {
168                (left, right)
169            } else {
170                (right, left)
171            };
172            (left.to_array(), right.to_array(), record.pipe.is_hadamard())
173        });
174        let mut spatial_ports = spatial_ports
175            .values()
176            .filter(|port| linked_positions.contains(&port.source))
177            .copied()
178            .collect::<Vec<_>>();
179        spatial_ports.sort_unstable_by_key(|port| port.source.to_array());
180        let mut blocks = linked_positions.into_iter().collect::<Vec<_>>();
181        blocks.sort_unstable_by_key(glam::IVec3::to_array);
182        Self {
183            blocks,
184            pipes,
185            spatial_ports,
186        }
187    }
188
189    pub(crate) fn temporal_templates(&self) -> impl Iterator<Item = TemplatePlan> + '_ {
190        self.pipes.iter().filter_map(|record| record.template)
191    }
192
193    pub(crate) fn spatial_walls(&self) -> impl Iterator<Item = SpatialPipeRef> + '_ {
194        self.pipes
195            .iter()
196            .filter(|record| record.pipe.dir().is_spatial() && record.pipe.is_hadamard())
197            .map(|record| {
198                let (src, dst) = record.pipe.endpoints();
199                SpatialPipeRef::new(src, dst)
200            })
201    }
202}

215    pub(crate) fn from_linked_input(source: &LinkedPlanInput) -> Self {
216        let mut layers: BTreeMap<i32, Vec<IVec3>> = BTreeMap::new();
217        for &pos in &source.blocks {
218            layers.entry(pos.z).or_default().push(pos);
219        }
220        let mut spatial_pipes_by_layer: BTreeMap<i32, Vec<&Pipe>> = BTreeMap::new();
221        for pipe in source
222            .pipes
223            .iter()
224            .filter(|record| record.pipe.dir().is_spatial())
225            .map(|record| &record.pipe)
226        {
227            spatial_pipes_by_layer
228                .entry(pipe.src().z)
229                .or_default()
230                .push(pipe);
231        }
232
233        let mut graph = DiGraph::new();
234        let mut node_by_block = crate::FxMap::default();
235        let mut node_by_temporal_pipe = crate::FxMap::default();
236        let mut node_by_spatial_port = crate::FxMap::default();
237        for (layer, mut block_positions) in layers {
238            block_positions.sort_by_key(glam::IVec3::to_array);
239            let spatial_pipes = spatial_pipes_by_layer
240                .get(&layer)
241                .map(Vec::as_slice)
242                .unwrap_or_default();
243            for component in spatial_components(&block_positions, spatial_pipes) {
244                let members = component
245                    .iter()
246                    .copied()
247                    .map(|pos| SourceBlockRef { pos })
248                    .collect::<Vec<_>>();
249                debug_assert!(members.windows(2).all(|pair| pair[0] <= pair[1]));
250                let node = graph.add_node(NodePlan {
251                    layer: 2 * i64::from(layer),
252                    incoming_cut: false,
253                    provenance: NodeProvenance::BlockComponent { members },
254                    template: None,
255                    walls: Vec::new(),
256                });
257                for pos in component {
258                    node_by_block.insert(pos, node);
259                }
260            }
261        }
262
263        // Use raw pipe endpoints: a virtual endpoint's owner may be in this
264        // component even though the wall itself is not part of it.
265        for wall in source.spatial_walls() {
266            if let (Some(&src), Some(&dst)) =
267                (node_by_block.get(&wall.src), node_by_block.get(&wall.dst))
268                && src == dst
269            {
270                graph[src].walls.push(wall);
271            }
272        }
273        for node in graph.node_weights_mut() {
274            node.walls.sort_unstable_by_key(|wall| wall.sort_key());
275        }
276
277        let mut edge_by_endpoints = crate::FxMap::default();
278        for record in source
279            .pipes
280            .iter()
281            .filter(|record| !record.pipe.dir().is_spatial())
282        {
283            let pipe = &record.pipe;
284            let (src_owner, dst_owner) = record.owners;
285            let src_node = node_by_block[&src_owner];
286            let dst_node = node_by_block[&dst_owner];
287            let (plan_source, plan_target) = temporal_edge_endpoints(&graph, src_node, dst_node);
288            let pipe_ref = temporal_pipe_ref(pipe);
289
290            if pipe.is_hadamard() {
291                let template = record
292                    .template
293                    .expect("temporal Hadamard link record carries its template ABI");
294                let pipe_node = graph.add_node(NodePlan {
295                    layer: temporal_hadamard_layer(&pipe_ref),
296                    incoming_cut: false,
297                    provenance: NodeProvenance::TemporalPipe { pipe: pipe_ref },
298                    template: Some(template),
299                    walls: Vec::new(),
300                });
301                node_by_temporal_pipe.insert(pipe_ref, pipe_node);
302                graph.add_edge(
303                    plan_source,
304                    pipe_node,
305                    EdgePlan {
306                        pipes: vec![pipe_ref],
307                        spatial_ports: Vec::new(),
308                    },
309                );
310                graph.add_edge(
311                    pipe_node,
312                    plan_target,
313                    EdgePlan {
314                        pipes: vec![pipe_ref],
315                        spatial_ports: Vec::new(),
316                    },
317                );
318                continue;
319            }
320
321            if let Some(&edge) = edge_by_endpoints.get(&(plan_source, plan_target)) {
322                graph
323                    .edge_weight_mut(edge)
324                    .expect("edge endpoint map references inserted edge")
325                    .pipes
326                    .push(pipe_ref);
327            } else {
328                let edge = graph.add_edge(
329                    plan_source,
330                    plan_target,
331                    EdgePlan {
332                        pipes: vec![pipe_ref],
333                        spatial_ports: Vec::new(),
334                    },
335                );
336                edge_by_endpoints.insert((plan_source, plan_target), edge);
337            }
338        }
339
340        for &port in &source.spatial_ports {
341            let cube = node_by_block[&port.source];
342            let virtual_port = graph.add_node(NodePlan {
343                layer: 2 * i64::from(port.source.z) + if port.is_input() { -1 } else { 1 },
344                incoming_cut: false,
345                provenance: NodeProvenance::SpatialPortSubstitution {
346                    source: port.source,
347                    role: port.role,
348                },
349                template: None,
350                walls: Vec::new(),
351            });
352            node_by_spatial_port.insert(port.source, virtual_port);
353            let (source, target) = if port.is_input() {
354                (virtual_port, cube)
355            } else {
356                (cube, virtual_port)
357            };
358            graph.add_edge(
359                source,
360                target,
361                EdgePlan {
362                    pipes: Vec::new(),
363                    spatial_ports: vec![port],
364                },
365            );
366        }
367
368        // Canonicalize each edge's coalesced pipe list: authoring order of the
369        // source pipes is not observable, and padding derives from this list in
370        // order (WF-4 pairing), so sorting here makes both the seam and its
371        // padding reproducible.
372        for edge in graph.edge_weights_mut() {
373            edge.pipes
374                .sort_unstable_by_key(|pipe| (pipe.src.to_array(), pipe.dst.to_array()));
375        }
376        for node in graph.node_indices().collect::<Vec<_>>() {
377            graph[node].incoming_cut = graph
378                .neighbors_directed(node, petgraph::Direction::Incoming)
379                .next()
380                .is_some();
381        }
382
383        // The instance detector pass appends nodes layer-major and relies on
384        // every plan edge ascending `layer` for that sort to be a topological
385        // order (lower/detector.rs); pin the invariant where edges are built.
386        debug_assert!(
387            graph
388                .edge_references()
389                .all(|edge| graph[edge.source()].layer <= graph[edge.target()].layer),
390            "plan edges must ascend layer: detector composition sorts layer-major"
391        );
392
393        Self {
394            graph,
395            node_by_block,
396            node_by_temporal_pipe,
397            node_by_spatial_port,
398        }
399    }

431    pub(crate) fn temporal_node_components(&self) -> Vec<Vec<NodeIndex>> {
432        let mut visited = vec![false; self.graph.node_count()];
433        let mut components = Vec::new();
434        for root in self.graph.node_indices() {
435            if visited[root.index()] {
436                continue;
437            }
438
439            let mut stack = vec![root];
440            visited[root.index()] = true;
441            let mut component = Vec::new();
442            while let Some(node) = stack.pop() {
443                component.push(node);
444                for neighbor in self
445                    .graph
446                    .neighbors_directed(node, petgraph::Outgoing)
447                    .chain(self.graph.neighbors_directed(node, petgraph::Incoming))
448                {
449                    if !visited[neighbor.index()] {
450                        visited[neighbor.index()] = true;
451                        stack.push(neighbor);
452                    }
453                }
454            }
455            components.push(component);
456        }
457
458        components
459    }

474fn temporal_edge_endpoints(
475    graph: &DiGraph<NodePlan, EdgePlan>,
476    src_node: NodeIndex,
477    dst_node: NodeIndex,
478) -> (NodeIndex, NodeIndex) {
479    let src = &graph[src_node];
480    let dst = &graph[dst_node];
481    if (src.layer, src_node.index()) <= (dst.layer, dst_node.index()) {
482        (src_node, dst_node)
483    } else {
484        (dst_node, src_node)
485    }
486}

488fn spatial_components(block_positions: &[IVec3], spatial_pipes: &[&Pipe]) -> Vec<Vec<IVec3>> {
489    let position_index: crate::FxMap<IVec3, usize> = block_positions
490        .iter()
491        .copied()
492        .enumerate()
493        .map(|(index, pos)| (pos, index))
494        .collect();
495    let mut components = UnionFind::new(block_positions.len());
496
497    for pipe in spatial_pipes {
498        let (src, dst) = pipe.endpoints();
499        if let (Some(&src_index), Some(&dst_index)) =
500            (position_index.get(&src), position_index.get(&dst))
501        {
502            components.union(src_index, dst_index);
503        }
504    }
505
506    let mut grouped: BTreeMap<usize, Vec<IVec3>> = BTreeMap::new();
507    for (index, &pos) in block_positions.iter().enumerate() {
508        grouped.entry(components.find(index)).or_default().push(pos);
509    }
510
511    let mut components: Vec<_> = grouped.into_values().collect();
512    components.sort_by_key(|component| component[0].to_array());
513    components
514}

516/// A canonical [`TemporalPipeRef`] for `pipe`, oriented `src.z <= dst.z` so
517/// authoring order (which endpoint the source `Pipe` names first) never leaks
518/// into the compiled artifact. Every consumer that cares about time order reads
519/// through this orientation or [`TemporalPipeRef::endpoints_by_z`]; keep this the
520/// single construction point so map keys (`node_by_temporal_pipe`, pipe
521/// templates) stay consistent — see [`super::observable::hadamard_temporal_pipe`].
522fn temporal_pipe_ref(pipe: &Pipe) -> TemporalPipeRef {
523    let (src, dst) = pipe.endpoints();
524    let (src, dst) = if src.z <= dst.z {
525        (src, dst)
526    } else {
527        (dst, src)
528    };
529    TemporalPipeRef {
530        src,
531        dst,
532        hadamard: pipe.is_hadamard(),
533    }
534}
535
536fn temporal_hadamard_layer(pipe: &TemporalPipeRef) -> i64 {
537    2 * i64::from(pipe.src.z.min(pipe.dst.z)) + 1
538}
539
540fn temporal_hadamard_origin(pipe: &TemporalPipeRef) -> IVec3 {
541    pipe.endpoints_by_z().0
542}
543
544fn temporal_hadamard_top_basis(graph: &BlockGraph, pipe: &TemporalPipeRef) -> Basis {
545    let (below, _) = pipe.endpoints_by_z();
546    let source = graph
547        .get_pipe(pipe.src, pipe.dst)
548        .expect("temporal hadamard comes from a source pipe");
549    graph.infer_pipe_basis_from_endpoint(source, below)[UDirection::Y.index()]
550        .expect("validated temporal pipe has a transverse basis")
551}

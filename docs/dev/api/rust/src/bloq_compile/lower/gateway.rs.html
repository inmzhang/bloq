46impl LocalSurfaceContext<'_> {
47    /// Only armless nonzero flow on a piped block needs global anchor facts.
48    pub(crate) fn needs_dynamic_anchor(
49        &self,
50        position: IVec3,
51        surface: &impl SurfaceSupport,
52    ) -> bool {
53        // Any nonzero arm rules out an anchor, including arms that reconstruct
54        // an initially zero center. With no arms, only the source center remains.
55        if surface.node_pauli(position) == Pauli::I {
56            return false;
57        }
58        let Some(compiled) = self.compiled.get(&position) else {
59            return false;
60        };
61        let connectivity = compiled.graph_connectivity;
62        if connectivity.is_isolated()
63            || (self
64                .ports
65                .get(&position)
66                .is_some_and(|port| port.is_spatial())
67                && surface.port_pauli(position) != Pauli::I)
68        {
69            return false;
70        }
71        Direction::iter()
72            .filter(|&direction| connectivity.has_pipe(direction))
73            .all(|direction| {
74                let Some((source, target)) =
75                    super::observable::incident_edge_positions(self.graph, position, direction)
76                else {
77                    return true;
78                };
79                // Forward support takes precedence even for Some(I). A reverse
80                // Hadamard flip does not affect whether the arm is identity.
81                surface
82                    .edge_pauli(source, target)
83                    .or_else(|| surface.edge_pauli(target, source))
84                    .is_none_or(|pauli| pauli == Pauli::I)
85            })
86    }

88    /// Bounded geometry must supply an anchor when `needs_dynamic_anchor` is
89    /// true. Without one, the fallback queries require a complete source graph.
90    pub(crate) fn resolve(
91        &self,
92        positions: &[IVec3],
93        surface: &impl SurfaceSupport,
94        selected: Option<(IVec3, bool)>,
95        kind: &StabilizerRowKind,
96        index: u32,
97        anchor: Option<BoundDynamicAnchor<'_>>,
98    ) -> Result<
99        (
100            Vec<InstanceMeasurement>,
101            Vec<InstanceBoundaryOperator>,
102            bool,
103        ),
104        CompileError,
105    > {
106        debug_assert!(positions.is_sorted_by_key(glam::IVec3::to_array));
107        debug_assert!(anchor.is_none() || positions.len() == 1);
108        let mut arms = crate::FxMap::default();
109        let selected_template = selected.map(|(position, selected)| {
110            debug_assert_eq!(positions, [position]);
111            let instances = self.selectives[&position];
112            let TemplateRef::Selective {
113                when_false,
114                when_true,
115            } = self.compiled[&position].template
116            else {
117                unreachable!("selected local gateway is a selective")
118            };
119            arms.insert(
120                position,
121                if selected {
122                    instances.when_true
123                } else {
124                    instances.when_false
125                },
126            );
127            (position, if selected { when_true } else { when_false })
128        });
129        let mut insertions =
130            super::observable::resolve_observable_insertions_at_positions_with_ports(
131                self.graph,
132                surface,
133                kind,
134                self.compiled,
135                self.pool,
136                positions,
137                index,
138                true,
139                self.ports,
140                selected_template,
141                anchor,
142            )?;
143        insertions.extend(
144            super::observable::resolve_spatial_port_observable_insertions(
145                surface,
146                self.ports,
147                self.port_templates,
148                self.pool,
149                positions,
150                index,
151                true,
152            )?,
153        );
154        // Each edge gateway belongs to its smaller source position. Borrow the
155        // complete surface so a proxy can bind every site without cloning it.
156        let wall_sign = super::observable::push_wall_observable_insertions(
157            self.graph,
158            self.wall_templates,
159            self.pool,
160            surface,
161            Some(positions),
162            index,
163            &mut insertions,
164        )?;
165        let (mut records, includes) = collect_assignment_content(
166            &insertions,
167            self.layout,
168            &arms,
169            self.instances,
170            anchor.and_then(|anchor| anchor.source_t),
171        )?;
172        let (pipe_records, pipe_sign) = pipe_crossing_content(
173            self.graph,
174            self.pipe_templates,
175            self.pool,
176            self.instances,
177            surface,
178            positions,
179            index,
180        )?;
181        records.extend(pipe_records);
182        let sign = matches!(kind, StabilizerRowKind::Logical)
183            && (wall_sign
184                ^ pipe_sign
185                ^ super::observable::port_transpose_sign(surface, self.ports, Some(positions)));
186        Ok((crate::xor_toggled_sorted(records), includes, sign))
187    }
188}

190/// A temporal H crossing contributes its realignment records and the H Y H = -Y offset.
191pub(super) fn pipe_crossing_content(
192    graph: &BlockGraph,
193    pipe_templates: &crate::FxMap<TemporalPipeRef, LoweringTemplateId>,
194    template_pool: &LoweringTemplatePool,
195    instance_by_site: &crate::FxMap<ChunkSiteSource, TemplateInstanceId>,
196    fixed: &impl SurfaceSupport,
197    positions: &[IVec3],
198    index: u32,
199) -> Result<(Vec<InstanceMeasurement>, bool), CompileError> {
200    let mut crossings: Vec<(TemporalPipeRef, Pauli)> = fixed
201        .owned_edges(Some(positions))
202        .filter_map(|((src, dst), edge_pauli)| {
203            hadamard_temporal_pipe(graph, src, dst).map(|pipe| {
204                let pauli = realignment_observable_pauli(graph, &pipe, fixed, edge_pauli);
205                (pipe, pauli)
206            })
207        })
208        .collect();
209    crossings.sort_by_key(|(pipe, _)| (pipe.src.to_array(), pipe.dst.to_array()));
210
211    let mut records: crate::FxSet<InstanceMeasurement> = crate::FxSet::default();
212    let mut sign = false;
213    for (pipe, pauli) in crossings {
214        if pauli == Pauli::I {
215            continue;
216        }
217        sign ^= pauli == Pauli::Y;
218        let template_id = pipe_templates
219            .get(&pipe)
220            .expect("crossed temporal pipe has a lowered template");
221        let gateway = &template_pool
222            .get(*template_id)
223            .expect("temporal pipe template exists")
224            .observable_gateway;
225        let local = LocalStabilizer::isolated(pauli);
226        let entry = gateway.lookup_ref(local).ok_or_else(|| {
227            CompileError::ObservableMissingGatewayEntry {
228                block_pos: pipe.src,
229                observable_index: index,
230                local_stabilizer: local.to_string(),
231            }
232        })?;
233        let instance = instance_by_site[&ChunkSiteSource::TemporalPipe(pipe)];
234        for chunk_meas in &entry.measurements {
235            xor_toggle(
236                &mut records,
237                chunk_meas
238                    .measurements
239                    .iter()
240                    .copied()
241                    .map(|measurement| InstanceMeasurement {
242                        instance,
243                        measurement,
244                    }),
245            );
246        }
247    }
248    let mut records: Vec<InstanceMeasurement> = records.into_iter().collect();
249    records.sort_unstable();
250    Ok((records, sign))
251}

253/// Relabel an assignment's template-local insertions into instance-global observable measurement
254/// records and observable boundary operators, naming each site's instance (the resolved arm
255/// instance for a selective site, else the block's own instance).
256pub(super) fn collect_assignment_content(
257    insertions: &[ObservableChunkInsertion],
258    layout: BlockLayout,
259    arm_instance_by_site: &crate::FxMap<IVec3, TemplateInstanceId>,
260    instance_by_site: &crate::FxMap<ChunkSiteSource, TemplateInstanceId>,
261    source_t: Option<TSourceGateway<'_>>,
262) -> Result<(Vec<InstanceMeasurement>, Vec<InstanceBoundaryOperator>), CompileError> {
263    let mut measurement_groups = Vec::new();
264    let mut includes: Vec<InstanceBoundaryOperator> = Vec::new();
265
266    for insertion in insertions {
267        let instance = match insertion.site {
268            ChunkSiteSource::Block(pos) => source_t
269                .filter(|source| source.source == pos)
270                .map(|source| source.instance)
271                .or_else(|| arm_instance_by_site.get(&pos).copied())
272                .unwrap_or_else(|| instance_by_site[&insertion.site]),
273            ChunkSiteSource::TemporalPipe(_)
274            | ChunkSiteSource::SpatialPipe(_)
275            | ChunkSiteSource::SpatialPort(_) => instance_by_site[&insertion.site],
276        };
277
278        if !insertion.measurements.is_empty() {
279            measurement_groups.push((instance, insertion.measurements.as_slice()));
280        }
281
282        let off = insertion_offset(layout, insertion.site)?;
283        for (face, operator) in [
284            (BoundaryFace::Input, &insertion.operator_in),
285            (BoundaryFace::Output, &insertion.operator_out),
286        ] {
287            if operator.is_empty() {
288                continue;
289            }
290            includes.push(InstanceBoundaryOperator {
291                instance,
292                face,
293                operator: operator.try_translated(off)?,
294            });
295        }
296    }
297
298    Ok((
299        xor_merge_assignment_measurements(measurement_groups),
300        includes,
301    ))
302}

304/// XOR-merge non-empty, non-overlapping sorted ranges without sorting every record.
305fn xor_merge_assignment_measurements(
306    mut groups: Vec<(TemplateInstanceId, &[u32])>,
307) -> Vec<InstanceMeasurement> {
308    groups.sort_unstable_by_key(|&(instance, group)| (instance, group[0]));
309    let mut merged = Vec::with_capacity(groups.iter().map(|(_, group)| group.len()).sum());
310    let mut previous = None;
311    for (instance, group) in groups {
312        debug_assert!(group.is_sorted());
313        for &measurement in group {
314            let measurement = InstanceMeasurement {
315                instance,
316                measurement,
317            };
318            debug_assert!(previous.is_none_or(|previous| previous <= measurement));
319            previous = Some(measurement);
320            if merged.last() == Some(&measurement) {
321                merged.pop();
322            } else {
323                merged.push(measurement);
324            }
325        }
326    }
327    merged
328}

330fn insertion_offset(
331    layout: BlockLayout,
332    source: ChunkSiteSource,
333) -> Result<glam::IVec2, CompileError> {
334    match source {
335        ChunkSiteSource::Block(pos) => layout.offset(pos),
336        ChunkSiteSource::TemporalPipe(pipe) => layout.offset(pipe.endpoints_by_z().0),
337        // The wall template uses its negative-axis endpoint's coordinates.
338        ChunkSiteSource::SpatialPipe(pipe) => layout.offset(pipe.src),
339        ChunkSiteSource::SpatialPort(port) => layout.offset(port),
340    }
341}

343#[cfg(test)]
344mod tests {
345    use super::*;
346    use crate::compile::CompiledTemplateInfo;
347    use crate::port::PortExpansion;
348    use crate::signature::Connectivity;
349    use bloq_graph::{Block, BlockKind, CubeKind, PortRole, Stabilizer};
350
351    #[test]
352    fn dynamic_anchors_require_nonzero_centers_without_incident_support() {
353        let center = IVec3::ZERO;
354        for height in [1, 2] {
355            let neighbor = height * IVec3::Z;
356            let mut graph = BlockGraph::new();
357            graph.add_block(
358                Block::new(center, BlockKind::Cube(CubeKind::ZXZ))
359                    .with_height(format!("{height}d").parse().unwrap())
360                    .unwrap(),
361            );
362            graph.add_block(Block::new(neighbor, BlockKind::Cube(CubeKind::ZXZ)));
363            for connectivity in [
364                Connectivity::ISOLATED,
365                Connectivity::ISOLATED.with_pipe(Direction::ZPLUS),
366                Connectivity::ISOLATED.with_hadamard(Direction::ZPLUS),
367            ] {
368                let compiled = CompiledTemplateMap::from_iter([(
369                    center,
370                    CompiledTemplateInfo {
371                        graph_connectivity: connectivity,
372                        template: TemplateRef::Fixed(LoweringTemplateId(0)),
373                    },
374                )]);
375                for (pauli, forward, reverse, spatial, expected) in [
376                    (Pauli::I, None, None, false, false),
377                    (Pauli::Z, None, None, false, true),
378                    (Pauli::Z, Some(Pauli::X), None, false, false),
379                    (Pauli::Z, None, Some(Pauli::Y), false, false),
380                    (Pauli::Z, None, Some(Pauli::I), false, true),
381                    (Pauli::Z, Some(Pauli::I), Some(Pauli::X), false, true),
382                    (Pauli::Z, None, None, true, false),
383                ] {
384                    let ports = if spatial {
385                        PortExpansionMap::from_iter([(
386                            center,
387                            PortExpansion {
388                                source: center,
389                                role: PortRole::Output,
390                                cube: Some((CubeKind::ZXZ, Default::default())),
391                                pipe_dir: Direction::ZMINUS,
392                                boundary_basis: CubeKind::ZXZ.y(),
393                                private_qubit: None,
394                            },
395                        )])
396                    } else {
397                        PortExpansionMap::default()
398                    };
399                    let context = LocalSurfaceContext {
400                        graph: &graph,
401                        compiled: &compiled,
402                        ports: &ports,
403                        port_templates: &Default::default(),
404                        wall_templates: &Default::default(),
405                        pipe_templates: &Default::default(),
406                        pool: &LoweringTemplatePool::default(),
407                        instances: &Default::default(),
408                        selectives: &Default::default(),
409                        layout: BlockLayout::new(3),
410                    };
411                    let mut surface = Stabilizer::from_interior_nodes([(center, pauli)]);
412                    surface.port_stabilizer.insert(center, Pauli::Y);
413                    for (edge, support) in
414                        [((center, neighbor), forward), ((neighbor, center), reverse)]
415                    {
416                        if let Some(support) = support {
417                            surface.interior_edges.insert(edge, support);
418                        }
419                    }
420                    assert_eq!(
421                        context.needs_dynamic_anchor(center, &surface),
422                        expected && !connectivity.is_isolated()
423                    );
424                    assert!(!context.needs_dynamic_anchor(IVec3::X, &surface));
425                }
426            }
427        }
428    }

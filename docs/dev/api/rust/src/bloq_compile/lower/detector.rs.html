18impl<'ctx> BloqLowerContext<'ctx> {
19    /// Resolve the templates each node member instantiates, with the offsets
20    /// and measurement-map sites needed to read their flows in place.
21    ///
22    /// The detector pass walks template flows directly instead of per-node
23    /// translated copies: the engine's offset-aware keys make that exact, and
24    /// it skips materializing two translated `PauliMap`s per flow per node.
25    fn detector_node_context(&self, plan_node: NodeIndex) -> DetectorNodeContext<'ctx> {
26        let node = &self.input.plan.graph()[plan_node];
27        let skip_unmatched_inputs = match &self.classified[plan_node.index()] {
28            NodeLowering::Selective { .. } => {
29                unreachable!("static detector lowering receives pinned selective templates")
30            }
31            // The intra-region seam already closed the escape template's
32            // cultivation-side consumers. Its output flows stay live here.
33            NodeLowering::TRegion { .. } => true,
34            NodeLowering::Fixed { .. }
35            | NodeLowering::TemporalPipe { .. }
36            | NodeLowering::SpatialPort { .. } => false,
37        };
38        let sources = self.node_instances[plan_node.index()]
39            .iter()
40            .map(|placed| MemberFlowSource {
41                template: &self.input.template_pool[placed.instance.template],
42                offset: placed.instance.offset,
43                instance_id: placed.instance.id,
44            })
45            .collect();
46        DetectorNodeContext {
47            plan_node: plan_node.index(),
48            layer: node.layer,
49            members: node.block_members(),
50            sources,
51            skip_unmatched_inputs,
52        }
53    }
54}

56struct DetectorNodeContext<'a> {
57    plan_node: usize,
58    layer: i64,
59    members: &'a [SourceBlockRef],
60    sources: Vec<MemberFlowSource<'a>>,
61    /// True only for a T region's escape instance at the outer level.
62    skip_unmatched_inputs: bool,
63}

65/// One node member's compiled template, placed at the member's XY offset.
66struct MemberFlowSource<'a> {
67    template: &'a LoweringTemplate,
68    offset: IVec2,
69    instance_id: TemplateInstanceId,
70}

72/// Composed boundary detectors grouped by the plan node the chains closed on.
73type ComposedDetectors = Vec<(usize, Vec<NodeDetector>)>;
74/// Composed restart syndromes grouped by the plan node the chains closed on.
75type ComposedRestarts = Vec<(usize, Vec<NodeRestart>)>;
76
77/// Pending side-table entries keyed by the plan node the chain closed on.
78struct DetectorLowerState<'c> {
79    detectors: ComposedDetectors,
80    restarts: ComposedRestarts,
81    engine: DetectorFlowEngine<'c>,
82}

84impl DetectorLowerState<'_> {
85    fn drain_node_chains(&mut self, plan_node: usize) {
86        let (detectors, restarts) = bloq_ir::lowering::drain_composed_chains(&mut self.engine);
87        if !detectors.is_empty() {
88            self.detectors.push((plan_node, detectors));
89        }
90        if !restarts.is_empty() {
91            self.restarts.push((plan_node, restarts));
92        }
93    }
94}

96// ==============================================================================
97// Boundary-only detector composition
98// ==============================================================================
99//
100// Detectors that close inside a single template are recovered downstream by the
101// backend (it instantiates `program_template.detectors`/`repeat_states` per
102// instance via a cheap offset + measurement relabel). So instance lowering only
103// composes the **boundary** detectors — flows the template left open that close
104// against a temporally adjacent node.
105//
106// Those open flows are precomputed per template as `boundary_flows` (see
107// `block/compile.rs::drain_boundary_flows`), already fused into mutually
108// independent chains. Here we replay only that residual through a per-component
109// engine in emission order, so an open created by a lower-z node is consumed by
110// the node above it. This pass composes nothing but `TopLevel` `NodeDetector`s
111// and allocates no loop state — a loop's recurrence is backend-owned, and its
112// sole cross-node round is resolved by the backend's post-`REPEAT` lookback.
113pub(super) fn attach_temporal_component_detectors(
114    plan: &LowerPlan,
115    context: &mut BloqLowerContext<'_>,
116) -> Result<(), CompileError> {
117    for mut component in plan.temporal_node_components() {
118        layer_major_order(&mut component, context)?;
119        attach_component_detectors(&component, context)?;
120    }
121    Ok(())
122}
123
124/// Layer-major, emit-order tie-broken order for one temporal component. Every
125/// plan edge ascends `layer`, so this is still a topological order — but plain
126/// emit order is not enough here: the engine keys open chains by 2D support only,
127/// and a bare toposort may interleave same-footprint branches out of z order (two
128/// T columns are DAG sources with no path between them, U17e), letting a lower
129/// column's creators clobber another column's still-open chains at the same XY
130/// offset. Layer-major ordering walks the component in time.
131fn layer_major_order(
132    nodes: &mut [NodeIndex],
133    context: &BloqLowerContext<'_>,
134) -> Result<(), CompileError> {
135    nodes.sort_by_cached_key(|node| {
136        (
137            context.input.plan.graph()[*node].layer,
138            context.emit_order[node.index()],
139        )
140    });
141
142    let mut first = 0;
143    while first < nodes.len() {
144        let layer = context.input.plan.graph()[nodes[first]].layer;
145        let end = nodes[first..]
146            .partition_point(|node| context.input.plan.graph()[*node].layer == layer)
147            + first;
148        order_same_layer_flows(&mut nodes[first..end], context, layer)?;
149        first = end;
150    }
151    Ok(())
152}
153
154/// A same-layer moving patch may produce a boundary still waiting to be
155/// consumed by its neighbor. Replay those consumers first; unrelated nodes
156/// retain emission order.
157fn order_same_layer_flows(
158    nodes: &mut [NodeIndex],
159    context: &BloqLowerContext<'_>,
160    layer: i64,
161) -> Result<(), CompileError> {
162    if nodes.len() < 2 {
163        return Ok(());
164    }
165    let mut consumers = crate::FxMap::<FlowKey<'_>, smallvec::SmallVec<[usize; 1]>>::default();
166    let mut produced = vec![Vec::new(); nodes.len()];
167    for (index, &node) in nodes.iter().enumerate() {
168        let node = context.detector_node_context(node);
169        for source in node.sources {
170            for flow in &source.template.program_template.boundary_flows {
171                if !flow.start.is_empty() {
172                    consumers
173                        .entry(FlowKey::try_new(&flow.start, source.offset)?)
174                        .or_default()
175                        .push(index);
176                }
177                if !flow.end.is_empty() {
178                    produced[index].push(FlowKey::try_new(&flow.end, source.offset)?);
179                }
180            }
181        }
182    }
183
184    let mut successors = vec![crate::FxSet::default(); nodes.len()];
185    let mut indegree = vec![0usize; nodes.len()];
186    for (producer, boundaries) in produced.iter().enumerate() {
187        for boundary in boundaries {
188            for &consumer in consumers.get(boundary).into_iter().flatten() {
189                if consumer != producer && successors[consumer].insert(producer) {
190                    indegree[producer] += 1;
191                }
192            }
193        }
194    }
195    let mut ready = indegree
196        .iter()
197        .enumerate()
198        .filter_map(|(index, &count)| (count == 0).then_some(index))
199        .collect::<std::collections::BTreeSet<_>>();
200    let mut order = Vec::with_capacity(nodes.len());
201    while let Some(index) = ready.pop_first() {
202        order.push(nodes[index]);
203        for &next in &successors[index] {
204            indegree[next] -= 1;
205            if indegree[next] == 0 {
206                ready.insert(next);
207            }
208        }
209    }
210    if order.len() != nodes.len() {
211        return Err(CompileError::NodeFlowCompositionFailed {
212            layer,
213            members: nodes
214                .iter()
215                .flat_map(|node| context.input.plan.graph()[*node].block_members())
216                .map(|member| member.pos)
217                .collect(),
218            source: bloq_circuit::FlowError::Composition(
219                "cyclic same-layer boundary-flow dependency".into(),
220            ),
221        });
222    }
223    nodes.copy_from_slice(&order);
224    Ok(())
225}

227/// Compose the already pinned proxy's boundary flows with their allocated ids.
228fn attach_component_detectors(
229    nodes: &[NodeIndex],
230    context: &mut BloqLowerContext<'_>,
231) -> Result<(), CompileError> {
232    let mut state = DetectorLowerState {
233        detectors: Vec::new(),
234        restarts: Vec::new(),
235        engine: DetectorFlowEngine::new(),
236    };
237    for &node in nodes {
238        let node_context = context.detector_node_context(node);
239        append_node_boundary_flows(&mut state, &node_context)?;
240    }
241    let remaining = state
242        .engine
243        .finish()
244        .map_err(|error| CompileError::DetectorFlowCompositionFailed { source: error })?;
245    debug_assert!(
246        remaining.is_empty(),
247        "completed detectors are drained into their closing node"
248    );
249    for (node, detectors) in state.detectors {
250        debug_assert!(context.node_detectors[node].is_empty());
251        context.node_detectors[node] = detectors;
252    }
253    for (node, restarts) in state.restarts {
254        debug_assert!(context.node_restarts[node].is_empty());
255        context.node_restarts[node] = restarts;
256    }
257    Ok(())
258}

260/// Append a node's boundary flows as one parallel group. Every member consumes
261/// flows from earlier groups before any member publishes its outputs.
262///
263/// A top-level T region contributes only its escape template. Its cultivation-side
264/// inputs were already closed inside the region and can be unmatched here.
265/// A top-open loop's residual arrives as a creator flow, resolved by the
266/// backend's post-`REPEAT` lookback.
267fn append_node_boundary_flows<'c>(
268    state: &mut DetectorLowerState<'c>,
269    node: &DetectorNodeContext<'c>,
270) -> Result<(), CompileError> {
271    if node.skip_unmatched_inputs {
272        let [escape] = node.sources.as_slice() else {
273            unreachable!("T plan node carries exactly its escape instance");
274        };
275        let flows = &escape.template.program_template.boundary_flows;
276        if !flows.is_empty() {
277            state
278                .engine
279                .append_group_skipping_unmatched_inputs(
280                    &[OffsetFlows::new(flows, escape.offset)],
281                    |_, &measurement| {
282                        DetectorTerm::Measurement(InstanceMeasurement {
283                            instance: escape.instance_id,
284                            measurement,
285                        })
286                    },
287                )
288                .map_err(|error| node_flow_error(node, error))?;
289        }
290    } else {
291        let sources: Vec<_> = node
292            .sources
293            .iter()
294            .filter(|source| !source.template.program_template.boundary_flows.is_empty())
295            .collect();
296        let parts: Vec<_> = sources
297            .iter()
298            .map(|source| {
299                OffsetFlows::new(
300                    &source.template.program_template.boundary_flows,
301                    source.offset,
302                )
303            })
304            .collect();
305        state
306            .engine
307            .append_group_allowing_stabilizer_transition(&parts, |part, &measurement| {
308                DetectorTerm::Measurement(InstanceMeasurement {
309                    instance: sources[part].instance_id,
310                    measurement,
311                })
312            })
313            .map_err(|error| node_flow_error(node, error))?;
314    }
315    // Classification (Discard drop / Restart -> `NodeRestart` / else
316    // `NodeDetector`) is shared with the U11 edit recompose so the marker
317    // discipline cannot drift between compile time and post-hoc seam edits.
318    state.drain_node_chains(node.plan_node);
319    Ok(())
320}

322/// Compose the seam between template instances of one `RepeatUntilSuccess`
323/// body: sources in emission order — cultivation, then escape. All
324/// completed chains here are the cultivation template's restart-flagged open
325/// Steane stabilizers closed by the escape template's first merge round, so
326/// they land as `NodeRestart`s (plus any plain detectors a future body shape
327/// might close). Deliberately no `engine.finish()`: the escape template's
328/// end-face creators legitimately stay open — the +Z neighbour closes them at
329/// top level (with the T source in skip mode dropping the Steane consumers
330/// this pass already resolved).
331pub(super) fn compose_instance_seam(
332    sources: &[(&LoweringTemplate, IVec2, TemplateInstanceId)],
333) -> Result<(Vec<NodeDetector>, Vec<NodeRestart>), CompileError> {
334    let mut state = DetectorLowerState {
335        detectors: Vec::new(),
336        restarts: Vec::new(),
337        engine: DetectorFlowEngine::new(),
338    };
339    for &(template, offset, instance_id) in sources {
340        let node = DetectorNodeContext {
341            plan_node: 0,
342            layer: 0,
343            members: &[],
344            skip_unmatched_inputs: false,
345            sources: vec![MemberFlowSource {
346                template,
347                offset,
348                instance_id,
349            }],
350        };
351        append_node_boundary_flows(&mut state, &node)?;
352    }
353    Ok((
354        state
355            .detectors
356            .into_iter()
357            .flat_map(|(_, detectors)| detectors)
358            .collect(),
359        state
360            .restarts
361            .into_iter()
362            .flat_map(|(_, restarts)| restarts)
363            .collect(),
364    ))
365}

367fn node_flow_error(node: &DetectorNodeContext<'_>, error: bloq_circuit::FlowError) -> CompileError {
368    CompileError::NodeFlowCompositionFailed {
369        layer: node.layer,
370        members: node.members.iter().map(|member| member.pos).collect(),
371        source: error,
372    }

376mod tests {
377    use bloq_circuit::{
378        Chunk, ChunkOrLoop, CoordCircuit, DetectorParity, Flow, FlowMarker, Pauli, PauliBasis,

386    /// One-measurement chunk carrying `flow`.
387    fn chunk_with_flow(coord: glam::IVec2, flow: Flow) -> ChunkOrLoop {
388        let mut circuit = CoordCircuit::new();
389        circuit.measure(PauliBasis::Z, [coord]);
390        ChunkOrLoop::Single(Box::new(Chunk {
391            circuit,
392            flows: vec![flow],
393        }))
394    }
395
396    fn template_with_flow(coord: glam::IVec2, flow: Flow) -> LoweringTemplate {
397        LoweringTemplate::from_chunks(vec![chunk_with_flow(coord, flow)], ObservableGateway::new())
398            .expect("single-flow template compiles")
399    }
400
401    /// The cross-template restart composition: a restart-flagged chain left
402    /// open by one template instance and closed by the next fuses at the seam
403    /// into a `NodeRestart` carrying both instances' measurements — never a
404    /// `NodeDetector`.
405    #[test]
406    fn cross_instance_restart_chain_becomes_node_restart() {
407        let seam = PauliMap::from_iter([(ivec2(0, 0), Pauli::Z)]);
408        let lower_template = template_with_flow(
409            ivec2(0, 0),
410            Flow::new(PauliMap::empty(), seam.clone())
411                .with_measurements([0])
412                .with_marker(FlowMarker::Restart),
413        );
414        let upper_template = template_with_flow(
415            ivec2(0, 0),
416            Flow::new(seam, PauliMap::empty()).with_measurements([0]),
417        );
418
419        let node_context = |template, plan_node, instance| DetectorNodeContext {
420            plan_node,
421            layer: 0,
422            members: &[],
423            skip_unmatched_inputs: false,
424            sources: vec![MemberFlowSource {
425                template,
426                offset: IVec2::ZERO,
427                instance_id: TemplateInstanceId(instance),
428            }],
429        };
430        let mut state = DetectorLowerState {
431            detectors: Vec::new(),
432            restarts: Vec::new(),
433            engine: DetectorFlowEngine::new(),
434        };
435        append_node_boundary_flows(&mut state, &node_context(&lower_template, 0, 0))
436            .expect("lower node appends");
437        append_node_boundary_flows(&mut state, &node_context(&upper_template, 1, 1))
438            .expect("upper node closes the seam");
439        assert!(
440            state.engine.finish().expect("no open flows").is_empty(),
441            "the fused chain drains at the closing node"
442        );
443
444        assert!(
445            state.detectors.is_empty(),
446            "no detector for a restart chain"
447        );
448        assert_eq!(state.restarts.len(), 1);
449        // The chain completes at the upper node, so the restart attaches there.
450        let (plan_node, restarts) = &state.restarts[0];
451        assert_eq!(*plan_node, 1);
452        assert_eq!(restarts.len(), 1);
453        assert_eq!(
454            restarts[0].parity,
455            DetectorParity::from_measurements([
456                InstanceMeasurement {
457                    instance: TemplateInstanceId(0),
458                    measurement: 0,
459                },
460                InstanceMeasurement {
461                    instance: TemplateInstanceId(1),
462                    measurement: 0,
463                },
464            ])
465        );
466    }

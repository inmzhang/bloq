12/// Build the `RepeatUntilSuccess` region for a T block: cultivation
13/// and escape quantum nodes joined by the intra-body seam (whose cross-template
14/// post-selected parities become the escape node's `NodeRestart`s), plus the two
15/// logical observables. Their predicted flips feed an explicit restart predicate;
16/// physical postselection parities remain attached to the quantum stages.
17pub(super) fn build_t_region(
18    pos: IVec3,
19    cultivation: PlacedInstance,
20    escape: PlacedInstance,
21    observable_base: u32,
22    remapper: &mut TemplateRemapper<'_>,
23) -> Result<RegionNode, CompileError> {
24    // The cultivation template's six restart-flagged open Steane chains close
25    // against the escape template's first merge round: cross-template restart
26    // parities (U17b §4), attached to the escape node inside the body.
27    let (seam_detectors, seam_restarts) = detector::compose_instance_seam(&[
28        (
29            &remapper.pool()[cultivation.template],
30            cultivation.offset,
31            cultivation.id,
32        ),
33        (&remapper.pool()[escape.template], escape.offset, escape.id),
34    ])?;
35
36    let mut body = SubGraph::new();
37    let mut cultivation_node = BloqNode::from_members(vec![SourceBlockRef { pos }]);
38    cultivation_node
39        .expect_quantum_mut()
40        .instances
41        .push(TemplateInstance::new(
42            cultivation.id,
43            remapper.remap(cultivation.template),
44            cultivation.offset,
45        ));
46    let cultivation_node = body.add_node(cultivation_node);
47
48    let mut escape_node = BloqNode::from_members(vec![SourceBlockRef { pos }]);
49    {
50        let quantum = escape_node.expect_quantum_mut();
51        quantum.instances.push(TemplateInstance::new(
52            escape.id,
53            remapper.remap(escape.template),
54            escape.offset,
55        ));
56        quantum.detectors = seam_detectors;
57        quantum.restarts = seam_restarts;
58    }
59    let escape_node = body.add_node(escape_node);
60    body.add_edge(cultivation_node, escape_node, BloqEdge::quantum(vec![]));
61
62    // GAP observables (U17 §5.1): one per escaped-patch logical, keyed in the
63    // escape gateway by the template-local Pauli at the +Z face. The
64    // Z-keyed entry takes the even slot, the X-keyed the odd one —
65    // deterministic regardless of gateway map order or the dual-family swap.
66    let escape_template = &remapper.pool()[escape.template];
67    let connectivity = crate::Connectivity::ISOLATED.with_pipe(bloq_graph::Direction::ZPLUS);
68    let mut observables = Vec::new();
69    for (slot, pauli) in [(0u32, bloq_graph::Pauli::Z), (1, bloq_graph::Pauli::X)] {
70        let key = crate::block::LocalStabilizer::new(pauli, connectivity);
71        let entry = escape_template
72            .observable_gateway
73            .lookup(key)
74            .expect("escape gateway carries both +Z-face logical entries");
75        let index = observable_base + slot;
76
77        // XOR-dedup the gateway measurements (duplicated records cancel),
78        // mirroring the static observable path's parity fold.
79        let measurements = crate::xor_toggled_sorted(
80            entry
81                .measurements
82                .iter()
83                .flat_map(|chunk| chunk.measurements.iter().copied()),
84        );
85        let generator = NodeProvenance::Generator { ordinal: index };
86        let observable = body.add_node(
87            BloqNode::classical(ClassicalNode::Observable {
88                index: Some(index),
89                measurements: measurements
90                    .into_iter()
91                    .map(|measurement| InstanceMeasurement {
92                        instance: escape.id,
93                        measurement,
94                    })
95                    .collect(),
96                operators: [
97                    (BoundaryFace::Input, &entry.operator_in),
98                    (BoundaryFace::Output, &entry.operator_out),
99                ]
100                .into_iter()
101                .map(|(face, operator)| {
102                    Ok(InstanceBoundaryOperator {
103                        instance: escape.id,
104                        face,
105                        operator: operator.try_translated(escape.offset)?,
106                    })
107                })
108                .collect::<Result<_, CompileError>>()?,
109            })
110            .with_provenance(generator),
111        );
112        body.add_edge(escape_node, observable, BloqEdge::Order);
113        observables.push(observable);
114    }
115
116    let restart = body.add_node(BloqNode::classical(ClassicalNode::Compute {
117        expr: ClassicalExpr::Or(Box::new([ClassicalExpr::In(0), ClassicalExpr::In(1)])),
118    }));
119    for (slot, observable) in observables.into_iter().enumerate() {
120        body.add_edge(observable, restart, BloqEdge::flip(slot as u32));
121    }
122
123    Ok(RegionNode::RepeatUntilSuccess {
124        body,
125        restart_condition: ClassicalExpr::In(0),
126        restart_source: Some(restart.into()),
127    })
128}

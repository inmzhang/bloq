29fn occupancy_pairs(
30    mut events: Vec<OccupancyEvent>,
31    mut allowed: impl FnMut(u32, u32, &[u32], Option<&crate::FxSet<u32>>, QubitRun) -> bool,
32) -> crate::FxSet<(u32, u32)> {
33    events.sort_unstable_by_key(|event| (event.y, event.parity, event.x));
34    let mut active = BTreeMap::<u32, usize>::new();
35    let mut pairs = crate::FxSet::default();
36    let mut users = SmallVec::<[u32; 8]>::new();
37    let mut first = 0;
38    while first < events.len() {
39        let event = events[first];
40        let mut end = first;
41        // Starts and ends at one coordinate change membership together.
42        // Counts retain a slot while another of its placements still covers it.
43        while end < events.len()
44            && (events[end].y, events[end].parity, events[end].x)
45                == (event.y, event.parity, event.x)
46        {
47            let update = events[end];
48            if update.start {
49                *active.entry(update.slot).or_default() += 1;
50            } else {
51                let count = active
52                    .get_mut(&update.slot)
53                    .expect("an ending run is active");
54                *count -= 1;
55                if *count == 0 {
56                    active.remove(&update.slot);
57                }
58            }
59            end += 1;
60        }
61        if end < events.len()
62            && (events[end].y, events[end].parity) == (event.y, event.parity)
63            && active.len() > 1
64        {
65            users.clear();
66            users.extend(active.keys().copied());
67            let user_slots =
68                (users.len() > 8).then(|| users.iter().copied().collect::<crate::FxSet<_>>());
69            let span = QubitRun {
70                y: event.y,
71                parity: event.parity,
72                start: event.x,
73                end: events[end].x,
74            };
75            for pair in users.windows(2) {
76                if allowed(pair[0], pair[1], &users, user_slots.as_ref(), span) {
77                    pairs.insert((pair[0], pair[1]));
78                }
79            }
80        }
81        first = end;
82    }
83    debug_assert!(active.is_empty(), "every footprint run has an end");
84    pairs
85}

87fn neighbor_in_users(
88    neighbors: &[usize],
89    users: &[u32],
90    user_slots: Option<&crate::FxSet<u32>>,
91) -> bool {
92    // Neighbor slots are sorted. Search the short user list when that costs
93    // less than scanning every neighbor (with an inline or hashed lookup).
94    let search_steps = (usize::BITS - neighbors.len().leading_zeros()) as usize;
95    if users.len().saturating_mul(search_steps) < neighbors.len() {
96        users
97            .iter()
98            .any(|&user| neighbors.binary_search(&(user as usize)).is_ok())
99    } else {
100        neighbors.iter().any(|&neighbor| {
101            let neighbor = neighbor as u32;
102            user_slots.map_or_else(
103                || users.contains(&neighbor),
104                |slots| slots.contains(&neighbor),
105            )
106        })
107    }
108}

149#[derive(Clone, Copy)]
150enum LiveDirection {
151    BeforeInputs,
152    AfterOutputs,
153}
154
155impl LiveDirection {
156    fn entry(self, uses: &LiveQubitUses) -> &QubitGuards {
157        match self {
158            Self::BeforeInputs => &uses.inputs,
159            Self::AfterOutputs => &uses.outputs,
160        }
161    }
162
163    fn exit(self, uses: &LiveQubitUses) -> &QubitGuards {
164        match self {
165            Self::BeforeInputs => &uses.outputs,
166            Self::AfterOutputs => &uses.inputs,
167        }
168    }
169
170    /// Orders `(near, far)` along the live state's flow.
171    fn orient<T>(self, near: T, far: T) -> (T, T) {
172        match self {
173            Self::BeforeInputs => (far, near),
174            Self::AfterOutputs => (near, far),
175        }
176    }
177}

179/// Memoized live boundary support of a top-level node.
180fn cache_live_uses(
181    bloq: &Bloq,
182    supports: &mut LiveSupportCache,
183    id: bloq_ir::BloqNodeId,
184) -> Result<(), crate::CompileError> {
185    if let Entry::Vacant(entry) = supports.nodes.entry(id) {
186        let support = if let Some(quantum) = bloq[id].try_quantum() {
187            let guards = instance_guards(bloq.top(), id);
188            let key = quantum
189                .instances
190                .iter()
191                .map(|instance| {
192                    (
193                        instance.template_id,
194                        instance.offset,
195                        guards.get(&instance.id).copied(),
196                    )
197                })
198                .collect::<Vec<_>>();
199            // Time and instance IDs do not change a quantum node's physical support.
200            match supports.placements.entry(key) {
201                Entry::Occupied(entry) => Rc::clone(entry.get()),
202                Entry::Vacant(entry) => {
203                    Rc::clone(entry.insert(Rc::new(quantum_live_uses(bloq, quantum, &guards)?)))
204                }
205            }
206        } else {
207            Rc::new(live_qubit_uses_in_level(bloq, bloq.top(), id, id)?)
208        };
209        entry.insert(support);
210    }
211    Ok(())
212}

214fn instance_guards(
215    level: &bloq_ir::SubGraph,
216    id: bloq_ir::BloqNodeId,
217) -> crate::FxMap<bloq_ir::lowering::TemplateInstanceId, bloq_ir::ValueRef> {
218    let mut guards = crate::FxMap::default();
219    for guard in &level[id].expect_quantum().guards {
220        let value = level
221            .value_inputs(id)
222            .find(|input| input.slot == guard.input)
223            .and_then(bloq_ir::ValueInput::value_ref)
224            .expect("compiled membership input exists");
225        guards.extend(guard.instances.iter().map(|&instance| (instance, value)));
226    }
227    guards
228}

230fn live_qubit_uses_in_level(
231    bloq: &Bloq,
232    level: &bloq_ir::SubGraph,
233    id: bloq_ir::BloqNodeId,
234    owner: bloq_ir::BloqNodeId,
235) -> Result<LiveQubitUses, crate::CompileError> {
236    let mut result = LiveQubitUses::default();
237    if let Some(region) = level[id].try_region() {
238        let activation = level[id].activation.map(|slot| {
239            level
240                .value_inputs(id)
241                .find(|input| input.slot == slot)
242                .and_then(bloq_ir::ValueInput::value_ref)
243                .expect("compiled region activation exists")
244        });
245        for (_, body) in region.bodies() {
246            if body.nodes().any(|(_, node)| guarded_node(node)) {
247                return Err(crate::CompileError::ConditionalRegionLivenessUnsupported {
248                    node: owner,
249                });
250            }
251            let children = body
252                .nodes()
253                .map(|(id, _)| Ok((id, live_qubit_uses_in_level(bloq, body, id, owner)?)))
254                .collect::<Result<crate::FxMap<_, _>, crate::CompileError>>()?;
255            for (&child, support) in &children {
256                for &qubit in support.uses.keys() {
257                    result.uses.entry(qubit).or_default().push(activation);
258                }
259                for (incoming, face, external) in [
260                    (true, &support.inputs, &mut result.inputs),
261                    (false, &support.outputs, &mut result.outputs),
262                ] {
263                    for &qubit in face.keys() {
264                        let internal = if incoming {
265                            body.incoming(child).any(|edge| {
266                                matches!(edge.edge, BloqEdge::Quantum(_))
267                                    && children[&edge.source].outputs.contains_key(&qubit)
268                            })
269                        } else {
270                            body.outgoing(child).any(|edge| {
271                                matches!(edge.edge, BloqEdge::Quantum(_))
272                                    && children[&edge.target].inputs.contains_key(&qubit)
273                            })
274                        };
275                        if !internal {
276                            external.entry(qubit).or_default().push(activation);
277                        }
278                    }
279                }
280            }
281        }
282        return Ok(result);
283    }
284    let Some(quantum) = level[id].try_quantum() else {
285        return Ok(result);
286    };
287    quantum_live_uses(bloq, quantum, &instance_guards(level, id))
288}

290fn quantum_live_uses(
291    bloq: &Bloq,
292    quantum: &bloq_ir::QuantumNode,
293    guards: &crate::FxMap<bloq_ir::lowering::TemplateInstanceId, bloq_ir::ValueRef>,
294) -> Result<LiveQubitUses, crate::CompileError> {
295    let mut result = LiveQubitUses::default();
296    for instance in &quantum.instances {
297        let guard = guards.get(&instance.id).copied();
298        let template = &bloq.templates()[instance.template_id];
299        for &qubit in template.qubits() {
300            let qubit = bloq_circuit::checked_translate_coordinate(qubit, instance.offset)?;
301            result.uses.entry(qubit).or_default().push(guard);
302        }
303        for flow in &template.boundary_flows {
304            for (map, support) in [
305                (&flow.start, &mut result.inputs),
306                (&flow.end, &mut result.outputs),
307            ] {
308                for (&qubit, _) in map.iter() {
309                    let qubit = bloq_circuit::checked_translate_coordinate(qubit, instance.offset)?;
310                    let alternatives = support.entry(qubit).or_default();
311                    if !alternatives.contains(&guard) {
312                        alternatives.push(guard);
313                    }
314                }
315            }
316        }
317    }
318    Ok(result)
319}

321fn guarded_era_boundary(
322    bloq: &Bloq,
323    endpoints: [(bloq_ir::BloqNodeId, bool); 2],
324    qubit: glam::IVec2,
325    require_live: bool,
326    neighbors: &EraNeighbors,
327    supports: &mut LiveSupportCache,
328    predicates: &mut bloq_ir::lowering::PredicateAnalysis<'_>,
329) -> Result<bool, crate::CompileError> {
330    let [(left, left_moving), (right, right_moving)] = endpoints;
331    let outgoing = &neighbors.successors[&left];
332    let incoming = &neighbors.predecessors[&right];
333    for id in [left, right]
334        .into_iter()
335        .chain(outgoing.iter().chain(incoming).copied())
336    {
337        cache_live_uses(bloq, supports, id)?;
338    }
339    let mut continuing = Vec::new();
340    for (endpoint, adjacent, direction) in [
341        (left, outgoing, LiveDirection::AfterOutputs),
342        (right, incoming, LiveDirection::BeforeInputs),
343    ] {
344        let Some(face) = direction.entry(&supports.nodes[&endpoint]).get(&qubit) else {
345            continue;
346        };
347        for next in adjacent {
348            if let Some(other) = direction.exit(&supports.nodes[next]).get(&qubit) {
349                for first in face {
350                    for second in other {
351                        continuing.push(first.iter().chain(second).copied().collect::<Vec<_>>());
352                    }
353                }
354            }
355        }
356    }
357    // A moving endpoint orders only a real teardown or initialization, never transit use.
358    let left_support = &supports.nodes[&left];
359    let right_support = &supports.nodes[&right];
360    for (moving, opposite) in [
361        (left_moving, &left_support.outputs),
362        (right_moving, &right_support.inputs),
363    ] {
364        if moving {
365            continuing.extend(
366                opposite
367                    .get(&qubit)
368                    .into_iter()
369                    .flatten()
370                    .map(|guard| guard.iter().copied().collect()),
371            );
372        }
373    }
374    let left_face = if left_moving {
375        &left_support.inputs
376    } else {
377        &left_support.uses
378    };
379    let right_face = if right_moving {
380        &right_support.outputs
381    } else {
382        &right_support.uses
383    };
384    let Some(left_guards) = left_face.get(&qubit) else {
385        return Ok(false);
386    };
387    let Some(right_guards) = right_face.get(&qubit) else {
388        return Ok(false);
389    };
390    for left_guard in left_guards {
391        for right_guard in right_guards {
392            let mut required = left_guard
393                .iter()
394                .chain(right_guard)
395                .copied()
396                .collect::<Vec<_>>();
397            if !coexists_outside(predicates, left, &required, &continuing)? {
398                continue;
399            }
400            if !require_live {
401                return Ok(true);
402            }
403            for guard in [
404                &left_support.inputs,
405                &left_support.outputs,
406                &right_support.inputs,
407                &right_support.outputs,
408            ]
409            .into_iter()
410            .filter_map(|face| face.get(&qubit))
411            .flatten()
412            {
413                let Some(guard) = guard else {
414                    return Ok(true);
415                };
416                required.push(*guard);
417                let live = coexists_outside(predicates, left, &required, &continuing)?;
418                required.pop();
419                if live {
420                    return Ok(true);
421                }
422            }
423        }
424    }
425    Ok(false)
426}

428/// Whether a candidate pair orders a data-bearing era boundary on some shared qubit.
429fn shares_live_boundary(
430    bloq: &Bloq,
431    endpoints: [(bloq_ir::BloqNodeId, bool); 2],
432    neighbors: &EraNeighbors,
433    supports: &mut LiveSupportCache,
434    predicates: &mut bloq_ir::lowering::PredicateAnalysis<'_>,
435) -> Result<bool, crate::CompileError> {
436    let [(from, _), (to, _)] = endpoints;
437    cache_live_uses(bloq, supports, from)?;
438    cache_live_uses(bloq, supports, to)?;
439    let (a, b) = (&supports.nodes[&from], &supports.nodes[&to]);
440    let mut qubits = a
441        .inputs
442        .keys()
443        .chain(a.outputs.keys())
444        .filter(|q| b.uses.contains_key(*q))
445        .chain(
446            b.inputs
447                .keys()
448                .chain(b.outputs.keys())
449                .filter(|q| a.uses.contains_key(*q)),
450        )
451        .copied()
452        .collect::<Vec<_>>();
453    qubits.sort_unstable_by_key(glam::IVec2::to_array);
454    qubits.dedup();
455    // Shared live support alone is not a witness: that qubit may still
456    // continue through a seam while only reset ancillas justify this pair.
457    for qubit in qubits {
458        if guarded_era_boundary(
459            bloq, endpoints, qubit, true, neighbors, supports, predicates,
460        )? {
461            return Ok(true);
462        }
463    }
464    Ok(false)
465}

490/// Whether `required` can hold while every `excluded` clause fails.
491fn coexists_outside(
492    predicates: &mut bloq_ir::lowering::PredicateAnalysis<'_>,
493    node: bloq_ir::BloqNodeId,
494    required: &[bloq_ir::ValueRef],
495    excluded: &[Vec<bloq_ir::ValueRef>],
496) -> Result<bool, crate::CompileError> {
497    // A required exclusion rules out the domain without lowering its guards.
498    if excluded
499        .iter()
500        .any(|clause| clause.iter().all(|guard| required.contains(guard)))
501    {
502        return Ok(false);
503    }
504    Ok(predicates
505        .coexist_without(required, excluded)
506        .map_err(
507            |source| bloq_ir::BloqValidationError::InvalidInstanceMergeStructure { node, source },
508        )?)
509}

511/// Follow one side of a live era, retaining the guards under which its outer boundary ends.
512#[expect(
513    clippy::too_many_arguments,
514    reason = "one era side shares the caller's support, predicate, and order state"
515)]
516fn walk_live_era(
517    bloq: &Bloq,
518    endpoint: bloq_ir::BloqNodeId,
519    reuser: bloq_ir::BloqNodeId,
520    neighbors: &crate::FxMap<bloq_ir::BloqNodeId, Vec<bloq_ir::BloqNodeId>>,
521    direction: LiveDirection,
522    supports: &mut LiveSupportCache,
523    predicates: &mut bloq_ir::lowering::PredicateAnalysis<'_>,
524    orders: &mut Vec<EraOrder>,
525) -> Result<Vec<EraFrontier>, crate::CompileError> {
526    use bloq_utils::boolean::BooleanDecisionDiagram;
527
528    for id in [endpoint, reuser] {
529        cache_live_uses(bloq, supports, id)?;
530    }
531    let (endpoint_support, reuser_support) = (
532        Rc::clone(&supports.nodes[&endpoint]),
533        Rc::clone(&supports.nodes[&reuser]),
534    );
535    let mut qubits = direction
536        .entry(&endpoint_support)
537        .keys()
538        .filter(|qubit| reuser_support.uses.contains_key(qubit))
539        .copied()
540        .collect::<Vec<_>>();
541    qubits.sort_unstable_by_key(glam::IVec2::to_array);
542    let mut work = BooleanDecisionDiagram::default();
543    let mut frontiers = Vec::new();
544    let coexist = |predicates: &mut bloq_ir::lowering::PredicateAnalysis<'_>,
545                   guards: &[bloq_ir::ValueRef]| {
546        predicates.can_coexist_values(guards).map_err(|source| {
547            bloq_ir::BloqValidationError::InvalidInstanceMergeStructure {
548                node: endpoint,
549                source,
550            }
551        })
552    };
553    for qubit in qubits {
554        let mut pending = Vec::new();
555        for input in &direction.entry(&endpoint_support)[&qubit] {
556            for usage in &reuser_support.uses[&qubit] {
557                pending.push((
558                    endpoint,
559                    input.iter().chain(usage).copied().collect::<Vec<_>>(),
560                ));
561            }
562        }
563        let mut visited = crate::FxSet::default();
564        while let Some((current, mut guards)) = pending.pop() {
565            work.charge(guards.len().saturating_add(1))?;
566            guards.sort_unstable();
567            guards.dedup();
568            if !visited.insert((current, guards.clone())) || !coexist(predicates, &guards)? {
569                continue;
570            }
571            let mut continuing = Vec::new();
572            let current_support = Rc::clone(&supports.nodes[&current]);
573            let entries = direction
574                .entry(&current_support)
575                .get(&qubit)
576                .map_or(&[][..], Vec::as_slice);
577            for &next in neighbors.get(&current).into_iter().flatten() {
578                cache_live_uses(bloq, supports, next)?;
579                let Some(outputs) = direction.exit(&supports.nodes[&next]).get(&qubit) else {
580                    continue;
581                };
582                for input in entries {
583                    for output in outputs {
584                        let clause = input.iter().chain(output).copied().collect::<Vec<_>>();
585                        let mut path_guards = guards.clone();
586                        path_guards.extend(&clause);
587                        work.charge(path_guards.len().saturating_add(1))?;
588                        // Reaching the opposite endpoint continues this same era, not a frontier.
589                        continuing.push(clause);
590                        if !coexist(predicates, &path_guards)? || next == reuser {
591                            continue;
592                        }
593                        let (before, after) = direction.orient(next, reuser);
594                        let (producer, consumer) = direction.orient(endpoint, next);
595                        orders.push(EraOrder {
596                            before,
597                            after,
598                            producer,
599                            consumer,
600                            reuser,
601                            qubit,
602                        });
603                        pending.push((next, path_guards));
604                    }
605                }
606            }
607            if coexists_outside(predicates, endpoint, &guards, &continuing)? {
608                frontiers.push(EraFrontier {
609                    node: current,
610                    qubit,
611                    required: guards,
612                    excluded: continuing,
613                });
614            }
615        }
616    }
617    Ok(frontiers)
618}

620/// Reverse complete eras atomically: the new terminal frontier must precede the old initial one.
621/// The graph-independent orders of an era reversal, in discovery order.
622///
623/// Exploring guards needs only live supports and predicates, so the same
624/// obligations stay valid while unrelated order edges are inserted. Only
625/// [`resolve_era_reversal`] reads graph paths.
626fn plan_live_era_reversal(
627    bloq: &Bloq,
628    from: bloq_ir::BloqNodeId,
629    to: bloq_ir::BloqNodeId,
630    neighbors: &EraNeighbors,
631    supports: &mut LiveSupportCache,
632    predicates: &mut bloq_ir::lowering::PredicateAnalysis<'_>,
633) -> Result<Vec<EraOrder>, crate::CompileError> {
634    let mut orders = Vec::new();
635    let initials = walk_live_era(
636        bloq,
637        from,
638        to,
639        &neighbors.predecessors,
640        LiveDirection::BeforeInputs,
641        supports,
642        predicates,
643        &mut orders,
644    )?;
645    let finals = walk_live_era(
646        bloq,
647        to,
648        from,
649        &neighbors.successors,
650        LiveDirection::AfterOutputs,
651        supports,
652        predicates,
653        &mut orders,
654    )?;
655    let mut terminals: crate::FxMap<_, Vec<_>> = crate::FxMap::default();
656    for terminal in &finals {
657        terminals.entry(terminal.qubit).or_default().push(terminal);
658    }
659    let mut work = bloq_utils::boolean::BooleanDecisionDiagram::default();
660    let mut excluded = Vec::new();
661    for initial in &initials {
662        for terminal in terminals.get(&initial.qubit).into_iter().flatten() {
663            work.charge(1)?;
664            if initial.node == terminal.node {
665                continue;
666            }
667            let required = initial
668                .required
669                .iter()
670                .chain(&terminal.required)
671                .copied()
672                .collect::<Vec<_>>();
673            excluded.clear();
674            excluded.extend(initial.excluded.iter().chain(&terminal.excluded).cloned());
675            work.charge(required.len().saturating_add(excluded.len()))?;
676            if !coexists_outside(predicates, from, &required, &excluded)? {
677                continue;
678            }
679            orders.push(EraOrder {
680                before: terminal.node,
681                after: initial.node,
682                producer: to,
683                consumer: terminal.node,
684                reuser: initial.node,
685                qubit: initial.qubit,
686            });
687        }
688    }
689    Ok(orders)
690}

692/// Check a reversal plan against the current graph and collect its missing edges.
693fn resolve_era_reversal(
694    bloq: &Bloq,
695    plan: &[EraOrder],
696    scratch: &mut bloq_ir::PathScratch,
697) -> Result<EraReversal, crate::CompileError> {
698    let mut paths = crate::FxMap::default();
699    let mut has_path = |from, to| {
700        *paths
701            .entry((from, to))
702            .or_insert_with(|| bloq.has_path_backwards_with_scratch(from, to, scratch))
703    };
704    let mut edges = crate::FxSet::default();
705    for order in plan {
706        if has_path(order.after, order.before) {
707            return Err(crate::CompileError::LiveQubitReuse {
708                producer: order.producer,
709                consumer: order.consumer,
710                reuser: order.reuser,
711                qubit: order.qubit,
712            });
713        }
714        if !has_path(order.before, order.after) {
715            edges.insert((order.before, order.after));
716        }
717    }
718    let mut edges = edges.into_iter().collect::<Vec<_>>();
719    edges.sort_unstable();
720    // One new edge closes a cycle only through the reverse path rejected above.
721    Ok(if edges.len() < 2 || bloq.is_acyclic_with_edges(&edges) {
722        EraReversal::Feasible(edges)
723    } else {
724        EraReversal::Cyclic
725    })
726}

728/// The live reuse that a cyclic reversal plan cannot schedule.
729///
730/// Blames the first order, in discovery order, whose edge closes a cycle with
731/// the graph and the plan's earlier edges.
732fn cyclic_reversal_error(
733    bloq: &Bloq,
734    plan: &[EraOrder],
735    scratch: &mut bloq_ir::PathScratch,
736) -> crate::CompileError {
737    let mut edges = Vec::new();
738    for order in plan {
739        let edge = (order.before, order.after);
740        if edges.contains(&edge) || bloq.has_path_backwards_with_scratch(edge.0, edge.1, scratch) {
741            continue;
742        }
743        edges.push(edge);
744        if !bloq.is_acyclic_with_edges(&edges) {
745            return crate::CompileError::LiveQubitReuse {
746                producer: order.producer,
747                consumer: order.consumer,
748                reuser: order.reuser,
749                qubit: order.qubit,
750            };
751        }
752    }
753    bloq_ir::BloqValidationError::CyclicGraph.into()
754}

756/// Induce occupancy `Order` edges between physical-qubit eras: a terminating
757/// use must precede a later source on the same qubit — teardown then reinit.
758/// Other qubits may continue through the same nodes' quantum seams.
759/// Detector/observable composition has already run over `Temporal` seams
760/// only, so these pure-ordering edges only constrain emission order — they
761/// never change emitted detectors or observables.
762///
763/// Only era boundaries (terminal → source) are ordered. Moving patches use
764/// intervals bounded by their members and owned temporal seam endpoints; their
765/// input-only/output-only live support identifies teardown/initialization.
766/// Transit and reset-only qubits do not establish moving era boundaries.
767///
768/// The Clifford proxy was the first emitter to need this: a T column's
769/// source node hovers a layer gap above the previous era's terminating block
770/// on the same footprint, and a bare toposort emitted the later era's port
771/// MPP before the earlier era's readout of the same qubits.
772pub(super) fn induce_occupancy_order_edges(bloq: &mut Bloq) -> Result<(), crate::CompileError> {
773    use bloq_ir::{BloqEdgeRef, BloqNodeId};
774
775    struct Footprint {
776        id: BloqNodeId,
777        z_min: i64,
778        z_max: i64,
779        placements: SmallVec<[(bloq_ir::TemplateId, glam::IVec2); 4]>,
780        moving: bool,
781    }
782
783    /// Seam-probe direction, and the neighbour a seam edge reaches in it.
784    #[derive(Clone, Copy, PartialEq)]
785    enum Dir {
786        Incoming,
787        Outgoing,
788    }
789    fn seam_edges<'a>(
790        bloq: &'a Bloq,
791        of: BloqNodeId,
792        direction: Dir,
793    ) -> impl Iterator<Item = BloqEdgeRef<'a>> {
794        let (incoming, outgoing) = match direction {
795            Dir::Incoming => (Some(bloq.incoming(of)), None),
796            Dir::Outgoing => (None, Some(bloq.outgoing(of))),
797        };
798        incoming
799            .into_iter()
800            .flatten()
801            .chain(outgoing.into_iter().flatten())
802    }
803
804    // A region occupies its bodies' members. This keeps its block worldlines
805    // ordered across eras even though the region node carries no members.
806    // Virtual spatial Ports also occupy qubits, just before/after their source
807    // layer. Omitting them lets a later input reset a still-live earlier patch.
808    // Pipe nodes still carry no footprint.
809    fn collect_member_positions(node: &BloqNode, positions: &mut Vec<(glam::IVec3, i64)>) {
810        positions.extend(
811            node.block_members()
812                .iter()
813                .map(|m| (m.pos, 2 * i64::from(m.pos.z))),
814        );
815        if let bloq_ir::NodeProvenance::SpatialPortSubstitution { source, .. } = node.provenance {
816            positions.push((source, node.layer()));
817        }
818        if let Some(region) = node.try_region() {
819            for (_, body) in region.bodies() {
820                for (_, body_node) in body.nodes() {
821                    collect_member_positions(body_node, positions);
822                }
823            }
824        }
825    }
826
827    // Keep first-appearance order in `placements`, including across region bodies.
828    fn collect_footprint_placements(
829        bloq: &Bloq,
830        node: &BloqNode,
831        placements: &mut SmallVec<[(bloq_ir::TemplateId, glam::IVec2); 4]>,
832        seen: &mut Option<crate::FxSet<(bloq_ir::TemplateId, glam::IVec2)>>,
833    ) {
834        if let Some(region) = node.try_region() {
835            for (_, body) in region.bodies() {
836                for (_, body_node) in body.nodes() {
837                    collect_footprint_placements(bloq, body_node, placements, seen);
838                }
839            }
840            return;
841        }
842        let Some(quantum) = node.try_quantum() else {
843            return;
844        };
845        for instance in &quantum.instances {
846            let placement = (instance.template_id, instance.offset);
847            if bloq.templates().get(instance.template_id).is_none() {
848                continue;
849            }
850            if let Some(seen) = seen {
851                if seen.insert(placement) {
852                    placements.push(placement);
853                }
854            } else if !placements.contains(&placement) {
855                if placements.len() == 4 {
856                    *seen = Some(placements.iter().copied().chain([placement]).collect());
857                }
858                placements.push(placement);
859            }
860        }
861    }
862
863    let mut footprints = Vec::new();
864    let mut positions = Vec::new();
865    for (id, node) in bloq.nodes() {
866        positions.clear();
867        collect_member_positions(node, &mut positions);
868        let Some(mut z_min) = positions.iter().map(|(_, layer)| *layer).min() else {
869            continue;
870        };
871        let mut z_max = positions
872            .iter()
873            .map(|(_, layer)| *layer)
874            .max()
875            .expect("z_min proved the member list is non-empty");
876        let endpoints = |direction| {
877            seam_edges(bloq, id, direction)
878                .filter_map(|edge| match edge.edge {
879                    BloqEdge::Quantum(edge) => Some(edge.pipes.as_slice()),
880                    _ => None,
881                })
882                .flatten()
883                .map(move |seam| match direction {
884                    Dir::Incoming => seam.pipe.dst,
885                    Dir::Outgoing => seam.pipe.src,
886                })
887        };
888        let endpoints: SmallVec<[glam::IVec3; 4]> = endpoints(Dir::Incoming)
889            .chain(endpoints(Dir::Outgoing))
890            .collect();
891        let moving = !endpoints.is_empty() && {
892            let member_columns: crate::FxSet<_> =
893                positions.iter().map(|(p, _)| p.truncate()).collect();
894            endpoints
895                .iter()
896                .any(|p| !member_columns.contains(&p.truncate()))
897        };
898        if moving {
899            // The displaced output face can outlive the block's origin layer.
900            for endpoint in endpoints {
901                z_min = z_min.min(2 * i64::from(endpoint.z));
902                z_max = z_max.max(2 * i64::from(endpoint.z));
903            }
904        }
905        let mut placements = SmallVec::new();
906        collect_footprint_placements(bloq, node, &mut placements, &mut None);
907        footprints.push(Footprint {
908            id,
909            z_min,
910            z_max,
911            placements,
912            moving,
913        });
914    }
915    // Active slots inherit this order, avoiding one sort per interval.
916    footprints.sort_unstable_by_key(|footprint| (footprint.z_min, footprint.id.0));
917
918    // Per qubit, order its users by z and constrain each *consecutive* pair:
919    // teardown must precede the next era's reinit. A pair already ordered
920    // through the IR (a continuous piped column, possibly routing through
921    // other columns' seams) needs no edge; what this catches is footprint
922    // reuse across an era gap with *no* connecting path — e.g. a T column
923    // whose port/region source hovers two layers above the previous era's
924    // terminating measurement (the U17g proxy hit this first), or a
925    // selective measurement freeing a qubit that an adjacent era's merged
926    // patch reinits (the CCZ factories: the merged patch has quantum seams
927    // from *other* columns, so any-seam probes wrongly treat it as ordered).
928    //
929    // Both endpoints are judged per qubit: `lower` must have no outgoing
930    // quantum seam carrying *this* qubit (its worldline ends there — a
931    // node-level any-seam probe wrongly treats a merged patch that measures
932    // one column out while others continue as mid-column), and `upper` must
933    // have no incoming quantum seam carrying it (its worldline starts there).
934    // The CCZ factories hit both: their merged patches carry seams for other
935    // columns across every era boundary.
936    //
937    // Moving patches include transit qubits. Their era boundaries require
938    // actual live input/output support as well as disjoint source intervals.
939    let slot_by_id: crate::FxMap<BloqNodeId, usize> = footprints
940        .iter()
941        .enumerate()
942        .map(|(slot, footprint)| (footprint.id, slot))
943        .collect();
944    let mut template_runs = crate::FxMap::default();
945    let mut events = Vec::new();
946    for (slot, footprint) in footprints.iter().enumerate() {
947        #[cfg(debug_assertions)]
948        let mut qubits = crate::FxSet::default();
949        for &(template_id, offset) in &footprint.placements {
950            let template = &bloq.templates()[template_id];
951            let local = template_runs
952                .entry(template_id)
953                .or_insert_with(|| template_qubit_runs(template.qubits()));
954            for &run in local.iter() {
955                let run = match run.translated(offset) {
956                    Ok(run) => run,
957                    Err(_) => {
958                        // Run order differs from template qubit order. Earlier
959                        // placements passed their endpoint checks, so re-scan
960                        // only this placement to retain the first old error.
961                        let error = template
962                            .qubits()
963                            .iter()
964                            .find_map(|&qubit| {
965                                bloq_circuit::checked_translate_coordinate(qubit, offset).err()
966                            })
967                            .expect("a run endpoint overflowed");
968                        return Err(error.into());
969                    }
970                };
971                append_run_events(&mut events, slot as u32, run);
972                #[cfg(debug_assertions)]
973                for x in run.start..run.end {
974                    qubits.insert(glam::IVec2::new(
975                        (2 * x + i64::from(run.parity)) as i32,
976                        run.y,
977                    ));
978                }
979            }
980        }
981        #[cfg(debug_assertions)]
982        debug_assert_eq!(
983            qubits,
984            bloq.node_qubits(
985                bloq.node(footprint.id)
986                    .expect("footprint node still exists")
987            )?,
988            "run collection preserves the node's qubit set"
989        );
990    }
991    if events.is_empty() {
992        return Ok(());
993    }
994    drop(template_runs);
995    // Per slot and direction, the footprint slots of its quantum neighbours
996    // (looking through member-less pipe nodes, which carry no footprint).
997    // Resolved once so the per-(qubit, pair) probe below never walks edges.
998    let neighbor_slots = |direction: Dir| -> Vec<Vec<usize>> {
999        let quantum_neighbors = |of: BloqNodeId| {
1000            seam_edges(bloq, of, direction)
1001                .filter(|edge| matches!(edge.edge, BloqEdge::Quantum(_)))
1002                .map(move |edge| match direction {
1003                    Dir::Incoming => edge.source,
1004                    Dir::Outgoing => edge.target,
1005                })
1006        };
1007        footprints
1008            .iter()
1009            .map(|footprint| {
1010                let mut slots = Vec::new();
1011                for next in quantum_neighbors(footprint.id) {
1012                    match slot_by_id.get(&next) {
1013                        Some(&slot) => slots.push(slot),
1014                        None => slots.extend(
1015                            quantum_neighbors(next).filter_map(|hop| slot_by_id.get(&hop).copied()),
1016                        ),
1017                    }
1018                }
1019                slots.sort_unstable();
1020                slots.dedup();
1021                slots
1022            })
1023            .collect()
1024    };
1025    let outgoing_neighbor_slots = neighbor_slots(Dir::Outgoing);
1026    let incoming_neighbor_slots = neighbor_slots(Dir::Incoming);
1027    // Whether the current qubit's worldline continues across a quantum seam of
1028    // the slot's node in `direction`: one of the quantum-neighbour footprint
1029    // slots also appears in the qubit's inverse user list.
1030    let worldline_continues =
1031        |slot: usize, users: &[u32], user_slots: Option<&crate::FxSet<u32>>, direction: Dir| {
1032            let neighbors = match direction {
1033                Dir::Incoming => &incoming_neighbor_slots[slot],
1034                Dir::Outgoing => &outgoing_neighbor_slots[slot],
1035            };
1036            neighbor_in_users(neighbors, users, user_slots)
1037        };
1038    // Membership is constant between run endpoints, so the same qubit filters
1039    // run once per interval. Sorted slots retain the old per-qubit user order;
1040    // duplicate slots only added self-windows, which always failed the z-gap.
1041    let conditional_members = footprints
1042        .iter()
1043        .map(|footprint| conditional_footprint(&bloq[footprint.id]))
1044        .collect::<Vec<_>>();
1045    let geometric_pair = |lower_slot: u32, upper_slot: u32| {
1046        let (lower_slot, upper_slot) = (lower_slot as usize, upper_slot as usize);
1047        let (lower, upper) = (&footprints[lower_slot], &footprints[upper_slot]);
1048        lower.z_max < upper.z_min
1049    };
1050    let neighbor_ids = |slots: &[Vec<usize>]| {
1051        footprints
1052            .iter()
1053            .enumerate()
1054            .map(|(slot, footprint)| {
1055                (
1056                    footprint.id,
1057                    slots[slot]
1058                        .iter()
1059                        .map(|&slot| footprints[slot].id)
1060                        .collect(),
1061                )
1062            })
1063            .collect()
1064    };
1065    let neighbors = EraNeighbors {
1066        predecessors: neighbor_ids(&incoming_neighbor_slots),
1067        successors: neighbor_ids(&outgoing_neighbor_slots),
1068    };
1069    let mut boundary_checks = Vec::new();
1070    // Conditional users can disappear; moving users may only touch transit qubits.
1071    let transparent =
1072        |slot: u32| conditional_members[slot as usize] || footprints[slot as usize].moving;
1073    let mut candidate_slots = occupancy_pairs(events, |lower, upper, users, user_slots, span| {
1074        let needs_boundary_check = users.iter().any(|&slot| transparent(slot));
1075        if needs_boundary_check && lower == users[0] {
1076            // Look through transparent runs so none hides a later era boundary.
1077            for (index, &lower) in users.iter().enumerate() {
1078                for upper_index in index + 1..users.len() {
1079                    if upper_index > index + 1 && !transparent(users[upper_index - 1]) {
1080                        break;
1081                    }
1082                    let upper = users[upper_index];
1083                    if geometric_pair(lower, upper) {
1084                        boundary_checks.push((lower, upper, span));
1085                    }
1086                }
1087            }
1088        }
1089        !needs_boundary_check
1090            && geometric_pair(lower, upper)
1091            && !worldline_continues(lower as usize, users, user_slots, Dir::Outgoing)
1092            && !worldline_continues(upper as usize, users, user_slots, Dir::Incoming)
1093    });
1094    let mut live_supports = LiveSupportCache::default();
1095    let mut predicates = bloq_ir::lowering::PredicateAnalysis::new(bloq.top());
1096    for (lower, upper, span) in boundary_checks {
1097        if candidate_slots.contains(&(lower, upper)) {
1098            continue;
1099        }
1100        for x in span.start..span.end {
1101            let qubit = glam::IVec2::new((2 * x + i64::from(span.parity)) as i32, span.y);
1102            if guarded_era_boundary(
1103                bloq,
1104                [lower, upper].map(|slot| {
1105                    let footprint = &footprints[slot as usize];
1106                    (footprint.id, footprint.moving)
1107                }),
1108                qubit,
1109                false,
1110                &neighbors,
1111                &mut live_supports,
1112                &mut predicates,
1113            )? {
1114                candidate_slots.insert((lower, upper));
1115                break;
1116            }
1117        }
1118    }
1119
1120    /// Whether an edge `from -> to` already exists. An existing edge appears
1121    /// in both `outgoing(from)` and `incoming(to)`, so scanning the two lists
1122    /// in lockstep and stopping as soon as either runs dry costs
1123    /// O(min(outdeg(from), indeg(to))) instead of a full out-list walk —
1124    /// `from` nodes can have enormous out-degree.
1125    fn has_edge(bloq: &Bloq, from: BloqNodeId, to: BloqNodeId) -> bool {
1126        let mut outgoing = bloq.outgoing(from);
1127        let mut incoming = bloq.incoming(to);
1128        loop {
1129            match (outgoing.next(), incoming.next()) {
1130                (Some(edge), _) if edge.target == to => return true,
1131                (_, Some(edge)) if edge.source == from => return true,
1132                (Some(_), Some(_)) => {}
1133                _ => return false,
1134            }
1135        }
1136    }
1137
1138    // Initial-T control flow can reverse source time; certify the candidate union.
1139    let mut candidate_pairs: Vec<_> = candidate_slots
1140        .into_iter()
1141        .map(|(lower, upper)| (footprints[lower as usize].id, footprints[upper as usize].id))
1142        .collect();
1143    candidate_pairs.retain(|&(from, to)| !has_edge(bloq, from, to));
1144    let initial_order = bloq
1145        .deterministic_emit_order()
1146        .expect("lowered IR is acyclic before occupancy ordering");
1147    let mut rank =
1148        vec![0usize; initial_order.iter().map(|id| id.0).max().unwrap_or(0) as usize + 1];
1149    for (index, id) in initial_order.into_iter().enumerate() {
1150        rank[id.0 as usize] = index;
1151    }
1152    // An acyclic union certifies every candidate and every insertion prefix.
1153    // Check it once instead of walking the graph for each backward candidate.
1154    // A cyclic union needs individual ordering decisions below.
1155    let batch_acyclic = candidate_pairs
1156        .iter()
1157        .all(|(from, to)| rank[from.0 as usize] < rank[to.0 as usize])
1158        || bloq.is_acyclic_with_edges(&candidate_pairs);
1159    // Reset-only overlaps are preferences; settle data-bearing boundaries first.
1160    let mut pairs = Vec::with_capacity(candidate_pairs.len());
1161    let mut scratch = bloq.path_scratch();
1162    let mut reversals = crate::FxMap::default();
1163    for (from, to) in candidate_pairs {
1164        let live = !batch_acyclic
1165            && (bloq[from].try_region().is_some()
1166                || bloq[to].try_region().is_some()
1167                || shares_live_boundary(
1168                    bloq,
1169                    [from, to].map(|id| (id, footprints[slot_by_id[&id]].moving)),
1170                    &neighbors,
1171                    &mut live_supports,
1172                    &mut predicates,
1173                )?);
1174        let forward = rank[from.0 as usize] < rank[to.0 as usize];
1175        // Prove required orientations before committing unrelated source-time preferences.
1176        let reversal_blocked = if live && !forward {
1177            let plan = plan_live_era_reversal(
1178                bloq,
1179                from,
1180                to,
1181                &neighbors,
1182                &mut live_supports,
1183                &mut predicates,
1184            )?;
1185            let blocked = match resolve_era_reversal(bloq, &plan, &mut scratch) {
1186                Ok(EraReversal::Feasible(_)) => false,
1187                Ok(EraReversal::Cyclic) | Err(crate::CompileError::LiveQubitReuse { .. }) => true,
1188                Err(error) => return Err(error),
1189            };
1190            reversals.insert((from, to), plan);
1191            blocked
1192        } else {
1193            false
1194        };
1195        pairs.push((!live, !reversal_blocked, !forward, from, to));
1196    }
1197    pairs.sort_unstable();
1198    // Forward edges cannot close a cycle until a backward edge is inserted.
1199    let mut ranked = true;
1200    for (_, _, backward, from, to) in pairs {
1201        if has_edge(bloq, from, to) {
1202            continue;
1203        }
1204        if batch_acyclic
1205            || (ranked && !backward)
1206            || !bloq.has_path_backwards_with_scratch(to, from, &mut scratch)
1207        {
1208            bloq.add_edge(from, to, BloqEdge::Order);
1209            ranked &= !backward;
1210        } else {
1211            ranked = false;
1212            // Inserted edges only change graph paths, so a probed plan stays valid.
1213            let plan = match reversals.remove(&(from, to)) {
1214                Some(plan) => plan,
1215                None => plan_live_era_reversal(
1216                    bloq,
1217                    from,
1218                    to,
1219                    &neighbors,
1220                    &mut live_supports,
1221                    &mut bloq_ir::lowering::PredicateAnalysis::new(bloq.top()),
1222                )?,
1223            };
1224            let EraReversal::Feasible(edges) = resolve_era_reversal(bloq, &plan, &mut scratch)?
1225            else {
1226                return Err(cyclic_reversal_error(bloq, &plan, &mut scratch));
1227            };
1228            for (before, after) in edges {
1229                bloq.add_edge(before, after, BloqEdge::Order);
1230            }
1231        }
1232    }
1233    Ok(())
1234}

1271    fn add_live_node(bloq: &mut Bloq, z: i32, input: bool, output: bool) -> BloqNodeId {
1272        let mut circuit = CoordCircuit::new();
1273        if !input {
1274            circuit.do_gate(GateType::RZ, [ivec2(0, 0)]).unwrap();
1275            circuit.tick();
1276        }
1277        let measurements = if output {
1278            circuit.do_gate(GateType::Z, [ivec2(0, 0)]).unwrap();
1279            Vec::new()
1280        } else {
1281            circuit.measure(bloq_circuit::PauliBasis::Z, [ivec2(0, 0)])
1282        };
1283        let face = |present| {
1284            if present {
1285                PauliMap::from_unique_entries([(ivec2(0, 0), Pauli::Z)])
1286            } else {
1287                PauliMap::empty()
1288            }
1289        };
1290        let template = bloq.add_template(BloqTemplate::with_parts(
1291            circuit,
1292            Vec::new(),
1293            Vec::new(),
1294            vec![Flow::new(face(input), face(output)).with_measurements(measurements)],
1295            Vec::new(),
1296        ));
1297        let mut node = BloqNode::from_members(vec![SourceBlockRef {
1298            pos: ivec3(0, 0, z),
1299        }]);
1300        node.expect_quantum_mut()
1301            .instances
1302            .push(TemplateInstance::new(
1303                TemplateInstanceId(bloq.node_count() as u32),
1304                template,
1305                ivec2(0, 0),
1306            ));
1307        bloq.add_node(node)
1308    }

1939    fn add_moving_patch(bloq: &mut Bloq, z: i32, returning: bool) -> (BloqNodeId, BloqNodeId) {
1940        let [source, target] = if returning {
1941            [ivec2(2, 0), ivec2(0, 0)]
1942        } else {
1943            [ivec2(0, 0), ivec2(2, 0)]
1944        };
1945        let mut circuit = CoordCircuit::new();
1946        circuit.do_gate(GateType::RZ, [target]).unwrap();
1947        circuit.do_gate(GateType::CX, [source, target]).unwrap();
1948        circuit.do_gate(GateType::CX, [target, source]).unwrap();
1949        circuit.measure(bloq_circuit::PauliBasis::Z, [source]);
1950        // A resettable transit ancilla is touched but carries no boundary state.
1951        circuit.do_gate(GateType::RZ, [ivec2(4, 0)]).unwrap();
1952        let face = |qubit| PauliMap::from_unique_entries([(qubit, Pauli::Z)]);
1953        let template = bloq.add_template(BloqTemplate::with_parts(
1954            circuit,
1955            Vec::new(),
1956            Vec::new(),
1957            vec![Flow::new(face(source), face(target))],
1958            Vec::new(),
1959        ));
1960        let mut node = BloqNode::from_members(vec![SourceBlockRef {
1961            pos: (source / 2).extend(z),
1962        }]);
1963        node.expect_quantum_mut()
1964            .instances
1965            .push(TemplateInstance::new(
1966                TemplateInstanceId(bloq.node_count() as u32),
1967                template,
1968                ivec2(0, 0),
1969            ));
1970        let moving = bloq.add_node(node);
1971        let next = add_block_node(bloq, (target / 2).extend(z + 2), target);
1972        bloq.add_edge(
1973            moving,
1974            next,
1975            BloqEdge::quantum(vec![bloq_ir::TemporalPipeRef {
1976                src: (target / 2).extend(z + 1),
1977                dst: (target / 2).extend(z + 2),
1978                hadamard: false,
1979            }]),
1980        );
1981        (moving, next)
1982    }

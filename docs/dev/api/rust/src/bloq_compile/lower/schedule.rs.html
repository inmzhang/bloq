28fn occupancy_pairs(
29    mut events: Vec<OccupancyEvent>,
30    mut allowed: impl FnMut(u32, u32, &[u32], Option<&crate::FxSet<u32>>, QubitRun) -> bool,
31) -> crate::FxSet<(u32, u32)> {
32    events.sort_unstable_by_key(|event| (event.y, event.parity, event.x));
33    let mut active = BTreeMap::<u32, usize>::new();
34    let mut pairs = crate::FxSet::default();
35    let mut users = SmallVec::<[u32; 8]>::new();
36    let mut first = 0;
37    while first < events.len() {
38        let event = events[first];
39        let mut end = first;
40        // Starts and ends at one coordinate change membership together.
41        // Counts retain a slot while another of its placements still covers it.
42        while end < events.len()
43            && (events[end].y, events[end].parity, events[end].x)
44                == (event.y, event.parity, event.x)
45        {
46            let update = events[end];
47            if update.start {
48                *active.entry(update.slot).or_default() += 1;
49            } else {
50                let count = active
51                    .get_mut(&update.slot)
52                    .expect("an ending run is active");
53                *count -= 1;
54                if *count == 0 {
55                    active.remove(&update.slot);
56                }
57            }
58            end += 1;
59        }
60        if end < events.len()
61            && (events[end].y, events[end].parity) == (event.y, event.parity)
62            && active.len() > 1
63        {
64            users.clear();
65            users.extend(active.keys().copied());
66            let user_slots =
67                (users.len() > 8).then(|| users.iter().copied().collect::<crate::FxSet<_>>());
68            let span = QubitRun {
69                y: event.y,
70                parity: event.parity,
71                start: event.x,
72                end: events[end].x,
73            };
74            for pair in users.windows(2) {
75                if allowed(pair[0], pair[1], &users, user_slots.as_ref(), span) {
76                    pairs.insert((pair[0], pair[1]));
77                }
78            }
79        }
80        first = end;
81    }
82    debug_assert!(active.is_empty(), "every footprint run has an end");
83    pairs
84}

86fn neighbor_in_users(
87    neighbors: &[usize],
88    users: &[u32],
89    user_slots: Option<&crate::FxSet<u32>>,
90) -> bool {
91    // Neighbor slots are sorted. Search the short user list when that costs
92    // less than scanning every neighbor (with an inline or hashed lookup).
93    let search_steps = (usize::BITS - neighbors.len().leading_zeros()) as usize;
94    if users.len().saturating_mul(search_steps) < neighbors.len() {
95        users
96            .iter()
97            .any(|&user| neighbors.binary_search(&(user as usize)).is_ok())
98    } else {
99        neighbors.iter().any(|&neighbor| {
100            let neighbor = neighbor as u32;
101            user_slots.map_or_else(
102                || users.contains(&neighbor),
103                |slots| slots.contains(&neighbor),
104            )
105        })
106    }
107}

125    node.activation.is_some()

132    guarded_node(node)

140#[derive(Clone, Copy)]
141enum LiveDirection {
142    BeforeInputs,
143    AfterOutputs,
144}
145
146impl LiveDirection {
147    fn entry(self, uses: &LiveQubitUses) -> &QubitGuards {
148        match self {
149            Self::BeforeInputs => &uses.inputs,
150            Self::AfterOutputs => &uses.outputs,
151        }
152    }
153
154    fn exit(self, uses: &LiveQubitUses) -> &QubitGuards {
155        match self {
156            Self::BeforeInputs => &uses.outputs,
157            Self::AfterOutputs => &uses.inputs,
158        }
159    }
160
161    /// Orders `(near, far)` along the live state's flow.
162    fn orient<T>(self, near: T, far: T) -> (T, T) {
163        match self {
164            Self::BeforeInputs => (far, near),
165            Self::AfterOutputs => (near, far),
166        }
167    }
168}

171fn cache_live_uses(

182fn live_qubit_uses_in_level(
183    bloq: &Bloq,
184    level: &bloq_ir::SubGraph,
185    id: bloq_ir::BloqNodeId,
186    owner: bloq_ir::BloqNodeId,
187) -> Result<LiveQubitUses, crate::CompileError> {
188    let mut result = LiveQubitUses::default();
189    if let Some(region) = level[id].try_region() {
190        let activation = level[id].activation.map(|slot| {
191            level
192                .value_inputs(id)
193                .find(|input| input.slot == slot)
194                .and_then(bloq_ir::ValueInput::value_ref)
195                .expect("compiled region activation exists")
196        });
197        for (_, body) in region.bodies() {
198            if body.nodes().any(|(_, node)| guarded_node(node)) {
199                return Err(crate::CompileError::ConditionalRegionLivenessUnsupported {
200                    node: owner,
201                });
202            }
203            let children = body
204                .nodes()
205                .map(|(id, _)| Ok((id, live_qubit_uses_in_level(bloq, body, id, owner)?)))
206                .collect::<Result<crate::FxMap<_, _>, crate::CompileError>>()?;
207            for (&child, support) in &children {
208                for &qubit in support.uses.keys() {
209                    result.uses.entry(qubit).or_default().push(activation);
210                }
211                for (incoming, face, external) in [
212                    (true, &support.inputs, &mut result.inputs),
213                    (false, &support.outputs, &mut result.outputs),
214                ] {
215                    for &qubit in face.keys() {
216                        let internal = if incoming {
217                            body.incoming(child).any(|edge| {
218                                matches!(edge.edge, BloqEdge::Quantum(_))
219                                    && children[&edge.source].outputs.contains_key(&qubit)
220                            })
221                        } else {
222                            body.outgoing(child).any(|edge| {
223                                matches!(edge.edge, BloqEdge::Quantum(_))
224                                    && children[&edge.target].inputs.contains_key(&qubit)
225                            })
226                        };
227                        if !internal {
228                            external.entry(qubit).or_default().push(activation);
229                        }
230                    }
231                }
232            }
233        }
234        return Ok(result);
235    }
236    let Some(quantum) = level[id].try_quantum() else {
237        return Ok(result);
238    };
239    let mut guards = crate::FxMap::default();
240    for guard in &quantum.guards {
241        let value = level
242            .value_inputs(id)
243            .find(|input| input.slot == guard.input)
244            .and_then(bloq_ir::ValueInput::value_ref)
245            .expect("compiled membership input exists");
246        guards.extend(guard.instances.iter().map(|&instance| (instance, value)));
247    }
248    for instance in &quantum.instances {
249        let guard = guards.get(&instance.id).copied();
250        let template = &bloq.templates()[instance.template_id];
251        for &qubit in template.qubits() {
252            let qubit = bloq_circuit::checked_translate_coordinate(qubit, instance.offset)?;
253            result.uses.entry(qubit).or_default().push(guard);
254        }
255        for flow in &template.boundary_flows {
256            for (map, support) in [
257                (&flow.start, &mut result.inputs),
258                (&flow.end, &mut result.outputs),
259            ] {
260                for (&qubit, _) in map.iter() {
261                    let qubit = bloq_circuit::checked_translate_coordinate(qubit, instance.offset)?;
262                    let alternatives = support.entry(qubit).or_default();
263                    if !alternatives.contains(&guard) {
264                        alternatives.push(guard);
265                    }
266                }
267            }
268        }
269    }
270    Ok(result)
271}

273fn guarded_era_boundary(
274    bloq: &Bloq,
275    left: bloq_ir::BloqNodeId,
276    right: bloq_ir::BloqNodeId,
277    qubit: glam::IVec2,
278    neighbors: &EraNeighbors,
279    supports: &mut crate::FxMap<bloq_ir::BloqNodeId, LiveQubitUses>,
280    predicates: &mut bloq_ir::lowering::PredicateAnalysis<'_>,
281) -> Result<bool, crate::CompileError> {
282    let outgoing = &neighbors.successors[&left];
283    let incoming = &neighbors.predecessors[&right];
284    for id in [left, right]
285        .into_iter()
286        .chain(outgoing.iter().chain(incoming).copied())
287    {
288        cache_live_uses(bloq, supports, id)?;
289    }
290    let mut continuing = Vec::new();
291    for (endpoint, adjacent, direction) in [
292        (left, outgoing, LiveDirection::AfterOutputs),
293        (right, incoming, LiveDirection::BeforeInputs),
294    ] {
295        let Some(face) = direction.entry(&supports[&endpoint]).get(&qubit) else {
296            continue;
297        };
298        for next in adjacent {
299            if let Some(other) = direction.exit(&supports[next]).get(&qubit) {
300                for first in face {
301                    for second in other {
302                        continuing.push(first.iter().chain(second).copied().collect::<Vec<_>>());
303                    }
304                }
305            }
306        }
307    }
308    let Some(left_guards) = supports[&left].uses.get(&qubit) else {
309        return Ok(false);
310    };
311    let Some(right_guards) = supports[&right].uses.get(&qubit) else {
312        return Ok(false);
313    };
314    for left_guard in left_guards {
315        for right_guard in right_guards {
316            let required = left_guard
317                .iter()
318                .chain(right_guard)
319                .copied()
320                .collect::<Vec<_>>();
321            if predicates
322                .coexist_without(&required, &continuing)
323                .map_err(
324                    |source| bloq_ir::BloqValidationError::InvalidInstanceMergeStructure {
325                        node: left,
326                        source,
327                    },
328                )?
329            {
330                return Ok(true);
331            }
332        }
333    }
334    Ok(false)
335}

337/// Source-z reversal may reorder complete eras. It cannot put another use
338/// inside a live era: retained boundary-flow support must survive along its
339/// temporal seams until consumed. Resettable syndrome ancillas impose no
340/// storage constraint when they do not occur on these faces.
341fn plan_live_era_reordering(
342    bloq: &Bloq,
343    endpoint: bloq_ir::BloqNodeId,
344    reuser: bloq_ir::BloqNodeId,
345    neighbors: &crate::FxMap<bloq_ir::BloqNodeId, Vec<bloq_ir::BloqNodeId>>,
346    direction: LiveDirection,
347    supports: &mut crate::FxMap<bloq_ir::BloqNodeId, LiveQubitUses>,
348    scratch: &mut bloq_ir::PathScratch,
349) -> Result<Vec<(bloq_ir::BloqNodeId, bloq_ir::BloqNodeId)>, crate::CompileError> {
350    use bloq_utils::boolean::BooleanDecisionDiagram;
351
352    for id in [endpoint, reuser] {
353        cache_live_uses(bloq, supports, id)?;
354    }
355    let mut qubits = direction
356        .entry(&supports[&endpoint])
357        .keys()
358        .filter(|qubit| supports[&reuser].uses.contains_key(qubit))
359        .copied()
360        .collect::<Vec<_>>();
361    qubits.sort_unstable_by_key(glam::IVec2::to_array);
362    let mut work = BooleanDecisionDiagram::default();
363    let mut ordering = crate::FxSet::default();
364    for qubit in qubits {
365        let mut pending = Vec::new();
366        for input in &direction.entry(&supports[&endpoint])[&qubit] {
367            for usage in &supports[&reuser].uses[&qubit] {
368                pending.push((
369                    endpoint,
370                    input.iter().chain(usage).copied().collect::<Vec<_>>(),
371                ));
372            }
373        }
374        let mut visited = crate::FxSet::default();
375        while let Some((current, mut guards)) = pending.pop() {
376            work.charge(guards.len().saturating_add(1))?;
377            guards.sort_unstable();
378            guards.dedup();
379            if !visited.insert((current, guards.clone())) {
380                continue;
381            }
382            let coexist = |guards: &[bloq_ir::ValueRef]| {
383                bloq.top()
384                    .predicates_can_coexist_values(guards)
385                    .map_err(
386                        |source| bloq_ir::BloqValidationError::InvalidInstanceMergeStructure {
387                            node: endpoint,
388                            source,
389                        },
390                    )
391            };
392            if !coexist(&guards)? {
393                continue;
394            }
395            for &next in neighbors.get(&current).into_iter().flatten() {
396                cache_live_uses(bloq, supports, next)?;
397                let Some(outputs) = direction.exit(&supports[&next]).get(&qubit) else {
398                    continue;
399                };
400                for output in outputs {
401                    let mut path_guards = guards.clone();
402                    path_guards.extend(output);
403                    work.charge(path_guards.len().saturating_add(1))?;
404                    if !coexist(&path_guards)? {
405                        continue;
406                    }
407                    if next == reuser {
408                        continue;
409                    }
410                    let (from, to) = direction.orient(next, reuser);
411                    if bloq.has_path_with_scratch(to, from, scratch) {
412                        let (producer, consumer) = direction.orient(endpoint, next);
413                        return Err(crate::CompileError::LiveQubitReuse {
414                            producer,
415                            consumer,
416                            reuser,
417                            qubit,
418                        });
419                    }
420                    if !bloq.has_path_with_scratch(from, to, scratch) {
421                        ordering.insert((from, to));
422                    }
423                    if let Some(inputs) = direction.entry(&supports[&next]).get(&qubit) {
424                        for input in inputs {
425                            let mut guards = path_guards.clone();
426                            guards.extend(input);
427                            pending.push((next, guards));
428                        }
429                    }
430                }
431            }
432        }
433    }
434    let mut ordering = ordering.into_iter().collect::<Vec<_>>();
435    ordering.sort_unstable();
436    Ok(ordering)
437}

439/// Induce occupancy `Order` edges: a *terminating* node (no
440/// outgoing quantum seam) that shares physical qubits with a later *source*
441/// node (no incoming quantum seam) must emit first — teardown then reinit.
442/// Detector/observable composition has already run over `Temporal` seams
443/// only, so these pure-ordering edges only constrain emission order — they
444/// never change emitted detectors or observables.
445///
446/// Only era boundaries (terminal → source) are ordered. Moving patches touch
447/// transit qubits in other columns, so source z alone cannot establish their
448/// reuse order.
449///
450/// The Clifford proxy was the first emitter to need this: a T column's
451/// source node hovers a layer gap above the previous era's terminating block
452/// on the same footprint, and a bare toposort emitted the later era's port
453/// MPP before the earlier era's readout of the same qubits.
454pub(super) fn induce_occupancy_order_edges(bloq: &mut Bloq) -> Result<(), crate::CompileError> {
455    use bloq_ir::{BloqEdgeRef, BloqNodeId};
456
457    struct Footprint {
458        id: BloqNodeId,
459        z_min: i64,
460        z_max: i64,
461        placements: SmallVec<[(bloq_ir::TemplateId, glam::IVec2); 4]>,
462        member_columns: crate::FxSet<glam::IVec2>,
463    }
464
465    /// Seam-probe direction, and the neighbour a seam edge reaches in it.
466    #[derive(Clone, Copy, PartialEq)]
467    enum Dir {
468        Incoming,
469        Outgoing,
470    }
471    fn seam_edges<'a>(
472        bloq: &'a Bloq,
473        of: BloqNodeId,
474        direction: Dir,
475    ) -> impl Iterator<Item = BloqEdgeRef<'a>> {
476        let (incoming, outgoing) = match direction {
477            Dir::Incoming => (Some(bloq.incoming(of)), None),
478            Dir::Outgoing => (None, Some(bloq.outgoing(of))),
479        };
480        incoming
481            .into_iter()
482            .flatten()
483            .chain(outgoing.into_iter().flatten())
484    }
485
486    // A region occupies its bodies' members. This keeps its block worldlines
487    // ordered across eras even though the region node carries no members.
488    // Virtual spatial Ports also occupy qubits, just before/after their source
489    // layer. Omitting them lets a later input reset a still-live earlier patch.
490    // Pipe nodes still carry no footprint.
491    fn collect_member_positions(node: &BloqNode, positions: &mut Vec<(glam::IVec3, i64)>) {
492        positions.extend(
493            node.block_members()
494                .iter()
495                .map(|m| (m.pos, 2 * i64::from(m.pos.z))),
496        );
497        if let bloq_ir::NodeProvenance::SpatialPortSubstitution { source, .. } = node.provenance {
498            positions.push((source, node.layer()));
499        }
500        if let Some(region) = node.try_region() {
501            for (_, body) in region.bodies() {
502                for (_, body_node) in body.nodes() {
503                    collect_member_positions(body_node, positions);
504                }
505            }
506        }
507    }
508
509    // Keep first-appearance order in `placements`, including across region bodies.
510    fn collect_footprint_placements(
511        bloq: &Bloq,
512        node: &BloqNode,
513        placements: &mut SmallVec<[(bloq_ir::TemplateId, glam::IVec2); 4]>,
514        seen: &mut Option<crate::FxSet<(bloq_ir::TemplateId, glam::IVec2)>>,
515    ) {
516        if let Some(region) = node.try_region() {
517            for (_, body) in region.bodies() {
518                for (_, body_node) in body.nodes() {
519                    collect_footprint_placements(bloq, body_node, placements, seen);
520                }
521            }
522            return;
523        }
524        let Some(quantum) = node.try_quantum() else {
525            return;
526        };
527        for instance in &quantum.instances {
528            let placement = (instance.template_id, instance.offset);
529            if bloq.templates().get(instance.template_id).is_none() {
530                continue;
531            }
532            if let Some(seen) = seen {
533                if seen.insert(placement) {
534                    placements.push(placement);
535                }
536            } else if !placements.contains(&placement) {
537                if placements.len() == 4 {
538                    *seen = Some(placements.iter().copied().chain([placement]).collect());
539                }
540                placements.push(placement);
541            }
542        }
543    }
544
545    let mut footprints = Vec::new();
546    let mut positions = Vec::new();
547    for (id, node) in bloq.nodes() {
548        positions.clear();
549        collect_member_positions(node, &mut positions);
550        let Some(z_min) = positions.iter().map(|(_, layer)| *layer).min() else {
551            continue;
552        };
553        let z_max = positions
554            .iter()
555            .map(|(_, layer)| *layer)
556            .max()
557            .expect("z_min proved the member list is non-empty");
558        let mut placements = SmallVec::new();
559        collect_footprint_placements(bloq, node, &mut placements, &mut None);
560        footprints.push(Footprint {
561            id,
562            z_min,
563            z_max,
564            placements,
565            member_columns: positions.iter().map(|(p, _)| p.truncate()).collect(),
566        });
567    }
568    // Active slots inherit this order, avoiding one sort per interval.
569    footprints.sort_unstable_by_key(|footprint| (footprint.z_min, footprint.id.0));
570
571    // Per qubit, order its users by z and constrain each *consecutive* pair:
572    // teardown must precede the next era's reinit. A pair already ordered
573    // through the IR (a continuous piped column, possibly routing through
574    // other columns' seams) needs no edge; what this catches is footprint
575    // reuse across an era gap with *no* connecting path — e.g. a T column
576    // whose port/region source hovers two layers above the previous era's
577    // terminating measurement (the U17g proxy hit this first), or a
578    // selective measurement freeing a qubit that an adjacent era's merged
579    // patch reinits (the CCZ factories: the merged patch has quantum seams
580    // from *other* columns, so any-seam probes wrongly treat it as ordered).
581    //
582    // Both endpoints are judged per qubit: `lower` must have no outgoing
583    // quantum seam carrying *this* qubit (its worldline ends there — a
584    // node-level any-seam probe wrongly treats a merged patch that measures
585    // one column out while others continue as mid-column), and `upper` must
586    // have no incoming quantum seam carrying it (its worldline starts there).
587    // The CCZ factories hit both: their merged patches carry seams for other
588    // columns across every era boundary.
589    //
590    // Exclude moving patches: their footprints include transit qubits, whose
591    // reuse order is not determined by source z. Displaced seam endpoints
592    // identify these nodes without the source graph.
593    let slot_by_id: crate::FxMap<BloqNodeId, usize> = footprints
594        .iter()
595        .enumerate()
596        .map(|(slot, footprint)| (footprint.id, slot))
597        .collect();
598    let mut template_runs = crate::FxMap::default();
599    let mut events = Vec::new();
600    for (slot, footprint) in footprints.iter().enumerate() {
601        #[cfg(debug_assertions)]
602        let mut qubits = crate::FxSet::default();
603        for &(template_id, offset) in &footprint.placements {
604            let template = &bloq.templates()[template_id];
605            let local = template_runs
606                .entry(template_id)
607                .or_insert_with(|| template_qubit_runs(template.qubits()));
608            for &run in local.iter() {
609                let run = match run.translated(offset) {
610                    Ok(run) => run,
611                    Err(_) => {
612                        // Run order differs from template qubit order. Earlier
613                        // placements passed their endpoint checks, so re-scan
614                        // only this placement to retain the first old error.
615                        let error = template
616                            .qubits()
617                            .iter()
618                            .find_map(|&qubit| {
619                                bloq_circuit::checked_translate_coordinate(qubit, offset).err()
620                            })
621                            .expect("a run endpoint overflowed");
622                        return Err(error.into());
623                    }
624                };
625                append_run_events(&mut events, slot as u32, run);
626                #[cfg(debug_assertions)]
627                for x in run.start..run.end {
628                    qubits.insert(glam::IVec2::new(
629                        (2 * x + i64::from(run.parity)) as i32,
630                        run.y,
631                    ));
632                }
633            }
634        }
635        #[cfg(debug_assertions)]
636        debug_assert_eq!(
637            qubits,
638            bloq.node_qubits(
639                bloq.node(footprint.id)
640                    .expect("footprint node still exists")
641            )?,
642            "run collection preserves the node's qubit set"
643        );
644    }
645    if events.is_empty() {
646        return Ok(());
647    }
648    drop(template_runs);
649    // Per slot and direction, the footprint slots of its quantum neighbours
650    // (looking through member-less pipe nodes, which carry no footprint).
651    // Resolved once so the per-(qubit, pair) probe below never walks edges.
652    let neighbor_slots = |direction: Dir| -> Vec<Vec<usize>> {
653        let quantum_neighbors = |of: BloqNodeId| {
654            seam_edges(bloq, of, direction)
655                .filter(|edge| matches!(edge.edge, BloqEdge::Quantum(_)))
656                .map(move |edge| match direction {
657                    Dir::Incoming => edge.source,
658                    Dir::Outgoing => edge.target,
659                })
660        };
661        footprints
662            .iter()
663            .map(|footprint| {
664                let mut slots = Vec::new();
665                for next in quantum_neighbors(footprint.id) {
666                    match slot_by_id.get(&next) {
667                        Some(&slot) => slots.push(slot),
668                        None => slots.extend(
669                            quantum_neighbors(next).filter_map(|hop| slot_by_id.get(&hop).copied()),
670                        ),
671                    }
672                }
673                slots.sort_unstable();
674                slots.dedup();
675                slots
676            })
677            .collect()
678    };
679    let outgoing_neighbor_slots = neighbor_slots(Dir::Outgoing);
680    let incoming_neighbor_slots = neighbor_slots(Dir::Incoming);
681    // Whether the current qubit's worldline continues across a quantum seam of
682    // the slot's node in `direction`: one of the quantum-neighbour footprint
683    // slots also appears in the qubit's inverse user list.
684    let worldline_continues =
685        |slot: usize, users: &[u32], user_slots: Option<&crate::FxSet<u32>>, direction: Dir| {
686            let neighbors = match direction {
687                Dir::Incoming => &incoming_neighbor_slots[slot],
688                Dir::Outgoing => &outgoing_neighbor_slots[slot],
689            };
690            neighbor_in_users(neighbors, users, user_slots)
691        };
692    // Computed once per slot: the probe is pure per-node but was previously
693    // re-evaluated for every (qubit, pair) window it appeared in.
694    let is_moving: Vec<bool> = (0..footprints.len())
695        .map(|slot| {
696            let footprint = &footprints[slot];
697            let displaced =
698                |endpoint: glam::IVec3| !footprint.member_columns.contains(&endpoint.truncate());
699            let seam_pipes = |direction: Dir| {
700                seam_edges(bloq, footprint.id, direction).flat_map(|edge| match edge.edge {
701                    BloqEdge::Quantum(edge) => edge.pipes.as_slice(),
702                    BloqEdge::Value { .. } | BloqEdge::Compose { .. } | BloqEdge::Order => &[],
703                })
704            };
705            seam_pipes(Dir::Outgoing).any(|seam| displaced(seam.pipe.src))
706                || seam_pipes(Dir::Incoming).any(|seam| displaced(seam.pipe.dst))
707        })
708        .collect();
709    // Membership is constant between run endpoints, so the same qubit filters
710    // run once per interval. Sorted slots retain the old per-qubit user order;
711    // duplicate slots only added self-windows, which always failed the z-gap.
712    let conditional_members = footprints
713        .iter()
714        .map(|footprint| conditional_footprint(&bloq[footprint.id]))
715        .collect::<Vec<_>>();
716    let geometric_pair = |lower_slot: u32, upper_slot: u32| {
717        let (lower_slot, upper_slot) = (lower_slot as usize, upper_slot as usize);
718        let (lower, upper) = (&footprints[lower_slot], &footprints[upper_slot]);
719        lower.z_max < upper.z_min && !is_moving[lower_slot] && !is_moving[upper_slot]
720    };
721    let neighbor_ids = |slots: &[Vec<usize>]| {
722        footprints
723            .iter()
724            .enumerate()
725            .map(|(slot, footprint)| {
726                (
727                    footprint.id,
728                    slots[slot]
729                        .iter()
730                        .map(|&slot| footprints[slot].id)
731                        .collect(),
732                )
733            })
734            .collect()
735    };
736    let neighbors = EraNeighbors {
737        predecessors: neighbor_ids(&incoming_neighbor_slots),
738        successors: neighbor_ids(&outgoing_neighbor_slots),
739    };
740    let mut conditional_checks = Vec::new();
741    let mut candidate_slots = occupancy_pairs(events, |lower, upper, users, user_slots, span| {
742        let conditional = users.iter().any(|&slot| conditional_members[slot as usize]);
743        if conditional && lower == users[0] {
744            // A conditional user can disappear. Keep candidates beyond every
745            // conditional run, then check actual carried boundary data under
746            // their guards. A seam belonging only to another arm cannot suppress
747            // a teardown-before-reset dependency.
748            for (index, &lower) in users.iter().enumerate() {
749                for upper_index in index + 1..users.len() {
750                    if upper_index > index + 1
751                        && !conditional_members[users[upper_index - 1] as usize]
752                    {
753                        break;
754                    }
755                    let upper = users[upper_index];
756                    if geometric_pair(lower, upper) {
757                        conditional_checks.push((lower, upper, span));
758                    }
759                }
760            }
761        }
762        !conditional
763            && geometric_pair(lower, upper)
764            && !worldline_continues(lower as usize, users, user_slots, Dir::Outgoing)
765            && !worldline_continues(upper as usize, users, user_slots, Dir::Incoming)
766    });
767    let mut live_supports = crate::FxMap::default();
768    let mut predicates = bloq_ir::lowering::PredicateAnalysis::new(bloq.top());
769    for (lower, upper, span) in conditional_checks {
770        if candidate_slots.contains(&(lower, upper)) {
771            continue;
772        }
773        for x in span.start..span.end {
774            let qubit = glam::IVec2::new((2 * x + i64::from(span.parity)) as i32, span.y);
775            if guarded_era_boundary(
776                bloq,
777                footprints[lower as usize].id,
778                footprints[upper as usize].id,
779                qubit,
780                &neighbors,
781                &mut live_supports,
782                &mut predicates,
783            )? {
784                candidate_slots.insert((lower, upper));
785                break;
786            }
787        }
788    }
789
790    /// Whether an edge `from -> to` already exists. An existing edge appears
791    /// in both `outgoing(from)` and `incoming(to)`, so scanning the two lists
792    /// in lockstep and stopping as soon as either runs dry costs
793    /// O(min(outdeg(from), indeg(to))) instead of a full out-list walk —
794    /// `from` nodes can have enormous out-degree.
795    fn has_edge(bloq: &Bloq, from: BloqNodeId, to: BloqNodeId) -> bool {
796        let mut outgoing = bloq.outgoing(from);
797        let mut incoming = bloq.incoming(to);
798        loop {
799            match (outgoing.next(), incoming.next()) {
800                (Some(edge), _) if edge.target == to => return true,
801                (_, Some(edge)) if edge.source == from => return true,
802                (Some(_), Some(_)) => {}
803                _ => return false,
804            }
805        }
806    }
807
808    // One topological order certifies most candidates without a graph walk: an
809    // edge that follows the order cannot close a cycle. Add all such edges
810    // first, then probe only candidates that oppose the order against the
811    // evolving graph. Initial-T control flow can genuinely point from a later-z
812    // footprint to an earlier-z one, so source time alone is not a certificate.
813    let mut candidate_pairs: Vec<_> = candidate_slots
814        .into_iter()
815        .map(|(lower, upper)| (footprints[lower as usize].id, footprints[upper as usize].id))
816        .collect();
817    candidate_pairs.sort_unstable_by_key(|(from, to)| (from.0, to.0));
818    let initial_order = bloq
819        .deterministic_emit_order()
820        .expect("lowered IR is acyclic before occupancy ordering");
821    let mut rank =
822        vec![0usize; initial_order.iter().map(|id| id.0).max().unwrap_or(0) as usize + 1];
823    for (index, id) in initial_order.into_iter().enumerate() {
824        rank[id.0 as usize] = index;
825    }
826    // An acyclic union certifies every candidate and every insertion prefix.
827    // Check it once instead of walking the graph for each backward candidate.
828    // If the union cycles, keep the existing per-edge decisions and order.
829    let batch_acyclic = candidate_pairs
830        .iter()
831        .all(|(from, to)| rank[from.0 as usize] < rank[to.0 as usize])
832        || bloq.is_acyclic_with_edges(&candidate_pairs);
833    let mut scratch = bloq.path_scratch();
834    for forward in [true, false] {
835        for &(from, to) in &candidate_pairs {
836            if (rank[from.0 as usize] < rank[to.0 as usize]) == forward && !has_edge(bloq, from, to)
837            {
838                if batch_acyclic || forward || !bloq.has_path_with_scratch(to, from, &mut scratch) {
839                    bloq.add_edge(from, to, BloqEdge::Order);
840                } else {
841                    for (from, to) in plan_live_era_reordering(
842                        bloq,
843                        from,
844                        to,
845                        &neighbors.predecessors,
846                        LiveDirection::BeforeInputs,
847                        &mut live_supports,
848                        &mut scratch,
849                    )? {
850                        bloq.add_edge(from, to, BloqEdge::Order);
851                    }
852                    for (before, after) in plan_live_era_reordering(
853                        bloq,
854                        to,
855                        from,
856                        &neighbors.successors,
857                        LiveDirection::AfterOutputs,
858                        &mut live_supports,
859                        &mut scratch,
860                    )? {
861                        bloq.add_edge(before, after, BloqEdge::Order);
862                    }
863                }
864            }
865        }
866    }
867    Ok(())
868}

16fn append_run_events(events: &mut Vec<OccupancyEvent>, slot: u32, run: QubitRun) {
17    for (x, start) in [(run.start, true), (run.end, false)] {
18        events.push(OccupancyEvent {
19            y: run.y,
20            parity: run.parity,
21            x,
22            slot,
23            start,
24        });
25    }

28fn occupancy_pairs(
29    mut events: Vec<OccupancyEvent>,
30    mut allowed: impl FnMut(u32, u32, &[u32], Option<&crate::FxSet<u32>>) -> bool,
31) -> crate::FxSet<(u32, u32)> {
32    events.sort_unstable_by_key(|event| (event.y, event.parity, event.x));
33    let mut active = BTreeMap::<u32, usize>::new();
34    let mut pairs = crate::FxSet::default();
35    let mut users = SmallVec::<[u32; 8]>::new();
36    let mut first = 0;
37    while first < events.len() {
38        let event = events[first];
39        let mut end = first;
40        // Starts and ends at one coordinate change membership together.
41        // Counts retain a slot while another of its placements still covers it.
42        while end < events.len()
43            && (events[end].y, events[end].parity, events[end].x)
44                == (event.y, event.parity, event.x)
45        {
46            let update = events[end];
47            if update.start {
48                *active.entry(update.slot).or_default() += 1;
49            } else {
50                let count = active
51                    .get_mut(&update.slot)
52                    .expect("an ending run is active");
53                *count -= 1;
54                if *count == 0 {
55                    active.remove(&update.slot);
56                }
57            }
58            end += 1;
59        }
60        if end < events.len()
61            && (events[end].y, events[end].parity) == (event.y, event.parity)
62            && active.len() > 1
63        {
64            users.clear();
65            users.extend(active.keys().copied());
66            let user_slots =
67                (users.len() > 8).then(|| users.iter().copied().collect::<crate::FxSet<_>>());
68            for pair in users.windows(2) {
69                if allowed(pair[0], pair[1], &users, user_slots.as_ref()) {
70                    pairs.insert((pair[0], pair[1]));
71                }
72            }
73        }
74        first = end;
75    }
76    debug_assert!(active.is_empty(), "every footprint run has an end");
77    pairs
78}

80fn neighbor_in_users(
81    neighbors: &[usize],
82    users: &[u32],
83    user_slots: Option<&crate::FxSet<u32>>,
84) -> bool {
85    // Neighbor slots are sorted. Search the short user list when that costs
86    // less than scanning every neighbor (with an inline or hashed lookup).
87    let search_steps = (usize::BITS - neighbors.len().leading_zeros()) as usize;
88    if users.len().saturating_mul(search_steps) < neighbors.len() {
89        users
90            .iter()
91            .any(|&user| neighbors.binary_search(&(user as usize)).is_ok())
92    } else {
93        neighbors.iter().any(|&neighbor| {
94            let neighbor = neighbor as u32;
95            user_slots.map_or_else(
96                || users.contains(&neighbor),
97                |slots| slots.contains(&neighbor),
98            )
99        })
100    }
101}

103/// Induce occupancy `Order` edges: a *terminating* node (no
104/// outgoing quantum seam) that shares physical qubits with a later *source*
105/// node (no incoming quantum seam) must emit first — teardown then reinit.
106/// Detector/observable composition has already run over `Temporal` seams
107/// only, so these pure-ordering edges only constrain emission order — they
108/// never change emitted detectors or observables.
109///
110/// Only era boundaries (terminal → source) are ordered. Mid-column walking
111/// nodes also touch other columns' qubits (a glide's rotation transits a
112/// neighbour cell's corner), but their member z-extent says nothing about
113/// *when* within the walk the shared qubit is used, so any z-derived edge
114/// between mid-column nodes can invert the real choreography
115/// (ghz_slide_then_glide breaks under a naive per-qubit z ordering). At an
116/// era boundary the direction is unambiguous.
117///
118/// The Clifford proxy was the first emitter to need this: a T column's
119/// source node hovers a layer gap above the previous era's terminating block
120/// on the same footprint, and a bare toposort emitted the later era's port
121/// MPP before the earlier era's readout of the same qubits.
122pub(super) fn induce_occupancy_order_edges(bloq: &mut Bloq) -> Result<(), crate::CompileError> {
123    use bloq_ir::{BloqEdgeRef, BloqNodeId};
124
125    struct Footprint {
126        id: BloqNodeId,
127        z_min: i64,
128        z_max: i64,
129        placements: SmallVec<[(bloq_ir::TemplateId, glam::IVec2); 4]>,
130        member_columns: crate::FxSet<glam::IVec2>,
131    }
132
133    /// Seam-probe direction, and the neighbour a seam edge reaches in it.
134    #[derive(Clone, Copy, PartialEq)]
135    enum Dir {
136        Incoming,
137        Outgoing,
138    }
139    fn seam_edges<'a>(
140        bloq: &'a Bloq,
141        of: BloqNodeId,
142        direction: Dir,
143    ) -> impl Iterator<Item = BloqEdgeRef<'a>> {
144        let (incoming, outgoing) = match direction {
145            Dir::Incoming => (Some(bloq.incoming(of)), None),
146            Dir::Outgoing => (None, Some(bloq.outgoing(of))),
147        };
148        incoming
149            .into_iter()
150            .flatten()
151            .chain(outgoing.into_iter().flatten())
152    }
153
154    // A region occupies its bodies' members. This keeps its block worldlines
155    // ordered across eras even though the region node carries no members.
156    // Virtual spatial Ports also occupy qubits, just before/after their source
157    // layer. Omitting them lets a later input reset a still-live earlier patch.
158    // Pipe nodes still carry no footprint.
159    fn collect_member_positions(node: &BloqNode, positions: &mut Vec<(glam::IVec3, i64)>) {
160        positions.extend(
161            node.block_members()
162                .iter()
163                .map(|m| (m.pos, 2 * i64::from(m.pos.z))),
164        );
165        if let bloq_ir::NodeProvenance::SpatialPortSubstitution { source, .. } = node.provenance {
166            positions.push((source, node.layer()));
167        }
168        if let Some(region) = node.try_region() {
169            for (_, body) in region.bodies() {
170                for (_, body_node) in body.nodes() {
171                    collect_member_positions(body_node, positions);
172                }
173            }
174        }
175    }
176
177    // Keep first-appearance order in `placements`, including across region bodies.
178    fn collect_footprint_placements(
179        bloq: &Bloq,
180        node: &BloqNode,
181        placements: &mut SmallVec<[(bloq_ir::TemplateId, glam::IVec2); 4]>,
182        seen: &mut Option<crate::FxSet<(bloq_ir::TemplateId, glam::IVec2)>>,
183    ) {
184        if let Some(region) = node.try_region() {
185            for (_, body) in region.bodies() {
186                for (_, body_node) in body.nodes() {
187                    collect_footprint_placements(bloq, body_node, placements, seen);
188                }
189            }
190            return;
191        }
192        let Some(quantum) = node.try_quantum() else {
193            return;
194        };
195        for instance in &quantum.instances {
196            let placement = (instance.template_id, instance.offset);
197            if bloq.templates().get(instance.template_id).is_none() {
198                continue;
199            }
200            if let Some(seen) = seen {
201                if seen.insert(placement) {
202                    placements.push(placement);
203                }
204            } else if !placements.contains(&placement) {
205                if placements.len() == 4 {
206                    *seen = Some(placements.iter().copied().chain([placement]).collect());
207                }
208                placements.push(placement);
209            }
210        }
211    }
212
213    let mut footprints = Vec::new();
214    let mut positions = Vec::new();
215    for (id, node) in bloq.nodes() {
216        positions.clear();
217        collect_member_positions(node, &mut positions);
218        let Some(z_min) = positions.iter().map(|(_, layer)| *layer).min() else {
219            continue;
220        };
221        let z_max = positions
222            .iter()
223            .map(|(_, layer)| *layer)
224            .max()
225            .expect("z_min proved the member list is non-empty");
226        let mut placements = SmallVec::new();
227        collect_footprint_placements(bloq, node, &mut placements, &mut None);
228        footprints.push(Footprint {
229            id,
230            z_min,
231            z_max,
232            placements,
233            member_columns: positions.iter().map(|(p, _)| p.truncate()).collect(),
234        });
235    }
236    // Active slots inherit this order, avoiding one sort per interval.
237    footprints.sort_unstable_by_key(|footprint| (footprint.z_min, footprint.id.0));
238
239    // Per qubit, order its users by z and constrain each *consecutive* pair:
240    // teardown must precede the next era's reinit. A pair already ordered
241    // through the IR (a continuous piped column, possibly routing through
242    // other columns' seams) needs no edge; what this catches is footprint
243    // reuse across an era gap with *no* connecting path — e.g. a T column
244    // whose port/region source hovers two layers above the previous era's
245    // terminating measurement (the U17g proxy hit this first), or a
246    // selective measurement freeing a qubit that an adjacent era's merged
247    // patch reinits (the CCZ factories: the merged patch has quantum seams
248    // from *other* columns, so any-seam probes wrongly treat it as ordered).
249    //
250    // Both endpoints are judged per qubit: `lower` must have no outgoing
251    // quantum seam carrying *this* qubit (its worldline ends there — a
252    // node-level any-seam probe wrongly treats a merged patch that measures
253    // one column out while others continue as mid-column), and `upper` must
254    // have no incoming quantum seam carrying it (its worldline starts there).
255    // The CCZ factories hit both: their merged patches carry seams for other
256    // columns across every era boundary.
257    //
258    // Walking nodes are excluded entirely: their layout footprint covers the
259    // whole transit, and member z says nothing about when *within* the walk a
260    // shared qubit is used. In ghz_slide_then_glide the glide endpoint's
261    // patch (member z=5) is live *before* the neighbour arm's slide transit
262    // (member z=3) crosses it, so any z-derived edge between them inverts the
263    // real choreography. A walk is detected by displacement between a node's
264    // member columns and its seam endpoints: a slide/glide block keeps its
265    // source member position but hands off (or was handed) the patch at a
266    // different xy column.
267    let slot_by_id: crate::FxMap<BloqNodeId, usize> = footprints
268        .iter()
269        .enumerate()
270        .map(|(slot, footprint)| (footprint.id, slot))
271        .collect();
272    let mut template_runs = crate::FxMap::default();
273    let mut events = Vec::new();
274    for (slot, footprint) in footprints.iter().enumerate() {
275        #[cfg(debug_assertions)]
276        let mut qubits = crate::FxSet::default();
277        for &(template_id, offset) in &footprint.placements {
278            let template = &bloq.templates()[template_id];
279            let local = template_runs
280                .entry(template_id)
281                .or_insert_with(|| template_qubit_runs(template.qubits()));
282            for &run in local.iter() {
283                let run = match run.translated(offset) {
284                    Ok(run) => run,
285                    Err(_) => {
286                        // Run order differs from template qubit order. Earlier
287                        // placements passed their endpoint checks, so re-scan
288                        // only this placement to retain the first old error.
289                        let error = template
290                            .qubits()
291                            .iter()
292                            .find_map(|&qubit| {
293                                bloq_circuit::checked_translate_coordinate(qubit, offset).err()
294                            })
295                            .expect("a run endpoint overflowed");
296                        return Err(error.into());
297                    }
298                };
299                append_run_events(&mut events, slot as u32, run);
300                #[cfg(debug_assertions)]
301                for x in run.start..run.end {
302                    qubits.insert(glam::IVec2::new(
303                        (2 * x + i64::from(run.parity)) as i32,
304                        run.y,
305                    ));
306                }
307            }
308        }
309        #[cfg(debug_assertions)]
310        debug_assert_eq!(
311            qubits,
312            bloq.node_qubits(
313                bloq.node(footprint.id)
314                    .expect("footprint node still exists")
315            )?,
316            "run collection preserves the node's qubit set"
317        );
318    }
319    if events.is_empty() {
320        return Ok(());
321    }
322    drop(template_runs);
323    // Per slot and direction, the footprint slots of its quantum neighbours
324    // (looking through member-less pipe nodes, which carry no footprint).
325    // Resolved once so the per-(qubit, pair) probe below never walks edges.
326    let neighbor_slots = |direction: Dir| -> Vec<Vec<usize>> {
327        let quantum_neighbors = |of: BloqNodeId| {
328            seam_edges(bloq, of, direction)
329                .filter(|edge| matches!(edge.edge, BloqEdge::Quantum(_)))
330                .map(move |edge| match direction {
331                    Dir::Incoming => edge.source,
332                    Dir::Outgoing => edge.target,
333                })
334        };
335        footprints
336            .iter()
337            .map(|footprint| {
338                let mut slots = Vec::new();
339                for next in quantum_neighbors(footprint.id) {
340                    match slot_by_id.get(&next) {
341                        Some(&slot) => slots.push(slot),
342                        None => slots.extend(
343                            quantum_neighbors(next).filter_map(|hop| slot_by_id.get(&hop).copied()),
344                        ),
345                    }
346                }
347                slots.sort_unstable();
348                slots.dedup();
349                slots
350            })
351            .collect()
352    };
353    let outgoing_neighbor_slots = neighbor_slots(Dir::Outgoing);
354    let incoming_neighbor_slots = neighbor_slots(Dir::Incoming);
355    // Whether the current qubit's worldline continues across a quantum seam of
356    // the slot's node in `direction`: one of the quantum-neighbour footprint
357    // slots also appears in the qubit's inverse user list.
358    let worldline_continues =
359        |slot: usize, users: &[u32], user_slots: Option<&crate::FxSet<u32>>, direction: Dir| {
360            let neighbors = match direction {
361                Dir::Incoming => &incoming_neighbor_slots[slot],
362                Dir::Outgoing => &outgoing_neighbor_slots[slot],
363            };
364            neighbor_in_users(neighbors, users, user_slots)
365        };
366    // Computed once per slot: the probe is pure per-node but was previously
367    // re-evaluated for every (qubit, pair) window it appeared in.
368    let is_walking: Vec<bool> = (0..footprints.len())
369        .map(|slot| {
370            let footprint = &footprints[slot];
371            let displaced =
372                |endpoint: glam::IVec3| !footprint.member_columns.contains(&endpoint.truncate());
373            let seam_pipes = |direction: Dir| {
374                seam_edges(bloq, footprint.id, direction).flat_map(|edge| match edge.edge {
375                    BloqEdge::Quantum(edge) => edge.pipes.as_slice(),
376                    BloqEdge::Value { .. } | BloqEdge::Compose { .. } | BloqEdge::Order => &[],
377                })
378            };
379            seam_pipes(Dir::Outgoing).any(|seam| displaced(seam.pipe.src))
380                || seam_pipes(Dir::Incoming).any(|seam| displaced(seam.pipe.dst))
381        })
382        .collect();
383    // Membership is constant between run endpoints, so the same qubit filters
384    // run once per interval. Sorted slots retain the old per-qubit user order;
385    // duplicate slots only added self-windows, which always failed the z-gap.
386    let candidate_slots = occupancy_pairs(events, |lower_slot, upper_slot, users, user_slots| {
387        let (lower_slot, upper_slot) = (lower_slot as usize, upper_slot as usize);
388        let (lower, upper) = (&footprints[lower_slot], &footprints[upper_slot]);
389        lower.z_max < upper.z_min
390            && !is_walking[lower_slot]
391            && !is_walking[upper_slot]
392            && !worldline_continues(lower_slot, users, user_slots, Dir::Outgoing)
393            && !worldline_continues(upper_slot, users, user_slots, Dir::Incoming)
394    });
395
396    /// Whether an edge `from -> to` already exists. An existing edge appears
397    /// in both `outgoing(from)` and `incoming(to)`, so scanning the two lists
398    /// in lockstep and stopping as soon as either runs dry costs
399    /// O(min(outdeg(from), indeg(to))) instead of a full out-list walk —
400    /// `from` nodes can have enormous out-degree.
401    fn has_edge(bloq: &Bloq, from: BloqNodeId, to: BloqNodeId) -> bool {
402        let mut outgoing = bloq.outgoing(from);
403        let mut incoming = bloq.incoming(to);
404        loop {
405            match (outgoing.next(), incoming.next()) {
406                (Some(edge), _) if edge.target == to => return true,
407                (_, Some(edge)) if edge.source == from => return true,
408                (Some(_), Some(_)) => {}
409                _ => return false,
410            }
411        }
412    }
413
414    // One topological order certifies most candidates without a graph walk: an
415    // edge that follows the order cannot close a cycle. Add all such edges
416    // first, then probe only candidates that oppose the order against the
417    // evolving graph. Initial-T control flow can genuinely point from a later-z
418    // footprint to an earlier-z one, so source time alone is not a certificate.
419    let mut candidate_pairs: Vec<_> = candidate_slots
420        .into_iter()
421        .map(|(lower, upper)| (footprints[lower as usize].id, footprints[upper as usize].id))
422        .collect();
423    candidate_pairs.sort_unstable_by_key(|(from, to)| (from.0, to.0));
424    let initial_order = bloq
425        .deterministic_emit_order()
426        .expect("lowered IR is acyclic before occupancy ordering");
427    let mut rank =
428        vec![0usize; initial_order.iter().map(|id| id.0).max().unwrap_or(0) as usize + 1];
429    for (index, id) in initial_order.into_iter().enumerate() {
430        rank[id.0 as usize] = index;
431    }
432    // An acyclic union certifies every candidate and every insertion prefix.
433    // Check it once instead of walking the graph for each backward candidate.
434    // If the union cycles, keep the existing per-edge decisions and order.
435    let batch_acyclic = candidate_pairs
436        .iter()
437        .all(|(from, to)| rank[from.0 as usize] < rank[to.0 as usize])
438        || bloq.is_acyclic_with_edges(&candidate_pairs);
439    let mut scratch = bloq.path_scratch();
440    for forward in [true, false] {
441        for &(from, to) in &candidate_pairs {
442            if (rank[from.0 as usize] < rank[to.0 as usize]) == forward
443                && !has_edge(bloq, from, to)
444                && (batch_acyclic || forward || !bloq.has_path_with_scratch(to, from, &mut scratch))
445            {
446                bloq.add_edge(from, to, BloqEdge::Order);
447            }
448        }
449    }
450    Ok(())
451}

522        let mut placements = vec![
523            (0, row.clone(), ivec2(-1, 2)),
524            (0, row.clone(), ivec2(1, 2)),
525            // These distinct placements touch in reduced x; they keep one slot live.
526            (0, vec![ivec2(-2, 7), ivec2(0, 7)], ivec2(0, 0)),
527            (0, vec![ivec2(2, 7), ivec2(4, 7)], ivec2(0, 0)),
528        ];
529        for slot in 1..12 {
530            placements.push((slot, row.clone(), ivec2(-1, 2)));
531        }
532        placements.extend([
533            (5, row, ivec2(1, 2)),
534            (6, vec![ivec2(0, 7), ivec2(2, 7)], ivec2(0, 0)),
535            // Touching slots do not share a physical qubit.
536            (14, vec![ivec2(8, 5)], ivec2(0, 0)),
537            (15, vec![ivec2(10, 5)], ivec2(0, 0)),
538            // Both lattice extremes have short runs and an i64 exclusive end.
539            (
540                12,
541                vec![ivec2(i32::MIN, 0), ivec2(i32::MAX, 0)],
542                ivec2(0, 0),
543            ),
544            (

547                ivec2(0, 0),
548            ),
549            // Equal rows but different x parity cannot intersect.
550            (13, vec![ivec2(-8, -1), ivec2(-6, -1)], ivec2(0, 0)),
551        ]);
552        placements.sort_by_key(|(slot, _, _)| *slot);
553        let mut events = Vec::new();
554        let mut oracle = BTreeMap::<(i32, i32), Vec<u32>>::new();

558                oracle.entry((global.x, global.y)).or_default().push(slot);
559            }
560            for run in template_qubit_runs(&qubits) {
561                append_run_events(&mut events, slot, run.translated(offset).unwrap());
562            }
563        }
564        let mut outgoing = vec![vec![]; 16];
565        let mut incoming = vec![vec![]; 16];
566        outgoing[0] = vec![2];
567        incoming[4] = vec![1];
568        for walking in [None, Some(2)] {
569            let allowed =
570                |lower: u32, upper: u32, users: &[u32], slots: Option<&crate::FxSet<u32>>| {
571                    lower != upper
572                        && Some(lower) != walking
573                        && Some(upper) != walking
574                        && !neighbor_in_users(&outgoing[lower as usize], users, slots)
575                        && !neighbor_in_users(&incoming[upper as usize], users, slots)
576                };
577            let mut expected = crate::FxSet::default();
578            for users in oracle.values() {
579                let slots = (users.len() > 8).then(|| users.iter().copied().collect());
580                for pair in users.windows(2) {
581                    if allowed(pair[0], pair[1], users, slots.as_ref()) {
582                        expected.insert((pair[0], pair[1]));
583                    }
584                }
585            }
586            let actual = occupancy_pairs(events.clone(), allowed);
587            assert_eq!(actual, expected, "walking={walking:?}");
588            assert!(
589                !actual.contains(&(14, 15)),
590                "touching endpoints are disjoint"
591            );
592        }
593        assert!(occupancy_pairs(Vec::new(), |_, _, _, _| true).is_empty());
594
595        // Isolated cases cannot have a missing or spurious pair hidden by a
596        // second shared coordinate elsewhere in the larger fixture.
597        let scenarios = [
598            vec![
599                (0, vec![ivec2(0, 0), ivec2(2, 0), ivec2(4, 0)], ivec2(0, 0)),
600                (0, vec![ivec2(0, 0), ivec2(2, 0), ivec2(4, 0)], ivec2(2, 0)),
601                (1, vec![ivec2(6, 0)], ivec2(0, 0)),
602            ],
603            vec![
604                (0, vec![ivec2(0, 0)], ivec2(0, 0)),
605                (1, vec![ivec2(2, 0)], ivec2(0, 0)),
606            ],
607            vec![
608                (0, vec![ivec2(i32::MAX, 0)], ivec2(0, 0)),
609                (1, vec![ivec2(i32::MAX, 0)], ivec2(0, 0)),
610            ],
611        ];
612        for placements in scenarios {
613            let mut events = Vec::new();
614            let mut oracle = BTreeMap::<(i32, i32), Vec<u32>>::new();
615            for (slot, qubits, offset) in placements {
616                for &qubit in &qubits {
617                    let global = bloq_circuit::checked_translate_coordinate(qubit, offset).unwrap();
618                    oracle.entry((global.x, global.y)).or_default().push(slot);
619                }
620                for run in template_qubit_runs(&qubits) {
621                    append_run_events(&mut events, slot, run.translated(offset).unwrap());
622                }
623            }
624            let expected = oracle
625                .values()
626                .flat_map(|users| users.windows(2))
627                .filter(|pair| pair[0] != pair[1])
628                .map(|pair| (pair[0], pair[1]))
629                .collect::<crate::FxSet<_>>();
630            assert_eq!(occupancy_pairs(events.clone(), |_, _, _, _| true), expected);
631            events.reverse();
632            assert_eq!(occupancy_pairs(events, |_, _, _, _| true), expected);
633        }
634    }

816    #[test]
817    fn batch_occupancy_order_matches_sequential_cycle_checks() {
818        // Every four-node DAG in node-id order, with every possible z order.
819        // The shared footprint makes consecutive z nodes occupancy candidates.
820        for z in itertools::Itertools::permutations(0..4, 4) {
821            for edges in 0..64 {
822                let mut bloq = Bloq::new();
823                let ids: Vec<_> = z
824                    .iter()
825                    .map(|&z| add_block_node(&mut bloq, ivec3(0, 0, z), ivec2(0, 0)))
826                    .collect();
827                let mut bit = 0;
828                for from in 0..4 {
829                    for to in from + 1..4 {
830                        if edges & (1 << bit) != 0 {
831                            bloq.add_edge(ids[from], ids[to], BloqEdge::Order);
832                        }
833                        bit += 1;
834                    }
835                }
836                let mut expected = bloq.clone();
837                let order = expected.deterministic_emit_order().unwrap();
838                let rank = |id| order.iter().position(|&node| node == id).unwrap();
839                let mut by_z = ids.clone();
840                by_z.sort_unstable_by_key(|id| z[id.0 as usize]);
841                let mut candidates: Vec<_> = by_z.windows(2).map(|w| (w[0], w[1])).collect();
842                candidates
843                    .sort_unstable_by_key(|&(from, to)| (rank(from) >= rank(to), from.0, to.0));
844                for (from, to) in candidates {
845                    if !expected.outgoing(from).any(|edge| edge.target == to)
846                        && !expected.has_path(to, from)
847                    {
848                        expected.add_edge(from, to, BloqEdge::Order);
849                    }
850                }
851
852                induce_occupancy_order_edges(&mut bloq).unwrap();
853
854                assert_eq!(
855                    order_edges(&bloq),
856                    order_edges(&expected),
857                    "z={z:?}, edges={edges}"
858                );
859                bloq.deterministic_emit_order().unwrap();
860            }
861        }
862    }

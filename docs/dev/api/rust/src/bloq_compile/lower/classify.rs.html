13    pub(super) template: LoweringTemplateId,

16/// How one plan node lowers, computed once by [`classify_nodes`] and matched by
17/// the instance-placement, detector, and node-materialization passes. Keeping
18/// selective/T dispatch in exactly one place means adding a new
19/// dynamically lowered block kind extends this enum and lets the compiler's
20/// exhaustiveness check find every pass that must handle it.
21#[derive(Debug)]
22pub(super) enum NodeLowering {
23    /// Ordinary spatial component: one fixed template instance per member, plus
24    /// one per spatial Hadamard wall standing between two of them.
25    Fixed {
26        members: Vec<(SourceBlockRef, LoweringTemplateId)>,
27        walls: Vec<WallLowering>,
28    },
29    /// Temporal Hadamard realignment pipe node: one instance at the lower
30    /// endpoint block's offset.
31    TemporalPipe {
32        pipe: TemporalPipeRef,
33        template: LoweringTemplateId,
34        origin: IVec3,
35    },
36    /// Positionless temporal Port derived from an authored spatial Port.
37    SpatialPort {
38        source: IVec3,
39        template: LoweringTemplateId,
40    },
41    /// Selective measurement block, lowered to two guarded per-basis instances.
42    Selective {
43        pos: IVec3,
44        when_true: LoweringTemplateId,
45        when_false: LoweringTemplateId,
46    },
47    /// T block, lowered to a `RepeatUntilSuccess` region holding the
48    /// cultivation + escape stage pair.
49    TRegion {
50        pos: IVec3,
51        cultivation: LoweringTemplateId,
52        escape: LoweringTemplateId,
53    },
54}

56/// Classify every plan node, indexed by `NodeIndex::index()`.
57///
58/// The sole-member invariant for dynamic blocks (their signatures forbid
59/// spatial pipes, so they never join a spatial component) is asserted here,
60/// once, instead of at each consuming pass.
61pub(super) fn classify_nodes(
62    plan: &LowerPlan,
63    compiled: &CompiledTemplateMap,
64    spatial_port_templates: &CompiledTemplateMap,
65    temporal_templates: &crate::FxMap<TemplatePlan, LoweringTemplateId>,
66    wall_templates: &crate::FxMap<SpatialPipeRef, LoweringTemplateId>,
67) -> Vec<NodeLowering> {
68    plan.graph()
69        .node_weights()
70        .map(|node| {
71            if let NodeProvenance::SpatialPortSubstitution { source, .. } = &node.provenance {
72                let source = *source;
73                let template = spatial_port_templates[&source]
74                    .template
75                    .observable_template()
76                    .expect("derived temporal Ports use fixed templates");
77                return NodeLowering::SpatialPort { source, template };
78            }
79            if let Some(template_plan) = node.template {
80                let template = *temporal_templates
81                    .get(&template_plan)
82                    .expect("temporal template plan was compiled before lowering");
83                let TemplatePlan { origin, .. } = template_plan;
84                let NodeProvenance::TemporalPipe { pipe } = &node.provenance else {
85                    unreachable!("temporal template nodes are backed by temporal pipe sources");
86                };
87                return NodeLowering::TemporalPipe {
88                    pipe: *pipe,
89                    template,
90                    origin,
91                };
92            }
93
94            let members = node.block_members();
95            if let [member] = members {
96                match compiled
97                    .get(&member.pos)
98                    .expect("plan members come from compiled blocks")
99                    .template
100                {
101                    TemplateRef::Selective {
102                        when_true,
103                        when_false,
104                    } => {
105                        return NodeLowering::Selective {
106                            pos: member.pos,
107                            when_true,
108                            when_false,
109                        };
110                    }
111                    TemplateRef::T {
112                        cultivation,
113                        escape,
114                    } => {
115                        return NodeLowering::TRegion {
116                            pos: member.pos,
117                            cultivation,
118                            escape,
119                        };
120                    }
121                    TemplateRef::Fixed(_) => {}
122                }
123            }
124
125            let members = members
126                .iter()
127                .map(|&member| {
128                    let template = match compiled
129                        .get(&member.pos)
130                        .expect("plan members come from compiled blocks")
131                        .template
132                    {
133                        TemplateRef::Fixed(id) => id,
134                        TemplateRef::Selective { .. } | TemplateRef::T { .. } => unreachable!(
135                            "a dynamic block never joins a spatial component \
136                             (its signature forbids spatial pipes)"
137                        ),
138                    };
139                    (member, template)
140                })
141                .collect();
142            let walls = node
143                .walls
144                .iter()
145                .map(|&pipe| WallLowering {
146                    pipe,
147                    template: *wall_templates
148                        .get(&pipe)
149                        .expect("spatial Hadamard walls are compiled before lowering"),
150                })
151                .collect();
152            NodeLowering::Fixed { members, walls }
153        })
154        .collect()
155}

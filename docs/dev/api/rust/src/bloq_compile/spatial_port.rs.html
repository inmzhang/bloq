11pub(crate) struct SpatialPortExpansion {
12    pub(crate) source: IVec3,
13    pub(crate) role: PortRole,
14    pub(crate) cube_kind: CubeKind,
15    pub(crate) height: CubeHeight,
16    /// Multiplex output in virtual-Port-local circuit coordinates.
17    pub(crate) output_qubit: Option<IVec2>,
18}
19
20impl SpatialPortExpansion {
21    pub(crate) const fn is_input(self) -> bool {
22        self.role.has_input_boundary()
23    }
24
25    pub(crate) const fn cube_pipe_dir(self) -> Direction {
26        match self.role {
27            PortRole::Input | PortRole::Multiplex => Direction::ZMINUS,
28            PortRole::Output => Direction::ZPLUS,
29            PortRole::Auto => unreachable!(),
30        }
31    }
32
33    pub(crate) const fn port_pipe_dir(self) -> Direction {
34        self.cube_pipe_dir().negate()
35    }
36
37    pub(crate) const fn boundary_basis(self) -> Basis {
38        self.cube_kind.y()
39    }
40}

42pub(crate) type SpatialPortExpansionMap = crate::FxMap<IVec3, SpatialPortExpansion>;
43
44/// Replace authored spatial Ports with their compiler-only cube half. The
45/// positionless temporal Port half is added by `LowerPlan`.
46pub(crate) fn expand_spatial_ports(
47    source: &BlockGraph,
48    distance: u32,
49    stabilizers: &StabilizerGenerators,
50) -> Result<(BlockGraph, SpatialPortExpansionMap), CompileError> {
51    let (graph, expansions) = expand_spatial_port_topology(source, distance)?;
52    let mut graph = graph.into_owned();
53    if expansions.is_empty() {
54        return Ok((graph, expansions));
55    }
56
57    graph.validate_structure()?;
58    graph = graph.with_analyzed_action_graph(stabilizers)?;
59    Ok((graph, expansions))
60}

62/// Expand spatial Ports without attaching semantic action dependencies.
63/// Definition-object compilation needs only this local physical topology; the
64/// root semantic preflight attaches the composed stabilizer rows separately.
65pub(crate) fn expand_spatial_port_topology(
66    source: &BlockGraph,
67    distance: u32,
68) -> Result<(Cow<'_, BlockGraph>, SpatialPortExpansionMap), CompileError> {
69    let mut expansions =
70        spatial_port_expansions(source, source.blocks().map(bloq_graph::Block::pos))?;
71    allocate_multiplex_outputs(source, distance, &mut expansions)?;
72    apply_spatial_port_expansions(source, expansions)
73}

75/// Expand only complete local sites, using the output coordinates reserved
76/// against the full source's occupancy. Halo blocks remain lookup context.
77pub(crate) fn expand_spatial_port_context<'a>(
78    source: &'a BlockGraph,
79    positions: &[IVec3],
80    global: &SpatialPortExpansionMap,
81) -> Result<(Cow<'a, BlockGraph>, SpatialPortExpansionMap), CompileError> {
82    let mut expansions = spatial_port_expansions(source, positions.iter().copied())?;
83    for expansion in expansions.values_mut() {
84        if expansion.role == PortRole::Multiplex {
85            expansion.output_qubit = global[&expansion.source].output_qubit;
86        }
87    }
88    apply_spatial_port_expansions(source, expansions)
89}

91fn spatial_port_expansions(
92    source: &BlockGraph,
93    positions: impl IntoIterator<Item = IVec3>,
94) -> Result<SpatialPortExpansionMap, CompileError> {
95    let mut expansions = SpatialPortExpansionMap::default();
96    for position in positions {
97        let block = source.get_block(position).expect("selected site exists");
98        if !block.kind().is_port() {
99            continue;
100        }
101        let pos = block.pos();
102        let Some(pipe) = source.pipes_at(pos).next() else {
103            continue;
104        };
105        if !pipe.dir().is_spatial() {
106            continue;
107        }
108        let neighbor = source
109            .get_endpoint_block(if pipe.src() == pos {
110                pipe.dst()
111            } else {
112                pipe.src()
113            })
114            .expect("Port pipe has a neighbor");
115        let role = block.port_role().expect("Port blocks expose a role");
116        debug_assert_ne!(
117            role,
118            PortRole::Auto,
119            "source validation requires a spatial role"
120        );
121        let BlockKind::Cube(_) = neighbor.kind() else {
122            return Err(CompileError::SpatialPortNeighborNotCube {
123                port: pos,
124                neighbor: neighbor.pos(),
125            });
126        };
127        if neighbor.height_cells() != 1 {
128            return Err(CompileError::TallSpatialPortUnsupported {
129                port: pos,
130                height: neighbor.height(),
131            });
132        }
133        let cube_kind = source
134            .infer_spatial_port_cube_kind(pos)
135            .ok_or(CompileError::SpatialPortCubeInferenceFailed { port: pos })?;
136        expansions.insert(
137            pos,
138            SpatialPortExpansion {
139                source: pos,
140                role,
141                cube_kind,
142                height: neighbor.height(),
143                output_qubit: None,
144            },
145        );
146    }
147
148    Ok(expansions)
149}

151fn apply_spatial_port_expansions(
152    source: &BlockGraph,
153    expansions: SpatialPortExpansionMap,
154) -> Result<(Cow<'_, BlockGraph>, SpatialPortExpansionMap), CompileError> {
155    if expansions.is_empty() {
156        return Ok((Cow::Borrowed(source), expansions));
157    }
158
159    let mut graph = source.clone();
160    // Geometry edits each refresh the whole action DAG. Its external inputs
161    // remain declared while the actions are detached, then one final lenient
162    // rebuild preserves the mutators' error and measurement metadata contract.
163    let actions = graph.actions();
164    graph.set_actions_lenient(Vec::new())?;
165    for expansion in expansions.values() {
166        graph.set_block_kind(expansion.source, BlockKind::Cube(expansion.cube_kind))?;
167        graph.set_cube_height(expansion.source, expansion.height)?;
168    }
169    graph.set_actions_lenient(actions)?;
170    Ok((Cow::Owned(graph), expansions))
171}

173pub(crate) fn allocate_multiplex_outputs(
174    source: &BlockGraph,
175    distance: u32,
176    expansions: &mut SpatialPortExpansionMap,
177) -> Result<(), CompileError> {
178    let mut ports = expansions
179        .values()
180        .filter(|port| port.role == PortRole::Multiplex)
181        .map(|port| port.source)
182        .collect::<Vec<_>>();
183    if ports.is_empty() {
184        return Ok(());
185    }
186    ports.sort_unstable_by_key(glam::IVec3::to_array);
187
188    let mut occupied: crate::FxSet<_> = crate::signature::occupied_branch_positions(source)?
189        .into_iter()
190        .map(IVec3::truncate)
191        .collect();
192    // Keep clear of compiler templates that spill one cell past source occupancy.
193    for cell in occupied.clone() {
194        for delta in [IVec2::X, IVec2::Y, IVec2::NEG_X, IVec2::NEG_Y] {
195            if let (Some(x), Some(y)) = (cell.x.checked_add(delta.x), cell.y.checked_add(delta.y)) {
196                occupied.insert(IVec2::new(x, y));
197            }
198        }
199    }
200    for port in ports {
201        let cell = port.truncate();
202        let limit = occupied.len().saturating_add(1);
203        let output = (1..=limit)
204            .filter_map(|radius| i32::try_from(radius).ok())
205            .flat_map(|radius| {
206                [
207                    IVec2::new(radius, 0),
208                    IVec2::new(0, radius),
209                    IVec2::new(-radius, 0),
210                    IVec2::new(0, -radius),
211                ]
212            })
213            .find_map(|delta| {
214                let candidate =
215                    IVec2::new(cell.x.checked_add(delta.x)?, cell.y.checked_add(delta.y)?);
216                if occupied.contains(&candidate) {
217                    return None;
218                }
219                block_xy_offset(candidate, distance).ok()?;
220                Some((candidate, block_xy_offset(delta, distance).ok()?))
221            });
222        let Some((cell, output)) = output else {
223            return Err(CompileError::MultiplexOutputCoordinateUnavailable { port });
224        };
225        occupied.insert(cell);
226        expansions
227            .get_mut(&port)
228            .expect("multiplex source came from expansion map")
229            .output_qubit = Some(output);
230    }
231    Ok(())
232}

234#[cfg(test)]
235mod tests {
236    use super::*;
237    use bloq_graph::{Action, Block, Expr, MeasureTarget, Pipe};
238
239    #[test]
240    fn batch_expansion_preserves_actions_inputs_and_lenient_error() {
241        let mut source = BlockGraph::new();
242        for x in [0, 10] {
243            let port = IVec3::new(x, 0, 0);
244            source.add_block(
245                Block::new(port, BlockKind::Port)
246                    .with_port_role(PortRole::Input)
247                    .unwrap(),
248            );
249            source.add_block(Block::new(port + IVec3::X, BlockKind::Cube(CubeKind::ZXZ)));
250            source.add_pipe(Pipe::new(port, Direction::XPLUS));
251        }
252        source
253            .set_actions_with_inputs(Vec::new(), ["external".to_owned()])
254            .unwrap();
255        let measure = Action::Measure {
256            target: MeasureTarget::Node(IVec3::new(11, 0, 0)),
257            name: "measured".into(),
258        };
259        let copy = Action::Let {
260            name: "copy".into(),
261            expr: Expr::Var("external".into()),
262        };
263        source
264            .set_actions_lenient(vec![measure.clone(), copy.clone()])
265            .unwrap();
266        assert!(source.action_graph_error().is_none());
267        let measurement = source
268            .action_graph()
269            .node_by_ordinal(0)
270            .unwrap()
271            .measurement;
272        assert!(measurement.is_some());
273
274        let (expanded, ports) = expand_spatial_port_topology(&source, 3).unwrap();
275        assert_eq!(ports.len(), 2);
276        assert_eq!(expanded.actions(), source.actions());
277        assert_eq!(
278            expanded.action_graph().inputs().collect::<Vec<_>>(),
279            ["external"]
280        );
281        assert!(expanded.action_graph_error().is_none());
282        assert_eq!(
283            expanded
284                .action_graph()
285                .node_by_ordinal(0)
286                .unwrap()
287                .measurement,
288            measurement
289        );
290
291        let mut degraded = source.clone();
292        degraded
293            .set_actions_lenient(vec![
294                measure,
295                copy,
296                Action::Let {
297                    name: "invalid".into(),
298                    expr: Expr::Var("missing".into()),
299                },
300            ])
301            .unwrap();
302        assert!(degraded.action_graph_error().is_some());
303        let (expanded, _) = expand_spatial_port_topology(&degraded, 3).unwrap();
304        assert_eq!(expanded.actions(), degraded.actions());
305        assert_eq!(
306            expanded.action_graph().inputs().collect::<Vec<_>>(),
307            ["external"]
308        );
309        assert_eq!(
310            expanded.action_graph_error().map(ToString::to_string),
311            degraded.action_graph_error().map(ToString::to_string)
312        );
313        assert_eq!(
314            expanded
315                .action_graph()
316                .node_by_ordinal(0)
317                .unwrap()
318                .measurement,
319            measurement
320        );
321        assert_eq!(
322            degraded.get_block(IVec3::ZERO).unwrap().kind(),
323            BlockKind::Port
324        );
325    }

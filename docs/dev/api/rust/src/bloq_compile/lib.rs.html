4//!

6//! convention, generates per-block circuit templates, composes detectors and

23/// A set usable as a GF(2) parity accumulator — see [`xor_toggle`].
24pub(crate) trait XorToggleSet<T> {
25    /// Insert `item` if absent, remove it if present.
26    fn toggle(&mut self, item: T);
27}
28
29impl<T: Eq + std::hash::Hash, S: std::hash::BuildHasher> XorToggleSet<T>
30    for std::collections::HashSet<T, S>
31{
32    fn toggle(&mut self, item: T) {
33        if !self.remove(&item) {
34            self.insert(item);
35        }
36    }
37}
38
39impl<T: Ord> XorToggleSet<T> for std::collections::BTreeSet<T> {
40    fn toggle(&mut self, item: T) {
41        if !self.remove(&item) {
42            self.insert(item);
43        }
44    }
45}

47/// XOR a stream of items into `set`: an item seen an even number of times
48/// cancels out of the parity. The one symmetric-difference kernel shared by
49/// observable, gateway, and output-frame folding.
50pub(crate) fn xor_toggle<T, S: XorToggleSet<T>>(set: &mut S, items: impl IntoIterator<Item = T>) {
51    for item in items {
52        set.toggle(item);
53    }
54}
55
56/// XOR `items` into a parity multiset, returning the survivors sorted for
57/// deterministic downstream ordering.
58///
59/// Sorting first and keeping one element per odd-length run is the same
60/// symmetric difference as [`xor_toggle`] into a set, minus the hashing — and
61/// the result had to be sorted regardless.
62pub(crate) fn xor_toggled_sorted<T: Copy + Ord>(items: impl IntoIterator<Item = T>) -> Vec<T> {
63    let mut all: Vec<T> = items.into_iter().collect();
64    all.sort_unstable();
65    all.chunk_by(|a, b| a == b)
66        .filter(|run| run.len() % 2 == 1)
67        .map(|run| run[0])
68        .collect()
69}

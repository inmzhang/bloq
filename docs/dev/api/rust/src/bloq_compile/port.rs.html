8/// One ideal boundary, with an optional spatial-to-temporal cube substitution.
9#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
10pub(crate) struct PortExpansion {
11    pub(crate) source: IVec3,
12    pub(crate) role: PortRole,
13    pub(crate) cube: Option<(CubeKind, CubeHeight)>,
14    pub(crate) pipe_dir: Direction,
15    pub(crate) boundary_basis: Basis,
16    /// Private input/output in boundary-template-local circuit coordinates.
17    pub(crate) private_qubit: Option<IVec2>,
18}
19
20impl PortExpansion {
21    pub(crate) const fn is_spatial(self) -> bool {
22        self.cube.is_some()
23    }
24
25    pub(crate) fn has_future_pipe(self) -> bool {
26        self.pipe_dir == Direction::ZPLUS
27    }
28
29    /// Pipe a spatial Port's derived cube opens toward its virtual temporal half.
30    pub(crate) fn cube_pipe_dir(self) -> Option<Direction> {
31        self.cube.map(|_| self.pipe_dir.negate())
32    }
33
34    pub(crate) fn needs_private_qubit(self) -> bool {
35        !self.is_spatial() || self.role == PortRole::Multiplex
36    }
37}
38
39pub(crate) type PortExpansionMap = crate::FxMap<IVec3, PortExpansion>;
40
41/// Plan ideal boundaries and replace spatial Ports with their cube half.
42/// `LowerPlan` adds the positionless temporal half of each spatial boundary.
43pub(crate) fn expand_port_topology(
44    source: &BlockGraph,
45    distance: u32,
46) -> Result<(Cow<'_, BlockGraph>, PortExpansionMap), CompileError> {
47    let mut expansions = port_expansions(source, source.blocks().map(bloq_graph::Block::pos))?;
48    allocate_private_port_qubits(source, distance, &mut expansions)?;
49    apply_port_expansions(source, expansions)
50}

52/// Expand only complete local sites, using the private coordinates reserved
53/// against the full source's occupancy. Halo blocks remain lookup context.
54pub(crate) fn expand_port_context<'a>(
55    source: &'a BlockGraph,
56    positions: &[IVec3],
57    global: &PortExpansionMap,
58) -> Result<(Cow<'a, BlockGraph>, PortExpansionMap), CompileError> {
59    let mut expansions = port_expansions(source, positions.iter().copied())?;
60    for expansion in expansions.values_mut() {
61        if expansion.needs_private_qubit() {
62            expansion.private_qubit = global[&expansion.source].private_qubit;
63        }
64    }
65    apply_port_expansions(source, expansions)
66}

68fn port_expansions(
69    source: &BlockGraph,
70    positions: impl IntoIterator<Item = IVec3>,
71) -> Result<PortExpansionMap, CompileError> {
72    let mut expansions = PortExpansionMap::default();
73    for position in positions {
74        let block = source.get_block(position).expect("selected site exists");
75        if !block.kind().is_port() {
76            continue;
77        }
78        let pos = block.pos();
79        let Some(pipe) = source.pipes_at(pos).next() else {
80            continue;
81        };
82        let neighbor = source
83            .get_endpoint_block(if pipe.src() == pos {
84                pipe.dst()
85            } else {
86                pipe.src()
87            })
88            .expect("Port pipe has a neighbor");
89        let role = block.port_role().expect("Port blocks expose a role");
90        if !pipe.dir().is_spatial() {
91            let pipe_dir = if neighbor.pos().z > pos.z {
92                Direction::ZPLUS
93            } else {
94                Direction::ZMINUS
95            };
96            let inferred = if pipe_dir == Direction::ZPLUS {
97                PortRole::Input
98            } else {
99                PortRole::Output
100            };
101            if role == PortRole::Auto || role == inferred {
102                continue;
103            }
104            let boundary_basis =
105                crate::signature::BlockSignature::temporal_neighbor_boundary_basis(block, source)
106                    .ok_or(CompileError::MissingBoundaryBasis {
107                    kind: BlockKind::Port,
108                })?;
109            expansions.insert(
110                pos,
111                PortExpansion {
112                    source: pos,
113                    role,
114                    cube: None,
115                    pipe_dir,
116                    boundary_basis,
117                    private_qubit: None,
118                },
119            );
120            continue;
121        }
122        debug_assert_ne!(
123            role,
124            PortRole::Auto,
125            "source validation requires a spatial role"
126        );
127        let BlockKind::Cube(_) = neighbor.kind() else {
128            return Err(CompileError::SpatialPortNeighborNotCube {
129                port: pos,
130                neighbor: neighbor.pos(),
131            });
132        };
133        if neighbor.height_cells() != 1 {
134            return Err(CompileError::TallSpatialPortUnsupported {
135                port: pos,
136                height: neighbor.height(),
137            });
138        }
139        let cube_kind = source
140            .infer_spatial_port_cube_kind(pos)
141            .ok_or(CompileError::SpatialPortCubeInferenceFailed { port: pos })?;
142        expansions.insert(
143            pos,
144            PortExpansion {
145                source: pos,
146                role,
147                cube: Some((cube_kind, neighbor.height())),
148                pipe_dir: if role.has_input_boundary() {
149                    Direction::ZPLUS
150                } else {
151                    Direction::ZMINUS
152                },
153                boundary_basis: cube_kind.y(),
154                private_qubit: None,
155            },
156        );
157    }
158
159    Ok(expansions)
160}

162fn apply_port_expansions(
163    source: &BlockGraph,
164    expansions: PortExpansionMap,
165) -> Result<(Cow<'_, BlockGraph>, PortExpansionMap), CompileError> {
166    if !expansions.values().any(|port| port.is_spatial()) {
167        return Ok((Cow::Borrowed(source), expansions));
168    }
169
170    let mut graph = source.clone();
171    // Geometry edits each refresh the whole action DAG. Its external inputs
172    // remain declared while the actions are detached, then one final lenient
173    // rebuild preserves the mutators' error and measurement metadata contract.
174    let actions = graph.actions();
175    graph.set_actions_lenient(Vec::new())?;
176    for expansion in expansions.values() {
177        if let Some((kind, height)) = expansion.cube {
178            graph.set_block_kind(expansion.source, BlockKind::Cube(kind))?;
179            graph.set_cube_height(expansion.source, height)?;
180        }
181    }
182    graph.set_actions_lenient(actions)?;
183    Ok((Cow::Owned(graph), expansions))
184}

186pub(crate) fn allocate_private_port_qubits(
187    source: &BlockGraph,
188    distance: u32,
189    expansions: &mut PortExpansionMap,
190) -> Result<(), CompileError> {
191    let mut ports = expansions
192        .values()
193        .filter(|port| port.needs_private_qubit())
194        .map(|port| port.source)
195        .collect::<Vec<_>>();
196    if ports.is_empty() {
197        return Ok(());
198    }
199    ports.sort_unstable_by_key(glam::IVec3::to_array);
200
201    let mut occupied: crate::FxSet<_> = crate::signature::occupied_branch_positions(source)?
202        .into_iter()
203        .map(IVec3::truncate)
204        .collect();
205    // Keep clear of compiler templates that spill one cell past source occupancy.
206    for cell in occupied.clone() {
207        for delta in [IVec2::X, IVec2::Y, IVec2::NEG_X, IVec2::NEG_Y] {
208            if let (Some(x), Some(y)) = (cell.x.checked_add(delta.x), cell.y.checked_add(delta.y)) {
209                occupied.insert(IVec2::new(x, y));
210            }
211        }
212    }
213    for port in ports {
214        let cell = port.truncate();
215        let limit = occupied.len().saturating_add(1);
216        let output = (1..=limit)
217            .filter_map(|radius| i32::try_from(radius).ok())
218            .flat_map(|radius| {
219                [
220                    IVec2::new(radius, 0),
221                    IVec2::new(0, radius),
222                    IVec2::new(-radius, 0),
223                    IVec2::new(0, -radius),
224                ]
225            })
226            .find_map(|delta| {
227                let candidate =
228                    IVec2::new(cell.x.checked_add(delta.x)?, cell.y.checked_add(delta.y)?);
229                if occupied.contains(&candidate) {
230                    return None;
231                }
232                block_xy_offset(candidate, distance).ok()?;
233                Some((candidate, block_xy_offset(delta, distance).ok()?))
234            });
235        let Some((cell, output)) = output else {
236            return Err(if expansions[&port].role == PortRole::Multiplex {
237                CompileError::MultiplexOutputCoordinateUnavailable { port }
238            } else {
239                CompileError::PrivatePortCoordinateUnavailable { port }
240            });
241        };
242        occupied.insert(cell);
243        expansions
244            .get_mut(&port)
245            .expect("private boundary came from expansion map")
246            .private_qubit = Some(output);
247    }
248    Ok(())
249}

251#[cfg(test)]
252mod tests {
253    use super::*;
254    use bloq_graph::{Action, Block, Expr, MeasureTarget, Pipe};
255
256    #[test]
257    fn batch_expansion_preserves_actions_inputs_and_lenient_error() {
258        let mut source = BlockGraph::new();
259        for x in [0, 10] {
260            let port = IVec3::new(x, 0, 0);
261            source.add_block(
262                Block::new(port, BlockKind::Port)
263                    .with_port_role(PortRole::Input)
264                    .unwrap(),
265            );
266            source.add_block(Block::new(port + IVec3::X, BlockKind::Cube(CubeKind::ZXZ)));
267            source.add_pipe(Pipe::new(port, Direction::XPLUS));
268        }
269        source
270            .set_actions_with_inputs(Vec::new(), ["external".to_owned()])
271            .unwrap();
272        let measure = Action::Measure {
273            target: MeasureTarget::Node(IVec3::new(11, 0, 0)),
274            name: "measured".into(),
275        };
276        let copy = Action::Let {
277            name: "copy".into(),
278            expr: Expr::Var("external".into()),
279        };
280        source
281            .set_actions_lenient(vec![measure.clone(), copy.clone()])
282            .unwrap();
283        assert!(source.action_graph_error().is_none());
284        let measurement = source
285            .action_graph()
286            .node_by_ordinal(0)
287            .unwrap()
288            .measurement;
289        assert!(measurement.is_some());
290
291        let (expanded, ports) = expand_port_topology(&source, 3).unwrap();
292        assert_eq!(ports.len(), 2);
293        assert_eq!(expanded.actions(), source.actions());
294        assert_eq!(
295            expanded.action_graph().inputs().collect::<Vec<_>>(),
296            ["external"]
297        );
298        assert!(expanded.action_graph_error().is_none());
299        assert_eq!(
300            expanded
301                .action_graph()
302                .node_by_ordinal(0)
303                .unwrap()
304                .measurement,
305            measurement
306        );
307
308        let mut degraded = source.clone();
309        degraded
310            .set_actions_lenient(vec![
311                measure,
312                copy,
313                Action::Let {
314                    name: "invalid".into(),
315                    expr: Expr::Var("missing".into()),
316                },
317            ])
318            .unwrap();
319        assert!(degraded.action_graph_error().is_some());
320        let (expanded, _) = expand_port_topology(&degraded, 3).unwrap();
321        assert_eq!(expanded.actions(), degraded.actions());
322        assert_eq!(
323            expanded.action_graph().inputs().collect::<Vec<_>>(),
324            ["external"]
325        );
326        assert_eq!(
327            expanded.action_graph_error().map(ToString::to_string),
328            degraded.action_graph_error().map(ToString::to_string)
329        );
330        assert_eq!(
331            expanded
332                .action_graph()
333                .node_by_ordinal(0)
334                .unwrap()
335                .measurement,
336            measurement
337        );
338        assert_eq!(
339            degraded.get_block(IVec3::ZERO).unwrap().kind(),
340            BlockKind::Port
341        );
342    }

27pub(super) struct AnchorIncidence {
28    positions: Vec<IVec3>,
29    indices: FxMap<IVec3, usize>,
30    edges: FxMap<(usize, usize), DecisionId>,
31    descents: FxMap<(usize, usize, bool), DecisionId>,
32}
33
34impl AnchorIncidence {
35    pub(super) fn new(
36        positions: Vec<IVec3>,
37        diagram: &mut BooleanDecisionDiagram,
38    ) -> Result<Self, CompileError> {
39        diagram.charge(positions.len().saturating_mul(8))?;
40        let indices = positions
41            .iter()
42            .enumerate()
43            .map(|(index, &position)| (position, index))
44            .collect();
45        Ok(Self {
46            positions,
47            indices,
48            edges: FxMap::default(),
49            descents: FxMap::default(),
50        })
51    }
52
53    pub(super) fn add_variant(
54        &mut self,
55        diagram: &mut BooleanDecisionDiagram,
56        position: IVec3,
57        graph: &BlockGraph,
58        guard: DecisionId,
59    ) -> Result<(), CompileError> {
60        diagram.charge(7)?;
61        let index = self.indices[&position];
62        let block = graph.get_block(position).expect("local center exists");
63        let connectivity = BlockSignature::graph_connectivity(block, graph);
64        for direction in Direction::iter() {
65            let Some((neighbor, hadamard)) =
66                crate::lower::physical_neighbor(graph, block, connectivity, direction)
67            else {
68                continue;
69            };
70            let neighbor_index = self.indices[&neighbor.pos()];
71            let value = self.edges.entry((index, neighbor_index)).or_insert(ZERO);
72            *value = diagram.apply(BooleanOp::Or, *value, guard)?;
73            if direction == Direction::ZMINUS {
74                debug_assert!(neighbor.pos().z < position.z);
75                let value = self
76                    .descents
77                    .entry((index, neighbor_index, hadamard))
78                    .or_insert(ZERO);
79                *value = diagram.apply(BooleanOp::Or, *value, guard)?;
80            }
81        }
82        Ok(())
83    }
84}

86/// One source-wide symbolic index; no selected full graph is retained here.
87pub(crate) struct GuardedDynamicAnchors {
88    sites: FxMap<IVec3, GuardedAnchor>,
89    case_limit: usize,
90}
91
92impl GuardedDynamicAnchors {
93    pub(crate) fn decisions_mut(&mut self) -> impl Iterator<Item = &mut DecisionId> {
94        self.sites.values_mut().flat_map(|site| {
95            std::iter::once(&mut site.dynamic)
96                .chain(site.sources.iter_mut().map(|(_, guard)| guard))
97        })
98    }

100    pub(crate) fn new(
101        topology: &mut GuardedTopology,
102        limits: ModuleCertificationLimits,
103    ) -> Result<Self, CompileError> {
104        let positions = topology
105            .sites
106            .keys()
107            .copied()
108            .map(IVec3::from_array)
109            .collect::<Vec<_>>();
110        let mut incidence = AnchorIncidence::new(positions, &mut topology.diagram)?;
111        for (&position, variants) in &topology.sites {
112            let position = IVec3::from_array(position);
113            for variant in variants {
114                incidence.add_variant(
115                    &mut topology.diagram,
116                    position,
117                    &variant.graph,
118                    variant.guard,
119                )?;
120            }
121        }
122
123        Self::from_incidence(topology, incidence, limits)
124    }

126    pub(super) fn from_incidence(
127        topology: &mut GuardedTopology,
128        incidence: AnchorIncidence,
129        limits: ModuleCertificationLimits,
130    ) -> Result<Self, CompileError> {
131        let AnchorIncidence {
132            positions,
133            indices,
134            edges,
135            descents,
136        } = incidence;
137
138        let mut reverse = vec![Vec::new(); positions.len()];
139        topology.diagram.charge(edges.len())?;
140        for ((source, target), guard) in edges {
141            reverse[target].push((source, guard));
142        }
143        let mut dynamic = vec![ZERO; positions.len()];
144        let mut pending = VecDeque::new();
145        // Source validation forbids dynamic blocks inside structural arms.
146        topology.diagram.charge(topology.source.block_count())?;
147        for block in topology
148            .source
149            .blocks()
150            .filter(|block| block.kind().is_dynamic())
151        {
152            let index = indices[&block.pos()];
153            if dynamic[index] != ONE {
154                dynamic[index] = ONE;
155                pending.push_back(index);
156            }
157        }
158        // Saturate paths of always-present reverse edges before composing
159        // guarded paths. TRUE then absorbs every conditional contribution.
160        while let Some(source) = pending.pop_front() {
161            topology
162                .diagram
163                .charge(reverse[source].len().saturating_add(1))?;
164            for &(target, edge) in &reverse[source] {
165                if edge == ONE && dynamic[target] != ONE {
166                    dynamic[target] = ONE;
167                    pending.push_back(target);
168                }
169            }
170        }
171        topology.diagram.charge(dynamic.len())?;
172        let mut queued = vec![false; positions.len()];
173        for (index, &root) in dynamic.iter().enumerate() {
174            if root == ONE {
175                queued[index] = true;
176                pending.push_back(index);
177            }
178        }
179        while let Some(source) = pending.pop_front() {
180            queued[source] = false;
181            topology
182                .diagram
183                .charge(reverse[source].len().saturating_add(1))?;
184            for &(target, edge) in &reverse[source] {
185                if dynamic[target] == ONE {
186                    continue;
187                }
188                let contribution = topology
189                    .diagram
190                    .apply(BooleanOp::And, edge, dynamic[source])?;
191                let next = topology
192                    .diagram
193                    .apply(BooleanOp::Or, dynamic[target], contribution)?;
194                if next != dynamic[target] {
195                    dynamic[target] = next;
196                    if !std::mem::replace(&mut queued[target], true) {
197                        pending.push_back(target);
198                    }
199                }
200            }
201        }
202
203        let mut incoming = vec![Vec::new(); positions.len()];
204        topology.diagram.charge(descents.len())?;
205        for ((source, target, flipped), guard) in descents {
206            incoming[source].push((target, flipped, guard));
207        }
208        topology.diagram.charge(topology.source.block_count())?;
209        let t_sources = topology
210            .source
211            .blocks()
212            .filter(|block| block.kind() == BlockKind::T)
213            .map(Block::pos)
214            .collect::<crate::FxSet<_>>();
215        let mut order = (0..positions.len()).collect::<Vec<_>>();
216        topology.diagram.charge(
217            positions
218                .len()
219                .saturating_mul(positions.len().max(1).ilog2() as usize + 1),
220        )?;
221        order.sort_unstable_by_key(|&index| {
222            let position = positions[index];
223            (position.z, position.x, position.y)
224        });
225        let mut lineages = vec![Vec::<((IVec3, bool), DecisionId)>::new(); positions.len()];
226        let mut coefficients = 0usize;
227        for index in order {
228            topology
229                .diagram
230                .charge(incoming[index].len().saturating_add(1))?;
231            let position = positions[index];
232            let mut sources = BTreeMap::<([i32; 3], bool), DecisionId>::new();
233            let mut add_source = |key,
234                                  contribution,
235                                  diagram: &mut bloq_utils::boolean::BooleanDecisionDiagram|
236             -> Result<(), CompileError> {
237                if !sources.contains_key(&key) {
238                    coefficients = add_resource(
239                        "guarded T lineage coefficients",
240                        coefficients,
241                        1,
242                        limits.max_witness_nodes,
243                    )?;
244                }
245                let guard = sources.entry(key).or_insert(ZERO);
246                *guard = diagram.apply(BooleanOp::Or, *guard, contribution)?;
247                Ok(())
248            };
249            if t_sources.contains(&position) {
250                add_source((position.to_array(), false), ONE, &mut topology.diagram)?;
251            } else {
252                for &(below, flipped, edge) in &incoming[index] {
253                    topology.diagram.charge(lineages[below].len())?;
254                    for &((source, previous_flip), guard) in &lineages[below] {
255                        let contribution = topology.diagram.apply(BooleanOp::And, edge, guard)?;
256                        if contribution == ZERO {
257                            continue;
258                        }
259                        add_source(
260                            (source.to_array(), flipped ^ previous_flip),
261                            contribution,
262                            &mut topology.diagram,
263                        )?;
264                    }
265                }
266            }
267            topology.diagram.charge(sources.len())?;
268            lineages[index] = sources
269                .into_iter()
270                .map(|((source, flipped), guard)| ((IVec3::from_array(source), flipped), guard))
271                .collect();
272        }
273        topology.diagram.charge(positions.len())?;
274        Ok(Self {
275            sites: positions
276                .into_iter()
277                .zip(dynamic)
278                .zip(lineages)
279                .map(|((position, dynamic), sources)| {
280                    (position, GuardedAnchor { dynamic, sources })
281                })
282                .collect(),
283            case_limit: limits.max_guarded_domain_size,
284        })
285    }

287    /// T lineage is a partial function, so intersect its mutually exclusive
288    /// alternatives directly instead of enumerating a one-hot Cartesian tuple.
289    pub(crate) fn cases(
290        &self,
291        topology: &mut GuardedTopology,
292        position: IVec3,
293        enabled: DecisionId,
294    ) -> Result<Vec<(DecisionId, DynamicAnchor)>, CompileError> {
295        let site = &self.sites[&position];
296        topology
297            .diagram
298            .charge(site.sources.len().saturating_add(3))?;
299        let mut remaining = topology
300            .diagram
301            .apply(BooleanOp::And, enabled, topology.domain)?;
302        let mut output = Vec::new();
303        let mut push = |guard, anchor| -> Result<(), CompileError> {
304            check_resource(
305                "local dynamic anchor patterns",
306                output.len().saturating_add(1),
307                self.case_limit,
308            )?;
309            output.push((guard, anchor));
310            Ok(())
311        };
312        for &(source, guard) in &site.sources {
313            let active = topology.diagram.apply(BooleanOp::And, remaining, guard)?;
314            if active == ZERO {
315                continue;
316            }
317            push(
318                active,
319                DynamicAnchor {
320                    touches_dynamic: true,
321                    source_t: Some(source),
322                },
323            )?;
324            let absent = topology.diagram.negate(guard)?;
325            remaining = topology.diagram.apply(BooleanOp::And, remaining, absent)?;
326        }
327        let dynamic = topology
328            .diagram
329            .apply(BooleanOp::And, remaining, site.dynamic)?;
330        if dynamic != ZERO {
331            push(
332                dynamic,
333                DynamicAnchor {
334                    touches_dynamic: true,
335                    source_t: None,
336                },
337            )?;
338        }
339        let static_guard = topology.diagram.negate(site.dynamic)?;
340        let static_guard = topology
341            .diagram
342            .apply(BooleanOp::And, remaining, static_guard)?;
343        if static_guard != ZERO {
344            push(static_guard, DynamicAnchor::default())?;
345        }
346        topology.diagram.charge(output.len())?;
347        for (guard, _) in &mut output {
348            *guard = topology.diagram.constrain(*guard, topology.domain)?;
349        }
350        Ok(output)
351    }

362    fn assert_matches_full_queries(source: &BlockGraph) {
363        let limits = ModuleCertificationLimits::DEFAULT;
364        let mut topology = GuardedTopology::new(source, limits).unwrap();
365        let anchors = GuardedDynamicAnchors::new(&mut topology, limits).unwrap();
366        let positions = topology.sites.keys().copied().collect::<Vec<_>>();
367        for position in positions {
368            let position = IVec3::from_array(position);
369            let enabled = topology.sites[&position.to_array()]
370                .iter()
371                .try_fold(ZERO, |guard, variant| {
372                    topology.diagram.apply(BooleanOp::Or, guard, variant.guard)
373                })
374                .unwrap();
375            for (guard, anchor) in anchors.cases(&mut topology, position, enabled).unwrap() {
376                let graph = topology.project(guard).unwrap();
377                let (graph, ports) =
378                    crate::spatial_port::expand_spatial_port_topology(&graph, 3).unwrap();
379                let compiled = graph
380                    .blocks()
381                    .map(|block| {
382                        let mut connectivity = BlockSignature::graph_connectivity(block, &graph);
383                        if let Some(port) = ports.get(&block.pos()) {
384                            connectivity = connectivity.with_pipe(port.cube_pipe_dir());
385                        }
386                        (
387                            block.pos(),
388                            CompiledTemplateInfo {
389                                graph_connectivity: connectivity,
390                                // These queries read connectivity and source kinds only.
391                                template: TemplateRef::Fixed(crate::block::LoweringTemplateId(0)),
392                            },
393                        )
394                    })
395                    .collect();
396                assert_eq!(
397                    anchor.touches_dynamic,
398                    crate::lower::patch_touches_dynamic_block(&graph, &compiled, position),
399                    "{position}"
400                );
401                assert_eq!(
402                    anchor.source_t,
403                    crate::lower::source_t_block(&graph, &compiled, position),
404                    "{position}"
405                );
406            }
407        }
408    }

528        let mut topology = GuardedTopology::new(&source, limits).unwrap();

534        let backward_only = ivec3(13, 0, 0);

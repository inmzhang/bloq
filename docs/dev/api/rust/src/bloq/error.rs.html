1//! Shared errors for workflows that cross crate boundaries.
2
3use std::error::Error as StdError;
4
5type BoxError = Box<dyn StdError + Send + Sync + 'static>;
6
7/// A result using the facade's [`Error`] by default.
8///
9/// `Result` means `Result<(), Error>`. An explicit error parameter remains
10/// available for functions that need a narrower error type.
11pub type Result<T = (), E = Error> = std::result::Result<T, E>;
12
13/// A small container retaining the typed error from a Bloq workflow.
14///
15/// Subsystem APIs keep their concrete errors. In a function returning [`Result`],
16/// `?` wraps those errors automatically. Use [`Self::downcast_ref`] to inspect
17/// the original error and [`StdError::source`] to traverse its causes.
18/// Application-specific errors can be wrapped with [`Self::new`].
19///
20/// ```
21/// use bloq::prelude::*;
22///
23/// fn circuit_text(source: &str) -> Result<String> {
24///     let graph = parse_blog_to_graph(source)?;
25///     let config = CompileConfig::try_new(3)?;
26///     let artifacts = compile_with(&graph, config)?;
27///     Ok(emit_bloq_stim(&artifacts.bloq)?)
28/// }
29///
30/// let error = circuit_text("invalid BLOG").unwrap_err();
31/// assert!(error.is::<bloq::graph::BlockGraphError>());
32/// ```
33#[derive(Debug, thiserror::Error)]
34#[error("{0}")]
35pub struct Error(#[source] BoxError);
36
37impl Error {
38    /// Wrap an owned error, retaining its type, message, and causes.
39    ///
40    /// ```
41    /// let error = bloq::Error::new("invalid".parse::<u32>().unwrap_err());
42    /// assert!(error.is::<std::num::ParseIntError>());
43    /// ```
44    pub fn new(error: impl StdError + Send + Sync + 'static) -> Self {
45        Self(Box::new(error))
46    }
47
48    /// Whether the original error has type `E`.
49    pub fn is<E: StdError + 'static>(&self) -> bool {
50        self.0.is::<E>()
51    }
52
53    /// Borrow the original error if it has type `E`.
54    pub fn downcast_ref<E: StdError + 'static>(&self) -> Option<&E> {
55        self.0.downcast_ref::<E>()
56    }
57}
58
59impl From<BoxError> for Error {
60    fn from(error: BoxError) -> Self {
61        Self(error)
62    }
63}
64
65// Keep automatic conversions without duplicating subsystem error enums.
66// A blanket From<E> would overlap From<Error> because Error implements StdError.
67macro_rules! impl_from {
68    ($($(#[$meta:meta])* $error:ty),* $(,)?) => {
69        $(
70            $(#[$meta])*
71            impl From<$error> for Error {
72                fn from(error: $error) -> Self {
73                    Self::new(error)
74                }
75            }
76        )*
77    };
78}
